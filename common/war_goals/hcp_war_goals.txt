hcp_wg_end_threat_harvesters = {
	casus_belli = hcp_cb_containment

	hide_if_no_cb = yes
	total_war = yes
	set_defender_wargoal = hcp_wg_resource_acquisition

	forbidden_peace_offers = {
		demand_surrender = ""
		surrender = ""
	}

	surrender_acceptance = -1000

	potential = {
		from = { 
			has_menace_perk = hcp_menp_galactic_harvest
		}
	}
	possible = {
		always = yes
	}

	ai_weight = {
		weight = 100
	}
}

hcp_wg_resource_acquisition = {
	casus_belli = hcp_cb_harvest
	hide_if_no_cb = yes
	total_war = yes
	set_defender_wargoal = hcp_wg_end_threat_harvesters

	galactic_empire_joins_defender = {
		from = { is_galactic_emperor = yes }
		imperial_authority >= 100
	}

	forbidden_peace_offers = {
		demand_surrender = ""
		surrender = ""
	}

	surrender_acceptance = -1000

	potential = {
		from = { NOT = { is_country_type = fallen_empire } }
		NAND = {
			is_galactic_emperor = yes
			from = { is_galactic_community_member = yes }
		}
	}
	possible = {
		custom_tooltip = {
			fail_text = cannot_use_against_emperor
			OR = {
				from = { is_galactic_emperor = no }
				is_galactic_community_member = no
			}
		}
		custom_tooltip = {
			fail_text = pax_galactica_active
			NAND = {
				is_active_resolution = resolution_emperor_pax_galactica
				is_galactic_community_member = yes
				from = { is_galactic_community_member = yes }
			}
		}
	}

	ai_weight = {
		weight = 100
	}
}
