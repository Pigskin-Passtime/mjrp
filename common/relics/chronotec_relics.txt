r_chronotec_standard = {
	activation_duration = @chronotec_triumph_duration
	portrait = "GFX_chronotec_relic_standard"
	sound = "relic_activation_extradimensional_warlock"

	resources = {
		category = relics
		# Activation cost
		cost = {
			unity = @chronotec_activation_cost
		}
	}

	ai_weight = {
		weight = 100
	}

	triggered_country_modifier = {
		potential = {
			always = yes
		}
		country_unity_produces_mult = 0.05
		all_technology_research_speed = 0.05
	}

	score = 1000

	active_effect = {
		# Standard wibbly wobbly timey stuff
		custom_tooltip = r_chronotec_standard_effect_tooltip
		if = {	# Leader is active - Chronocharges
			limit = {
				hidden_trigger = {
					is_same_value = event_target:chronotec_actual_owner_country
				}
			}
			custom_tooltip = r_chronotec_standard_leader_effect_tooltip
		}
		custom_tooltip = relic_triumph_cooldown
		hidden_effect = {
			add_modifier = {
				modifier = "relic_activation_cooldown"
				days = @chronotec_triumph_duration
			}
			if = {
				limit = {
					NOT = {
						has_country_flag = chronotec_activated_relic
					}
				}
				set_country_flag = chronotec_activated_relic
				country_event = {
					id = chronotec_relic.10
					days = 0
				}
			}
			else = {
				country_event = {
					id = chronotec_relic.50
					days = 0
				}
			}
		}
	}

	# Possible check for activation
	possible = {
		always = yes
		inline_script = {
			script = relics/activation_checks
			RELIC = r_chronotec_standard
		}
	}
}
