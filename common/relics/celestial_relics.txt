r_celestial_compass_alive = {
	activation_duration = @triumph_duration
	portrait = "GFX_celestial_compass_alive"
	sound = "relic_activation_extradimensional_warlock"

	resources = {
		category = relics
		cost = {
			energy = 2500
		}
	}

	triggered_country_modifier = {
		potential = {
			always = yes
		}
		country_physics_tech_research_speed = 0.10
		ship_base_speed_mult = 0.15
	}

	score = 1000

	active_effect = {
		custom_tooltip = relic_compass_alive_effect
		custom_tooltip = relic_triumph_cooldown
		hidden_effect = {
			country_event = { id = celestial_setup.15 }
			set_country_flag = celestial_compass_used
		}
	}

	# Possible check for activation
	possible = {
		NOT = { has_country_flag = celestial_compass_used }
	}
}
