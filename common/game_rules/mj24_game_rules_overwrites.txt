# this = leader
can_leader_get_council_trait = {
	can_leader_get_council_trait_trigger = yes
	NOT = {
		has_leader_flag = no_council_traits				# mod jam overwrite
	}
}

# this = leader
can_leader_get_normal_trait = {
	NOR = {
		AND = {
			exists = owner
			owner = { is_gestalt = yes }
			OR = {
				is_councilor = yes
				has_leader_flag = restored_node
			}
		}
		AND = {
			has_global_flag = game_started
			is_ruler = yes
			exists = owner
			owner = {
				is_autocracy = yes
			}
			NOT = { has_leader_flag = ignore_autocracy_ruler_trait_restriction }
		}
		is_heir = yes
	}
	NOT = {
		has_leader_flag = no_normal_traits			# mod jam overwrite
	}
}

# this = leader
can_leader_get_subclass_trait = {
	NAND = {
		exists = owner
		owner = { is_gestalt = yes }
		OR = {
			is_councilor = yes
			has_leader_flag = restored_node
		}
		has_leader_flag = no_subclass_traits		# mod jam overwrite
	}
	has_any_subclass = no
}
