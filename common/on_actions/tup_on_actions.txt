# Triggers when the game starts
on_game_start = {
	events = {
		#tup_events.1
	}
}

# this = country
on_yearly_pulse_country = {
	events = {
		tup_crisis_events.306
		tup_crisis_events.307
		tup_crisis_events.308
		tup_crisis_events.309
	}
}

# this = country
on_five_year_pulse_country = {
	events = {
		tup_crisis_events.320
	}
}

# A Megastructure has been built
# Root = Country
# From = Megastructure
# FromFrom = System
# FromFromFrom = Fleet
on_megastructure_built = {
	events = {
		tup_crisis_events.310
	}
}

# THIS = System: system being occupied
# FROM = Country: Conqueror of the system
# FROMFROM = Country: Original owner of the system
on_system_occupied = {
	events = {
		tup_crisis_events.419 # utopic Dyson Swarm
		tup_crisis_events.420 # Unification Engine
	}
}


# This = owner of fleet 1 (winner)
# From = owner of fleet 2 (loser)
# FromFrom = fleet 1
# FromFromFrom = fleet 2
# This fires whenever a fleet is finished with a combat through any means (emergency ftl, destroyed, disabled)
on_space_battle_won = {
	events = {
		tup_situation_events.205
	}
}
