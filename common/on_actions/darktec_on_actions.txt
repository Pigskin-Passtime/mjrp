# Triggers when the game starts
on_game_start = {
	events = {
		darktec_misc.1	# Post-Baryonic - Sets mod flag
	}
}

on_game_start_country = {
	events = {
		darktec_misc.2	# Post-Baryonic - Removes unwanted flags if someone cheeses
	}
}

# No scope, like on_game_start
# Does not run when loading MP saves due to OOS concerns
on_single_player_save_game_load = {
	events = {
		darktec_crisis_endings.3010	# Post-Baryonic ending - Spawns a dimensional horror somewhere in the galaxy
	}
}

# No scope, like on_game_start
on_monthly_pulse = {
	events = {
	}
}

# No scope, like on_game_start
on_yearly_pulse = {
	events = {
		darktec_crisis_objectives.40	# Post-Baryonic Crisis objective - Retain Vassals
		darktec_crisis_objectives.50	# Post-Baryonic Crisis objective - Retain Megas
		darktec_crisis_objectives.60	# Post-Baryonic Crisis objective - Retain Galcom respect
	}
}

# this = country
on_monthly_pulse_country = {
	events = {
		darktec_misc.10					# Post-Baryonic Crisis Progression - Checks entropy levels to allow levels up or not
		darktec_crisis_objectives.10	# Post-Baryonic Crisis objective - Checks entropy produced to give the reward (and substract it later)

		darktec_mechanics.5001			# Post-Baryonic Crisis - Bypass in case an empire synth ascends so they don't lose the entrophereal trait
	}
}

# this = country
on_yearly_pulse_country = {
	events = {
		darktec_mechanics.6003			# Post-Baryonic Crisis - Void Threadware handler
	}
}

# this = country
on_bi_yearly_pulse_country = {
	events = {
		darktec_mechanics.1021			# Post-Baryonic Crisis - Saves entrophereal action dates
		darktec_mechanics.10			# Post-Baryonic Crisis - Entrophereal action sorting (growth/assim)
	}
}

# No scope, like on_game_start
on_decade_pulse = {
	events = {
		darktec_crisis_endings.7010	# Post-Baryonic Crisis ending - The collapse accelerates
		darktec_crisis_endings.7020	# Post-Baryonic Crisis ending - The collapse is terminal
	}
}

#A building construction has been completed.
# This = Planet
on_building_complete = {
	events = {
		darktec_crisis.27501	# Post-Baryonic Crisis process - built first anchor
		darktec_crisis.2101		# Post-Baryonic Crisis process - built first storage
		darktec_crisis.40301	# Post-Baryonic Crisis process - built first portal

		darktec_crisis_objectives.30	# Post-Baryonic Crisis objective - Built an umbral building
	}
}

# A building construction has finished, replacing another building.
# This = Planet
on_building_replaced = {
	events = {
		darktec_crisis.27501	# Post-Baryonic Crisis process - built first anchor
		darktec_crisis.2101		# Post-Baryonic Crisis process - built first storage
		darktec_crisis.40301	# Post-Baryonic Crisis process - built first portal
	}
}


# A building construction has been completed, which is an upgrade of previous building.
# This = Planet
on_building_upgraded = {
	events = {
		darktec_crisis_objectives.31	# Upgraded an umbral building
	}
}

# A colony has been destroyed by bombardment. Called after the planet's colony status has been cleared
# This = Planet
# From = Bombarder
on_colony_destroyed_by_bombardment = {
	events = {
		darktec_crisis_objectives.71	# Post-Baryonic Crisis objective - Harvested a world
	}
}

# Planet has taken damage from orbital bombardment - Damage is applied daily
# This = Planet
# From = Bombarder (country)
on_planet_bombarded = {
	events = {
		darktec_ships.200
	}
}

# Executed whenever a pop is killed due to bombardment damages
# This = Planet where the pop was bombed to death
# From = Country which is raining down fire and brimstone
# FromFrom = Biggest fleet bombarding the planet (the one we used the bombardment stance from)
# local_kill_pop_amount = amount of pop killed
on_pop_bombed_to_death = {
	events = {
		darktec_ships.150
	}
}

# Scope: Ship Event

# From = Planet

		darktec_crisis.2030		# Post-Baryonic Crisis - First Umbral Ship created

	}
}
