	portrait = "GFX_hcp_menp_standardised_orbital_mining"

hcp_menp_efficient_metallurgy = {
	portrait = "GFX_hcp_menp_efficient_metallurgy"

	on_unlock = {
		if = {
			limit = {
				is_catalytic_empire = no
				is_machine_empire = no
			}
			custom_tooltip = hcp_unlock_efficient_metallurgy_minerals
		}
		else_if = {
			limit = {
				is_catalytic_empire = no
				is_machine_empire = yes
			}
			custom_tooltip = hcp_unlock_efficient_metallurgy_minerals_machine
		}
		else_if = {
			limit = {
				is_catalytic_empire = yes
				is_machine_empire = no
			}
			custom_tooltip = hcp_unlock_efficient_metallurgy_food
		}
		else_if = {
			limit = {
				is_catalytic_empire = yes
				is_machine_empire = yes
			}
			custom_tooltip = hcp_unlock_efficient_metallurgy_food_machine
		}
	}
}
