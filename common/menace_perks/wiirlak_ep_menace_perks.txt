# Bloomform trait
wiirlak_ep_menap_terraforming_4 = {
	portrait = GFX_wiirlak_ep_menap_terraforming_4

	on_unlock = {
		custom_tooltip = wiirlak_ep_menap_terraforming_4_tooltip

		hidden_effect = {
			every_owned_species = {
				set_habitability_trait = {
					trait = trait_wiirlak_ep_pc_bloom_preference
				}
			}
		}

		tooltip = {
			custom_tooltip = wiirlak_ep_menap_modifier_1_tooltip
			add_modifier = {
				modifier = wiirlak_ep_modifier_perk_4
			}
		}
		
		tooltip = {
			custom_tooltip = wiirlak_ep_menap_modifier_1_galaxy_tooltip
			add_modifier = {
				modifier = wiirlak_ep_modifier_perk_4_galaxy
			}
		}
	}
}
