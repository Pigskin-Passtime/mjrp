	event_chain = wiirlak_ep_evergrowth_protocol_chain

		# Requirements has to be done to continue
		if = {
			limit = {

			}
		}

		# Otherwise, relaunch itself
		else = {
			abort_special_project = { type = WIIRLAK_EP_FINAL_PROJECT_1 }
			enable_special_project = { name = WIIRLAK_EP_FINAL_PROJECT_1 }
		}
	}
}
