special_project = {
	key = "hcp_NO_BLACK_HOLE"
	picture = GFX_evt_black_hole
	icon = "gfx/interface/icons/situation_log/situation_log_radar.dds"
	event_chain = hcp_harvesters_crisis_chain
	cost = 2000
	tech_department = physics_technology
	timelimit = -1
	event_scope = country_event

	abort_trigger = {
		hcp_has_valid_silo_site = { PROJECT = "hcp_NO_BLACK_HOLE" }
	}

	on_cancel = {
		hcp_target_valid_silo_site = { EVENT = hcp_harvesters_crisis.101 }
		abort_special_project = {
			type = hcp_NO_BLACK_HOLE
		}
	}

	on_success = {
		hcp_black_hole_discovered = { EVENT = hcp_harvesters_crisis.102 }
	}
}
