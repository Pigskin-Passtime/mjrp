ds_pr_level_1 = {
	required_crisis_currency = 0
	allow = {
		custom_tooltip = {
			success_text = ds_pr_level_1_req_tt
			fail_text =	ds_pr_level_1_req
			has_country_flag = ds_pr_crisis_level_1
		}
		hidden_trigger = {
			NOT = { has_active_event = { dsPR.5 } }
		}
	}
	perks = {
		menp_ds_pr_objectives
		menp_ds_pr_ethereal
		menp_ds_pr_unbreakable
	}
	on_unlock = {
		hidden_effect = {
			owner = {
				add_event_chain_counter = {
					event_chain = "ds_pr_crisis_chain"
					counter = "ds_pr_crisis_counter"
					amount = 1
				}
			}
		}
	}
}
