mgo_crisis_occultism_level_1 = {
	# REQUIREMENTS
	required_crisis_currency = 0

	# REWARDS
	perks = {
		occultist_alternative_approach
		occultist_ancient_stories 
		occultist_galactic_history
	}

	on_unlock = {
		set_country_flag = player_crisis_chosen
		set_variable = {
			which = mgo_has_pacified_demons
			value = 0
		}
	}
}

mgo_crisis_occultism_level_5 = {
	# REQUIREMENTS
	allow = {
		custom_tooltip = {
			success_text =	mgo_crisis_level_5_req_clear_tooltip
			fail_text =	mgo_crisis_level_5_req_tooltip
			has_country_flag = mgo_crisis_has_summoned_demons
		}
		is_subject = no
	}
	required_crisis_currency = 10000

	# REWARDS
	perks = {
		occultist_occult_rituals_3
		occultist_flavor_rituals_2
		occultist_greater_evil 
		occultist_endbringer
	}

	on_unlock = {
		owner = {
			if = {
				limit = { has_federation = yes }
				leave_alliance = { override_requirements = yes }
			}
			tooltip = {
				if = {
					limit = {
						is_galactic_community_formed = yes
						is_galactic_community_member = yes
					}
					remove_from_galactic_community = yes
				}
			}
		}
		hidden_effect = {
			declare_crisis_effect = yes
		}
	}
}
