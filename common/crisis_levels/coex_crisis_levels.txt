coex_crisis_coexistence_level_5 = {
	# REQUIREMENTS
	allow = {
		custom_tooltip = {
			success_text = coex_coexistence_level_5_req_clear_tooltip
			fail_text = coex_coexistence_level_5_req_tooltip
			has_megastructure = coex_enlightenment_array_3
			has_country_flag = coex_crisis_special_project_4_complete
		}
		is_subject = no
	}
	required_crisis_currency = 45000
	# REWARDS
	perks = {
		menp_coex_features_5
		menp_coex_enlightenment_5
	}
	on_unlock = {
		owner = {
			if = {
				limit = {
					has_federation = yes
				}
				leave_alliance = {
					override_requirements = yes
				}
			}
			tooltip = {
				if = {
					limit = {
						is_galactic_community_formed = yes
						is_galactic_community_member = yes
					}
					remove_from_galactic_community = yes
				}
			}
		}
		hidden_effect = {
			# force Declare Crisis resolution through an appropriate party
			declare_crisis_effect = yes
			owner = {
				#end_event_chain = become_the_crisis_chain
				country_event = {
					id = coex_crisis.7120
				}
			}
		}
	}
}
