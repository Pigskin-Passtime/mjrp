hcp_crisis_harvesters_level_1 = {
	# REQUIREMENTS
	required_crisis_currency = 0

	# REWARDS
	perks = {
		hcp_menp_prospector_drone
		hcp_menp_mining_efficiency
		hcp_menp_standardised_orbital_mining
		hcp_menp_infinite_storage
	}

	on_unlock = {
		hidden_effect = {
			owner = {
				country_event = { id = hcp_harvesters_crisis.100 }
				give_technology = {
					tech = hcp_tech_prospector_drone
					message = no
				}
				refresh_auto_generated_ship_designs = yes
				add_research_option = tech_mining_drone_weapon_1
			}
		}
	}
}
