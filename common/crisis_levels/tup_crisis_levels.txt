tup_crisis_utopiaproject_level_1 = {
	# REQUIREMENTS
	required_crisis_currency = 0

	# REWARDS
	perks = {
		tup_crisis_features_1_utopic_energy
		tup_crisis_features_1_leader_adjustments
	}

	on_unlock = {
		add_utopiaproject_level_1_perks = yes
		hidden_effect = {
			owner = {
				country_event = { id = tup_crisis_events.100 } 
			}
		}
	}
}

			fail_text =	tup_crisis_level_5_req3_tooltip
			has_country_flag = tup_unification_engine_test_3
		}
		custom_tooltip = {
			success_text = tup_crisis_level_5_req4_clear_tooltip
			fail_text =	tup_crisis_level_5_req4_tooltip
			has_country_flag = tup_unification_engine_test_4
		}
		custom_tooltip = {
			success_text = tup_crisis_level_5_req5_clear_tooltip
			fail_text =	tup_crisis_level_5_req5_tooltip
			has_country_flag = tup_unification_engine_test_5
		}
		custom_tooltip = {
			success_text = tup_crisis_level_5_req_clear_tooltip
			fail_text =	tup_crisis_level_5_req_tooltip
			has_megastructure = tup_unification_engine_1
		}
		is_subject = no
	}
	required_crisis_currency = 0

	# REWARDS
	perks = {
		tup_crisis_features_5_unification_engine_effects
	}

	on_unlock = {
		add_utopiaproject_level_5_perks = yes
		hidden_effect = {
			owner = {
				country_event = { id = tup_crisis_events.500 } 
			}

			declare_crisis_effect = yes
			every_playable_country = {
				limit = {
					is_galactic_community_member = no
					NOR = {
						is_same_value = owner
						is_at_war_with = owner
					}
				}
				join_war_on_side = {
					war = event_target:crisis_war
					side = event_target:community_war_leader
				}
				country_event = { id = tup_crisis_events.502 days = 30 } 
			}
		}
	}

}
