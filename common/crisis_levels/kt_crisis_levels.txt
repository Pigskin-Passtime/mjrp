kt_crisis_killing_thought_level_1 = {
	required_crisis_currency = 0

	perks = {
		kt_menp_everwar
		kt_menp_killing_momentum
		kt_menp_warlust
		kt_menp_grand_armada
	}

	on_unlock = {
		owner = {
			begin_event_chain = {
				event_chain = kt_killing_thought_chain
				target = this
			}
			add_event_chain_counter = {
				event_chain = "kt_killing_thought_chain"
				counter = "kt_crisis_level_reached"
				amount = 1
			}
			country_event = { id = kt.1 }
		}
	}
}
