cwis_crisis_level_1 = {
	# REQUIREMENTS
	required_crisis_currency = 0

	# REWARDS
	perks = {
		cwis_menp_objectives
		cwis_menp_modifier_1
		cwis_menp_edict_1
		cwis_menp_espionage_operations_1
	}

	on_unlock = {
		hidden_effect = {
			owner = {
				country_event = { id = cwis.7100 }
			}
		}
	}
}
