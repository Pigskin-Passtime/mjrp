veu_crisis_veu_level_1 = {
	# REQUIREMENTS
	required_crisis_currency = 0

	# REWARDS
	perks = {
		menp_veu_features_1
		menp_veu_buildings_1
	}

	on_unlock = {
			hidden_effect = {
			owner = {
				country_event = { id = veu_crisis.1001 }
			}
		}
	}
}

veu_crisis_veu_level_2 = {
	# REQUIREMENTS
	allow = {
		custom_tooltip = {
			success_text = veu_level_2_req_clear_tooltip
			fail_text =	veu_level_2_req_tooltip
		}
		has_country_flag = veu_lvl2_unlocked
	}
	required_crisis_currency = 500

	# REWARDS
	perks = {
		menp_veu_features_2
		menp_veu_ship_1
	}

	on_unlock = {
			hidden_effect = {
			owner = {
				add_event_chain_counter = {
					event_chain = "veu_chain"
					counter = "veu_level_reached"
					amount = 1
				}
				country_event = { id = veu_crisis.1002 }
			}
		}
	}
}

veu_crisis_veu_level_5 = {
	# REQUIREMENTS
	allow = {
		custom_tooltip = {
			success_text =	veu_level_5_req_clear_tooltip
			fail_text =	veu_level_5_req_tooltip
		}
		has_country_flag = veu_lvl5_unlocked
		is_subject = no
	}
	required_crisis_currency = 3000

	# REWARDS
	perks = {
		menp_veu_features_5
	}

	on_unlock = {
		hidden_effect = {
			# force declare crisis + trigger crisis war
			declare_crisis_effect = yes

			owner = {
				add_event_chain_counter = {
					event_chain = "veu_chain"
					counter = "veu_level_reached"
					amount = 1
				}
				end_event_chain = veu_chain
				country_event = { id = veu_crisis.1005 }
			}
		}
		
	}
}
