crisis_mjarchitect_level_1 = {
	# REQUIREMENTS
	required_crisis_currency = 0
	allow = {
		custom_tooltip = {
			success_text = crisis_mjarchitect_level_1_req_clear_tt
			fail_text =	crisis_mjarchitect_level_1_req_tt
			has_country_flag = mjarchitect_initial_chain_complete
		}
	}

	# REWARDS
	perks = {
		mjarchitect_crisis_perk_1
		mjarchitect_crisis_perk_2
		mjarchitect_crisis_perk_3
	}

	on_unlock = {
		hidden_effect = {
			owner = {
				create_message = {
					type = MESSAGE_MJARCHITECT_LEVELUP
					localization = MESSAGE_MJARCHITECT_LEVELUP
					days = 30
					target = root
				}
			}
		}
	}
}
