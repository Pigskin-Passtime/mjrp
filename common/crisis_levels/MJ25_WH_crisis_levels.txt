MJ25_WH_crisis_whispers_level_1 = {

	required_crisis_currency = 0

	perks = {
		MJ25_WH_menp_sound
		MJ25_WH_menp_attuned_arrays
		MJ25_WH_menp_congregational
		MJ25_WH_menp_translators
	}

	on_unlock = {
		hidden_effect = {
			owner = {
				begin_event_chain = { event_chain = MJ25_WH_whispers_crisis_chain target = this }
				add_event_chain_counter = {
					event_chain = "MJ25_WH_whispers_crisis_chain"
					counter = "crisis_level_reached"
					amount = 1
				}
				country_event = { id = MJ25_WH_whispers.1 days = 3 }
				country_event = { id = MJ25_WH_whispers.2000 days = 180 random = 360 }
			}
		}
	}

}

MJ25_WH_crisis_whispers_level_5 = {

	required_crisis_currency = 47200

	perks = {
		MJ25_WH_menp_we_are
		MJ25_WH_menp_stars_scream
		MJ25_WH_menp_starsong_tide
		MJ25_WH_menp_massa
	}

	allow = {
		custom_tooltip = {
			success_text = MJ25_WH_site_4_done
			fail_text = MJ25_WH_site_4_required
			AND = {
				has_country_flag = MJ25_WH_crisis_site_4_project_complete
				has_country_flag = MJ25_WH_finished_hue_situation
			}
		}
	}

	on_unlock = {
		hidden_effect = {
			declare_crisis_effect = yes
			every_playable_country = {
				limit = {
					is_galactic_community_member = no
					NOR = {
						is_same_value = owner
						is_at_war_with = owner
					}
				}
				join_war_on_side = {
					war = event_target:crisis_war
					side = event_target:community_war_leader
				}
			}
			owner = {
				add_event_chain_counter = {
					event_chain = "MJ25_WH_whispers_crisis_chain"
					counter = "crisis_level_reached"
					amount = 1
				}
				country_event = { id = MJ25_WH_whispers.40 days = 3 }
				# country_event = { id = MJ25_WH.1160 days = 15 }
				country_event = { id = MJ25_WH.1120 }
			}
		}
	}

}
