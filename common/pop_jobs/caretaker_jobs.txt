	swappable_data = {

			icon = high_monitor
		}
	}

	possible_pre_triggers = {
		has_owner = yes

		is_being_purged = no

	}
	possible_precalc = can_fill_ruler_job
	possible = {
		NOT = {
			has_trait = trait_mechanical
		}
		species = {
			is_same_value = owner_main_species
		}
	}
	# on purpose extremely high, low job number
	resources = {
		category = planet_caretaker_monitors
		produces = {
			# society_research = 300
			unity = 700
		}
	}
	
	triggered_country_modifier = {
		potential = {
			exists = owner
			owner = {
				has_valid_civic = civic_monitor_cult
			}
		}
		country_base_influence_produces_add = 5
	}
	planet_modifier = {
		logistic_growth_mult = 10
	}
	triggered_planet_modifier = {
		potential = {
			always = yes
		}
		planet_amenities_add = 100000
		mult = value:scripted_modifier_mult|MODIFIER|pop_job_amenities_mult|
	}
	weight = {
		weight = @ruler_job_weight
		mult = value:job_weights_modifier|JOB|politician|RESOURCE|unity|
		mult = value:scripted_modifier_job_weight_mult|MODIFIER|pop_job_amenities_mult|
		modifier = {
			factor = 0.1
			has_citizenship_rights = no
			NOT = {
				has_trait = trait_mechanical
			}
		}
		# crisis purge
		modifier = {
			factor = 0.01
			exists = planet
			exists = planet.controller
			planet.controller = {
				OR = {
					is_country_type = swarm
					is_country_type = ai_empire
				}
			}
		}
	}
	inline_script = "jobs/automodding_priority_charisma"
	promotion = {
		time = @standard_promotion_time
		effect = {
			pop_amount_change_category = {
				CATEGORY = ruler
				AMOUNT = local_pop_amount
			}
		}
	}
}
