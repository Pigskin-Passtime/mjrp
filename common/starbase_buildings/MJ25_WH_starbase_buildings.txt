MJ25_WH_thrumming_choirhall = {
	icon = "GFX_spaceport_building_MJ25_WH_thrumming_choirhall"
	construction_days = 720

	potential = {
		exists = owner
		owner = {
			has_menace_perk = MJ25_WH_menp_echoic
		}
		is_normal_starbase = yes
	}

	abort_construction_trigger = {
		exists = owner
		NOT = {
			owner = {
				has_country_flag = MJ25_WH_can_have_choirhall
			}
		}
	}

	destroy_trigger = {
		exists = owner
		NOT = {
			owner = {
				has_country_flag = MJ25_WH_can_have_choirhall
			}
		}
	}

	resources = {
		category = starbase_buildings
		cost = {
			alloys = 500
			MJ25_WH_void_whispers = 360
		}

		produces = {
			MJ25_WH_void_whispers = 15
		}

		upkeep = {
			energy = 5
		}
	}

	equipped_component = "MJ25_WH_STARBASE_AURA_FIRE_RATE"

	ai_weight = {
		weight = 100
	}

	on_finished = {
		starbase_event = { id = MJ25_WH_whispers.300 days = 90 }
		set_timed_starbase_flag = {
			flag = MJ25_WH_thrumming_active
			days = 89
		}	
		if = {
			limit = {
				solar_system = {
					NOT = {
						has_star_flag = MJ25_WH_songflare_system
					}
				}
			}
			solar_system = {
				set_star_flag = MJ25_WH_songflare_system
				set_star_flag = MJ25_WH_songflare_system@root.owner
				every_system_planet = {
					limit = { is_a_habitable_planet_megastructure = yes }
					set_planet_flag = MJ25_WH_songflare_planet@root.owner
				}
				system_event = { id = MJ25_WH_whispers.332 }
			}
		}
	}

	on_destroyed = {
		solar_system = { 
			system_event = { id = MJ25_WH_whispers.320 days = 60 }
		}
	}
}
