		modifier = {
			add = owner.value:veurcc_job_number
			desc = situation_wasteextractor_jobs_tt

		modifier = {
			add = 0.8

		}

		modifier = {
			add = 1.4
			desc = situation_toxic_knights_frugal_tt
			current_situation_approach = veu_situation_drastic
		}
	}
