eos_to_war = {
    complete_icon_frame = GFX_situation_outcome_frame_red
    complete_icon = GFX_situation_outcome_revolt

    picture = {
        trigger = {
            always = yes
        }

        picture = GFX_advisor_event_background
    }

    monthly_progress = {
        #approx 1 year
        base = 4
    }

    category = neutral

    stages = {
        stage1 = {
            icon = GFX_situation_stage_1
            icon_background = GFX_situation_stage_frame_blue
            end = 100
        }
    }

    approach = {
        icon = GFX_situation_approach_this_is_fine
        icon_background = GFX_situation_approach_bg_yellow

        name = eos_for_glory
    }

    on_start = {
        owner = {
            set_country_flag = eos_for_the_sake_of_it
            every_neighbor_country = {
                root.owner = {
                    check_casus_belli_valid = {
                        target = prev
                        type = eos_cb_for_the_sake_of_it
                    }
                }
            }
        }
    }

    #will never happen
    on_fail = {
        destroy_situation = this
    }

    on_progress_complete = {
        owner = {
            hidden_effect = {
                if = {
                    limit = {
                        NOT = {
                            any_war = {
                                is_war_participant = {
                                    who = prev
                                    side = attackers
                                }
                                using_war_goal = {
                                    owner = prev
                                    type = eos_wg_for_the_sake_of_it
                                }
                            }
                        }
                    }
                    eos_find_war_target_all = yes
                    if = {
                        limit = {
                            exists = event_target:war_target
                        }
                        declare_war = {
                            target = event_target:war_target
                            attacker_war_goal = eos_wg_for_the_sake_of_it
                        }
                    }
                    else = {
                        add_modifier = {
                            modifier = eos_war_not_started
                            years = 5
                        }
                        log = "failed_to_start_war"
                    }
                }
            }
            remove_country_flag = eos_for_the_sake_of_it
            every_neighbor_country = {
                root.owner = {
                    check_casus_belli_valid = {
                        target = prev
                        type = eos_cb_for_the_sake_of_it
                    }
                }
            }
        }
        destroy_situation = this
    }
}
