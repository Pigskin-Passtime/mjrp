hcp_situation_harvest_planet = {
	complete_icon_frame = GFX_situation_outcome_frame_green
	complete_icon = GFX_situation_exploding_planet
	picture = GFX_evt_mining_station

	stages = {
		stage_1 = {
			icon = GFX_situation_stage_1
			icon_background = GFX_situation_stage_frame_green
			end = 1000
			custom_tooltip = hcp_stage_harvesting
		}
	}

	category = positive
	initial_progress = 0

	on_progress_complete = {
		custom_tooltip = hcp_harvest_finish_tooltip
		hidden_effect = {
			situation_event = { id = hcp_harvesters_situations.2 }
		}
	}
	on_monthly = {
		events = {
			hcp_harvesters_situations.1 #Reduces Planet Size and grants reward
		}
	}
	on_abort = {
		target = {
			remove_planet_flag = being_devoured
			remove_modifier = hcp_being_harvested_modifier
			solar_system = {
				if = {
					limit = {
						any_system_megastructure = {
							has_megastructure_flag = hcp_harvest_@prevprev
						}
					}
					prev = { remove_planet_flag = megastructure }
					every_system_megastructure = {
						limit = {
							has_megastructure_flag = hcp_harvest_@prevprev
						}
						remove_megastructure = this
					}
				}
			}
		}
	}
	approach = {#Harvest
		name = hcp_approach_harvest
		on_select = {
			custom_tooltip = hcp_planet_harvest_effect
		}
		default = yes
		icon = GFX_leader_trait_mining_focus
		icon_background = GFX_situation_approach_bg_green
		ai_weight = {
			base = 10
		}
	}
	approach = {#Dismantle Orbital Quarrier
		name = hcp_approach_dismantle
		on_select = {
			custom_tooltip = hcp_harvester_dismantle_effect
			owner = {
				add_resource = { hcp_mining_efficiency = -1000 }
			}
			abort_situation = this
		}
		icon = GFX_situation_approach_influence
		icon_background = GFX_situation_approach_bg_yellow
		ai_weight = {
			base = 10
		}
	}
	monthly_progress = {
		base = 0
		modifier = {
			add = value:terravore_progress
			desc = hcp_string_harvesting_progress
		}
	}

	abort_trigger = {
		OR = {
			NOT = { exists = target.solar_system.owner }
			target.solar_system.owner = { NOT = { is_same_value = root.owner } }
		}
	}
}
