defined_text = {
    name = TFOGetShroudCondensers
    text = {
        localization_key = TFOGetShroudCondensersZero
        trigger = {
            has_technology = TFO_tech_shroud_condensers
            check_variable = {
                which = TFO_sr_shroud_condensers_current
                value = 0
            }
        }
    }
    text = {
        localization_key = TFOGetShroudCondensersPositive
        trigger = {
            has_technology = TFO_tech_shroud_condensers
            check_variable = {
                which = TFO_sr_shroud_condensers_current
                value > 0
            }
        }
    }
    text = {
        localization_key = TFOGetShroudCondensersMax
        trigger = {
            has_technology = TFO_tech_shroud_condensers
            check_variable = {
                which = TFO_sr_shroud_condensers_current
                value >= this.TFO_sr_shroud_condensers_max
            }
        }
    }

    default = TFOGetShroudCondensersDefault
}
