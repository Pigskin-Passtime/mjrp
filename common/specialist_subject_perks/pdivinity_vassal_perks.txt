pdivinity_acolyte_1_shroud_storm = {
    icon = "GFX_specialist_pdivinity_acolyte_1_shroud_storm"
    activate_effect = {
        target = {
			every_owned_starbase = {
				limit = { is_normal_starbase = yes }
				pdivinity_add_shroud_storm_vassal_array = yes
			}
		}
    }

    deactivate_effect = {
        target = {
            every_owned_starbase = {
                pdivinity_remove_shroud_storm_vassal_array = yes
            }
        }
    }

    desc = {
        description = pdivinity_acolyte_1_shield_desc
    }
}
