#maybe give it high sensor range and a ship aura to make enemy ships take more damage? (so its still similar to arctrellis but instead of reducing enemy dps you're increasing your own)
lbd_dathnak_specialist_3_arctrellis = {
	icon = "GFX_specialist_perk_lbd_dathnak_specialist_3_arctrellis"

	desc = {
		description = lbd_dathnak_specialist_3_arctrellis_desc
	}

	activate_effect = {
		target = {
			create_ship_design = { design = "NAME_LBD_DATHNAK_Arctrellis" }
			last_created_design = {
				set_design_flag = lbd_dathnak_specialist_arctrellis
			}
			add_ship_design = last_created_design
		}
	}

	deactivate_effect = {
		target = {
			every_owned_design = {
				limit = { has_design_flag = lbd_dathnak_specialist_arctrellis }
				save_event_target_as = remove_this_design
			}
			remove_ship_design = event_target:remove_this_design
		}
	}
}
