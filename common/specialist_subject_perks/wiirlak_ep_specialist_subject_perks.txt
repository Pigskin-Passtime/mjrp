	icon = GFX_specialist_perk_scholarium_1_discovery
	activate_effect = {
		target = {
			if = {
				limit = { NOT = { has_technology = wiirlak_ep_tech_energy } }

				give_technology = {
					tech = wiirlak_ep_tech_energy
					message = yes
				}
			}
			if = {
				limit = {
					NOR = {
						has_technology = wiirlak_ep_tech_growth
						has_tech_option = wiirlak_ep_tech_growth
					}
				}

				add_research_option = wiirlak_ep_tech_growth
			}
			if = {
				limit = {
					NOR = {
						has_technology = wiirlak_ep_tech_defense
						has_tech_option = wiirlak_ep_tech_defense
					}
				}

				add_research_option = wiirlak_ep_tech_defense
			}
		}
	}

	desc = {
		description = wiirlak_ep_1_insight_desc
	}
}
