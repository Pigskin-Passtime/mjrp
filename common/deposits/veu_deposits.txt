d_veu_bombing_blocker = {
	icon = d_radioactive_wasteland

	time = 360

	is_for_colonizable = yes
	category = deposit_cat_blockers

	potential = { always = no }

	can_be_cleared = {
			exists = owner
			owner = {
				has_ascension_perk = veu_ap_veu
				}
		}


	planet_modifier = {
		planet_max_districts_add = -1
		pop_environment_tolerance = -0.1
	}

	drop_weight = {
		weight = 0
	}

	on_cleared = {
		if = {
			limit = {
				exists = owner

					has_ascension_perk = veu_ap_veu
				}
			}
			owner = {
				add_resource = {
					veu_waste = 5
				}
			}
		}
	}

	resources = {
		category = deposit_blockers
		cost = {
			energy = 300
		}
	}

	ai_weight = {
		weight = 2
	}
}
