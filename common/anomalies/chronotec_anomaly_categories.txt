# Irregular Energy Emmisions mimic (star)
chronotec_fake_anomaly_strange_energy = {
	desc = chronotec_fake_anomaly_strange_energy_desc
	picture = "GFX_evt_ship_in_orbit"
	level = @chronotec_anomaly_level
	max_once_global = yes

	spawn_chance = {
		factor = @chronotec_anomaly_factor
		modifier = {
			add = 0.1
			is_star = yes
		}
		modifier = {	# Avoids it appearing
			factor = 0
			OR = {
				has_global_flag = chronotec_anomaly_spawned
				NOT = {
					has_global_flag = chronotec_allowed
				}
				has_global_flag = chronotec_system_spawned
				from.owner = {
					is_ai = yes
				}
				exists = event_target:chronotec_gate_system
				# Wrapper edge case fixer
				has_global_flag = fixed_random
				has_global_flag = random_player
				has_global_flag = player_choice
				has_global_flag = spawn_all
			}
		}
	}

	on_spawn = {
		set_global_flag = chronotec_anomaly_spawned
	}

	on_success = chronotec_narrative.1001
}
