darktec_crisobj_immediate_conquer_worlds = {	# darktec_crisis_objectives.70
	potential = {
		has_crisis_level = darktec_crisis_postbaryonic_level_4
	}
	reward = {
		base = 100000
		divide = trigger:num_galaxy_systems 	#on huge, this is 100. On smallest, it is 500
		complex_trigger_modifier = { 			#fewer worlds => more resource from taking one
			trigger = check_galaxy_setup_value
			parameters = { setting = habitable_worlds_scale }
			mode = divide
		}
		round_to = 50 								#we don't want 127.5 resource
		min = 100									#just in case
	}
}
