######## SCIENCE SHIP

ds_pr_menu_science = {
	allow = {
		custom_tooltip = {
			fail_text = ds_pr_menu_ship_cost
			has_resource = { type = alloys amount >= @ds_pr_ui_ship_req }
		}
	}
	effect = {
		custom_tooltip = ds_pr_menu_science_tt
		add_resource = { alloys = @ds_pr_ui_ship_cost }
		hidden_effect = {
			random_owned_fleet = {
				limit = { exists = leader }
				create_fleet = {
					effect = {
						set_owner = root
						create_ship = {
							name = random
							random_existing_design = science
							graphical_culture = root
						}
						set_location = { target = prev }
					}
				}
			}
		}
	}
}

######## ARMY

ds_pr_menu_army = {
	allow = {
		custom_tooltip = {
			fail_text = ds_pr_menu_army_cost
			has_resource = { type = minerals amount >= @ds_pr_ui_army_req }
		}
	}
	effect = {
		custom_tooltip = ds_pr_menu_army_tt
		add_resource = { minerals = @ds_pr_ui_army_cost }
		hidden_effect = {
			random_owned_fleet = {
				limit = { exists = leader }
				create_fleet = {
					effect = {
						set_owner = root
						while = {
							count = 2
							create_army_transport = {
								ship_name = random
								graphical_culture = root
								army_name = "Extradimensional Raiders"
								army_type = ds_pr_dark_army
								species = owner_main_species
							}
						}
						set_location = { target = prev }
					}
				}
			}
		}
	}
}

######## LEADER

# Cost is handled in event
ds_pr_menu_leader = {
	allow = {
		custom_tooltip = {
			fail_text = ds_pr_interface_limit
			NOT = { event_target:ds_pr_crisis_country = { has_active_event = { dsPR.140 } } }
		}
		custom_tooltip = {
			fail_text = ds_pr_leader_cd
			NOR = { event_target:ds_pr_crisis_country = { has_country_flag = ds_pr_leader_added } }
		}
	}
	effect = {
		custom_tooltip = ds_pr_menu_leader_tt
		hidden_effect = { event_target:ds_pr_crisis_country = { country_event = { id = dsPR.140 } } }
	}
}
