celestial_relic_button1 = {

			}

		custom_tooltip_fail = {

		}

			text = celestial_emporium_quest_selected
			from = {
				NOR = {

			hidden_effect = {

		}
	}
}

# Store Items
celestial_store_effect_1 = {
	potential = {
		exists = from
		from = { has_country_flag = cosmic_emporium_tech_1 }
	}
	allow = {
		from = {
			if = {
				limit = {
					check_variable = {
						which = celestial_emporium_ec_income_x10
						value > 3000
					}
				}
				resource_stockpile_compare = {
					resource = energy
					value >= celestial_emporium_ec_income_x10
				}
			}
			else = {
				resource_stockpile_compare = {
					resource = energy
					value >= 3000
				}
			}
			custom_tooltip_fail = {
				text = celestial_emporium_already_bought
				NOT = { has_country_flag = cosmic_emporium_slot_1_bought }
			}
			custom_tooltip_fail = {
				text = celestial_emporium_cooldown_active1
				OR = {
					NOT = { has_country_flag = cosmic_emporium_cooldown }
					has_country_flag = cosmic_emporium_slot_1_bought
				}
			}
		}
	}
	effect = {
		custom_tooltip = celestial_store_effect_1_tt
		from = {
			set_timed_country_flag = {
				flag = cosmic_emporium_cooldown
				days = 1081
			}
			hidden_effect = {
				# Set flags for the timer in the custom menu
				set_country_flag = cosmic_emporium_cooldown_years_3
				country_event = {
					id = celestial_setup.23
					days = 360
				}
			}
			set_country_flag = cosmic_emporium_slot_1_bought

			cosmic_emporium_tech_1_effect = yes
		}
	}
}
