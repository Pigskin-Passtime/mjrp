# Relic Buttons
celestial_relic_button1 = {
	potential = {
		is_scope_type = country
		NOT = {
			has_country_flag = celestial_relic_quest1_completed
		}
	}
	allow = {
		custom_tooltip_fail = {
			text = celestial_emporium_quest_selected_already
			NOT = { has_country_flag = celestial_relic_quest1_selected }
		}
		custom_tooltip_fail = {
			text = celestial_emporium_quest_selected
			NOR = {
				has_country_flag = celestial_relic_quest2_selected
				has_country_flag = celestial_relic_quest3_selected
				has_country_flag = celestial_relic_quest4_selected
				has_country_flag = celestial_relic_quest5_selected
				has_country_flag = celestial_relic_quest6_selected
			}
		}
	}
	effect = {
		custom_tooltip = celestial_relic_1
		hidden_effect = {
			reset_cosmic_emporium_flags = yes
			set_country_flag = celestial_emporium_relic_1
		}
	}
}

celestial_store_effect_1 = {
	potential = {
		has_country_flag = cosmic_emporium_tech_1
	}
	allow = {
		custom_tooltip_fail = {
			text = celestial_emporium_already_bought
			NOT = { has_country_flag = cosmic_emporium_slot_1_bought }
		}
		custom_tooltip_fail = {
			text = celestial_emporium_cooldown_active
			OR = {
				NOT = { has_country_flag = cosmic_emporium_cooldown }
				has_country_flag = cosmic_emporium_slot_1_bought
			}
		}
	}
	effect = {
		set_timed_country_flag = {
			flag = cosmic_emporium_cooldown
			days = 1080
		}
		set_country_flag = cosmic_emporium_slot_1_bought

		cosmic_emporium_tech_1_effect = yes
		custom_tooltip = celestial_store_effect_1_tt
	}
}
