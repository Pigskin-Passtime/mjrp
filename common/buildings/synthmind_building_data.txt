building_Automated_Command_System = {
	icon = Automated_Command_System
	position_priority = 1
	empire_limit = 1
	base_buildtime = 1800
	
	resources = {
		category = planet_buildings
		cost = {
			minerals = 1000
			alloys = 1000
		}
		upkeep = {
			rare_crystals = 1
			exotic_gases = 1
			volatile_motes = 1
			alloys = 5
			energy = 50
		}
	}

	country_modifier = {
        country_naval_cap_add = 100
    }

    triggered_planet_modifier = {
        potential = {
            exists = owner
            owner = { is_regular_empire = yes } 
        }
        modifier = {
            job_soldier_add = 3
            job_enforcer_add = 3
            planet_housing_add = 10
        }
    } 
    
    triggered_planet_modifier = {
        potential = {
	    exists = owner
            owner = { is_gestalt = yes }
        }
        modifier = {
            job_warrior_drone_add = 7
            planet_housing_add = 10
        }
    }

    prerequisites = {
        "tech_AI_Command_System"
    }

    allow = {
        has_major_upgraded_capital = yes
    }

    destroy_trigger = {
		OR = {
			has_modifier = slave_colony
			has_modifier = resort_colony
		}
    }
}
