		}
		upkeep = {
			energy = @b1_upkeep
		}
	}
	planet_modifier = {
		job_monitor_add = @b1_jobs
		planet_crime_mult = -0.1

	}
	#prerequisites = {
	#	"tech_cultural_heritage"
	#}

		building_monitor_station_2
	}
	#show_tech_unlock_if = {
	#	is_gestalt = no

	#	is_megacorp = no
	#}
	# triggered_desc = {

	# }

			id = caretaker_origin_event.1200
		}
	}
	on_destroy = {
		planet_event = {
			id = caretaker_origin_event.1210
		}
	}
	ai_weight = {
		weight = 0
		modifier = {
			weight = 5
			NOR = {
				has_building = building_monitor_station_1
				has_building = building_monitor_station_2

			}
			owner = {
				has_monthly_income = {
					resource = consumer_goods
					value > 10
				}
			}
		}
	}
}
