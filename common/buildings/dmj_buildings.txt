dmj_reinhardist_temple = {
	can_build = no
	can_demolish = no
	can_be_ruined = no

	planet_limit = 1

	category = government

	potential = {
		exists = owner
		owner = {
			OR = {
				has_country_flag = owns_reinhard
				any_owned_leader = {
					is_same_value = event_target:dmj_daniel
				}
			}
		}
	}

	destroy_trigger = {
		exists = owner
		owner = {
			NOR = {
				has_country_flag = owns_reinhard
				any_owned_leader = {
					is_same_value = event_target:dmj_daniel
				}
			}
		}
	}

	planet_modifier = {
		job_reinhardist_devotee_add = 2
		job_reinhardist_devotee_per_pop = 0.05
		pop_ethic_authoritarian_attraction_mult = 1
		pop_ethic_spiritualist_attraction_mult = 1
		pop_ethic_militarist_attraction_mult = 1
		pop_ethic_egalitarian_attraction_mult = -0.75
		pop_ethic_materialist_attraction_mult = -0.75
		pop_ethic_pacifist_attraction_mult = -0.75
	}
}
