@entity_x = -20
@entity_y = -20
dtm_theophany_throne_1 = {
	entity = "slaver_01_payback_warship_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	build_type = inside_gravity_well
	entity_offset = {
		x = @entity_x
		y = @entity_y
	}
	build_time = 3600
	resources = {
		category = megastructures
		cost = {
			alloys = 10000
		}
		upkeep = {
			energy = 10
		}
	}
	country_modifier = {
		shroud_delve_cooldown = -0.25
		psionic_pop_bonus_workforce_mult = 0.125
	}
	construction_blocks_and_blocked_by = self_type
	prerequisites = {
		"tech_dtm_theophany_sanctum_1"
	}
	show_prereqs = yes
	potential = {
		NOT = {
			has_country_flag = dtm_has_built_megastructure
		}
		has_first_contact_dlc = yes
		is_country_type = default
		NOR = {
			any_owned_megastructure = {
				is_megastructure_type = dtm_theophany_throne_1
			}
		}
	}
	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "dtm_requires_capitol"
			any_system_planet = {
				is_same_value = prev.from.capital_scope
			}
		}
	}
	placement_rules = {
		#is_same_value = from.capital_star.solar_system
		planet_possible = {
			custom_tooltip = {
				fail_text = "BUILD_MEGASTRUCTURE_PLANET_TARGET_NEEDED"				# these loc lines are inverted
				always = no
			}
		}
		# planet_possible = {
		# 	is_colony = yes
		# 	OR = {
		# 		is_planet_class = pc_dtm_shardworld
		# 		is_capital = yes
		# 	}
		# }
	}
	on_build_complete = {
		from = {
			set_country_flag = dtm_has_built_megastructure
			set_country_flag = has_built_or_repaired_megastructure
			country_event = {
				id = dtm_theophany_stages.51
			}
		}
	}
}

	upgrade_from = {

	}
	construction_blocks_and_blocked_by = self_type
	prerequisites = {
		"tech_dtm_theophany_sanctum_2"
	}
	show_prereqs = yes
	country_modifier = {
		shroud_delve_cooldown = -0.5
		psionic_pop_bonus_workforce_mult = 0.5
	}
	potential = {
	}
	possible = {

			}
		}
	}
