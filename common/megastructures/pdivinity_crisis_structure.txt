            energy = 10000
            sr_zro = 5000
        }
    }

    country_modifier = {
        # unsure what else
        country_sr_zro_produces_mult = 0.25
        weapon_type_pdivinity_psionic_weapon_damage_mult = 0.2
        country_resource_max_energy_add = 25000
        country_resource_max_sr_zro_add = 10000
    }

    potential = {
        has_ascension_perk = ap_pdivinity
    }

    upgrade_from = {
        pdivinity_shroud_needle_1
    }

    on_build_complete = {
        fromfrom = {
            set_megastructure_flag = pdivinity_shroud_needle_flag_stage_2

            create_ambient_object = {
                type = "psi_storm_1"
                location = fromfrom
            }
            last_created_ambient_object = {
                set_ambient_object_flag = pdivinity_megastructure_psi_storm_1
                set_location = {
                    target = prev
                    distance = 0
                    angle = random
                }
            }
        }
        from = {
            country_event = { id = pdivinity_needle.2 }
        }
    }
}
