darktec_sphere_0 = {
	entity = "darktec_bald_blackhole_phase_00_entity"
	construction_entity = "darktec_bald_blackhole_phase_00_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	build_type = inside_gravity_well

	entity_offset = { x = 0 y = 0 }
	build_time = 1800
	resources = {
		category = megastructures
		cost = {
			sr_dark_matter = 10000
			darktec_contained_entropy = 20000
		}

		upkeep = {
			darktec_contained_entropy = 5
		}
	}

	custom_tooltip_requirements = "MEGASTRUCTURE_TOOLTIP_REQUIREMENTS_ONE_PER_COUNTRY"

	prerequisites = { tech_mine_dark_matter }
	show_prereqs = no

	potential = {
		has_crisis_level = darktec_crisis_postbaryonic_level_5
	}

	possible = {
		hidden_trigger = {
			exists = starbase
			from = {
				NOT = {
					has_country_flag = darktec_built_sphere
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
	}

	#placement_rules = {
		#always = yes
		#planet_possible = {
		#	custom_tooltip = {
		#		fail_text = "must_build_around_star"
		#		is_star = yes
		#	}
		#	custom_tooltip = {
		#		fail_text = "requires_no_anomaly"
		#		NOT = { has_anomaly = yes }
		#	}
		#}
	#}

	# root = system
	# from = country
	ai_weight = {
		factor = 15

		modifier = {
			factor = 0.1
			starbase = { NOT = { has_starbase_size >= starbase_starfortress } }
		}

		modifier = {
			factor = 0.1
			any_neighbor_system = {
				exists = owner
				NOT = {
					owner = { is_same_value = from }
				}
			}
		}
	}

	on_build_start = {
	}
	on_build_cancel = {
	}
	on_build_complete = {
		from = {
			set_country_flag = darktec_built_sphere
		}
		create_ambient_object = {
			type = darktec_sphere_00_entity
			location = fromfrom
		}
	}
}
