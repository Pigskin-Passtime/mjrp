hcp_orbital_quarrier = {
	entity = "ancient_drone_station_entity"
	construction_entity = "ancient_drone_station_entity"
	portrait = "GFX_hcp_orbital_quarrier_construction_background"
	place_entity_on_planet_plane = no
	entity_offset = { x = 0 y = 0 }
	build_time = 360
	show_in_outliner = no
	resources = {
		category = megastructures
		cost = {
			trigger = {
				country_uses_bio_ships = no
			}
			alloys = 500
		}
		cost = {
			trigger = {
				country_uses_bio_ships = yes
			}
			alloys = 500
			mult = 0.5
		}
		cost = {
			trigger = {
				country_uses_bio_ships = yes
			}
			food = 500
			mult = @halved_alloy_to_food_cost_ratio
		}
		upkeep = {
			energy = 15
		}
	}

	construction_blocks_and_blocked_by = none

	potential = {
		has_crisis_perk =  hcp_menp_planetary_quarrying
	}

	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
	}

	placement_rules = {
		planet_possible = {
			custom_tooltip = {
				fail_text = "hcp_requires_not_capital"
				is_capital = no
			}
			custom_tooltip = {
				fail_text = "hcp_requires_not_gas_giant"
				NOT = {
					is_planet_class = pc_gas_giant
				}
			}
			custom_tooltip = {
				fail_text = "hcp_requires_no_deposit"
				OR = {
					has_planet_flag = hcp_has_prospected_deposit
					NOR = {
						has_deposit_for = shipclass_mining_station
						has_deposit_for = shipclass_research_station
					}
				}
			}
			custom_tooltip = {
				fail_text = "requires_no_anomaly"
				NOT = { has_anomaly = yes }
			}
			custom_tooltip = {
				fail_text = "hcp_requires_min_size"
				planet_size > 10
			}
			custom_tooltip = {
				fail_text = "requires_not_star"
				is_star = no
			}
			custom_tooltip = {
				fail_text = "requires_not_astral_scar"
				is_astral_scar = no
			}
			custom_tooltip = {
				fail_text = "requires_not_ring_world"
				is_ringworld = no
			}
			custom_tooltip = {
				fail_text = "requires_no_existing_megastructure"
				NOT = { has_planet_flag = megastructure }
			}
		}
	}

	on_build_complete = {
		fromfrom= {
			set_megastructure_flag = hcp_harvest_@planet
			planet = {
				set_planet_flag = megastructure
				set_timed_planet_flag = {
					days = 360
					flag = recently_eaten_planet
				}
				hidden_effect = {
					set_planet_flag = being_devoured
					add_modifier = {
						modifier = hcp_being_harvested_modifier
						days = -1
					}
					export_trigger_value_to_variable = {
						trigger = planet_size
						variable = num_districts_terravore
					}
					subtract_variable = {
						which = num_districts_terravore
						value = 5
					}
				}
				prev.owner = {
					start_situation = {
						type = hcp_situation_harvest_planet
						target = prev
					}
				}
				if = {
					limit = { has_orbital_station = yes }
					orbital_station = {
						dismantle = yes
					}
				}
			}
		}
	}
}
