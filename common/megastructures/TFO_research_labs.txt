#Negative-Mass Particle Accelerator
TFO_mega_research_station_3 = {
    entity = TFO_negative_mass_particle_accelerator
    construction_entity = construction_platform_entity
    portrait = GFX_TFO_megastructure_NMPA_background
    place_entity_on_planet_plane = yes
    entity_offset = { x = 0 y = 0 }
    scales_with_planet = yes
    show_galactic_map_icon = yes
    show_in_outliner = yes

    construction_blocks_and_blocked_by = none

    upgrade_from = {
        TFO_mega_research_station_2
    }

    potential = {
        has_country_flag = TFO_can_build_NMPA
    }

    possible = {
        custom_tooltip = {
            fail_text = TFO_requires_mega_engineering_tech
            from = {
                has_technology = tech_mega_engineering
            }
        }
        custom_tooltip = {
            fail_text = TFO_requires_shroud_condensers_tech_unknown
            from = {
                has_country_flag = TFO_unlock_shroud_condensers
            }
        }
        custom_tooltip = {
            fail_text = TFO_requires_shroud_condensers_tech
            from = {
                OR = {
                    AND = {
                        has_country_flag = TFO_unlock_shroud_condensers
                        has_technology = TFO_tech_shroud_condensers
                    }
                    NOR = {
                        has_country_flag = TFO_unlock_shroud_condensers
                        has_technology = TFO_tech_shroud_condensers
                    }
                }
            }
        }
    }

    #Fake modifier, used to display mod resource stockpile
    country_modifier = {
        country_TFO_sr_shroud_condensers_produces_mult = 0.33
        TFO_display_shroud_condensers_scripted_modifier = 1
    }

    build_time = 1800
    resources = {
        category = megastructures
        cost = {
            TFO_sr_shroud_condensers = 2500
            alloys = 5000
            unity = 10000

            trigger = {
                has_technology = TFO_tech_shroud_condensers
            }
        }
        cost = {
            alloys = 5000
            unity = 10000
            trigger = {
                NOT = {
                    has_technology = TFO_tech_shroud_condensers
                }
            }
        }
        produces = {
            trigger = {
                exists = owner
                owner = {
                    has_technology = TFO_tech_shroud_condensers
                }
            }
            TFO_sr_shroud_condensers = @TFO_mega_research_station_3_production
        }
        upkeep = {
            trigger = {
                exists = owner
                owner = {
                    has_technology = TFO_tech_shroud_condensers
                }
            }
            TFO_sr_shroud_condensers = @TFO_mega_research_station_3_upkeep
        }
    }
}
