coex_enlightenment_array_0 = {
	entity = "coex_enlightenment_array_stage_0_entity"
	construction_entity = "coex_enlightenment_array_stage_0_entity"
	portrait = "GFX_megastructure_construction_background"
	build_time = 1200
	place_entity_on_planet_plane = no
	entity_offset = {
		x = -27
		y = -27
	}
	resources = {
		category = megastructures
		#TODO: add cost for bioshipsets
		cost = {
			alloys = 6000
		}
	}
	country_modifier = {
		coex_brain_relay_megastructures_produces_mult = 1
	}
	# upgrade_from = {
	# 	coex_enlightenment_array_ruined
	# }
	potential = {
		has_menace_perk = menp_coex_mega_2
		NOT = {
			any_owned_megastructure = {
					has_megastructure_flag = coex_enlightenment_array
				}
			}
	}
	placement_rules = {
		planet_possible = {
			custom_tooltip = {
				fail_text = "requires_survey_not_habitable"
				is_surveyed = {
					# prevent leaking habitability information
					who = prev.from
					status = yes
				}
				colonizable_planet = no
			}
			custom_tooltip = {
				fail_text = "requires_no_anomaly"
				NOT = {
					has_anomaly = yes
				}
			}
			custom_tooltip = {
				fail_text = "requires_not_minor_planetary_body"
				NOR = {
					is_asteroid = yes
					is_moon = yes
				}
			}
			custom_tooltip = {
				fail_text = "must_build_around_star"
				is_star = yes
			}
			custom_tooltip = {
				fail_text = "requires_not_astral_scar"
				is_astral_scar = no
			}
			custom_tooltip = {
				fail_text = "requires_not_ring_world"
				is_ringworld = no
			}
			custom_tooltip = {
				fail_text = "requires_no_existing_megastructure"
				NOT = {
					has_planet_flag = megastructure
				}
			}
		}
	}
	on_build_complete = {
		fromfrom = {
			solar_system.star = {
				create_ambient_object = {
					type = "crisis_sphere_1"
					location = this
				}
				last_created_ambient_object = {
					set_ambient_object_flag = crisis_sphere_1_system_effect
					set_location = {
						target = prev
						distance = 0
						angle = random
					}
				}
			}
			every_country = {
				limit = {
					has_opinion_modifier = {
						who = from
						modifier = coex_affected_by_brain_relay_0
					}
				}
				remove_opinion_modifier = {
					who = from
					modifier = coex_affected_by_brain_relay_0
				}
				add_opinion_modifier = {
					who = from
					modifier = coex_affected_by_brain_relay_1
				}
			}
			set_megastructure_flag = coex_enlightenment_array
		}
	}
}
