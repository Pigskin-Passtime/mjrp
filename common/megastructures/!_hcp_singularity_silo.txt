hcp_singularity_silo_1 = {
	entity = "dyson_sphere_phase_02_entity"
	construction_entity = "dyson_sphere_phase_02_entity"
	portrait = "GFX_megastructure_construction_background"
	build_time = 3600
	resources = {
		category = megastructures
		cost = {
			alloys = @hcp_singularity_silo_1_cost
		}
		cost = {
			trigger = {
				country_uses_bio_ships = no
			}
			alloys = @hcp_singularity_silo_1_cost
		}
		cost = {
			trigger = {
				country_uses_bio_ships = yes
			}
			alloys = @hcp_singularity_silo_1_cost
			mult = 0.5
		}
		cost = {
			trigger = {
				country_uses_bio_ships = yes
			}
			food = @hcp_singularity_silo_1_cost
			mult = @halved_alloy_to_food_cost_ratio
		}

		upkeep = {
			energy = @hcp_singularity_silo_1_upkeep
		}
	}

	construction_blocks_and_blocked_by = self_type

	country_modifier = {
		country_resource_max_add = @hcp_singularity_silo_1_capacity
	}

	custom_tooltip_requirements = "MEGASTRUCTURE_TOOLTIP_REQUIREMENTS_MATTER_DECOMPRESSOR"

	potential = {
		has_crisis_perk = hcp_menp_infinite_storage
	}

	possible = {
		hidden_trigger = {
			exists = starbase
		}
		custom_tooltip = {
			fail_text = "requires_inside_border"
			is_inside_border = from
		}
		custom_tooltip = {
			fail_text = "requires_surveyed_system"
			NOT = {
				any_system_planet = {
					is_surveyed = {
						who = prev.from
						status = no
					}
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_no_habitable_planets"
			NOT = {
				any_system_planet = {
					colonizable_planet = yes
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_black_hole"
			is_star_class = sc_black_hole
		}
		custom_tooltip = {
			fail_text = "hcp_requires_stable_system"
			count_system_planet = {
				count < 2
				limit = {
					is_planet_class = pc_black_hole
				}
			}
		}
		custom_tooltip = {
			fail_text = "hcp_requires_no_existing_singularity_silo"
			from = {
				OR = {
					NOT = { has_country_flag = hcp_built_singularity_silo }
					has_special_project = hcp_REBUILD_SINGULARITY_SILO_1
					has_special_project = hcp_REBUILD_SINGULARITY_SILO_2
				}
			}
		}
		custom_tooltip = {
			fail_text = "requires_no_existing_megastructure"
			has_no_non_gate_megastructure = yes
			NOT = {
				any_system_planet = {
					is_planet_class = pc_cosmogenesis_world
				}
			}
		}
	}

	placement_rules = {
		planet_possible = {
			custom_tooltip = {
				fail_text = "must_build_around_star"
				is_star = yes
			}
			custom_tooltip = {
				fail_text = "requires_no_anomaly"
				NOT = { has_anomaly = yes }
			}
			is_star_class = sc_black_hole
		}
	}

	# root = system
	# from = country
	ai_weight = {
		factor = 15

		modifier = {
			factor = 0.1
			starbase = { NOT = { has_starbase_size >= starbase_starfortress } }
		}

		modifier = {
			factor = 0.1
			any_neighbor_system = {
				exists = owner
				NOT = {
					owner = { is_same_value = from }
				}
			}
		}
	}

	on_build_complete = {
		set_star_flag = hcp_singularity_silo_@from
		star = {
			set_planet_flag = has_megastructure
		}
		from = {
			set_country_flag = hcp_built_singularity_silo
			set_country_flag = hcp_built_singularity_silo_stage_1
			if = {
				limit = { has_special_project = hcp_BUILD_SINGULARITY_SILO_LEVEL_1 }
				complete_special_project = { type = hcp_BUILD_SINGULARITY_SILO_LEVEL_1 }
			}
			if = {
				limit = { has_special_project = hcp_REBUILD_SINGULARITY_SILO_1 }
				complete_special_project = { type = hcp_REBUILD_SINGULARITY_SILO_1 }
			}
			if = {
				limit = { has_special_project = hcp_REBUILD_SINGULARITY_SILO_2 }
				complete_special_project = { type = hcp_REBUILD_SINGULARITY_SILO_2 }
			}
		}
		fromfrom = {
			set_megastructure_flag = hcp_crisis_silo
		}
		fromfrom.planet = {
			if = {
				limit = { has_orbital_station = yes }
				orbital_station = {
					dismantle = yes
				}
			}
		}
	}
}
