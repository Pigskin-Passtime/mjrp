spawn_eos_leader = {
    create_leader = {
        species = root.species
        skill = 5
        background_ethic = $ethic$
        class = $class$

        #added stuff
        name = "Emobodiement Of Society"
        event_leader = yes
        immortal = yes
        gender = indeterminable
        hide_age = yes
        randomize_traits = no


        traits = {
            #traits go here
            #Do we even need this?
            #trait = leader_trait_eos_base

            #unncomment this when all traits are ready
            trait = eos_leader_trait_bonus_$ethic$
            trait = eos_leader_trait_planet_$ethic$

            trait = eos_leader_trait_ethic_$ethic$
        }
        effect = {
            save_global_event_target_as = eos_leader
            set_leader_flag = legendary_leader
            set_leader_flag = eos
            set_owner = root
            exile_leader_as = eos_leader
        }
    }
}

clone_eos_as = {
    event_target:eos_leader_prev = {
        change_background_ethic = $ethic$

        remove_trait = eos_leader_trait_ethic_$ethic_prev$
        add_trait = eos_leader_trait_ethic_$ethic$

                #Unncomment when additional traits are ready
                remove_trait =eos_leader_trait_planet$ethic_prev$
                add_trait = eos_leader_trait_planet$ethic$

                remove_trait = eos_leader_trait_bonus$ethic_prev$
                add_trait = eos_leader_trait_bonus$ethic$

                save_global_event_target_as = eos_leader
                set_leader_flag = legendary_leader
                set_leader_flag = eos
                change_leader_class = $class$
            }
        }
