hcp_harvest_star_system = {
	if = {
		limit = {
			any_system_megastructure = {
				exists = this
				has_megastructure_flag = hcp_crisis_silo
			}
			any_playable_country = {
				NOT = {
					is_same_value = event_target:crisis_country
				}
				has_ascension_perk = hcp_ap_harvesters
			}
		}
		every_playable_country = {
			limit = {
				OR = {
					has_special_project = hcp_STEAL_SINGULARITY_SILO_1
					has_special_project = hcp_STEAL_SINGULARITY_SILO_2
				}
			}
			country_event = {
				id = hcp_harvesters_crisis.1015
			}
		}
		every_playable_country = {
			limit = {
				prev = {
					has_star_flag = hcp_singularity_silo_@prev
				}
			}
			country_event = {
				id = hcp_harvesters_crisis.1000
			}
			prev = {
				remove_star_flag = hcp_singularity_silo_@prev
			}
		}
	}
	if = {
		limit = { exists = event_target:crisis_country }
		fire_on_action = {
			on_action = on_destroy_star_system
			scopes = { from = event_target:crisis_country }
		}
	}
	else = {
		fire_on_action = { on_action = on_destroy_star_system }
	}
	from.solar_system = {
		set_asteroid_belt = {
			radius = 0
		}
	}
	every_system_planet = {
		limit = {
			is_colony = yes
		}
		if = {
			limit = {
				OR = {
					root.from.owner = {
						has_policy_flag = hcp_evacuate_everyone
					}
					owner = {
						is_same_value = root.from.owner
					}
				}
			}
			every_owned_pop_group = {
				limit = {
					OR = {
						is_pre_sapient = no
						has_trait = trait_nascent_stage
					}
					NOR = {
						has_trait = trait_pathogenic_genes
						has_trait = trait_self_modified
					}
					if = {
						limit = {
							has_virtual_species_trait = yes
						}
						exists = owner
						owner = {
							is_fallen_empire = no
							NOT = {
								has_active_tradition = tr_virtuality_adopt
							}
						}
					}
				}
				resettle_pop_group = {
					POP_GROUP = this
					PLANET = event_target:hcp_relocate_target
					PERCENTAGE = 100%
				}
			}
			if = {
				limit = {
					root.owner = {
						has_policy_flag = pre_sapients_protect
					}
				}
				every_owned_pop_group = {
					limit = {
						is_pre_sapient = yes
					}
					resettle_pop_group = {
						POP_GROUP = this
						PLANET = event_target:hcp_relocate_target
						PERCENTAGE = 100%
					}
				}
			}
		}
		else = {
			every_owned_pop_group = {
				limit = {
					is_lithoid = yes
				}
				root.from.owner = {
					add_resource = {
						minerals = 1
						multiplier = prev.trigger:pop_group_size
					}
				}
			}
			every_owned_pop_group = {
				limit = {
					is_infernal = yes
				}
				root.from.owner = {
					add_resource = {
						alloys = 1
						multiplier = prev.trigger:pop_group_size
					}
				}
			}
			every_owned_pop_group = {
				limit = {
					is_lithoid = no
					is_organic_species = yes
				}
				root.from.owner = {
					add_resource = {
						food = 1
						multiplier = prev.trigger:pop_group_size
					}
				}
			}
			every_owned_pop_group = {
				limit = {
					species = {
						OR = {
							is_archetype = MACHINE
							is_archetype = ROBOT
						}
					}
				}
				root.from.owner = {
					add_resource = {
						alloys = 1
						multiplier = prev.trigger:pop_group_size
					}
				}
			}
		}
		destroy_colony = yes
	}
	set_star_flag = star_cracked
	set_star_flag = hcp_system_harvested
	set_star_class = hcp_sc_harvested_system
	every_system_ambient_object = {
		limit = {
			# Preserve Unbidden crisis effects
			NOR = {
				has_ambient_object_flag = extradimensional_system_effect
				has_ambient_object_flag = extradimensional_system_effect_2
			}
		}
		destroy_ambient_object = this
	}
	every_system_planet = {
		limit = { is_primary_star = yes }
		root.from.owner = {
			hcp_resources_from_planet_size = yes
		}
		change_pc = hcp_pc_harvested_system
		clear_planet_modifiers = yes
	}
	# Handle Contingency Machine Worlds
	every_system_planet = {
		limit = {
			is_planet_class = pc_ai
			NOT = { has_planet_flag = machine_lair }
		}
		destroy_machine_world = yes
	}
	random_system_planet = {
		limit = {
			is_planet_class = pc_ai
			has_planet_flag = machine_lair
		}
		from.owner = { save_event_target_as = final_machine_world_destroyer }
		stop_crisis_sound = yes
		planet_event = { id = crisis.2046 }
	}
	every_system_planet = {
		limit = {
			exists = orbital_station
		}
		orbital_station = { dismantle = yes }
	}
	every_system_planet = {
		clear_deposits = yes
		clear_planet_modifiers = yes
		prevent_anomaly = yes
	}
	every_system_planet = {
		limit = { is_primary_star = no }
		root.from.owner = {
			hcp_resources_from_planet_size = yes
		}
		remove_planet = yes
	}
	every_fleet_in_system = {
		limit = {
			NOR = {
				is_same_value = from
				# except Extradimensionals
				AND = {
					exists = owner
					owner = {
						OR = {
							is_country_type = portal_holder
							is_country_type = extradimensional
							is_country_type = extradimensional_2
							is_country_type = extradimensional_3
						}
					}
				}
				# except biggest monsters
				any_owned_ship = {
					OR = {
						is_ship_size = behemoth_04
						is_ship_size = voidspawn_boss
					}
				}
			}
		}
		if = {
			limit = {
				exists = event_target:crisis_country #i.e. not final destruction of all matter
				is_mobile = yes
			}
			set_mia = mia_return_home
		}
		else = {
			root.from.owner = {
				hcp_resources_from_ship_size = yes
			}
			destroy_fleet = this
		}
	}
	every_system_megastructure = {
		limit = {
			NOR = {
				is_megastructure_type = lgate_base
				is_megastructure_type = gateway_ness_ruined
				is_megastructure_type = gateway_ness_restored
			}
		}
		root.from.owner = {
			add_resource = {
				alloys = 10000
			}
		}
		remove_megastructure = this
	}
}

hcp_resources_from_ship_size = {
	if = {
		limit = {
			prev.owner = {
				country_uses_bio_ships = no
			}
		}
		add_resource = {
			alloys = 1
			multiplier = prev.value:hcp_harvest_star_system_resources|TRIGGER|fleet_size|
		}
	}
	else = {
		add_resource = {
			food = 1
			multiplier = prev.value:hcp_harvest_star_system_resources|TRIGGER|fleet_size|
		}
	}
}

hcp_replicate_stellar_harvester = {
	$OWNER$ = {
		if = {
			limit = {
				any_owned_ship = {
					is_ship_size = hcp_stellar_harvester
				}
			}
			random_owned_ship = {
				limit = {
					is_ship_size = hcp_stellar_harvester
				}
				save_event_target_as = hcp_target_harvester
			}
		}
		else = {
			owner = {
				random_owned_design = {
					limit = {
						is_ship_size = hcp_stellar_harvester
					}
					save_event_target_as = hcp_target_harvester
				}
			}
		}
	}
	create_fleet = {
		name = "hcp_NAME_Stellar_Harvester"
		settings = {
			can_disband = no
		}
		effect = {
			set_owner = $OWNER$
			set_fleet_flag = hcp_stellar_harvester
			create_ship = {
				name = "hcp_NAME_Stellar_Harvester"
				design = event_target:hcp_target_harvester #"hcp_NAME_Stellar_Harvester"
			}
			set_location = {
				target = $TARGET$
				distance = 10
			}
			set_fleet_stance = passive
			set_variable = {
				which = hcp_harvester_ability_progress
				value = 0
			}
		}
	}
	last_created_fleet = {
		save_event_target_as = hcp_new_harvester
		owner = {
			save_event_target_as = hcp_new_harvester_owner
			if = {
				limit = {
					is_variable_set = hcp_num_stellar_harvesters
				}
				add_variable = {
					which = hcp_num_stellar_harvesters
					value = 1
				}
			}
			else = {
				set_variable = {
					which = hcp_num_stellar_harvesters
					value = 1
				}
			}
		}
		if = {
			limit = {
				any_playable_country = {
					has_event_chain = hcp_anti_harvesters_chain
					has_country_flag = hcp_has_anti_harvesters_chain_@event_target:hcp_new_harvester_owner
				}
			}
			every_playable_country = {
				limit = {
					has_event_chain = hcp_anti_harvesters_chain
					has_country_flag = hcp_has_anti_harvesters_chain_@event_target:hcp_new_harvester_owner
				}
				event_target:hcp_new_harvester = {
					enable_special_project = {
						name = hcp_DESTROY_STELLAR_HARVESTER
						location = this
						owner = prev
					}
				}
			}
		}
		if = {
			limit = {
				switch = {
					trigger = galaxy_size
					huge = { owner = { check_variable = { which = hcp_num_stellar_harvesters value >= @hcp_outward_bound_emergency_start_huge } } }
					large = { owner = { check_variable = { which = hcp_num_stellar_harvesters value >= @hcp_outward_bound_emergency_start_large } } }
					medium = { owner = { check_variable = { which = hcp_num_stellar_harvesters value >= @hcp_outward_bound_emergency_start_medium } } }
					small = { owner = { check_variable = { which = hcp_num_stellar_harvesters value >= @hcp_outward_bound_emergency_start_small } } }
					tiny = { owner = { check_variable = { which = hcp_num_stellar_harvesters value >= @hcp_outward_bound_emergency_start_tiny } } }
				}
				any_playable_country = {
					NOR = {
						is_same_value = event_target:hcp_new_harvester_owner
						any_situation = {
							is_situation_type = hcp_situation_outward_bound_dummy
							target = { is_same_value = event_target:hcp_new_harvester_owner }
						}
					}
				}
			}
			every_playable_country = {
				limit = {
					NOT = { is_same_value = event_target:hcp_new_harvester_owner }
					has_communications = event_target:hcp_new_harvester_owner
					NOT = {
						any_situation = {
							is_situation_type = hcp_situation_outward_bound_dummy
							target = { is_same_value = event_target:hcp_new_harvester_owner }
						}
					}
				}
				hcp_start_outward_bound_dummy_situation = { EVENT = hcp_harvesters_situations.2001 }
			}
		}
	}
}

hcp_start_outward_bound_dummy_situation = {
	switch = {
		trigger = galaxy_size
		huge = { set_variable = { which = hcp_outward_bound_required_progress value = @hcp_outward_bound_required_progress_huge } }
		large = { set_variable = { which = hcp_outward_bound_required_progress value = @hcp_outward_bound_required_progress_large } }
		medium = { set_variable = { which = hcp_outward_bound_required_progress value = @hcp_outward_bound_required_progress_medium } }
		small = { set_variable = { which = hcp_outward_bound_required_progress value = @hcp_outward_bound_required_progress_small } }
		tiny = { set_variable = { which = hcp_outward_bound_required_progress value = @hcp_outward_bound_required_progress_tiny } }
	}
	start_situation = {
		type = hcp_situation_outward_bound_dummy
		target = root.owner
		effect = {
			situation_event = {
				id = $EVENT$
				[[SCOPE]
				scopes = {
					from = 	$SCOPE$
				}
				]
			}
		}
	}
}

			initializer = "hcp_back_up_black_hole"

						id = $EVENT$

							from = prev
						}
					}
				}
			}
		}
		if = {
			limit = { last_created_system = { NOT = { has_hyperlane_to = prev } } }

		else = {
			add_extra_hyperlane_to_spawned_system_effect = yes
		}
	}
}

hcp_flag_departure_points_per_galaxy_size = {
	if = {
		limit = {
			NOT = { has_global_flag = hcp_departure_points_set }
		}
		switch = {
			trigger = galaxy_size
			tiny = { hcp_flag_departure_points = { NUM = @hcp_outward_bound_required_progress_tiny } }
			small = { hcp_flag_departure_points = { NUM = @hcp_outward_bound_required_progress_small } }
			medium = { hcp_flag_departure_points = { NUM = @hcp_outward_bound_required_progress_medium } }
			large = { hcp_flag_departure_points = { NUM = @hcp_outward_bound_required_progress_large } }
			huge = { hcp_flag_departure_points = { NUM = @hcp_outward_bound_required_progress_huge } }
		}
		set_global_flag = hcp_departure_points_set
	}
}


hcp_flag_departure_points = {
	while = {
		count = $NUM$
		random_system = {
			limit = {
				is_rim_system = yes
				NOR = {
					has_star_flag = hcp_depart_galaxy_location
					any_rim_system = {
						has_star_flag = hcp_depart_galaxy_location
						distance = {
							source = prev
							min_distance >= 5
							max_distance <= 60
							type= euclidean
							use_bypasses = no
							same_solar_system = no
						}
					}
				}
			}
			set_star_flag = hcp_depart_galaxy_location
		}
	}
	if = {
		limit = {
			count_rim_system = {
				count < $NUM$
				limit = {
					has_star_flag = hcp_depart_galaxy_location
				}
			}
		}
		while = {
			limit = {
				count_rim_system = {
					count < $NUM$
					limit = {
						has_star_flag = hcp_depart_galaxy_location
					}
				}
			}
			random_system = {
				limit = {
					is_rim_system = yes
					NOR = {
						has_star_flag = hcp_depart_galaxy_location
					}
				}
				set_star_flag = hcp_depart_galaxy_location
			}
		}
	}
}

hcp_highlight_departure_points = {

		limit = {

		}
		set_star_flag = hcp_depart_galaxy_location_@root
		star = {
			enable_special_project = {
				name = hcp_DEPARTURE_POINT
				owner = root
			}
		}
	}
}
