mjarchitect_create_entity_country = {
	create_country = {
		name = MJARCHITECT_NAME
		species = last_created_species
		flag = { 
			icon = {
				category = "mjarchitect"
				file = "mjarchitect_flag.dds"
			}
			background = {
				category = "backgrounds"
				file = "circle.dds"
			}
			colors = {
				"dark_red" 
				"intense_burgundy"
				"null"
				"null"
			}
		}
		type = mjarchitect_entity_country
		name_list = ROOT
		auto_delete = no
		effect = {
			save_global_event_target_as = mjarchitect_entity_country
			set_country_flag = mjarchitect_entity_country
			establish_communications_no_message = root
			set_faction_hostility = {
				set_friendly = yes
				target = root
			}
		}
	}
}

mjarchitect_convert_percentage_pops = {
	while = {
				limit = {
					pop_amount_percentage = {
						percentage < $PERCENT$
						limit = {
							has_trait = trait_mjarchitect_conduit_of_ruin
						}
					}
				}
				random_owned_pop_group = {
					limit = {
						is_sapient = yes
					}
					modify_species = {
						species = this
						add_trait = trait_mjarchitect_conduit_of_ruin
						add_traits_at_start_of_list = yes
					}
				}
	}
}

mjarchitect_veil_cloud_map_effect = {
	solar_system = {
		set_star_flag = mjarchitect_beacon_afflicted_system
		add_modifier = {
			modifier = $MODIFIER$
		}
		# Temporarily set system ownership to Architect
		starbase.owner = {
			save_event_target_as = mjarchitect_system_owner
		}
		if = {
				limit = {
					exists = starbase
				}
				starbase.solar_system = {
					save_event_target_as = mjarchitect_beacon_afflicted_system
				}
				starbase.owner = {
					save_event_target_as = mjarchitect_system_owner
					root = { save_event_target_as = mjarchitect_beacon_builder }
				}
				starbase = { set_owner = event_target:mjarchitect_entity_country }
				# Re-set owner [must be done 1 day later, as must trigger after fog machine enabled]
				system_event = {
					id = mjarchitect.0
					days = 1
				}
			}
			else = {
				create_starbase = {
					size = starbase_outpost
					owner = event_target:mjarchitect_entity_country
				}
				system_event = {
					id = mjarchitect.1
					days = 1
				}
		}
		# Finds all systems within set radius
		every_neighbor_system_euclidean = {
			limit = {
				distance = {
					source = prev
					max_distance <= $RADIUS$
					type = euclidean
				}
			}
			set_star_flag = mjarchitect_beacon_afflicted_system
			add_modifier = {
				modifier = $MODIFIER$
			}
			# GFX
			# every_ambient_object = {
			# 	limit = { has_ambient_object_flag = mjarchitect_veil_beacon_effect }
			# 	destroy_ambient_object = this
			# }
			# random_system_planet = {
			# 	limit = {
			# 		is_star = yes
			# 	}
			# 	create_ambient_object = {
			# 		type = mjarchitect_veil_effect_2
			# 		location = this
			# 		use_3d_location = yes
			# 	}
			# 	last_created_ambient_object = {	
			# 		set_ambient_object_flag = mjarchitect_veil_beacon_effect
			# 	}
			# }
			every_system_colony = {
				limit = {
					any_owned_pop_group = {
						is_sapient = yes
					}
				}
				# Change Pops to have the Veiled trait
				random_list = {
					10 = {
						mjarchitect_convert_percentage_pops = {
							PERCENT = 0.75
						}
					}
					20 = {
						mjarchitect_convert_percentage_pops = {
							PERCENT = 0.66
						}
					}
					30 = {
						mjarchitect_convert_percentage_pops = {
							PERCENT = 0.50
						}
					}
					30 = {
						mjarchitect_convert_percentage_pops = {
							PERCENT = 0.33
						}
					}
					10 = {
						mjarchitect_convert_percentage_pops = {
							PERCENT = 0.25
						}
					}
				}
				# Low chance to turn into Veiled World # todo! if not artificial
				random_list = {
					80 = {}
					20 = {
						change_pc = pc_mjarchitect_veiled
					}
				}
			}
			# Temporarily set system owners to Architect, saving previous owner
			if = {
				limit = {
					exists = starbase
				}
				starbase.solar_system = {
					save_event_target_as = mjarchitect_beacon_afflicted_system
				}
				starbase.owner = {
					save_event_target_as = mjarchitect_system_owner
					root = { save_event_target_as = mjarchitect_beacon_builder }
					country_event = {
						id = mjarchitect.7
					}
				}
				starbase = { set_owner = event_target:mjarchitect_entity_country }
				# Re-set owner [must be done 1 day later, as must trigger after fog machine enabled]
				system_event = {
					id = mjarchitect.0
					days = 1
				}
			}
			else = {
				create_starbase = {
					size = starbase_outpost
					owner = event_target:mjarchitect_entity_country
				}
				system_event = {
					id = mjarchitect.1
					days = 1
				}
			}
		}
		# Also set all systems with existing gfx + adjacent beacon to the Architect, so that they
		# keep their existing storms
		every_system = {
			limit = {
				has_megastructure = mjarchitect_celestial_beacon
			}
			every_neighbor_system_euclidean = {
				limit = {
						distance = {
							source = prev
							max_distance <= $RADIUS$
							type = euclidean
						}
				}
				mjarchitect_update_miasma_systems = yes
			}
			mjarchitect_update_miasma_systems = yes
		}
		# Enable fog machine
		event_target:mjarchitect_entity_country = {
			activate_fog_machine = mjarchitect_veil_storm
			fog_machine_auto_tracking = yes
			fog_machine_auto_tracking = no
		}
	}
}

mjarchitect_check_miasma = {
	every_system = {
			limit = {
				has_star_flag = mjarchitect_beacon_afflicted_system
				OR = {
					any_neighbor_system_euclidean = {
						distance = {
							source = prev
							max_distance <= $RADIUS$
							type = euclidean
						}
						has_megastructure = mjarchitect_celestial_beacon
					}
					has_star_flag = mjarchitect_beacon_afflicted_system_piercer
				}

			}
			mjarchitect_update_miasma_systems = yes
	}

	event_target:mjarchitect_entity_country = {
			activate_fog_machine = mjarchitect_veil_storm
			fog_machine_auto_tracking = yes
			fog_machine_auto_tracking = no
	}

	every_system = {
			limit = {
				has_star_flag = mjarchitect_beacon_afflicted_system
				NOR = {
					any_neighbor_system_euclidean = {
						distance = {
							source = prev
							max_distance <= $RADIUS$
							type = euclidean
						}
						has_megastructure = mjarchitect_celestial_beacon
					}
					has_star_flag = mjarchitect_beacon_afflicted_system_piercer
				}
			}
			remove_star_flag = mjarchitect_beacon_afflicted_system
			# every_ambient_object = {
			# 	limit = { has_ambient_object_flag = mjarchitect_veil_beacon_effect }
			# 	destroy_ambient_object = this
			# }
			remove_modifier = mjarchitect_veiled_system
			remove_modifier = mjarchitect_veiled_system_piercer
	}
}

mjarchitect_update_sacrifical_unhappiness = {
	if = {
		limit = {
			has_modifier = mjarchitect_planet_sacrifices_penalties
		}
		remove_modifier = mjarchitect_planet_sacrifices_penalties
	}
	add_modifier = {
		modifier = mjarchitect_planet_sacrifices_penalties
		days = 360
		multiplier = $EFFECT_MULT$
		time_multiplier = $LENGTH_MULT$
	}
}

mjarchitect_resource_reward_effect_hidden = {
		hidden_effect = {
			# Cognitive Matter
			set_variable = {
				which = mjarchitect_cm_gained
				value = $COGNITIVE_MATTER_VALUE$
			}
			multiply_variable = {
				which = mjarchitect_cm_gained
				value = $COGNITIVE_MATTER_MULT$
			}
			add_resource = {
				mjarchitect_cognitive_matter = 1
				mult = mjarchitect_cm_gained
			}
			
			# Satisfaction
			set_variable = {
				which = mjarchitect_satisfaction_gained
				value = $SATISFACTION_VALUE$
			}
			multiply_variable = {
				which = mjarchitect_satisfaction_gained
				value = $SATISFACTION_MULT$
			}
			if = {
				limit = {
					check_variable = {
						which = mjarchitect_satisfaction_gained
						value > $SATISFACTION_CAP$
					}
				}
				set_variable = {
					which = mjarchitect_satisfaction_gained
					value = $SATISFACTION_CAP$
				}
			}

			change_variable = {
				which = mjarchitect_satisfaction
				value = mjarchitect_satisfaction_gained
			}
			change_variable = {
				which = mjarchitect_satisfaction_display
				value = mjarchitect_satisfaction_gained
			}
		}
}

mjarchitect_create_request = {
	# mjarchitect_set_request_timers = {
	# 	YEARS = $TIME$
	# }
	mjarchitect_set_request_timers_static = yes
	# Set variable for retriggering the gatekeeper to be 5 days before the flag runs out
	set_variable = {
		which = mjarchitect_request_length
		value = 360
	}
	multiply_variable = {
		which = mjarchitect_request_length
		value = $TIME$
	}
	subtract_variable = {
		which = mjarchitect_request_length
		value = 5
	}

	set_variable = {
		which = mjarchitect_request_event_cooldown
		value = mjarchitect_request_length
	}
	subtract_variable = {
		which = mjarchitect_request_event_cooldown
		value = 1
	}
	# Set request vars and flag
	set_timed_country_flag = {
		flag = mjarchitect_request_$NUM$_active
		years = $TIME$
	}
	set_timed_country_flag = {
		flag = mjarchitect_request_active
		days = mjarchitect_request_event_cooldown
	}

	set_variable = {
		which = mjarchitect_request_target_num
		value = $TARGET$
	}
	set_variable = {
		which = mjarchitect_request_current_num
		value = 0
	}
	country_event = {
		id = mjarchitect_ui.110
		days = mjarchitect_request_length
	}
	create_message = {
		type = MESSAGE_MJARCHITECT_NEW_REQUEST
		localization = MESSAGE_MJARCHITECT_NEW_REQUEST
		days = 60
		target = root
	}

	# BELOW are values to compare against when request time is over #

	export_resource_stockpile_to_variable = {
		resource = mjarchitect_cognitive_matter
		variable = mjarchitect_num_cm_initial
	}

	set_variable = {
		which = mjarchitect_num_beacons_initial
		value = mjarchitect_num_beacons
	}

	set_variable = {
		which = mjarchitect_num_thralls_initial
		value = 0
	}
	every_subject = {
		limit = {
			any_agreement = {
				agreement_preset = preset_mjarchitect_voidthrall
			}
		}
		owner = {
			change_variable = {
				which = mjarchitect_num_thralls_initial
				value = 1
			}
		}
	}

	set_variable = {
		which = mjarchitect_num_veiled_worlds_new
		value = 0
	}

	set_variable = {
		which = mjarchitect_wars_won_new
		value = 0
	}

	set_variable = {
		which = mjarchitect_techs_initial
		value = trigger:num_researched_techs
	}

	set_variable = {
		which = mjarchitect_physics_income_initial
		value = trigger:stored_physics_points
	}
}

mjarchitect_check_satisfaction = {
		if = {
			limit = {
				check_variable = {
					which = mjarchitect_satisfaction_display
					value > 150
				}
			}
			set_variable = {
				which = mjarchitect_satisfaction_display
				value = 150
			}
		}
		else_if = {
			limit = {
				check_variable = {
					which = mjarchitect_satisfaction_display
					value < -150
				}
			}
			set_variable = {
				which = mjarchitect_satisfaction_display
				value = -150
			}
		}

		if = {
			limit = {
				check_variable = {
					which = mjarchitect_satisfaction
					value > 300
				}
			}
