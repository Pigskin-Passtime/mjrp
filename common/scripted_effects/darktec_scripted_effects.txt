					species = {

	darktec_void_collapse_galaxy_effect = {
		if = {
			limit = {
				galaxy_size = tiny
			}
			while = { #Destroy ~10% random stars, for good measure
				count = 18
				random_system = {
					destroy_star_system = yes
				}
			}
		}
		else_if = {
			limit = {
				galaxy_size = small
			}
			while = { #Destroy ~10% random stars, for good measure
				count = 36
				random_system = {
					destroy_star_system = yes
				}
			}
		}
		else_if = {
			limit = {
				galaxy_size = medium
			}
			while = { #Destroy ~10% random stars, for good measure
				count = 52
				random_system = {
					destroy_star_system = yes
				}
			}
		}
		else_if = {
			limit = {
				galaxy_size = large
			}
			while = { #Destroy ~10% random stars, for good measure
				count = 70
				random_system = {
					destroy_star_system = yes
				}
			}
		}
		else_if = {
			limit = {
				galaxy_size = huge
			}
			while = { #Destroy ~10% random stars, for good measure
				count = 88
