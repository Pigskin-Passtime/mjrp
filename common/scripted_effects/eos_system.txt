        #test
        initializer = eos_system_initializer
        max_jumps = 1
        direction = corewards
        min_distance = 100
    }
}

# scope (this = planet, prev = system)
eos_adapt_ethic_planet = {
    if = {
        limit = {
            event_target:owner = {
                has_ethic = ethic_$ethic$
            }
        }
        change_pc = {
            class = pc_barren
            inherit_entity = no
        }
        set_deposit = eos_d_unity_1
    }
    else_if = {
        limit = {
            event_target:owner = {
                has_ethic = ethic_fanatic_$ethic$
            }
        }
        change_pc = {
            class = pc_molten
            inherit_entity = no
        }
        set_deposit = eos_d_unity_2
    }
}

eos_restore_ethic_planet = {
    if = {
        limit = {
            event_target:owner_prev = {
                OR = {
                    has_ethic = ethic_$ethic$
                    has_ethic = ethic_fanatic_$ethic$
                }
            }
        }
        eos_restore_ethic_planet_raw = yes
    }
}

eos_restore_ethic_planet_raw = {
    if = {
        limit = {
            NOT = {
                is_planet_class = pc_shattered
            }
        }
        change_pc = {
            class = pc_shattered
            inherit_entity = no
        }
        clear_deposits = yes
    }
}
