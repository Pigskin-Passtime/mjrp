ds_pr_crisis_effect = {
	# Will add the GalCom leaders as war_leader, otherwise the strongest opponent
	if = {
		limit = {
			has_galactic_emperor = yes
			galactic_emperor = { has_country_flag = ds_pr_anti_crisis }
		}
		galactic_emperor = { save_event_target_as = war_leader }
	}
	else_if = {
		limit = {
			has_galactic_emperor = no
			has_galactic_custodian = yes
			galactic_custodian = { has_country_flag = ds_pr_anti_crisis }
		}
		galactic_custodian = { save_event_target_as = war_leader }
	}
	else_if = {
		limit = {
			any_playable_country = {
				has_country_flag = ds_pr_anti_crisis
				relative_power = {
					who = prev
					value >= superior
					category = fleet
				}
			}
		}
		random_playable_country = {
			limit = {
				has_country_flag = ds_pr_anti_crisis
				relative_power = {
					who = prev
					value >= superior
					category = fleet
				}
			}
			save_event_target_as = war_leader
		}
	}
	else_if = {
		limit = {
			any_playable_country = {
				has_country_flag = ds_pr_anti_crisis
				relative_power = {
					who = prev
					value >= equivalent
					category = all
				}
			}
		}
		random_playable_country = {
			limit = {
				has_country_flag = ds_pr_anti_crisis
				relative_power = {
					who = prev
					value >= equivalent
					category = all
				}
			}
			save_event_target_as = war_leader
		}
	}
	else = {
		random_playable_country = {
			limit = {
				has_country_flag = ds_pr_anti_crisis
				is_subject = no
			}
			save_event_target_as = war_leader
		}
	}
	# War
	event_target:war_leader = {
		declare_war = {
			target = event_target:ds_pr_crisis_country
			name = {
				key = "NAME_Declared_Crisis_War"
				variable_string = "\\[This.MainDefender.GetAdj\\]"
			}
			attacker_war_goal = "wg_declared_crisis"
		}
	}
	every_playable_country = {
		limit = {
			has_country_flag = ds_pr_anti_crisis
			NOT = { is_same_value = event_target:war_leader }
		}
		join_war = event_target:war_leader
	}
	# Notifications
	every_playable_country = {
		limit = { has_crisis_level = ds_pr_level_5 }
		country_event = { id = dsPR.55 }
	}
	every_playable_country = {
		limit = { NOT = { has_crisis_level = ds_pr_level_5 } }
		country_event = { id = dsPR.60 }
	}
}

# Scope = star
ds_pr_crisis_breach = {
	change_variable = { which = ds_pr_breach_var value = 10 } # Modifier has 1%
	add_modifier = { modifier = ds_pr_reality_breach mult = ds_pr_breach_var }
	if = {
		limit = { check_variable = { which = ds_pr_breach_var value >= 100 } }
		prev = {
			set_star_flag = ds_pr_breach_star
			set_star_class = sc_ds_pr_rift
		}
		set_planet_entity = { entity = ds_pr_rift_star_entity }
		clear_variable = ds_pr_breach_var
		clear_global_event_target = ds_pr_breach_target@root
	}
}

ds_pr_crisis_trait_limit = {
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		variable = ds_pr_pops_all_var
	}
	multiply_variable = { which = ds_pr_pops_all_var value = 0.5 } # 50% of the population
	export_trigger_value_to_variable = {
		trigger = count_owned_pop_amount
		parameters = { limit = { has_trait = trait_ds_exd_phantom } }
		variable = ds_pr_pops_trait_var
	}
	if = {
		limit = {
			check_variable_arithmetic = {
				which = ds_pr_pops_all_var
				subtract = ds_pr_pops_trait_var
				value > 0
			}
		}
		country_event = { id = dsPR.31 days = 15 }
	}
	clear_variable = ds_pr_pops_all_var
	clear_variable = ds_pr_pops_trait_var
}
