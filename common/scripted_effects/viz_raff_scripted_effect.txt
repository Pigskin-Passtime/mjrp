spawn_viz_raff = {
	set_global_flag = viz_raff_system_spawned
	spawn_system = {
		initializer = viz_raff_blasted_ruins_init_01
	}
	last_created_system = {
		log = "spawned [This.GetName]"
	}
}

increment_viz_raff_hidden_power = {
	if = {
		limit = {
			NOT = {
				is_variable_set = viz_raff_power_gained_hidden
			}
		}
		set_variable = {
			which = viz_raff_power_gained_hidden
			value = 0
		}
	}
	change_variable = {
		which = viz_raff_power_gained_hidden
		value = 10
	}
}
