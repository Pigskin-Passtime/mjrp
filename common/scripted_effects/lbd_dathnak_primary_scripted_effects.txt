lbd_declare_new_xenon_war = {
	
	random_country = {
		limit = {
			not = {
				is_same_empire = event_target:lbd_dathnak_xenon_crisis_country
			}
			
			event_target:lbd_dathnak_xenon_crisis_country = {
				can_declare_war = {
					target = prev
					attacker_war_goal = wg_lbd_dathnak_xenon_crisis_containment
				}
			}
		}
event_target:lbd_dathnak_xenon_crisis_country = {
declare_war = {
	target = prev
	attacker_war_goal = wg_lbd_dathnak_xenon_crisis_containment
	effect = {
		save_global_event_target_as = lbd_dathnak_current_xenon_war
	}
}

}
save_global_event_target_as = lbd_dathnak_xenon_current_main_combatant
		weights = {
			base =5
			modifier = {
				is_ai = no
				factor = 250
			}
			modifier = {
				pop_amount > 0
			num_ships > 0
			 add = 200
			}
		}
	}
	lbd_dathnak_add_to_xenon_war = yes
}
lbd_dathnak_add_to_xenon_war = {
	if = {
		limit = {
			exists = event_target:lbd_dathnak_current_xenon_war
		}
		every_country = {
		limit = {
			# OR = {
				pop_amount > 0
				num_ships >0
			# }
			NOR = {
				is_same_empire = event_target:lbd_dathnak_xenon_crisis_country
				is_at_war_with = event_target:lbd_dathnak_xenon_crisis_country
			}

		}
		join_war_on_side = {
			war = event_target:lbd_dathnak_current_xenon_war
			side = defenders
		}

	}	}
}
