generate_random_variable_value = {
	set_variable = {
		which = random_number
		value = 0
	}
	set_variable = {
		which = random_number_temp
		value = 0
	}
	set_variable = {
		which = random_number_digits
		value = 1
	}
	while = {
		count = 2
		random_list = {
			1 = {
				set_variable = {
					which = random_number_temp
					value = 0
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 1
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 2
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 3
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 4
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 5
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 6
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 7
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 8
				}
			}
			1 = {
				set_variable = {
					which = random_number_temp
					value = 9
				}
			}
		}
		multiply_variable = {
			which = random_number_temp
			value = random_number_digits
		}
		change_variable = {
			which = random_number
			value = random_number_temp
		}
		multiply_variable = {
			which = random_number_digits
			value = 10
		}
	}
	set_variable = {
		which = random_number_temp
		value = 0
	}
	set_variable = {
		which = random_number_digits
		value = 0
	}
}

tup_increase_buy_utopic_energy_costs = {
	if = {
		limit = {
			check_variable = {
				which = tup_utopic_energy_cost
				value < 10000
			}
		}
		change_variable = {
				which = tup_utopic_energy_cost
				value = 500
			}
	}
	else_if = {
		limit = {
			check_variable = {
				which = tup_utopic_energy_cost
				value < 25000
			}
		}
		change_variable = {
			which = tup_utopic_energy_cost
			value = 750
		}
	}
	else_if = {
		limit = {
			check_variable = {
				which = tup_utopic_energy_cost
				value < 50000
			}
		}
		change_variable = {
			which = tup_utopic_energy_cost
			value = 1000
		}
	}
	else = {
		change_variable = {
			which = tup_utopic_energy_cost
			value = 5000
		}
	}
}

tup_convert_vassal_to_loyal_vassal = {
	set_country_flag = tup_loyal_vassal

	every_agreement = {
		limit = {
			agreement_preset = preset_vassal
		}
		set_agreement_preset = {
			preset = tup_preset_loyal_vassal
		}
	}

	every_playable_country = {
		remove_opinion_modifier = {
			who = FROM
			modifier = tup_opinion_discord
		}
		remove_opinion_modifier = {
			who = FROM
			modifier = tup_opinion_brainwasher
		}
		remove_opinion_modifier = {
			who = FROM
			modifier = tup_opinion_crisis_test_1
		}
		remove_opinion_modifier = {
			who = FROM
			modifier = tup_opinion_crisis_test_2
		}
		remove_opinion_modifier = {
			who = FROM
			modifier = tup_opinion_crisis_test_3
		}
		remove_opinion_modifier = {
			who = FROM
			modifier = tup_opinion_crisis_test_4
		}
		remove_opinion_modifier = {
			who = FROM
			modifier = tup_opinion_crisis_existential_threat
		}
	}

	add_modifier = {
		modifier = tup_utopia_society
		years = -1
	}
	if = {
		limit = {
			is_egalitarian = no
		}
		shift_ethic = ethic_egalitarian

	every_owned_pop_group = {
