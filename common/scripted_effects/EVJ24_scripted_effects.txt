EVJ24_solar_eruption = {
	# Call this on the star, PLANET SCOPE
	
	if = {
		limit = { exists = space_owner }
	
		# Create a message
		
		space_owner = {
			create_message = { 
				type = EVJ24_STAR_ERUPTED
				localization = EVJ24_STAR_ERUPTED_MESSAGE 
				days = 30 
				target = prev
				
				variable = {
					type = name
					localization = SYSTEM
					scope = prev
				}
			}
		}
		
		# Destroys stations, send fleets MIA
		
		solar_system = {
			every_fleet_in_system = {
				if = {
					limit = {
						OR = {
							is_ship_class = shipclass_transport
							is_ship_class = shipclass_science_ship
							is_ship_class = shipclass_colonizer
							is_ship_class = shipclass_constructor
							is_ship_class = shipclass_military
						}
						NOT = { has_fleet_flag = EVJ24_avatar_of_light_fleet } 
					}
					set_mia = mia_return_home
				}
				if = {
					limit = {
						OR = {
							is_ship_class = shipclass_observation_station
							is_ship_class = shipclass_research_station
							is_ship_class = shipclass_mining_station
							is_ship_class = shipclass_military_station
							is_ship_class = shipclass_starbase
						}
					}
					destroy_fleet = this
				}
			}
		}
		
		# Make a BIIIG explosion
		
		create_ambient_object = {
			type = "star_explosion"
			play_animation_once = yes
			location = this
		}
		
		# Planets receive devastation, gain blockers
		
		solar_system = {
			every_system_planet = {
				limit = { is_colonizable = yes }
				add_planet_devastation = 80
				add_deposit = d_city_ruins
				add_deposit = d_ruined_district
				add_deposit = d_ruined_district
				add_deposit = d_ruined_district
			}
		}
		
		# Make a nebula
		
		solar_system = {
			if = {
				limit = { any_system_ambient_object = { NOT = { is_ambient_object_type = EVJ24_rare_nebula_1 } } }
				create_ambient_object = {
					type = "EVJ24_rare_nebula_1"
					location = this
				}
			}
		}
	
	}
}

EVJ24_AOL_tantrum = {

	# Execute in any scope, as long as the Avatar of Light exists.
	
	# The AOL sets their sights on the player that owns the Jailer and their star.
	
	event_target:EVJ24_jailer_leader = {
		owner = {
			every_system_within_border = {
				set_star_flag = EVJ24_AOL_target
			}
		}
	}
	
	# Waltz up to star and make it explode.
	
	event_target:EVJ24_avatar_of_light_fleet = {
		if = {
			limit = { NOT = { has_fleet_flag = EVJ24_throwing_tantrum } }
			set_fleet_flag = EVJ24_throwing_tantrum
			queue_actions = {
				repeat = {
					find_closest_system = {
						trigger = {
							id = AOL.$TAG$.gotosystem
							has_star_flag = EVJ24_AOL_target
							star = { NOT = { is_planet_class = pc_t_star } }
						}
						found_system = {
							move_to = this
						}
					}
					find_closest_planet = {
						trigger = {
							id = AOL.$TAG$.gotostar
							is_star = yes
							NOT = { is_planet_class = pc_t_star }
						}
						found_planet = {
							move_to = this
							orbit_planet = this
							wait = {
								duration = 90
							}
							effect = {
								id = AOL.$TAG$.blowupstar
								if = {
									limit = {
										event_target:EVJ24_avatar_of_light_fleet = { 
											NOT = { has_fleet_flag = EVJ24_AOL_interrupted }
										}
									}
									solar_system = { remove_star_flag = EVJ24_AOL_target }
									EVJ24_solar_eruption = yes
									EVJ24_make_t_dwarf = yes
								}
							}
						}
					}
					wait = {
						duration = 90
					}
				}
			}
		}
	}
}

			}
			colors = {
				"orange"
				"black"
				"null"
				"null"
			}
		}
	}
	last_created_country = {
		save_global_event_target_as = EVJ24_AOL_minions
		set_country_flag = EVJ24_AOL_minions
		set_faction_hostility = {
			target = event_target:EVJ24_avatar_of_light
			set_hostile = no
			set_neutral = no
			set_friendly = yes
		}
		establish_communications_no_message = root
		every_country = {
			limit = {
				intel_level = {
					system = root.solar_system
					level > low
				}
			}
			establish_communications_no_message = prev
		}
	}	
	
	export_trigger_value_to_variable = {
		trigger = fleet_power
		variable = EVJ24_country_fleet_power
		rounded = yes
	}
	set_variable = { which = EVJ24_country_fleet_power value = value:EVJ24_AOL_num_minions }

	#divide_variable = { which = EVJ24_country_fleet_power value = 2 }
	#round_variable_to_closest = { which = EVJ24_country_fleet_power value = 1 }

	while = {
		count = this.EVJ24_country_fleet_power
		random_planet_within_border = {
			limit = { NOT = { is_same_value = owner.capital_scope } }
			create_fleet = {
				name = "NAME_AOL_minions"
				settings = {
					spawn_debris = no
					is_boss = no
				}
				effect = {
					set_owner = event_target:EVJ24_AOL_minions
					create_ship = {
						name = ""
						design = "NAME_Avatar_of_Light_Minion"
						
						effect = {
							set_ship_flag = EVJ24_AOL_minion_ship
							add_modifier = { modifier = EVJ24_avatar_of_light_minion_debuff days = -1 }
						}
					}
					set_location = {
						target = prev
						distance = 60
						angle = random
					}
					set_fleet_stance = aggressive
					set_aggro_range_measure_from = return_point
					set_aggro_range = 2000
					set_fleet_flag = EVJ24_AOL_minion_fleet
