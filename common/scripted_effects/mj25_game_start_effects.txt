generate_mining_and_research_stations = {
	optimize_memory
	every_system_within_border = {
		limit = {
			any_system_planet = {
				OR = {
					AND = {
						has_deposit_for = shipclass_mining_station
						has_mining_station = no
						has_any_rare_strategic_resource = no
					}
					AND = {
						has_deposit_for = shipclass_research_station
						has_research_station = no
						has_any_rare_strategic_resource = no
					}
				}
			}
		}
		every_system_planet = {
			limit = {
				has_deposit_for = shipclass_mining_station
				has_mining_station = no
				has_any_rare_strategic_resource = no
				is_colony = no
			}
			create_mining_station = {
				owner = ROOT
			}
		}
		every_system_planet = {
			limit = {
				has_deposit_for = shipclass_research_station
				has_research_station = no
				has_any_rare_strategic_resource = no
				is_colony = no
			}
			create_research_station = {
				owner = ROOT
			}
		}
	}
}

		which = num_empires_setup
		value = 1
	}

		limit = {
			count_playable_country = {
				count < this.num_empires_setup

					has_country_flag = not_setup
				}
			}
		}

		log = "passed setup flag with [This.GetName]"
		switch = {
			trigger = has_global_flag

				generate_starting_borders_actual = {
					JUMPS_FROM_CAPITAL = 3

				}

					JUMPS_FROM_CAPITAL = 5

generate_starting_borders_actual = {							# $JUMPS_FROM_CAPITAL$ - how adjacent systems the effect will try to take
	optimize_memory												# $EXTRA_JUMPS_SURVEYED$ - how systems out from their borders will be surveyed 
																# $EXTRA_JUMPS_VISITED$ - how many systems out from their borders will be visited (eg by a science ship or fleet)
#	every_system = {											# generate outposts
#		limit = {
#			distance = {
#				source = root.capital_scope.solar_system
#				min_jumps = 1
#				max_jumps = $JUMPS_FROM_CAPITAL$
#				type = hyperlane
#			}
#			valid_for_border_generation = yes
#		}
#		root = { set_visited = prev }
#		set_surveyed = {
#			surveyed = yes
#			surveyor = root
#		}
#		create_starbase = {
#			owner = root
#			size = starbase_outpost
#		}
#	}

	while = {
		count = $JUMPS_FROM_CAPITAL$
		every_country = {
			limit = {
				is_country_type = default
			}
			every_system_within_border = {
				limit = {
					is_on_border = yes
					any_neighbor_system = {
						valid_for_border_generation = yes
					}
				}
				every_neighbor_system = {
					limit = {
						valid_for_border_generation = yes
						#print_scope_trigger = yes
						distance = {
							source = prevprev.capital_scope.solar_system
							min_jumps = 1
							max_jumps = $JUMPS_FROM_CAPITAL$
							type = hyperlane
						}
					}
					prevprev = {
						set_visited = prev }
						set_surveyed = {
							surveyed = yes
							surveyor = root
						}
					create_starbase = {
						owner = prevprev
						size = starbase_outpost
					}
				}
			}
			set_country_flag = borders_generated
		}
	}
	every_country = {
		limit = {
			is_country_type = default
		}
		every_system = {										# set system surveyed
			limit = {
				distance = {
					source = prev.capital_scope.solar_system
					min_jumps = $JUMPS_FROM_CAPITAL$
					max_jumps = $EXTRA_JUMPS_SURVEYED$
					type = hyperlane
				}
			}
			prev = { set_visited = prev }
			set_surveyed = {
				surveyed = yes
				surveyor = prev
			}
		}

		every_system = {										# set system visited
			limit = {
				distance = {
					source = prev.capital_scope.solar_system
					min_jumps = $EXTRA_JUMPS_SURVEYED$
					max_jumps = $EXTRA_JUMPS_VISITED$
					type = hyperlane
				}
			}
			prev = { set_visited = prev }
			set_surveyed = {
				surveyed = yes
				surveyor = prev
			}
		}
	}
	set_global_flag = ready_to_generate_empires
}

generate_starting_planets_actual = {
	optimize_memory
	root = {
		generate_mining_and_research_stations = yes
		while = {
			limit = {
				num_owned_planets <= $NUM_COLONIES$
				NOR = {
					has_origin = origin_shattered_ring
					#has_origin = origin_clone_army
				}
			}
			if = {
				limit = {
					any_system_within_border = {
						any_system_planet = {
							is_colonizable = yes
						}
					}
				}
				random_system_within_border = {
					limit = {
						any_system_planet = {
							is_colonizable = yes
						}
					}
					random_system_planet = {
						limit = {
							is_colonizable = yes
						}
						generate_colony = yes
					}
				}
			}
			else = {
				random_system_within_border = {
					limit = {
						any_system_planet = {
							is_colony = no
							is_star = no
							is_asteroid = no
						}
					}
					random_system_planet = {
						limit = {
							is_star = no
							is_asteroid = no
							is_colony = no
						}
						set_planet_size = 18
						change_pc = root.species
						reroll_planet = yes
						generate_colony = yes
					}
				}
			}
			every_owned_planet = {
				clear_blockers = yes
				#calculate_modifier = yes
			}
		}
	}
}

generate_start_selection_dates = {
	set_variable = {			# mid game dates
		which = mid_game_setup_value_temp
		value = mid_game_setup_value
	}
	subtract_variable = {
		which = mid_game_setup_value_temp
		value = $years$
	}
	set_variable = {
		which = mid_game_start_$years$
		value = mid_game_setup_value_temp
	}
	if = {
		limit = {
			check_variable = {
				which = mid_game_start_$years$
				value <= 0
			}
		}
		set_variable = {
			which = mid_game_start_$years$
			value = 1
		}
	}
	add_variable = {
		which = mid_game_start_$years$
		value = 2200
	}
	set_variable = {		# end game dates
		which = end_game_setup_value_temp
		value = end_game_setup_value
	}
	subtract_variable = {
		which = end_game_setup_value_temp
		value = $years$
	}
	set_variable = {
		which = end_game_start_$years$
		value = end_game_setup_value_temp
	}
	if = {
		limit = {
			check_variable = {
				which = end_game_start_$years$
				value <= 0
			}
		}
		set_variable = {
			which = end_game_start_$years$
			value = 1
		}
	}
	add_variable = {
		which = end_game_start_$years$
		value = 2200
	}
	set_variable = {		# victory year dates
		which = victory_year_setup_value_temp
		value = victory_year_setup_value

		which = victory_year_setup_value_temp

		which = victory_year_$years$
		value = victory_year_setup_value_temp

		limit = {
			check_variable = {
				which = victory_year_$years$
				value <= 0
			}
		}
		set_variable = {
			which = victory_year_$years$
			value = 1
		}
	}
}
