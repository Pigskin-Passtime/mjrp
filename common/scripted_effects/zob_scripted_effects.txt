spawn_zob = {
	spawn_system = { 
		min_distance = 20
		max_distance = 50
		initializer = "zob_system"
		hyperlane = yes
	}
	last_created_system = {
		ordered_neighbor_system_euclidean = {
			limit = {
				not = {
					has_hyperlane_to = prev 
				}
			}
			add_hyperlane = {
				from = this
				to = prev
			}
	  	}
		ordered_neighbor_system_euclidean = {
			limit = {
				not = {
					has_hyperlane_to = prev 
				}
			}
			add_hyperlane = {
				from = this
				to = prev
			}
	  	}
	}
	set_global_flag = zob_system_spawned
	log = "Spawned Zobs Home System"
	###Maybe figure out nebula object on galaxy map
}

	optimize_memory
	if = {
		limit = {
			has_trait = $trait$
		}
		set_variable = {
			which = zob_$trait$
			value = 1	
		}
		owner = {
			change_variable = {
				which = zob_$trait$
				value = 1
			}
		}
	}
}

#If the leaders local variable is 0 but they have the trait, they recently gained it. Saves variable, and increments the country level tracker for collected traits
update_zob_variable = {
    optimize_memory
	if = {
		limit = {
			exists = owner
			check_variable = {
				which = zob_$trait$
				value = 0
			}
			has_trait = $trait$
		}
		set_variable = {
			which = zob_$trait$
			value = 1
		}
		owner = {
			change_variable = {
				which = zob_$trait$
				value = 1
			}
		}
	}
	if = {
		limit = {
			exists = owner
			owner = {
				check_variable = { which = zob_$trait$ value > 0 }
			}
		}
		set_variable = {
			which = $trait$
			value = owner.zob_$trait$
		}
		if = {
			limit = {
				check_variable = { which = zob_$trait$ value > 0 }
			}
			subtract_variable = {
				which = $trait$
				value = this.zob_$trait$
			}
		}
	}
}
