#To be called directly from TFO.2 event (to avoid code duplication)
TFO_event_2_add_trait_and_start_chain = {
    if = {
        limit = {
            exists = fromfromfrom
            exists = fromfromfrom.leader
        }

        fromfromfrom.leader = {
            add_trait = leader_trait_carefree
        }
    }

    hidden_effect = {
        country_event = {
            id = TFO.3
            days = 1
            scopes = {
                from = from
            }
        }
    }
}

#Leader scope
TFO_set_leader_traits = {
    remove_all_traits = yes
    if = {
        limit = {
            owner = {
                is_gestalt = no
            }
        }

        add_trait_no_notify = subclass_scientist_councilor
        add_trait_no_notify = leader_trait_inquisitive_2
        add_trait_no_notify = leader_trait_expertise_particles_2
        add_trait_no_notify = leader_trait_expertise_field_manipulation_2
        add_trait_no_notify = leader_trait_expertise_voidcraft_2
        add_trait_no_notify = leader_trait_spark_of_genius_2
    }
    else = {
        add_trait_no_notify = subclass_scientist_governor
        add_trait_no_notify = leader_trait_collaborator_2
        add_trait_no_notify = leader_trait_manufacturer_2
        add_trait_no_notify = TFO_leader_trait_philosopher
        add_trait_no_notify = leader_trait_physics_focus_2
        add_trait_no_notify = leader_trait_adventurous_spirit_3
    }
}

#Leader scope
TFO_add_leader_destiny_trait = {
    add_skill_without_trait_selection = 1
    if = {
        limit = {
            owner = {
                is_gestalt = no
            }
        }

        add_trait = TFO_leader_trait_relict_from_the_past
    }
    else = {
        add_trait = TFO_leader_trait_relict_from_the_past_gestalt
    }
}
