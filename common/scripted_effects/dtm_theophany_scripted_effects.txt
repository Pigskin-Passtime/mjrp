dtm_theophany_increase_corruption_planet = {
	if = {
		limit = {
			has_modifier = dtm_theophany_planet_corruption
		}
		remove_modifier = dtm_theophany_planet_corruption
	}
	if = {
		limit = {
			NOT = {
				is_variable_set = dtm_theophany_planet_corruption
			}
		}
		set_variable = {
			which = dtm_theophany_planet_corruption
			value = $AMOUNT$
		}
	}
	else = {
		change_variable = {
			which = dtm_theophany_planet_corruption
			value = $AMOUNT$
		}
	}
	add_modifier = {
		modifier = dtm_theophany_planet_corruption
		mult = this.dtm_theophany_planet_corruption
	}
}

dtm_theophany_create_shroud_fleet_from_country = {
	random_country = {
		limit = {
			is_country_type = default
			is_at_war_with = root.owner
		}
		create_leader = {
			class = commander
			species = owner_main_species
			name = random
			skill = 4
			effect = {
				save_event_target_as = dtm_theophany_create_shroud_fleet_from_country_leader
			}
		}
		last_created_leader = {
			set_owner = event_target:dtm_theophany_shroud_country
		}
		create_fleet_from_naval_cap = $FRACTION$
		last_created_fleet = {
			set_name = NAME_Shroud_Manifestation
			set_owner = event_target:dtm_theophany_shroud_country
			set_location = {
				target = $LOCATION$
				distance = 300
				angle = random
			}
			set_fleet_stance = aggressive
			set_aggro_range = 500
			set_aggro_range_measure_from = self
			# set_fleet_bombardment_stance = indiscriminate
			assign_leader = event_target:dtm_theophany_create_shroud_fleet_from_country_leader
		}
	}
}

dtm_theophany_create_shroud_fleet_fly_to_center = {
	last_created_fleet = {
		clear_orders = yes
		clear_fleet_actions = THIS		# fleet scope	
		queue_actions = {
			find_closest_system = {
				trigger = {
					id = dtm_theophany_create_shroud_fleet_from_country_$ID$
					has_star_flag = dtm_theophany_endgame_center
				}
				found_system = {
					move_to = this
					orbit_planet = star
				}
			}
		}
	}
}

dtm_theophany_corrupt_planet_final = {
	dtm_theophany_increase_corruption_planet = {
		AMOUNT = 100
	}
	if = {
		limit = {
			habitable_planet = yes
			NOT = {
				is_planet_class = pc_dtm_shardworld
			}
		}
		change_pc = pc_shrouded
		# this is for all, not just the corruptor
		every_playable_country = {
			limit = {
				has_menace_perk = menp_dtm_theophany_3d
			}
			country_event = {
				id = dtm_theophany_objectives.6
			}
		}
	}
	if = {
		limit = {
			has_global_flag = dtm_theophany_unlocked_theophany_corruption_3
		}
		solar_system = {
			save_event_target_as = solar_system
			inline_script = {
				script = dtm_theophany/start_system_corruption
				AMOUNT = 20
				SYSTEM = event_target:solar_system
			}
		}
	}
}

dtm_theophany_corrupt_system_final = {
	set_star_flag = dtm_theophany_corruption_complete
	add_modifier = {
		modifier = dtm_theophany_system_corruption_final		#this way we show the full modifer 
	}
	solar_system = {
		create_nebula = {
			radius = 1
			effect = {
				star = {
					create_ambient_object = {
						type = dtm_theophany_space_storm_1
						location = root.target.star
						scale = 1
						use_3d_location = yes
						#entity_scale_to_size = yes
					}
					last_created_ambient_object = {
						#set_ambient_object_flag = space_storm_object
						set_location = {
							target = root.target.star
							distance = 0
							angle = random
						}
					}
				}
			}
		}
		if = {
			limit = {
				has_global_flag = dtm_theophany_unlocked_theophany_corruption_3
			}
			every_neighbor_system = {
				solar_system = {
					save_event_target_as = solar_system
					inline_script = {
						script = dtm_theophany/start_system_corruption
						AMOUNT = 30
						SYSTEM = event_target:solar_system
					}
				}
			}
		}
	}
	# this is for all, not just the corruptor
	every_playable_country = {
		limit = {
			has_menace_perk = menp_dtm_theophany_3d
		}
		country_event = {
			id = dtm_theophany_objectives.7
		}
	}
}
