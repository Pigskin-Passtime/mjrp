spawn_wp02 = {
	if = {
		limit = {
			NOT = {
				has_global_flag = wp02_system_spawned
			}
		}
		set_global_flag = wp02_system_spawned
		spawn_system = {
			initializer = wp02_initializer_01
		}
		last_created_system = {
			log = "spawned [This.GetName]"
		}
	}
}
