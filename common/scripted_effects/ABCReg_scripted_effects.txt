

ABCReg_spawn_bubbles_leader = {
    if = {
        limit = {
            NOT = {
                has_global_flag = ABCReg_system_spawned
            }
        }
        set_global_flag = ABCReg_system_spawned
        spawn_system = {
            min_distance >= 10
            max_distance <= 30
            max_jumps = 1
            initializer = ABCReg_limbo
		}
	    last_created_system = {
            log = "spawned \\[This.GetName\\]"
        }
	}
}
