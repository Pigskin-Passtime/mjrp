enable_storm_cluster = {
	if = {
		limit = {
			NOT = { has_global_flag = scluster_allowed }
		}
		set_global_flag = scluster_allowed
	}
}

disable_storm_cluster = {
	if = {
		limit = {
			has_global_flag = scluster_allowed
		}
		remove_global_flag = scluster_allowed
	}
}

set_storm_cluster_variables = {
	event_target:global_event_country = {
		[[MAX_MEGASTRUCTURES]
			set_variable = {
				which = scluster_megastructures_max
				value = $MAX_MEGASTRUCTURES$
			}
		]
		[[MAX_GUARDIANS]
			set_variable = {
				which = scluster_guardians_max
				value = $MAX_GUARDIANS$
			}
		]
	}
}

scluster_trigger_great_khan = {
	if = {
		limit = {
			NOT = { has_global_flag = marauder_crisis_ongoing }
		}
		random_country = {
			limit = {
				is_country_type = dormant_marauders
				has_country_flag = scluster_marauders
			}
			set_global_flag = horde_triggered
			country_event = {
				id = achievement.1
			}
		}
	}
}

scluster_clear_orbital_deposits = {
	every_deposit = {
		limit = { scluster_has_orbital_deposit_category_$CAT$ = yes }
		remove_deposit = yes
	}
	if = {
		limit = { any_moon = { has_planet_flag = research_habitat } }
	}
}

scluster_validate_habitat_flags = {
	if = {
		limit = {
			is_planet_class = pc_habitat
			# Any way to exclude habitats that have used decision_prospect_void?
		}
		# has_planet_flag = energy_habitat
		# has_planet_flag = mining_habitat
		# has_planet_flag = research_habitat
		if = {
			limit = {
				orbit = {
					OR = {
						has_deposit_for = shipclass_research_station
						scluster_has_orbital_deposit_category_zro = yes
						scluster_has_orbital_deposit_category_dark_matter = yes
						has_deposit = d_living_metal_deposit
						has_deposit = d_nanites_deposit
						has_deposit = d_artifacts_1
						has_deposit = d_artifacts_2
						has_deposit = d_artifacts_3
					}
				}
			}
			set_planet_flag = research_habitat
		}
		else = {
			remove_planet_flag = research_habitat
			if = {
				limit = {
					orbit = {
						OR = {
							scluster_has_orbital_deposit_category_minerals = yes
							scluster_has_orbital_deposit_category_alloys = yes
						}
					}
				}
				set_planet_flag = mining_habitat
				remove_planet_flag = energy_habitat
			}
			else = {
				remove_planet_flag = mining_habitat
				limit = {
					orbit = {
						scluster_has_orbital_deposit_category_energy = yes
					}
				}
				set_planet_flag = energy_habitat
			}
		}
	}
	# validate buildings & districts?
}

scluster_clear_invalid_with_energy_deposits = {
	if = {
		limit = {
			OR = {
				has_deposit_for = shipclass_research_station
				scluster_has_orbital_deposit_category_minerals = yes
				scluster_has_orbital_deposit_category_alloys = yes
			}
		}
		clear_deposits = yes
	}
}

	if = {

				scluster_has_orbital_deposit_category_alloys = yes
			}
		}
		clear_deposits = yes
