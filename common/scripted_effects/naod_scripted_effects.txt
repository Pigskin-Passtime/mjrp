spawn_naod = {
	if = {
		limit = {
			has_global_flag = naod_allowed
			NOT = {
				has_global_flag = naodeus_system_spawned
			}
		}
		set_global_flag = naodeus_system_spawned
		spawn_system = {
			initializer = naod_talil_system
		}
		last_created_system = {
			log = "spawned \\[This.GetName\\]"
		}
	}
}
