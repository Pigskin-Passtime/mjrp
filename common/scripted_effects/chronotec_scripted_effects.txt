# Mandatory leader debug/spawn system
spawn_chronotec = {
	set_global_flag = chronotec_system_spawned
	# As this effect would be spawned either by menu or manual
	# at system level in theory...
	root = {
		save_event_target_as = chronotec_spawn_country
	}
	# Redundancy in case its forcespawned
	set_global_flag = chronotec_anomaly_spawned

	# Redundancy - creates the country/leader if they dont exist
		# Country Created
		if = {
			limit = {
				NOT = {
					exists = event_target:chronotec_chronolord_country
				}
			}
			create_country = {
				name = "NAME_chronotec_chronolord"
				type = chronotec_event_country
				flag = {
					icon = {
						category = "spherical"
						file = "flag_spherical_20.dds"
					}
					background= {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors={
						"purple"
						"black"
						"null"
						"null"
					}
				}
				effect = {
					set_country_flag = suppress_first_contact_comms
					save_global_event_target_as = chronotec_chronolord_country
				}
			}
		}
		# Leader created
		if = {
			limit = {
				NOT = {
					exists = event_target:chronotec_chronolord_leader
				}
			}
			event_target:chronotec_chronolord_country = {
				create_species = {
					name = "NAME_Unknown"
					class = PARAGON
					portrait = "chronotec_chronolord_portrait_standard"
					namelist = "PRT1"
					traits = random
					effect = {
						save_event_target_as = chronotec_chronolord_species
					}
				}
				create_leader = {
					species = event_target:chronotec_chronolord_species
					event_leader = yes
					class = scientist
					name = "NAME_chronotec_chronolord"
					skill = 10
					immortal = yes
					background_ethic = ethic_materialist
					randomize_traits = no
					skip_background_generation = yes
					custom_catch_phrase = chronotec_chronolord_catch_phrase
					custom_description = chronotec_chronolord_backstory
					traits = {
						trait = leader_trait_physics_focus_3
						trait = leader_trait_collaborator_3
						trait = leader_trait_chronotec_chronolord_infusion_neutral
						trait = subclass_scientist_governor
					}
					gender = indeterminable
					set_age = 0
					hide_age = yes
					effect = {
						save_global_event_target_as = chronotec_chronolord_leader
						set_leader_flag = legendary_leader
						set_leader_flag = chronotec_chronolord_leader
						set_leader_flag = immune_to_negative_traits
						set_leader_flag = leader_death_events_blocked
						set_owner = event_target:chronotec_chronolord_country
						exile_leader_as = chronotec_chronolord_leader
					}
				}
			}
		}
		# /
	# /
	
	# Dynamic scope
	[[SCOPE]
		$SCOPE$ = {
			save_event_target_as = chronotec_spawn_scope
		}
	]
	# Fallback scope to the capital
	[[!SCOPE]
		save_event_target_as = chronotec_spawn_scope
	]

	# Spawns the system
		event_target:chronotec_spawn_scope = {	# Redundancy to be safe
			spawn_system = {
				min_distance = 5
				max_distance = 15
				max_jumps = 0
				initializer = "chronotec_chronolord_system"
			}
		}	
	# /

	# Debug surveys the system
	[[SURVEY]
		last_created_system = {	# Scoping to the created system
			every_system_planet = {
				set_surveyed = {
					surveyed = yes
					surveyor = event_target:chronotec_spawn_country
				}
			}
		}
	]
	[[!SURVEY]

	]

	# Bypass digsite, adds the leader immediately but allows investigating it
	[[DIGSITE_DEBUG]
		every_archaeological_site = {
			limit = {
				is_site_type = site_chronotec_world
			}
			set_archaeology_flag = chronotec_digsite_debug
		}
	]
	[[!DIGSITE_DEBUG]
	]

	# Force spawns the leader
	[[LEADER_DEBUG]
		root = {
			country_event = {
				id = chronotec_narrative.30
			}
		}
	]
	[[!LEADER_DEBUG]
	]

	# In case we want to add extra sauce dynamically to the effect
	$EXTRA_EFFECT$
}

	# Usage:
		# For standard spawning (adds the system)
		#spawn_chronotec = {
		#	SCOPE = scope
		#	SURVEY = no
		#	DIGSITE_DEBUG = no
		#	LEADER_DEBUG = no
		#	EXTRA_EFFECT = ""
		#}
		# Debug the system entirely (appears link to capital) and spawn the leader
		#spawn_chronotec = {
		#	SURVEY = no
		#	DIGSITE_DEBUG = yes
		#	LEADER_DEBUG = yes
		#	EXTRA_EFFECT = ""
		#}
	# /

	# Shortcut
		chronotec_test_spawn_effect = {
			random_system = {
				limit = {
					NOT = {
						is_in_cluster = leader_spawn
					}
				}
				spawn_chronotec = {
					SURVEY = yes
					SCOPE = this
					DIGSITE_DEBUG = yes
					LEADER_DEBUG = yes
					EXTRA_EFFECT = ""
				}
				create_cluster = {
					id = leader_spawn
					radius = 150
					center = THIS
				}
			}
		}
	# /
# /

chronotec_convert_into_relic_effect = {
	remove_country_flag = chronotec_chronolord_is_leader
	set_country_flag = chronotec_chronolord_is_relic
	# Adds the relic
	add_relic = r_chronotec_standard
	set_saved_date = {	# Charge timer
		key = chronotec_charge_timer
		days_from_present = @chronotec_charge_timer
		expires = @chronotec_charge_timer
	}
	# Repeatable event
	country_event = {
		id = chronotec_relic.20
		days = @chronotec_charge_timer
	}
	# Handles the leader
	event_target:chronotec_chronolord_leader = {
		exile_leader_as = chronotec_chronolord_is_relic
	}
	#set_saved_date = {	# Transformation timer
	#	key = chronotec_transformation_timer
	#	days_from_present = @chronotec_charge_timer
	#	expires = 3180060
	#}
}

chronotec_convert_into_leader_effect = {
	create_fleet = {
		name = "NAME_Other_Science_Ship"
		effect = {
			set_owner = prev
			create_ship = {
				name = random
				random_existing_design = science
			}
			set_leader = chronotec_chronolord_is_relic
			unassign_leader = this
			destroy_fleet = this
		}
	}
	# Redundancy
	remove_country_flag = chronotec_chronolord_is_relic
	set_country_flag = chronotec_chronolord_is_leader
	# Removes the relic
	if = {
		limit = {
			has_relic = r_chronotec_standard
		}
		remove_relic = r_chronotec_standard
	}
	#set_saved_date = {	# Transformation timer
	#	key = chronotec_transformation_timer
	#	days_from_present = @chronotec_charge_timer
	#	expires = 3180060
	#}
}

			owner = {
				country_event = {
					id = chronotec_relic.30
				}
				remove_country_flag = chronotec_chronolord_is_relic
				if = {
					limit = {
						has_relic = r_chronotec_standard

				}
				if = {

							has_relic = r_chronotec_perfected

			}
			remove_trait = leader_trait_chronotec_chronolord_infusion_1
