begin_crisis_replicators = {
	set_global_flag = mid_game_crisis_happened 			# this must be set and not removed
	set_timed_global_flag = {										# this must be set and not removed
		flag = mid_game_crisis_happened_recently
		years = 12
	}
	# REPLICATORS
	set_global_flag = replicators_in_progress 				# must be removed when crisis is resolved
	set_global_flag = replicators_fired 							# (optional) only set this flag if you want your crisis to fire only once
	set_global_flag = midgame_crisis_ongoing				# must be removed when crisis is resolved
	country_event = { id = replicators.1 days = 150 random = 730 }
}
