hskv_spawn = {
	if = {
		limit = {
			NOT = {
				has_global_flag = hskv_system_spawned
			}
		}
		set_global_flag = hskv_system_spawned
		spawn_system = {
			initializer = hskv_azure_dream_system
		}
		# TODO, clean up all log files
		# This is the initalization for the mod
		random_country = {
			limit = {
				is_country_type = fallen_empire
				OR = {
					has_ethic = ethic_fanatic_xenophobe
					has_ethic = ethic_fanatic_spiritualist
					has_ethic = ethic_fanatic_materialist
				}
			}
			# Sorry MrFreake, While there is probably a way to do this without a global event,
			# it is everywhere in my loc and elsewhere.
			# So I am spending my global event target currency here because
			# I'll go crazy trying to set the event target of this every single time
			# even if its for events that don't trigger much
			set_country_flag = hskv_target_fe_country
			save_global_event_target_as = hskv_target_fe_country
		}
		last_created_system = {
			log = "spawned [This.GetName]"
		}
	}
}

hskv_create_fe_aux_defense_country = {
	if = {
		limit = {
			NOT = {
				any_country = {
					has_country_flag = hskv_fe_aux_defense_country
				}
			}
		}
		create_country = {
			name = NAME_hskv_fe_empire
			type = faction
			species = event_target:hskv_target_fe_country
			flag = event_target:hskv_target_fe_country
			effect = {
				save_event_target_as = hskv_fe_aux_defense_country
				set_country_flag = hskv_fe_aux_defense_country
			}
		}
	}
}

hskv_create_shield_defense_force = {
	event_target:hskv_fe_aux_defense_country = {
		every_system = {
			limit = {
				has_star_flag = hskv_azure_dream_system
			}
			save_event_target_as = hskv_spawn_point
			every_system_planet = {
				limit = {
					has_planet_flag = hskv_azure_prison_planet
				}
				save_event_target_as = hskv_spawn_point
			}
		}

		create_fleet = {
			name = "NAME_Pirate_Fleet"
			settings = {
				spawn_debris = no
			}
			effect = {
				set_owner = event_target:hskv_fe_aux_defense_country
				while = {
					count = 5
					create_ship = {
						name = random
						design = "NAME_Gamma"
						graphical_culture = "fallen_empire_04" #Hack, I know I am using the same one for all three valid FEs
					}
				}
				set_location = event_target:hskv_spawn_point
				set_fleet_stance = aggressive
				set_aggro_range_measure_from = self
				set_aggro_range = 1000
				set_fleet_flag = hskv_shield_defense_force_fleet
			}
		}
	}
}

hskv_create_vault_defense_force ={
	event_target:hskv_fe_aux_defense_country = {
		every_system = {
			limit = {
				has_star_flag = hskv_dawns_end_system
			}
			random_system_planet = {
				limit = {
					has_planet_flag = hskv_spawn_point
				}
				save_event_target_as = hskv_spawn_point
			}
		}

		owner = {
			create_leader = {
				class = commander
				species = owner_main_species
				name = random
				skill = 10
				leader_age_min = 25
				leader_age_max = 45
			}
		}

		create_fleet = {
			name = "NAME_Pirate_Fleet"
			settings = {
				spawn_debris = yes
			}
			effect = {
				set_owner = event_target:hskv_fe_aux_defense_country
				
				while = {
					count = 2
					create_ship = {
						name = random
						design = "NAME_Gamma"
						graphical_culture = "fallen_empire_04"
					}
				}
				while = {
					count = 1
					create_ship = {
						name = random
						design = "NAME_Beta"
						graphical_culture = "fallen_empire_04"
					}
				}
				set_location = event_target:hskv_spawn_point
				set_fleet_stance = aggressive
				set_aggro_range_measure_from = self
				set_aggro_range = 1000
				set_fleet_flag = hskv_vault_defense_force_fleet
			}
		}
		last_created_fleet = { assign_leader = last_created_leader }
	}
}
