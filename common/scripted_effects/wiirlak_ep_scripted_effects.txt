# Planet Scope; event_target:wiirlak_ep_crisis_reward as the Crisis - Bloom the World and reward the Objectives
wiirlak_ep_bloom_world = {
	if = {
		limit = { is_terraforming = yes }

		stop_terraform_process = yes
	}
	if = {
		limit = { wiirlak_ep_adv_planet = yes }

		event_target:wiirlak_ep_crisis_reward = {
			complete_crisis_objective = wiirlak_ep_crisobj_blooming_adv_pc
		}
	}
	if = {
		limit = {
			exists = OWNER
			NOT = { OWNER = { is_same_empire = event_target:wiirlak_ep_crisis_reward } }
		}
		
		event_target:wiirlak_ep_crisis_reward = {
			complete_crisis_objective = wiirlak_ep_crisobj_blooming_outside_border
		}
	}
	event_target:wiirlak_ep_crisis_reward = {
		complete_crisis_objective = wiirlak_ep_crisobj_blooming_planet
	}
	change_pc = wiirlak_ep_pc_bloom
}




# Country Scope - Spread the trait in this Country
wiirlak_ep_spread_trait_country = {
	if = {
		limit = { wiirlak_ep_valid_country_trait_spreading = yes }

		wiirlak_ep_spread_trait_random_pop = yes
	}
	else_if = {
		limit = {
			count_owned_species = {
				count < trigger:num_species
				limit = { wiirlak_ep_valid_country_trait_spreading = yes }
			}
		}
				
		wiirlak_ep_spread_trait_random_pop = yes
	}
}
# Country/Planet Scope - Change Habitability to Bloom Preference for a random pop group
wiirlak_ep_spread_trait_random_pop = {
	random_owned_pop_group = {
		limit = { NOT = { has_trait = trait_wiirlak_ep_pc_bloom_preference } }

		species = {
			set_habitability_trait = {
				trait = trait_wiirlak_ep_pc_bloom_preference
			}
		}
	}
}
# Country/Planet Scope - Change Habitability to Bloom Preference for every pop group
wiirlak_ep_spread_trait_every_pop = {
	every_owned_pop_group = {

		species = {
			set_habitability_trait = {
				trait = trait_wiirlak_ep_pc_bloom_preference
			}
		}
	}
}
# Country/Planet Scope - Change Habitability to Bloom Preference for the main species
wiirlak_ep_spread_trait_main_pop = {

# Planet Scope - Spawn the Evergrowth Boon deposit on Bloom World
wiirlak_ep_spawn_deposit = {
	hidden_effect = {
		if = {
			limit = {
				is_planet_class = wiirlak_ep_pc_bloom
				NOT = { has_deposit = wiirlak_ep_deposit }
			}

			add_deposit = wiirlak_ep_deposit
		}
		wiirlak_ep_deposit_bloom = yes
	}
}

# Planet Scope - Spawn deposits on Bloom World (since Terraforming remove all of them)
wiirlak_ep_deposit_bloom = {
	restore_saved_deposits = yes

	add_deposit_if_missing = { DEPOSIT = d_fungal_caves }
	add_deposit_if_missing = { DEPOSIT = d_fungal_forest }
	add_deposit_if_missing = { DEPOSIT = d_lichen_fields }
	add_deposit_if_missing = { DEPOSIT = d_boggy_fens }
	add_deposit_if_missing = { DEPOSIT = d_lush_jungle }
	add_deposit_if_missing = { DEPOSIT = d_teeming_reef }

	add_deposit_if_missing = { DEPOSIT = d_fungal_study_zone }
	
	
	# Good Deposit are randomized
	random_list = {
		10 = {
			modifier = {
				subtract = 10
				has_biogenesis_dlc = yes
			}
			add_deposit_if_missing = { DEPOSIT = d_bogplants }
		}
		10 = {
			modifier = {
				subtract = 10
				has_paragon_dlc = yes
			}
			add_deposit_if_missing = { DEPOSIT = d_hostile_flora }
		}
		10 = {
			modifier = {
				subtract = 10
				has_paragon_dlc = yes
			}
			add_deposit_if_missing = { DEPOSIT = d_hostile_fauna }
		}
		10 = {
			add_deposit_if_missing = { DEPOSIT = d_fuming_bog }
		}
		10 = {
			add_deposit_if_missing = { DEPOSIT = d_hyperfertile_valley }
		}
		10 = {
			add_deposit_if_missing = { DEPOSIT = d_cave_shroom_veins }
		}
		10 = {
			add_deposit_if_missing = { DEPOSIT = d_spore_vents }
		}
		10 = {
			add_deposit_if_missing = { DEPOSIT = d_underground_farm }
		}
	}
	

	# Blockers; only once
	if = {
		limit = { NOT = { has_planet_flag = wiirlak_ep_flag_blockers_set } }

		set_planet_flag = wiirlak_ep_flag_blockers_set

		add_deposit = d_dangerous_wildlife_blocker
		add_deposit = d_dense_jungle
		add_deposit = d_noxious_swamp
		add_deposit = d_toxic_kelp
	}
}


# Country Scope - Set the acceptation variable by AMOUNT
wiirlak_ep_acceptation = {
	if = {
		limit = { is_variable_set = wiirlak_ep_var_acceptation }
		
		change_variable = {
			which = wiirlak_ep_var_acceptation
			value = $AMOUNT$
		}
	}
	else = {
		set_variable = {
			which = wiirlak_ep_var_acceptation
			value = $AMOUNT$
		}
	}
}


# Country Scope - THIS will be vassalized, PREV will be OVERLORD
wiirlak_ep_set_vassal = {
	if = {
		limit = { has_overlord_dlc = yes }

		set_subject_of = {
			allow_instant_negotiation = yes
			preset = wiirlak_ep_preset_evergrowth
			who = PREV
		}
	}
	else = {
		set_subject_of = {
			allow_instant_negotiation = yes
			preset = wiirlak_ep_preset_subject
			who = PREV
		}
	}
}


# Global Scope - Prepare the Blooming
wiirlak_ep_blooming_worlds_start = {
	every_galaxy_planet = {
		limit = { is_colonizable = yes }

		add_modifier = {
			modifier = wiirlak_ep_modifier_blooming_warning				days = 3630
		}
	}
}
# Global Scope - Stop the Blooming
wiirlak_ep_blooming_worlds_stop = {
	every_galaxy_planet = {
		limit = { has_modifier = wiirlak_ep_modifier_blooming_warning }

		remove_modifier = wiirlak_ep_modifier_blooming_warning
	}
}
# Global Scope - The Evergrowth is there
wiirlak_ep_blooming_worlds_finish = {
	every_galaxy_planet = {
		limit = {
			NOT = { is_planet_class = wiirlak_ep_pc_bloom }

			# Don't touch artificial planets
			is_planet_class = pc_habitat
			planet_is_habitat_equivalent = no

			is_planet_class = pc_ringworld_habitable
			is_ringworld = no
			planet_is_ring_world_equivalent = no

			is_planet_class = pc_cosmogenesis_world


			OR = {
				habitable_planet = yes
				is_colonizable = yes
				is_colony = yes
				is_terraforming_candidate = yes
				is_planet_class = pc_toxic
			}
		}

		change_pc = wiirlak_ep_pc_bloom
		solar_system = { remove_system_terraforming_candidates = yes }
	}
}

		remove_trait = leader_trait_evergrowth_myco_admin
