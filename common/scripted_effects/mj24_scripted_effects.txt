}

spawn_skrand = {
	set_global_flag = skrand_system_spawned
	set_global_flag = gateway_dude_found
	spawn_system = {
		initializer = mj24_abandoned_gateways_01
	}
	every_system = {
		limit = {
			has_star_flag = abandoned_gateway
			any_system_megastructure = {
				has_megastructure_flag = paragon_gateway_dude_homegate_flag
			}
		}
		random_system_megastructure = {
			limit = {
				has_megastructure_flag = paragon_gateway_dude_homegate_flag
			}
			remove_megastructure_flag = paragon_gateway_dude_homegate_flag
			clear_global_event_target = paragon_gateway_dude_homegate
		}
	}
	last_created_system = {
		random_system_megastructure = {
			set_megastructure_flag = paragon_gateway_dude_homegate_flag
			save_global_event_target_as = paragon_gateway_dude_homegate
		}
		log = "spawned [This.GetName]"
	}
}
