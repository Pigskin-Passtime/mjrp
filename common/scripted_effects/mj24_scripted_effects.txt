
remove_all_spawning_flags = {
	if = {
		limit = {
			has_global_flag = fixed_random
		}
		remove_global_flag = fixed_random
	}
	if = {
		limit = {
			has_global_flag = random_player
		}
		remove_global_flag = random_player
	}
	if = {
		limit = {
			has_global_flag = player_choice
		}
		remove_global_flag = player_choice
	}
	if = {
		limit = {
			has_global_flag = spawn_all
		}
		remove_global_flag = spawn_all
	}
}
