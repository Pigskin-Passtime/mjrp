### Request cooldown depending on current mood [empire scope]
caretaker_set_request_cooldown = {
	if = {
		limit = {
			caretaker_mood_is_pleased_check = yes
		}
		set_timed_country_flag = {
			flag = caretaker_resource_request_cooldown
			days = 180
		}
	}
	else_if = {
		limit = {
			caretaker_mood_is_content_check = yes
		}
		set_timed_country_flag = {
			flag = caretaker_resource_request_cooldown
			days = 360
		}
	}
	else_if = {
		limit = {
			caretaker_mood_is_annoyed_check = yes
		}
		set_timed_country_flag = {
			flag = caretaker_resource_request_cooldown
			days = 720
		}
	}
	else_if = {
		limit = {
			caretaker_mood_is_angry_check = yes
		}
		set_timed_country_flag = {
			flag = caretaker_resource_request_cooldown
			days = 1800
		}
	}
	else = {
		set_timed_country_flag = {
			flag = caretaker_resource_request_cooldown
			days = 720
		}
	}
}

# Given a species, create a mod with trait_personal_caretaker and apply to all pops, leaders
# and armies that have the original species and the same owner as the scope object.
# Scope: Something that has a species and an owner
create_pci_species_mod = {
	species = {
		create_species = {
			is_mod = yes
			name = this
			plural = this
			class = this
			portrait = this
			traits = this
			homeworld = this
			traits = {
				ideal_planet_class = this
				trait = trait_personal_caretaker
			}
		}
		save_event_target_as = changing_species
	}
	owner = {
		every_owned_pop_group = {
			limit = {
				is_exact_same_species = event_target:changing_species
			}
			change_species = last_created_species
		}
		every_owned_leader = {
			limit = {
				is_exact_same_species = event_target:changing_species
			}
			change_species = last_created_species
		}
		every_pool_leader = {
			limit = {
				is_exact_same_species = event_target:changing_species
			}
			change_species = last_created_species
		}
		every_owned_army = {
			limit = {
				exists = species
				is_exact_same_species = event_target:changing_species
			}
			change_species = last_created_species
		}
		every_owned_ship = {
			limit = {
				is_ship_class = shipclass_colonizer
				is_exact_same_species = event_target:changing_species
			}
			change_species = last_created_species
		}
	}
}

### setups initial population for seceded Caretaker [planet scope]
# CARETAKER_SPECIES = species of Caretaker drones
# PARENT_EMPIRE_SPECIES = species of the empire Caretaker secedes from
# <!!!> this effect should run before 'caretaker_seceded_empire_setup_capital' to avoid ruined buildins
caretaker_seceded_empire_setup_initial_population = {
	# kill all robots - we'll recreate them as $CARETAKER_SPECIES$ drones
	every_owned_pop_group = {
		limit = {
			OR = {
				has_trait = trait_machine_unit
				has_trait = trait_mechanical
			}
		}
		kill_all_pop = yes
	}
	# remove extra biological pops
	while = {
		limit = {
			pop_amount > 400
		}
		kill_single_pop = yes
	}
	# if we don't have enough bio pops, add them
	while = {
		limit = {
			pop_amount < 400
		}
		create_pop = {
			species = $PARENT_EMPIRE_SPECIES$
		}
	}
	# add RS drones
	while = {
		count = 70
		create_pop = {
			species = $CARETAKER_SPECIES$
		}
	}
}

### removes caretaker deposits/modifiers and 'takes away' some pops [planet scope]
caretaker_secession_planets_clean_up = {
	if = {
		limit = {
			caretaker_presence_on_planet = yes
		}
		caretaker_core_clean_up = yes
		caretaker_hub_clean_up = yes
		if = {
			limit = {
				pop_amount > 1000
			}
			while = {
				count = 5
				kill_single_pop = yes
			}
		}
		else_if = {
			limit = {
				pop_amount > 800
			}
			while = {
				count = 4
				kill_single_pop = yes
			}
		}
		else_if = {
			limit = {
				pop_amount > 600
			}
			while = {
				count = 3
				kill_single_pop = yes
			}
		}
		else_if = {
			limit = {
				pop_amount > 400
			}
			while = {
				count = 2
				kill_single_pop = yes
			}
		}
		else = {
			kill_single_pop = yes
		}
	}
}

caretaker_change_mood = {
	hidden_effect = {
		if = {
			limit = {
				any_situation = {
					is_situation_type = caretaker_mood_situation
				}
			}
			random_situation = {
				limit = {
					is_situation_type = caretaker_mood_situation
				}
				set_variable = {
					which = caretaker_mood_situation_value
					value = $VALUE$
				}
				# if negative, continue
				if = {
					limit = {
						check_variable_arithmetic = {
							which = caretaker_mood_situation_value
							value < 0
						}
					}
					add_situation_progress = $VALUE$
				}
				# if would increase by more than 120
				# decrease it to excatly 120
				# else add
				else = {
					export_trigger_value_to_variable = {
						trigger = situation_progress
						variable = situation_progress
					}
					set_variable = {
						which = situation_progress_future
						value = caretaker_mood_situation_value
					}
					change_variable = {
						which = situation_progress_future
						value = situation_progress
					}
					if = {
						limit = {
							check_variable_arithmetic = {
								which = situation_progress_future
								value > 120
							}
						}
						change_variable = {
							which = situation_progress_future
							value = -120
						}
						multiply_variable = {
							value = -1
							which = situation_progress_future
						}
						change_variable = {
							which = caretaker_mood_situation_value
							value = situation_progress_future
						}
						add_situation_progress = caretaker_mood_situation_value
					}
					else = {
						add_situation_progress = $VALUE$
					}
					clear_variable = situation_progress_future
					clear_variable = situation_progress
				}
			}
		}
	}
	# thsi wont be synced but looks better than situation stuff
	add_resource = {
		caretaker_mood = $VALUE$
	}
}
