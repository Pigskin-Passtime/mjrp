pdivinity_cult_star_flags = {
    event_target:pdivinity_closest_system = {
        set_star_flag = pdivinity_cult_star
        if = {
            limit = {
                any_neighbor_system = {
                    exists = starbase
                    starbase.owner = {
                        is_same_value = event_target:pdivinity_cult_defender
                    }
                    NOT = { has_star_flag = pdivinity_cult_star }
                }
            }
            random_neighbor_system = {
                limit = {
                    exists = starbase
                    starbase.owner = {
                        is_same_value = event_target:pdivinity_cult_defender
                    }
                    NOT = { has_star_flag = pdivinity_cult_star }
                }
                save_event_target_as = pdivinity_closest_system
            }
        }
        else = {
            closest_system = {
                limit = {
                    exists = starbase
                    starbase.owner = {
                        is_same_value = event_target:pdivinity_cult_defender
                    }
                    NOT = { has_star_flag = pdivinity_cult_star }
                }
                save_event_target_as = pdivinity_closest_system
            }
        }
    }
}

pdivinity_shroud_home_system = {
    set_star_flag = pdivinity_star_shrouded
    set_star_class = sc_pdivinity_shrouded_hole
    every_system_planet = {
        limit = { is_star = yes }
        create_ambient_object = {
            type = "pdivinity_shrouding_star"
            play_animation_once = yes
            location = this
        }
        last_created_ambient_object = {
            set_location = {
                target = prev
                distance = 0
                angle = random
            }
        }
        change_pc = pc_pdivinity_shrouded_hole
    }
    every_system_planet = {
        limit = {
            is_star = no
            is_asteroid = no
            NOR = {
                is_planet_class = pc_shattered
                is_planet_class = pc_shattered_2
                is_planet_class = pc_shielded
                is_planet_class = pc_ringworld_habitable_damaged
                is_planet_class = pc_ringworld_tech_damaged
                is_planet_class = pc_ringworld_seam_damaged
                is_colony = yes
            }
        }
        if = {
            limit = { is_colony = no }
            if = {
                limit = { is_planet_class = pc_habitat }
                spawn_habitat_cracker_effect = yes
            }
            else_if = {
                limit = { is_ringworld = yes }
                spawn_ringworld_cracker_effect = yes
                change_pc = pc_ringworld_habitable_damaged
            }
            else = {
                change_pc = pc_shrouded
            }
        }
        else_if = {
            limit = {
                is_ringworld = yes
                is_colony = no
            }
            spawn_ringworld_cracker_effect = yes
            change_pc = pc_ringworld_habitable_damaged
        }
        else_if = {
            limit = {
                is_planet_class = pc_ringworld_tech
                is_colony = no
            }
            spawn_ringworld_cracker_effect = yes
            change_pc = pc_ringworld_tech_damaged
        }
        else_if = {
            limit = {
                is_planet_class = pc_ringworld_seam
                is_colony = no
            }
            spawn_ringworld_cracker_effect = yes
            change_pc = pc_ringworld_seam_damaged
        }
        else = {
            change_pc = pc_shrouded
        }
    }
    star = {
        create_ambient_object = {
            type = "pdivinity_shrouded_system"
            location = this
        }
        last_created_ambient_object = {
            set_ambient_object_flag = pdivinity_shrouded_system_effect
            set_location = {
                target = prev
                distance = 0
                angle = random
            }
        }
    }
    star = {
        create_ambient_object = {
            type = "space_storm_1"
            location = this
        }
        last_created_ambient_object = {
            set_ambient_object_flag = pdivinity_storm
            set_location = {
                target = prev
                distance = 0
                angle = random
            }
        }
    }
    if = {
        limit = {
            has_cosmic_storms_dlc = yes
        }
        star = {
            create_ambient_object = {
                type = "shroud_storm"
                location = this
            }
            last_created_ambient_object = {
                set_ambient_object_flag = pdivinity_shroud_storm
                set_location = {
                    target = prev
                    distance = 0
                    angle = random
                }
            }
        }
    }
}

pdivinity_shroud_star_system = {
    if = {
        limit = { exists = event_target:pdivinity_crisis_empire }
        fire_on_action = {
            on_action = on_destroy_star_system
            scopes = { from = event_target:pdivinity_crisis_empire }
        }
    }
    set_star_flag = pdivinity_star_shrouded
    set_star_class = sc_pdivinity_shrouded_hole
    every_system_ambient_object = {
        limit = {
            # Preserve Unbidden crisis effects
            NOR = {
                has_ambient_object_flag = extradimensional_system_effect
                has_ambient_object_flag = extradimensional_system_effect_2
            }
        }
        destroy_ambient_object = this
    }
    every_system_planet = {
        limit = { is_star = yes }
        create_ambient_object = {
            type = "pdivinity_shrouding_star"
            play_animation_once = yes
            location = this
        }
        last_created_ambient_object = {
            set_location = {
                target = prev
                distance = 0
                angle = random
            }
        }
        change_pc = pc_pdivinity_shrouded_hole
    }
    # Handle Contingency Machine Worlds
    every_system_planet = {
        limit = {
            is_planet_class = pc_ai
            NOT = { has_planet_flag = machine_lair }
        }
        destroy_machine_world = yes
    }
    random_system_planet = {
        limit = {
            is_planet_class = pc_ai
            has_planet_flag = machine_lair
        }
        from.owner = { save_event_target_as = final_machine_world_destroyer }
        stop_crisis_sound = yes
        planet_event = { id = crisis.2046 }
    }
    every_system_planet = {
        limit = {
            is_star = no
            is_asteroid = no
            NOR = {
                is_planet_class = pc_shattered
                is_planet_class = pc_shattered_2
                is_planet_class = pc_shielded
                is_planet_class = pc_ringworld_habitable_damaged
                is_planet_class = pc_ringworld_tech_damaged
                is_planet_class = pc_ringworld_seam_damaged
            }
        }
        if = {
            limit = { is_planet_class = pc_habitat } # uncolonized
            spawn_habitat_cracker_effect = yes
        }
        if = {
            limit = { is_colony = yes }
            if = {
                limit = {
                    exists = orbital_defence
                }
                destroy_fleet = orbital_defence
            }
            remove_all_buildings = yes
            destroy_colony = yes
            if = {
                limit = { is_planet_class = pc_habitat }
                spawn_habitat_cracker_effect = yes
            }
            else_if = {
                limit = { is_ringworld = yes }
                spawn_ringworld_cracker_effect = yes
                change_pc = pc_ringworld_habitable_damaged
            }
            else = {
                change_pc = pc_shrouded
            }
        }
        else_if = {
            limit = { is_ringworld = yes }
            spawn_ringworld_cracker_effect = yes
            change_pc = pc_ringworld_habitable_damaged
        }
        else_if = {
            limit = { is_planet_class = pc_ringworld_tech }
            spawn_ringworld_cracker_effect = yes
            change_pc = pc_ringworld_tech_damaged
        }
        else_if = {
            limit = { is_planet_class = pc_ringworld_seam }
            spawn_ringworld_cracker_effect = yes
            change_pc = pc_ringworld_seam_damaged
        }
        else = {
            change_pc = pc_shrouded
        }
    }
    every_system_planet = {
        limit = {
            exists = orbital_station
        }
        orbital_station = { dismantle = yes }
    }
    every_system_planet = {
        clear_deposits = yes
        clear_planet_modifiers = yes
        prevent_anomaly = yes
    }
    every_system_planet = {
        limit = { is_asteroid = yes }
        clear_deposits = yes
    }
    every_fleet_in_system = {
        limit = {
            NOR = {
                is_same_value = from
                # except Extradimensionals
                AND = {
                    exists = owner
                    owner = {
                        OR = {
                            is_country_type = portal_holder
                            is_country_type = extradimensional
                            is_country_type = extradimensional_2
                            is_country_type = extradimensional_3
                        }
                    }
                }
                # except biggest monsters
                any_owned_ship = {
                    OR = {
                        is_ship_size = behemoth_04
                        is_ship_size = voidspawn_boss
                    }
                }
            }
        }
        ### nothing happens to yours
        if = {
            limit = {
                exists = event_target:pdivinity_crisis_empire #i.e. not final destruction of all matter
                is_mobile = yes
            }
        }
        else = {
            destroy_fleet = this
        }
    }
    every_system_megastructure = {
        set_ruined_megastructure = yes
    }
    star = {
        create_ambient_object = {
            type = "pdivinity_shrouded_system"
            location = this
        }
        last_created_ambient_object = {
            set_ambient_object_flag = pdivinity_shrouded_system_effect
            set_location = {
                target = prev
                distance = 0
                angle = random
            }
        }
    }
    star = {
        create_ambient_object = {
            type = "pdivinity_extradimensional_portal"
            location = this
        }
        last_created_ambient_object = {
            set_ambient_object_flag = pdivinity_extradimensional_portal_effect
            set_location = {
                target = prev
                distance = 0
                angle = random
            }
        }
    }
    star = {
        create_ambient_object = {
            type = "pdivinity_psi_storm_2"
            location = this
        }
        last_created_ambient_object = {
            set_ambient_object_flag = pdivinity_storm
            set_location = {
                target = prev
                distance = 0
                angle = random
            }
        }
    }
    #if = {

    #           }
    #       }
    #   }
    #}
    create_fleet = {
        name = "NAME_pdivinity_dimensional_portal"
        settings = {
            is_boss = yes
            uses_naval_capacity = no
            spawn_debris = no
            can_change_leader = no
            can_change_composition = no

        effect = {

pdivinity_add_shroud_storm_array = {
    fleet = {
        every_controlled_ship = {
            limit = {
                is_ship_class = shipclass_starbase
                NOT = { has_component = "PDIVINITY_SHROUD_STORM_COMPONENT" }
            }
            starbase = {
                add_starbase_component = {
                    component = "PDIVINITY_SHROUD_STORM_COMPONENT"
                }
            }
        }
    }
}
