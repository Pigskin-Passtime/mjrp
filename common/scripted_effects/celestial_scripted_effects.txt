spawn_celestial = {
	if = {
		limit = {
			NOT = { has_global_flag = celestial_system_spawned }
		}
		set_global_flag = celestial_system_spawned

			initializer = celestial_wayfarer_start_system
		}
		last_created_system = {
			log = "spawned [This.GetName]"
		}
	}
}

create_celestial_wayfarer = {
	create_country = {
		name = "NAME_Unknown"
		type = paragon_event_country
		auto_delete = yes
		flag = {
			icon = {
				category = "special"
				file = "unknown.dds"
			}
			background = {
				category = "backgrounds"
				file = "00_solid.dds"
			}
			colors = {
				"black"
				"black"
				"null"
				"null"
			}
		}
		effect = {
			save_global_event_target_as = celestial_country
		}
	}
	event_target:celestial_country = {
		# Setup
		set_country_flag = celestial_country_flag
		set_faction_hostility = {
			target = root.owner
			set_hostile = no
			set_neutral = yes
			set_friendly = no
		}
		
		# Species
		create_species = {
			name = "NAME_Unknown"
			class = REP
			portrait = celestial_leader # Celestial Wayfarer Portrait
			namelist = "PRT1"
			traits = {
				trait = trait_quick_learners
				trait = trait_talented
			}
			effect = {
				save_event_target_as = celestial_species
			}
		}
	}
	# Leader
	if = { # Astral Planes installed
		limit = { has_astral_planes_dlc = yes }
		event_target:celestial_country = {
				create_leader = {
					class = scientist
					species = event_target:celestial_species
					name = "NAME_Wayfarer"
					skill = 7
					randomize_traits = no
					gender = indeterminable
					event_leader = yes
					skip_background_generation = yes
					background_ethic = ethic_materialist
					custom_catch_phrase = celestial_catch_phrase
					custom_description = desc_celestial
					set_age = 210
					immortal = yes
					traits = {
						trait = leader_trait_frontier_adventurer
						trait = leader_trait_celestial_vagabond_astralplanes
						trait = leader_trait_celestial_eccentric
						trait = leader_trait_academic_astral_rift_expert_2
						trait = leader_trait_salvager
						trait = trait_ruler_charismatic_2
						trait = subclass_scientist_explorer
					}
					effect = {
						set_leader_flag = legendary_leader
						set_leader_flag = celestial
						set_leader_flag = immune_to_negative_traits # todo: check if it should be removed
						save_global_event_target_as = celestial_leader
						set_owner = root
						exile_leader_as = celestial_leader
					}
				}
		}
	}
	else = { # Astral Planes not installed
		event_target:celestial_country = {
				create_leader = {
					class = scientist
					species = event_target:celestial_species
					name = "NAME_Wayfarer"
					skill = 7
					randomize_traits = no
					gender = indeterminable
					event_leader = yes
					skip_background_generation = yes
					background_ethic = ethic_materialist
					custom_catch_phrase = celestial_catch_phrase
					custom_description = desc_celestial
					set_age = 210
					immortal = yes
					traits = {
						trait = leader_trait_frontier_adventurer
						trait = leader_trait_celestial_vagabond
						trait = leader_trait_celestial_eccentric
						trait = leader_trait_ancient_knowledge
						trait = leader_trait_salvager
						trait = trait_ruler_charismatic_2
						trait = subclass_scientist_explorer
					}
					effect = {
						set_leader_flag = legendary_leader
						set_leader_flag = celestial
						set_leader_flag = immune_to_negative_traits # todo: check if it should be removed
						save_global_event_target_as = celestial_leader
						set_owner = root
						exile_leader_as = celestial_leader
					}
				}
		}
	}
	create_fleet = {
					name = "NAME_TheWayfarer"
					effect = {
						set_owner = event_target:celestial_country
						create_ship = {
							name = "NAME_NomadsPillage"
							design = "NAME_NomadsShip"
							graphical_culture = "extra_dimensional_01"
							prefix = no
							upgradable = no
						}
						last_created_ship = {
							set_ship_flag = iss_wayfarer
							save_global_event_target_as = iss_wayfarer
							set_disable_at_health = 0.05
						}
						set_location = {
							target = event_target:wayfarer_target_planet # Check
							distance = 10
							angle = random
						}
						set_fleet_stance = passive
						# assign_leader = event_target:celestial_leader
						save_global_event_target_as = celestial_fleet
					}
					settings = {
						can_upgrade = yes
						can_change_composition = no
						can_change_leader = no
						spawn_debris = no
					}
	}
}

			set_planet_flag = celestial_wayfarer_portal

			create_archaeological_site = site_$CELESTIAL_MISSION_TYPE$

				type = "psi_storm_1"
				location = this
				entity_offset_height = { min = -15 max = -15 }
				entity_scale_to_size = yes
				scale = 0.1
			}
			last_created_ambient_object = {	
				set_ambient_object_flag = celestial_temp_vortex_gfx
				set_location = {
					target = prev
					distance = 0
					angle = random
				}
			}
		}
	}
}

clear_celestial_mission_vortex = { 	# Fire on system scope
	every_system_planet = {
		limit = {
			has_planet_flag = celestial_wayfarer_portal
		}
		remove_planet = yes
	}
	every_system_ambient_object = {
		limit = {
			has_ambient_object_flag = celestial_temp_vortex_gfx
		}
		destroy_ambient_object = this
	}
}

# Spectral Beacon
#	- Allows 1 more use of the [todo] decision
#	- The decision adds a special modifier to the planet
#	- Upon finishing it, 3-5 special Extradimensional Pops are spawned with powerful traits
#	- But other Pops can randomly gain the Not of This World/Portal Sickness traits
celestial_beacon_effect = {
	change_variable = {
		which = celestial_beacon_decision_limit
		value = 1
	}
