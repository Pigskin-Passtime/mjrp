# Util
eos_dispatch_alignment_event = {
    event_target:eos_leader = {
        switch = {
            trigger = has_ethic
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = militarist
            }
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = pacifist
            }
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = xenophile
            }
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = xenophobe
            }
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = authoritarian
            }
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = egalitarian
            }
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = spiritualist
            }
            inline_script = {
                script = eos_dispatch_alignment_events
                alignment = $alignment$
                ethic = materialist
            }
        }
    }
}
