create_reinhardist_rebel_country = {
	create_country = {
		name = random
		type = default
		species = event_target:dmj_original_species
		ignore_initial_colony_error = yes
		day_zero_contact = no
		ethos = { 
			ethic = ethic_fanatic_authoritarian
			ethic = ethic_spiritualist
		}
		authority = auth_imperial
		civics = {
			civic = civic_imperial_cult
			civic = civic_dystopian_society
			civic = civic_crusader_spirit
			civic = civic_psionic_sovereign
		}
		flag = random
		origin = origin_separatists
		effect = {
			set_male_ruler_title = RT_GOD_EMPEROR
			set_male_heir_title = HT_SAINT
			set_female_ruler_title = RT_GOD_EMPRESS
			set_female_heir_title = HT_SAINT_FEMALE
			save_global_event_target_as = dmj_reinhardist_uprising
			set_country_flag = dmj_reinhardist_empire
		}
	}
}
