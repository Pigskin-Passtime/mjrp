spawn_riftgate_starter = {
	random_system = {
		save_event_target_as = riftgate_spawn_system
	}
	event_target:riftgate_spawn_system = {
		if = {
			limit = {
				has_astral_planes_dlc = yes
			}
			spawn_system = {
				min_distance = 10
				max_distance = 30
				max_jumps = 0
				initializer = "riftgate_starting_system"
			}
			if = {
				limit = { last_created_system = { NOT = { has_hyperlane_to = prev } } }
					add_hyperlane = {
					from = this
					to = last_created_system
				}
			}
			else = {
				add_extra_hyperlane_to_spawned_system_effect = yes
			}
		}
		else = {
			spawn_system = {
				min_distance = 10
				max_distance = 30
				max_jumps = 0
				initializer = "riftgate_starting_system_nodlc"
			}
			if = {
				limit = { last_created_system = { NOT = { has_hyperlane_to = prev } } }
					add_hyperlane = {
					from = this
					to = last_created_system
				}
			}
			else = {
				add_extra_hyperlane_to_spawned_system_effect = yes
			}
		}
