decision_chronotec_chronoacceleration = {
	owned_planets_only = yes
	sound = event_mystic_reveal
	icon = decision_chronotec_chronoacceleration

	enactment_time = 1

	potential = {
		exists = leader
		leader = {
			chronotec_has_normal_infusion = yes	
		}
		hidden_trigger = {
			has_global_flag = chronotec_advanced
		}
	}

	allow = {
		custom_tooltip = {
			fail_text = decision_chronotec_cd
			hidden_trigger = {	# Lets avoid cheesing the leader
				exists = owner
				owner = {
					NOT = {
						has_country_flag = chronotec_chronoacceleration_locked
					}
				}
			}
		}
	}

	resources = {
		category = decisions
		cost = {
			unity = 1000
		}
	}

	effect = {
		# Normal effects, normal backlash
		custom_tooltip = decision_chronotec_chronoacceleration_tooltip
		add_modifier = {
			modifier = pm_chronotec_chronoacceleration
			years = 5
		}
		hidden_effect = {
			
			planet_event = {	# Triggers the negative
				id = chronotec_leader.50001
				days = @chronotec_charge_timer
			}
			owner = {
				set_saved_date = {	# Decision timer
					key = chronotec_decision_timer
					days_from_present = @chronotec_decision_cd_timer
					expires = @chronotec_decision_cd_timer
				}
				set_timed_country_flag = {
					flag = chronotec_chronoacceleration_locked
					years = 10
				}
			}
			# Reduce 1 charge to the leader
			event_target:chronotec_chronolord_leader = {
				chronotec_decrease_charges_effect = yes
			}
		}
	}

	ai_weight = {
		weight = 5
	}
}
