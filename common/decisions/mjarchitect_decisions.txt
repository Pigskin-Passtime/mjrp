
decision_mjarchitect_sacrifice = {
	owned_planets_only = yes
	icon = decision_mjarchitect_sacrifice

	enactment_time = 30
	resources = {
		category = decisions
		cost = {
			unity = 20
			mult = value:mjarchitect_local_pops_sacrifice_cost_mult
		}
	}
	
	potential = {
		exists = owner
		owner = {
			has_crisis_perk = mjarchitect_crisis_perk_2
		}
	}
	allow = {
		custom_tooltip = {
			fail_text = decision_mjarchitect_sacrifice_req
			pop_amount > 500
		}
	}
	
	effect = {
		hidden_effect = {
			# Count number of Pops pre-sacrifice
			set_variable = {
				which = mjarchitect_local_pops_before_killing
				value = trigger:pop_amount
			}
			# Kill Pops
			every_owned_pop_job = {
				kill_assigned_pop_amount = {
					percentage = 0.10
				}
			}
			# Count number of Pops post-sacrifice
			set_variable = {
				which = mjarchitect_local_pops_after_killing
				value = trigger:pop_amount
			}
			# Find number of Pops killed, provide CM accordingly
			set_variable = {
				which = mjarchitect_num_pops_sacrificed
				value = mjarchitect_local_pops_before_killing
			}
			subtract_variable = {
				which = mjarchitect_num_pops_sacrificed
				value = mjarchitect_local_pops_after_killing
			}
			# divide_variable = {
			# 	which = mjarchitect_num_pops_sacrificed
			# 	value = 100
			# }

			owner = {
				change_variable = {
					which = mjarchitect_num_sacrificed_new
					value = root.mjarchitect_num_pops_sacrificed
				}
			}
		}
		owner = {
			mjarchitect_resource_reward_effect = {
				SATISFACTION_VALUE = 0.05
				SATISFACTION_MULT = root.mjarchitect_num_pops_sacrificed
				SATISFACTION_CAP = 10
				COGNITIVE_MATTER_VALUE = 0.5
				COGNITIVE_MATTER_MULT = root.jarchitect_num_pops_sacrificed
			}
			hidden_effect = {
				set_timed_country_flag = {
					flag = mjarchitect_recently_sacrificed
					years = 5
				}
			}
		}

		if = {
			limit = {
				has_modifier = mjarchitect_planet_sacrifices_penalties
			}
			custom_tooltip = decision_mjarchitect_sacrifice_tt_successive
		}
		else = {
			custom_tooltip = decision_mjarchitect_sacrifice_tt_initial
		}
		
		hidden_effect = {
			change_variable = {
				which = mjarchitect_recent_sacrifices_local
				value = 1
			}
			mjarchitect_update_sacrifical_unhappiness = {
				EFFECT_MULT = mjarchitect_recent_sacrifices_local
				LENGTH_MULT = value:mjarchitect_sacrifice_penalty_length_mult
			}

			if = {
				limit = {
					NOT = {
						has_planet_flag = mjarchitect_sacrifice_tally_loop_ongoing
					}
				}
				planet_event = {
					id = mjarchitect.2
					days = 1800
				}
				set_planet_flag = mjarchitect_sacrifice_tally_loop_ongoing
			}
		}
	}
	
	ai_weight = {
		weight = 0
	}
}

decision_mjarchitect_veil_world = {
	# owned_planets_only = yes
	icon = decision_mjarchitect_veil_world

	enactment_time = 180
	resources = {
		category = decisions
		cost = {
			trigger = {
				NOT = {
					owner = { has_country_flag = mjarchitect_blessing_6_unlocked }
				}
			}
			influence = 75
		}
		cost = {
			trigger = {
				owner = { has_country_flag = mjarchitect_blessing_6_unlocked }
			}
			influence = 25
		}
	}
	
	potential = {
		exists = controller
		controller = {
			has_crisis_perk = mjarchitect_crisis_perk_8
		}
	}
	allow = {
		NOT = {
			is_planet_class = pc_mjarchitect_veiled
		}
	}
	
	effect = {
		hidden_effect = {
		# Find number of Pops changed
			set_variable = {
				which = mjarchitect_num_pops_affected
				value = trigger:pop_amount
			}
			divide_variable = {
				which = mjarchitect_num_pops_affected
				value = 2
			}
			# divide_variable = {
			# 	which = mjarchitect_num_pops_affected
			# 	value = 100
			# }
		}
		# Satisfaction reward
		owner = {
			hidden_effect = {
				set_timed_country_flag = {
					flag = mjarchitect_recently_formed_hollow_world
					years = 5
				}
				change_variable = {
					which = mjarchitect_num_veiled_worlds_new
					value = 1
				}
				complete_crisis_objective = mjarchitect_crisobj_markofruin
			}
			mjarchitect_satisfaction_reward_effect = {
				SATISFACTION_VALUE = 0.1
				SATISFACTION_MULT = root.mjarchitect_num_pops_affected
				SATISFACTION_CAP = 30
			}
		}
		custom_tooltip = decision_mjarchitect_veil_world_tt
		hidden_effect = {
			if = {
				limit  = {
					is_artificial = no
				}
				change_pc = pc_mjarchitect_veiled
			}
			else = {}
			# Convert ~50% of the Pops to have the Veiled trait
			mjarchitect_convert_percentage_pops = {
				PERCENT = 0.50
			}
		}
	}
	
	ai_weight = {
		weight = 0
	}
}
