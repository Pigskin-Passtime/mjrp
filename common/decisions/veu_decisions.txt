# recycling stuff
veu_decision_recycle_colony = {
	icon = decision_burning_settlement
	sound = event_structural_collapse

	resources = {
		category = decisions
	}
	owned_planets_only = yes

	potential = {
		owner = {
			has_ascension_perk = veu_ap_veu
		}
		is_capital = no
		NOT = {
			has_planet_flag = being_recycled
		}
		NOT = {
			has_planet_flag = veu_yard_flag
		}
		NOT = { has_designation = col_nanotech_world }
	}

	allow = {
		owner = {
		has_menace_perk = menp_veu_features_4
		}
	}

	effect = {
		custom_tooltip = veu_decision_recycle_colony_effect_desc
		set_timed_planet_flag = {
			days = 100
			flag = recently_recycled_planet
		}
		hidden_effect = {
			set_planet_flag = colony_event
			set_planet_flag = being_recycled
			add_modifier = {
				modifier = being_recycled_modifier
				days = -1
			}
			export_trigger_value_to_variable = {
				trigger = planet_size
				variable = num_districts_recycling
			}
			export_trigger_value_to_variable = {
				trigger = count_deposits
				parameters = {
					type = d_veu_recycling_devastation
				}
				variable = num_recycling_blockers
			}
			if = {
				limit = {
					is_variable_set = num_recycling_blockers
				}
				subtract_variable = {
					which = num_districts_recycling
					value = num_recycling_blockers
				}
			}
		}
		owner = {
			start_situation = {
				type = veu_recycle_planet_situation
				target = root
			}
		}
	}

	ai_weight = 0
}
