# Bloom Propagation
# +50% Food Production and +0.5 Mycogrowth per Farmer
wiirlak_ep_decision_planet_boost = {
	owned_planets_only = yes
	sound = event_alien_nature
	icon = wilderness_planet_expand

	resources = {
		category = decisions
		inline_script = {
			script = "wiirlak_ep_food_cost"
			AMOUNT = 1000
			MULT = 1
			MYCO = 100
		}
	}

	potential = {
		owner = { has_menace_perk = wiirlak_ep_menap_food_2 }
		owner = { NOT = { has_menace_perk = wiirlak_ep_menap_food_3 } }
		is_planet_class = wiirlak_ep_pc_bloom
	}

	allow = {
		NOT = { has_modifier = wiirlak_ep_modifier_decision_planet_boost }
	}

	effect = {
		add_modifier = {
			modifier = wiirlak_ep_modifier_decision_planet_boost
			days = 3600
		}

		# Failsafe
		wiirlak_ep_spawn_deposit = yes
	}

	ai_weight = {
		weight = 50

		# don't encourage growth if there is no free housing
		modifier = {
			factor = 0
			free_housing <= 0
		}
	}
}


# Bloom Propagation
# +50% Food Production and +0.5 Mycogrowth per Farmer; Apply Sector-wide
# If not Bloom Worlds, will start Terraforming
wiirlak_ep_decision_planet_boost_sector = {
	owned_planets_only = yes
	sound = event_alien_nature
	icon = wilderness_planet_expand

	resources = {
		category = decisions
		inline_script = {
			script = "wiirlak_ep_food_cost"
			AMOUNT = 3000
			MULT = 1
			MYCO = 250
		}
	}

	potential = {
		owner = { has_menace_perk = wiirlak_ep_menap_food_3 }
		is_planet_class = wiirlak_ep_pc_bloom
	}

	allow = {
		NOT = { has_modifier = wiirlak_ep_modifier_decision_planet_boost }
		custom_tooltip = {
			fail_text = wiirlak_ep_decision_planet_boost_sector_sector_fail
			success_text = wiirlak_ep_decision_planet_boost_sector_sector_success
			is_in_frontier_space = no
		}
		custom_tooltip = {
			fail_text = wiirlak_ep_decision_planet_boost_sector_sector_leader_fail
			success_text = wiirlak_ep_decision_planet_boost_sector_sector_leader_success
			sector.sector_capital = {
				exists = leader
			}
		}
	}

	effect = {
		tooltip = {
			custom_tooltip = wiirlak_ep_decision_planet_boost_sector_tooltip
			add_modifier = {
				modifier = wiirlak_ep_modifier_decision_planet_boost
				days = 3600
			}
		}

		hidden_effect = {
			sector = {
				every_owned_planet = {
					if = {
						limit = { is_planet_class = wiirlak_ep_pc_bloom }

						hidden_effect = { remove_modifier = wiirlak_ep_modifier_decision_planet_boost }
						add_modifier = {
							modifier = wiirlak_ep_modifier_decision_planet_boost
							days = 3600
						}

						# Failsafe
						wiirlak_ep_spawn_deposit = yes
					}
					else_if = {
						limit = { is_terraforming = no }

						start_terraform_process = wiirlak_ep_pc_bloom
					}
				}
			}
		}
	}

	ai_weight = {
		weight = 5

		# don't encourage growth if there is no free housing
		modifier = {
			factor = 0
			free_housing <= 0
		}

		modifier = {
			factor = 100
			sector = {
				any_owned_planet = { NOT = { is_planet_class = wiirlak_ep_pc_bloom } }
			}
		}
	}
}
