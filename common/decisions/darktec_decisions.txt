# Taint world
darktec_decision_taint_world_menial = {
	owned_planets_only = yes
	sound = event_mystic_reveal_light
	icon = darktec_decision_taint_world_menial

	enactment_time = 180

	resources = {
		category = decisions
		cost = {
			influence = 15
		}
	}

	potential = {
		always = no	# DISCARDED
		exists = owner
		owner = {
			has_ascension_perk = darktec_ap_postbaryonic
			has_crisis_level = darktec_crisis_postbaryonic_level_4
			has_country_flag = darktec_entropic_taint
		}
		if = {
			limit = {
				NOT = {
					has_modifier = darktec_pm_tainted_worker
				}
			}
			NOR = {
				has_modifier = darktec_pm_tainted_worker
				has_modifier = darktec_pm_tainted_specialist
				has_modifier = darktec_pm_tainted_utility
				has_modifier = darktec_pm_tainted_pops
			}
		}
	}

	effect = {
		if = {
			limit = {
				hidden_trigger = {
					NOT = {
						has_modifier = darktec_pm_tainted_worker
					}
				}
			}
			custom_tooltip = darktec_decision_taint_world_menial_tooltip
			
			hidden_effect = {
				set_variable = {
					which = darktec_tainted_level
					value = 1
				}
			}
			add_modifier = {
				modifier = darktec_pm_tainted_worker
				mult = darktec_tainted_level
			}
		}
		else = {
			custom_tooltip = darktec_decision_taint_world_menial_repeated_tooltip
			hidden_effect = {
				change_variable = {
					which = darktec_tainted_level
					value = 1
				}
			}
			remove_modifier = darktec_pm_tainted_worker
			add_modifier = {
				modifier = darktec_pm_tainted_worker
				mult = darktec_tainted_level
			}
		}
		owner = {
			complete_crisis_objective = darktec_crisobj_immediate_taint_world
		}
	}

	ai_weight = {
		weight = 0
	}
}

		hidden_effect = {

					remove_modifier = darktec_pm_tainted_worker
				}
				darktec_pm_tainted_specialist = {
					remove_modifier = darktec_pm_tainted_specialist
				}
				darktec_pm_tainted_utility = {
					remove_modifier = darktec_pm_tainted_utility
				}
				darktec_pm_tainted_pops = {
					remove_modifier = darktec_pm_tainted_pops
				}
			}
			clear_variable = darktec_tainted_level
		}
		add_planet_devastation = 30
	}

	ai_weight = {
		weight = 0
	}
}
