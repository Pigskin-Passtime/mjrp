dmj_trigger = { # Hard requirements, if you don't fulfill these, you can't go through the recruitment event chain at all
	has_utopia = yes
	has_paragon_dlc = yes
	has_astral_planes_dlc = yes
	is_ai = no # Because I don't think they're smart enough to go through the event chain
	NOT = { has_country_flag = dmj_sol_country }
	NOT = { is_human_species = yes }
	NOT = {
		any_system = {
			has_star_flag = sol_system
		}
	}
	NOT = { has_country_flag = dmj_freed_reinhard }
}

dmj_soft_trigger = { # Soft requirements, if you don't fulfill these, you can still go through the recruitment event chain, but you will not have the option to recruit
	is_gestalt = no
	is_homicidal = no
}
