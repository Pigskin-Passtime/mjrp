				6 = { event_target:nsc_frebooter_empire_admiral_market_target = { has_resource = { type = energy amount >= 1200 } } }
				7 = { event_target:nsc_frebooter_empire_admiral_market_target = { has_resource = { type = energy amount >= 1400 } } }
				8 = { event_target:nsc_frebooter_empire_admiral_market_target = { has_resource = { type = energy amount >= 1600 } } }
				9 = { event_target:nsc_frebooter_empire_admiral_market_target = { has_resource = { type = energy amount >= 1800 } } }
				10 = { event_target:nsc_frebooter_empire_admiral_market_target = { has_resource = { type = energy amount >= 2000 } } }
			}
			#if the player uses some kinds of mod to increase max leader level...

				limit = { has_skill > 10 }
				event_target:nsc_frebooter_empire_admiral_market_target = { has_resource = { type = energy amount >= 2200 } }
