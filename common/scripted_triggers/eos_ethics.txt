eos_pop_alignment = {
    switch = {
        trigger = pop_has_ethic
        inline_script = {
            script = $inline_body$
            ethic = militarist
        }
        inline_script = {
            script = $inline_body$
            ethic = pacifist
        }
        inline_script = {
            script = $inline_body$
            ethic = xenophile
        }
        inline_script = {
            script = $inline_body$
            ethic = xenophobe
        }
        inline_script = {
            script = $inline_body$
            ethic = authoritarian
        }
        inline_script = {
            script = $inline_body$
            ethic = egalitarian
        }
        inline_script = {
            script = $inline_body$
            ethic = spiritualist
        }
        inline_script = {
            script = $inline_body$
            ethic = materialist
        }
        default = {
            log = "pop has fanatic ethic"
            always = yes
        }
    }
}

    eos_pop_alignment = {
        inline_body = eos_ethic_compatible
    }
}

        inline_body = eos_ethic_not_compatible
    }
}
