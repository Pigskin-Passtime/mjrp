	}

scluster_can_have_physics_deposits = {
	# Used by: nothing
	# Necessary because: 
	OR = {
		can_have_physics_deposits = yes # common/scripted_triggers/00_scripted_triggers.txt
		AND = {
			is_colonizable = no
			OR = {
				has_modifier = high_gravity
				has_modifier = low_gravity
				has_modifier = strong_magnetic_field
			}
			scluster_is_terraforming_candidate = no
		}
	}
}

scluster_is_terraforming_candidate = {
	# Used by: `scluster_init_final_pass` in [common\scripted_effects\scluster_init_effects.txt], `scluster_can_have_<physics/society/engineering>_deposits`, `scluster_planet_available_to_modify`

scluster_planet_available_to_modify = {
	# Used by: `scluster_init_final_pass`
	# Necessary because: 
	colonizable_planet = no
	scluster_is_terraforming_candidate = no
	has_anomaly = no
	is_preventing_anomaly = no
	NOR = {
		has_planet_flag = ignore_startup_effect
		has_planet_flag = starting_deposit
		# has_deposit_for = shipclass_mining_station
		# has_deposit_for = shipclass_research_station
		exists = archaeological_site
	}
}
