    is_ai = no
    has_ascension_perk = ap_become_the_crisis

    has_global_flag = TFO_allowed

    has_country_flag = TFO_ready_second_encounter
    NOT = {
        has_global_flag = TFO_leader_spawned
    }
}

        has_ethic = ethic_fanatic_xenophile
        has_ethic = ethic_pacifist
        has_ethic = ethic_fanatic_pacifist

    }
}

#Country scope
TFO_can_unlock_mega = {
    exists = event_target:TFO_leader
    event_target:TFO_leader = {
        is_owned_by = this
    }

    NOT = {
        any_archaeological_site = {
            exists = excavator_fleet
            excavator_fleet = {
                exists = leader
                leader = {
                    is_same_value = event_target:TFO_leader
                }
            }
        }
    }

    has_country_flag = crisis_special_project_1_complete
}
