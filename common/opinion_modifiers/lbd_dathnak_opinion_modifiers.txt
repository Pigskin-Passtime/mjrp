opinion_lbd_dathnak_baldarak = {
	trigger = {
		has_country_flag = lbd_dathnak_country
		
		from = {
			species = {
				is_species = "Baldarak"
			}
			NOT = {
				has_country_flag = lbd_dathnak_country
			}
		}
	}
	opinion = {
		base = 120
		modifier = {
			has_country_flag = lbd_dathnak_original_crisis_awoken
			add = -40
		}
	}
}
