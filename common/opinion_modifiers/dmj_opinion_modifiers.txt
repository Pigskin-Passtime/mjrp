opinion_reinhardist_uprising = {
	opinion = {
		base = -200
	}
}

opinion_reinhardist_uprising_originator = {
	opinion = {
		base = -200
	}
}

triggered_opinion_dmj_freed_reinhard = {
	trigger = {
		has_global_flag = dmj_system_spawned
		OR = {
			AND = {
				exists = event_target:dmj_sol_country
				is_same_value = event_target:dmj_sol_country
			}
			has_country_flag = humans_early_space_age
			has_country_flag = humans_machine_age
			has_country_flag = humans_late_medieval_age
		}
		species = {
			is_human_species = yes
		}
		NOT = { has_country_flag = dmj_freed_reinhard }
		FROM = { has_country_flag = dmj_freed_reinhard }
	}

	opinion = {
		base = -100
	}
}
