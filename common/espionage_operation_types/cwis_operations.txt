cwis_operation_culture_link_deployment = {
	target = none
	categories = { op_cat_subterfuge }
	picture = GFX_evt_spymaster
	desc = {
		text = cwis_operation_culture_link_deployment_desc
	}
	stages = 1

	resources = {
		category = operations
		cost = {
			influence = 25
		}
		upkeep = {
			energy = 30
			unity = 5
			information_seal = 1
		}
	}

	spy_power_cost = 2
	potential = {
		owner = { has_crisis_level = cwis_crisis_level_1 }
		target = {
			or = {
				is_country_type = default
				is_country_type = primitive
			}
			not = { has_country_flag = cwis_culture_link_deployed }
		}
	}
	allow = {
		custom_tooltip = {
			is_running_espionage_operation = no
			fail_text = operation_one_at_a_time
		}
	}
	stage = {
		difficulty = 0
		icon = GFX_espionage_chapter_icon_document
		event = cwis.5001
	}

	on_roll_failed = {
		cwis_espionage_operation_on_roll_failed = yes
	}
	on_create = {
		target = {
			country_event = {
				id=cwis.5000
			}
		}
	}
}
