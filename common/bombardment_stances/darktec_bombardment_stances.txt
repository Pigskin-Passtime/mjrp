darktec_bombardment_entropy = {
	trigger = {
		owner = {
			has_crisis_level = darktec_crisis_postbaryonic_level_5
		}
	}

	default = no

	stop_when_armies_dead = no
	stop_when_ground_combat = no
	abduct_pops = no

	planet_damage = 1.5
	army_damage = 2.0

	kill_pop_chance = {
		base = 1.0
	}
	min_pops_to_kill_pop = 0
	kill_pop_amount = { min = 0 max = 200 }

	# root = fleet
	# from = planet
	ai_weight = {
		weight = 10
		modifier = {
			factor = 0.01
			exists = from
			from = {
				owner = { NOT = { is_hostile = root.owner } }
				NAND = {
					controller = { is_crisis_faction = yes }
					planet_garrison_strength > 0
				}
			}
			exists = owner
			owner = { is_crisis_faction = no }
		}
	}
}
