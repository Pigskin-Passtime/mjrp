	power = 0
	
	scripted_action_button_possible = {
		event_target:celestial_leader.owner = {
			has_resource = { type = energy amount >= 5000 }
			NOT = { has_modifier = celestial_used_distortion }
		}
	}
	
	scripted_action_possible = {
		is_star = yes
		exists = event_target:celestial_leader

		}
	}
	
	scripted_action_icon = GFX_fleet_order_button_special_project
	scripted_action_name = on_celestial_target_planet_reached

}
