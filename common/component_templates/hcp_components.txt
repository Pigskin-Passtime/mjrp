utility_component_template = {
	key = "hcp_SHIP_STELLAR_HARVESTER_REPLICATE"
	size = large
	icon = GFX_starbase_shipyard
	icon_frame = 1
	power = 0

	potential = {
		is_ship_size = hcp_stellar_harvester
	}
	component_set = "hcp_SHIP_STELLAR_HARVESTER_REPLICATE"

	scripted_action = {
		scope = self

        possible = {
			always = yes
		}

		button_clickable = {
			custom_tooltip = {
				fail_text = hcp_stellar_harvester_retrofit_can_upgrade
				is_scope_set = owner
				owner = {
					OR = {
						root = {
							can_be_upgraded = yes
						}
						count_owned_design = {
							limit = {
								is_ship_size = hcp_stellar_harvester
							}
							count > 1
						}
					}
				}
			}
			custom_tooltip = {
				fail_text = hcp_stellar_harvester_retrofit_in_progress
				is_scope_set = owner
				owner = {
					NOT = {
						has_active_event = {
							hcp_harvesters_crisis.725
						}
					}
				}
			}
		}

		tooltip = hcp_stellar_harvester_retrofit_ship_button

		}

		}
