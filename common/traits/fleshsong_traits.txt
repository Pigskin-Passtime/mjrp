trait_fleshsong_example = {
	icon = "gfx/interface/icons/traits/trait_fleshsong_example.dds"
	cost = 1
	species_potential_add = {
		exists = from
		from = {
			has_technology = tech_sing_fleshsong
		}
	}
	species_possible_remove = {
		exists = from
		from = {
			OR = { 
				AND = { 
					has_technology = tech_sing_fleshsong 
					can_remove_beneficial_genetic_traits = yes
				}
				has_technology = tech_cure_fleshsong 
			}
		}
	}
	potential_crossbreeding_chance = 1.0	# 1.0 = 100% chance of being considered for new traits when forming half-species. does not guarantee the trait will be added if it costs points.
	allowed_archetypes = { BIOLOGICAL }
	modifier = {
		planet_jobs_food_produces_mult = 0.15
	}
	slave_cost = {
		energy = 500
	}
	assembly_score = {
		modifier = {
			add = 1.5
			from = { has_farming_designation = yes }
		}
		modifier = {
			add = 0.5
			from = { has_rural_designation = yes }
		}
	}
}
