soj_trait_module_military = {
	cost = 0
	inline_script = {
		script = trait/icon_psionic
		CLASS = leader
		ICON = "GFX_leader_trait_psionic_communion"
		RARITY = paragon
		COUNCIL = yes
		TIER = none
	}
    # TODO set modifiers
	planet_modifier = {
		planet_stability_add = 10
	}
	sector_modifier = {
		planet_stability_add = 5
	}
    triggered_councilor_modifier = {
		potential = {
			NOT = { has_leader_flag = legendary_leader_coma }
			is_variable_set = paragon_origin_story_ruling
		}
		planet_stability_add = 0.75
		planet_jobs_unity_produces_mult = 0.015
		mult = this.paragon_origin_story_ruling
	}
	triggered_self_modifier = {
		potential = {
			NOT = { has_leader_flag = legendary_leader_coma }
		}
		leader_lifespan_add = 40
	}
	triggered_self_modifier = {
		potential = {
			has_leader_flag = legendary_leader_coma
		}
		species_leader_exp_gain = -2.0
		leader_lifespan_add = 300
	}

    inline_script = {
		script = paragon/leader_resource_produces
		RESOURCE = food
		AMOUNT = 30
	}

	leader_class = { official }
	initial = no
	randomized = no
}
