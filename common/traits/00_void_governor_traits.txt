leader_trait_time_warper = {
	#negative = no
	inline_script = {
		script =  trait/icon
		CLASS = leader
		ICON = "GFX_leader_trait_intemporal"
		RARITY = paragon
		COUNCIL = no
		TIER = none
	}

	#custom_tooltip_with_modifiers = leader_trait_time_warper_tt ###NOT NEEDED LOOKS BAD
	modifier = {
		pop_growth_speed = 0.15
		#planet_pop_assembly_mult = 0.15 ###covered by alfrays code below
		#planet_pop_assembly_organic_mult = 0.15 ###covered by alfrays code below
		planet_jobs_produces_mult = 0.15
		planet_army_build_speed_mult = 0.15
		planet_building_build_speed_mult = 0.15
		planet_clear_blocker_time_mult = 0.15
		pop_demotion_time_mult = 0.15
		trade_value_mult = 0.15
		#
		planet_pops_upkeep_mult = 0.15
		planet_structures_upkeep_mult = 0.15
		planet_jobs_upkeep_mult = 0.15
		#planet_districts_upkeep_mult = 0.15 structures counts both
	}
	
	triggered_planet_modifier = {
   	 potential = {
				OR = {
					FROM={
   	     	   		NOT = { exists = assembling_species }
    	    		assembling_species = { is_robotic_species = yes }
    	    	}
			}
    	}
    	planet_pop_assembly_mult  = 0.15
	}

	triggered_planet_modifier = {
    potential = {
		FROM = {
   	    assembling_species = { is_robotic_species = no }
   	 }
	}
    planet_pop_assembly_organic_mult  = 0.15
}
	immortal_leaders = yes
	leader_class = { official }
	initial = no
	randomized = no
}
