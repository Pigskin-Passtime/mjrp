cwis_ap_culture_weapon = {
	potential = {
		OR = {
			has_machine_age_dlc = yes
			has_nemesis = yes
			has_biogenesis_dlc = yes
		}
		not = { has_country_flag = player_crisis_chosen }
	}

	possible = {
		custom_tooltip = {
			fail_text = "requires_ascension_perks_3"
			num_ascension_perks > 2
		}
		custom_tooltip = {
			fail_text = "requires_independence"
			is_subject = no
		}
		custom_tooltip = {
			fail_text = "must_not_be_custodian"
			is_galactic_custodian = no
		}
		custom_tooltip = {
			fail_text = "must_not_be_emperor"
			is_galactic_emperor = no
		}
		custom_tooltip = {
			fail_text = "must_have_enough_decryption"
			OR = {
				has_tradition = tr_subterfuge_adopt
				has_tech_option = tech_decryption_1
				check_modifier_value = {
					modifier = intel_decryption_add
					value > 1
				}
			}
		}
	}

	on_enabled = {
		activate_crisis_progression = culture_weapon_path
		hidden_effect = {
			set_country_flag = player_crisis_chosen
		}
	}

	ai_weight = {
		factor = 7.5
		modifier = {
			factor = 1.5
			has_ascension_perk = ap_technological_ascendancy
		}
		modifier = {
			factor = 1.5
			has_ascension_perk = ap_enigmatic_engineering
		}
		modifier = {
			factor = 0.25
			is_gestalt = yes
		}
		modifier = {
			factor = 0
			galaxy_percentage < 0.1
		}
		modifier = {
			factor = 2
			galaxy_percentage > 0.2
		}
	}
}
