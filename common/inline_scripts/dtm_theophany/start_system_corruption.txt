hidden_effect = {
	solar_system = {
		if = {
			limit = {
				has_star_flag = dtm_theophany_corruption_complete
			}
			# nothing
		}
		else_if = {
			limit = {
				has_star_flag = dtm_theophany_corruption_started
			}
			event_target:global_event_country = {
			#owner = {
				random_situation = {
					limit = {
						is_situation_type = dtm_theophany_corruption_system
						has_situation_flag = dtm_theophany_corruption_system@prev
					}
					add_situation_progress = $AMOUNT$
				}
			}
			dtm_theophany_increase_corruption_system = {
				AMOUNT = $AMOUNT$
			}
		}
		else = {
			event_target:global_event_country = {
			#owner = {
				start_situation = {
					type = dtm_theophany_corruption_system
					target = prev
					effect = {
						set_situation_flag = dtm_theophany_corruption_system@prev
						add_situation_progress = $AMOUNT$
					}
				}
			}
			dtm_theophany_increase_corruption_system = {
				AMOUNT = $AMOUNT$
			}
		}
	}
}
