# Handle the Cost - Auto switch Food and Minerals

# Food
cost = {
	trigger = {
		OWNER = {
			country_uses_food = yes
			has_technology = wiirlak_ep_tech_bloom_colonization_2
		}
	}

	food = $AMOUNT$
	wiirlak_ep_resource_mycogrowth = $MYCO$
	multiplier = 0.5
}
cost = {
	trigger = {
		OWNER = {
			country_uses_food = yes
			NOT = { has_technology = wiirlak_ep_tech_bloom_colonization_2 }
		}
	}

	food = $AMOUNT$
	wiirlak_ep_resource_mycogrowth = $MYCO$
	multiplier = 1
}
