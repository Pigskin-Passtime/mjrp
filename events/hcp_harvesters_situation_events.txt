situation_event = { #Every 6 months event harvesting a planet
	id = hcp_harvesters_situations.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		target = { NOT = { has_planet_flag = recently_eaten_planet } }
	}

	immediate = {
		target = {
			hcp_harvest_planet = yes
		}
	}
}

# Colony Harvested
situation_event = {
	id = hcp_harvesters_situations.2
	title = "hcp_harvesters_situations.2.name"
	desc = "hcp_harvesters_situations.2.desc"
	picture = GFX_evt_shattered_planet
	show_sound = event_super_explosion
	situation = this
	location = target

	is_triggered_only = yes
	trigger = {
		owner = {
			has_crisis_perk = hcp_menp_planetary_quarrying
		}
	}

	immediate = {
		target = {
			if = {
				limit = {
					has_modifier = holy_planet
				}
				planet_event = { id = planet_destruction.605 }
			}
			while = {
				limit = {
					planet_size > 5
				}
				hcp_harvest_planet = yes
			}
			every_owned_pop_group = {
				resettle_pop_group = {
					POP_GROUP = this
					PLANET = root.owner.capital_scope
					PERCENTAGE = 100%
				}
			}
			solar_system = {
				if = {
					limit = {
						any_system_megastructure = {
							owner = { is_same_value = root.owner }
							has_megastructure_flag = hcp_harvest_@prevprev
						}
					}
					prev = { remove_planet_flag = megastructure }
					every_system_megastructure = {
						limit = {
							has_megastructure_flag = hcp_harvest_@prevprev
						}
						remove_megastructure = this
					}
				}
			}
			remove_modifier = hcp_being_harvested_modifier
			change_pc = pc_asteroid
			clear_deposits = yes
			locked_random_list = {
				3 = { add_deposit = d_minerals_1 }
				1 = { add_deposit = d_alloys_1 }
			}
		}
		owner = { complete_crisis_objective = hcp_crisobj_harvest_planets }
	}

	option = {
		name = hcp_harvesters_situations.2.a
		custom_tooltip = hcp_harvesters_situations.2.a.tooltip
		hidden_effect = {
			destroy_situation = this
		}
	}
}

# Master and Dummy Situation Synchronization
situation_event = {
	id = hcp_harvesters_situations.10
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		target = {
			any_situation = {
				is_situation_type = hcp_situation_outward_bound_master
				NOT = { situation_progress = root.trigger:situation_progress }
			}
		}
	}

	immediate = {
		target = {
			random_situation = {
				limit = {
					is_situation_type = hcp_situation_outward_bound_master
				}
				root = {
					add_situation_progress = value:hcp_dummy_situation_progress_corrector
				}
			}
		}
	}
}

								has_star_flag = hcp_singularity_silo_@root.owner
							}
						}
					}
				}

		}
		hidden_effect = {

					id = hcp_harvesters_crisis.800

				}
			}
		}
	}
	option = {
		name = hcp_harvesters_situations.1005.b
		trigger = {
			owner = {
				any_owned_planet = {

						solar_system = {
							has_star_flag = hcp_singularity_silo_@root.owner
						}
					}
				}
			}
		}
	}

	after = {
		owner = {
			set_country_flag = hcp_finished_outward_bound_situation
			hidden_effect = {
				abort_special_project = {
					type = hcp_DEPARTURE_POINT
				}
			}
		}
