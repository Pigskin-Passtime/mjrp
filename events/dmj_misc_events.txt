situation_event = { # Select from a random list of unrest events to execute on a random planet
	id = dmj_misc.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		owner = {
			any_owned_planet = {
				is_under_colonization = no
				has_ground_combat = no
				is_occupied_flag = no
				OR = {
					leader = {
						has_trait = leader_trait_dmj_reinhardist_devotee
					}
					sector = {
						leader = {
							has_trait = leader_trait_dmj_reinhardist_devotee
						}
					}
					has_building = dmj_reinhardist_temple
					pop_percentage = {
						percentage >= 0.25
						limit = {
							exists = pop_faction
							pop_faction = { is_same_value = root.target }
						}
						exclude = { can_think = no }
					}
				}
			}
		}
	}
	immediate = {
		owner = {
			random_owned_planet = {
				limit = {
					is_under_colonization = no
					has_ground_combat = no
					is_occupied_flag = no
					OR = {
						leader = {
							has_trait = leader_trait_dmj_reinhardist_devotee
						}
						sector = {
							leader = {
								has_trait = leader_trait_dmj_reinhardist_devotee
							}
						}
						has_building = dmj_reinhardist_temple
						pop_percentage = {
							percentage >= 0.25
							limit = {
								exists = pop_faction
								pop_faction = { is_same_value = root.target }
							}
							exclude = { can_think = no }
						}
					}
				}
				if = {
					limit = {
						root = { situation_progress < 30 }
					}
					random_list = {
						33 = {
							planet_event = { id = unrest.120 }
						}
						33 = {
							planet_event = { id = unrest.115 }
						}
						33 = {
							planet_event = { id = unrest.150 }
						}
					}
				}
				else = {
					random_list = {
						75 = {
							random_list = {
								33 = {
									planet_event = { id = unrest.120 }
								}
								33 = {
									planet_event = { id = unrest.115 }
								}
								33 = {
									planet_event = { id = unrest.150 }
								}
							}
						}
						25 = {
							modifier = {
								factor = 0
								has_planet_flag = recent_unrest_event
							}
							modifier = {
								factor = 5
								planet_stability < @stabilitylevel2
							}
							random_list = {
								33 = {
									planet_event = { id = dmj_misc.21 }
								}
								33 = {
									planet_event = { id = dmj_misc.22 }
								}
								33 = {
									modifier = {
										factor = 0
										OR = {
											leader = {
												has_trait = leader_trait_dmj_reinhardist_devotee
											}
											sector = {
												leader = {
													has_trait = leader_trait_dmj_reinhardist_devotee
												}
											}
											has_building = dmj_reinhardist_temple
											pop_percentage = {
												percentage > 0.4
												limit = {
													exists = pop_faction
													pop_faction = { is_pop_faction_type = reinhardist }
												}
												exclude = { can_think = no }
											}
										}
									}
									planet_event = { id = dmj_misc.23 }
								}
							}
						}
					}
				}
			}			
		}
	}
}
