# Portrait species change handler - on_modification_complete
# Whenever you change the portrait of an entrophereal species... it gives that portrait back
# This = Country
# From = Species ( Post Modification )
# FromFrom = Species ( Original )
# Prev = Planet
country_event = {	# on_modification_complete
	id = darktec_mechanics.50
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		OR = {
			fromfrom = {
				has_trait = trait_darktec_entro_main
				OR = {
					species_portrait = darktec_humanoid_01
					species_portrait = darktec_beast_01
					species_portrait = darktec_insectoid_01
					species_portrait = darktec_molluscoid_01
					species_portrait = darktec_plantoid_01
					species_portrait = darktec_lithoid_01
					species_portrait = darktec_machine_01
				}
			}
			from = {
				has_trait = trait_darktec_entro_main
				NOR = {
					species_portrait = darktec_humanoid_01
					species_portrait = darktec_beast_01
					species_portrait = darktec_insectoid_01
					species_portrait = darktec_molluscoid_01
					species_portrait = darktec_plantoid_01
					species_portrait = darktec_lithoid_01
					species_portrait = darktec_machine_01
				}
			}
		}
	}

	immediate = {
		fromfrom = {
			save_event_target_as = darktec_original_species
		}
		from = {
			save_event_target_as = darktec_changed_species
		}
		event_target:darktec_changed_species = {
			darktec_transform_species_portrait = yes
		}
		every_owned_leader = {
			limit = {
				species = {
					is_same_value = event_target:darktec_changed_species
				}
			}
			darktec_transform_leader_portrait = yes
		}
	}
}

# root = country
# From = leader
country_event = {	# on_leader_level_up
	id = darktec_mechanics.6001
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		from = {
			species = {
				has_trait = trait_darktec_purity_ingrained_entrogenomics
			}
			num_negative_traits > 1
		}
	}

	immediate = {
		random = {
			chance = 25
			set_variable = {
				which = darktec_leader_skill_dm_cost
				value = from.trigger:has_base_skill
			}
			multiply_variable = {
				which = darktec_leader_skill_dm_cost
				value = 200
			}
			if = {
				limit = {
					resource_stockpile_compare = {
						resource = sr_dark_matter
						value > darktec_leader_skill_dm_cost
					}
				}
				add_resource = {
					sr_dark_matter = -1
					mult = darktec_leader_skill_dm_cost
				}
				from = {
					remove_all_negative_traits = yes
				}
			}
		}
	}
}

		if = {

				}

			set_variable_to_random_value = {
				which = darktec_void_randomness_researcher
				min = 1
				max = 100
				rounded = yes
			}
			set_variable_to_random_value = {
				which = darktec_void_randomness_bureaucrat
				min = 1
				max = 100
				rounded = yes
			}
		}
	}

# Synthetic evolution fixer - As i can't (and should not) edit synthevo events, in case an empire ascends after doing the phase 4 of the crisis and adopt the transformation path
country_event = {	# on_monthly_pulse_country
	id = darktec_mechanics.5001
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		# Synthetically ascended
		OR = {
			has_ascension_perk = ap_synthetic_evolution
			has_tradition = tr_synthetics_adopt
			has_country_flag = synthetic_empire
		}
		# Post-baryonic crisis
		has_ascension_perk = darktec_ap_postbaryonic
		has_country_flag = darktec_entropic_transformation
		# Species have changed but do not have the trait
		owner_main_species = {
			OR = {
				has_trait = trait_mechanical
				has_trait = trait_machine_unit
			}
			NOT = {
				has_trait = trait_darktec_entro_main
			}
		}
		NOT = {
			has_country_flag = darktec_fixed_synthevo
		}
	}

	immediate = {
		set_country_flag = darktec_fixed_synthevo
		owner_main_species = {
			save_event_target_as = darktec_changed_species
		}
		event_target:darktec_changed_species = {
			darktec_transform_species_portrait = yes
		}
		every_owned_leader = {
			limit = {
				species = {
					is_same_value = event_target:darktec_changed_species
				}
			}
			darktec_transform_leader_portrait = yes
		}
	}
}

		while = {

						value = 0
					}

					}

			}

						has_component = DARKTEC_SHIP_UTILITY_COLLAPSER

					years = 1
				}
				export_trigger_value_to_variable = {
					trigger = has_shield_hp
					variable = dt_s_e
				}
				root = {
					change_variable = {
						which = darktec_detonation_effect
						value = prev.dt_s_e
					}
				}

					which = dt_s_e
					value = 0

				add_modifier = {	# Debuff on itself
					modifier = darktec_modifier_fleet_collapser

					multiplier = -1
				}
			}
		}
		# Operate the modifier
		multiply_variable = {	# 100k shields = 5k splash damage
			which = darktec_detonation_effect
			value = 0.05
		}
		if = {
			limit = {
				check_variable = {
					which = darktec_detonation_effect
					value > 10000
				}
			}
			set_variable = {
				which = darktec_detonation_effect
				value = 10000
			}

					NOR = {
						has_component = DARKTEC_SHIP_UTILITY_COLLAPSER
						has_component = DARKTEC_SHIP_UTILITY_COLLAPSER_ORGANIC
					}
				}
				damage_ship = root.darktec_detonation_effect
			}
		}

			sound = darktec_shield_collapse
