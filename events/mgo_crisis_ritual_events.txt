### Demonic Corruption Finisher
situation_event = {
	id = mgo_crisis_rituals.007
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		owner = {
			if = {
				limit = {
					NOT = {
						is_variable_set = mgo_corruption_value
					}
				}
				set_variable = {
					which = mgo_corruption_value
					value = 0
				}
			}

			set_variable = {
				which = mgo_corruption_value_helper
				value = value:total_occult_knowledge
			}
			multiply_variable = {
				which = mgo_corruption_value_helper
				value = 0.0001
			}
			change_variable = {
				which = mgo_corruption_value_helper
				value = 1
			}
			round_variable = mgo_corruption_value_helper

			change_variable = {
				which = mgo_corruption_value
				value = mgo_corruption_value_helper
			}
			add_modifier = {
				modifier = mgo_crisis_infernal_demonic_corruption
				mult = mgo_corruption_value
			}
		}
	}

	after = {
		destroy_situation = this
		owner = {
			set_timed_country_flag = {
				flag = mgo_demonic_corruption_cooldown
				years = 10
			}
			country_event = { id = mgo_crisis_perks.001 }
			country_event = {
				id = mgo_crisis_rituals.000
				days = 360
				random = 360
			}
		}
	}

### Scrying Finisher
situation_event = {
	id = mgo_crisis_rituals.008
	is_triggered_only = yes
	hide_window = yes

	after = {
		owner = {
			country_event = { id = mgo_crisis_perks.001 }
			country_event = {
				id = mgo_crisis_rituals.000
				days = 360
				random = 360
			}
			set_variable = {
				which = precognition_value
				value = value:total_occult_knowledge
			}
			divide_variable = {
				which = precognition_value
				value = 3333
			}
			change_variable = {
				which = precognition_value
				value = mgo_crisis_rituals_counter
			}
			ceiling_variable = precognition_value
			while = {
				count = precognition_value
				random_list = {
					1 = {
						add_random_research_option = {
							add_progress = 0.25
						}
					}
					1 = {
						random_country = {
							limit = {
								is_default_or_fallen = yes
							}
							save_event_target_as = intel_target
						}
						add_intel = {
							who = event_target:intel_target
							amount = 10
						}
					}
					1 = {
						random_system = {
							limit = {
								is_surveyed = {
									who = THIS
									status = no
								}
							}
							set_surveyed = {
								surveyed = yes
								surveyor = prev
							}
						}
					}
				}
			}
		}
		destroy_situation = this
	}
}
