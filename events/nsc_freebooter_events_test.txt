country_event = {
	id = nsc_freebooter_test.4
	title = "nsc_freebooter.4.name"
	desc = "nsc_freebooter.4.desc"
	picture = GFX_evt_surrender
	is_triggered_only = yes

	immediate = {
		create_leader = {
			name = random
			species = root.species
			gender = male
			skill = random
			class = general
			set_age = 25
			effect = {
				save_event_target_as = nsc_freebooter_the_duel_evt_the_opponent_target
			}
		}
	}

	option = {
		name = "GOOD"
		event_target:nsc_freebooter_the_duel_evt_the_opponent_target = { kill_leader = { show_notification = yes } }
	}
	option = {
		name = "DIG"
		kill_leader = {
			type = ruler
			heir = no
			show_notification = yes
		}
		if = {
			limit = { exists = heir }
			kill_leader = {
				type = ruler
				heir = yes
				show_notification = yes
			}
		}
		event_target:nsc_freebooter_the_duel_evt_the_opponent_target = {
			add_trait = nsc_freebooter_leader_trait_ruler_great_khan
			root = {
				clone_leader = {
					target = prev
					class = admiral
					traits = { trait = nsc_freebooter_leader_trait_great_khan }
					skill = 5 #less than the original khan since the leader was a general
				}
				set_leader = prev
			}
		}
	}
}
