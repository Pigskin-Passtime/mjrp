namespace = hskv_gamestart

event = {
	id = hskv_gamestart.1
	is_triggered_only = yes
	hide_window = yes

	immediate = {
		if = {
			limit = {
				# Restrictions that prevent Thayo's Spawning such as requring a valid FE
				any_country = {
					is_country_type = fallen_empire
					OR = {
						has_ethic = ethic_fanatic_xenophobe
						has_ethic = ethic_fanatic_spiritualist
						has_ethic = ethic_fanatic_materialist
					}
				}
			}
			set_global_flag = hskv_allowed
		}
		else = {
			log = "No Valid Fallen Empire for Thayo Everdawn spawn"
		}
	}
}
