# Great Host Located
country_event = {
	id = mjarchitect_unlock.45
	title = "mjarchitect_unlock.45.name"
	desc = "mjarchitect_unlock.45.desc"
	picture = GFX_evt_astral_rift_microverse
	show_sound = event_death_cult
	location = event_target:mjarchitect_great_host

	is_triggered_only = yes

	after = {
		mjarchitect_satisfaction_reward_effect_static = {
			SATISFACTION_VALUE = 50
		}
	}

	immediate = {
		random_system_within_border = {
			save_event_target_as = mjarchitect_host_spawn_system
		}
		event_target:mjarchitect_host_spawn_system = {
			spawn_system = {
				min_distance = 10
				max_distance = 30
				max_jumps = 0
				initializer = "mjarchitect_great_host_system"
				is_discovered = yes
			}
			last_created_system = {
				save_event_target_as = mjarchitect_great_host
			}
			if = {
				limit = { last_created_system = { NOT = { has_hyperlane_to = prev } } }
					add_hyperlane = {
						from = this
						to = last_created_system
					}
				}
			else = {
				add_extra_hyperlane_to_spawned_system_effect = yes
			}
		}
	}

	option = {
		name = mjarchitect_unlock.45.a
		set_country_flag = mjarchitect_great_host_located
		create_point_of_interest = {
			id = mjarchitect_great_host
			name = "mjarchitect_great_host_poi"
			desc = "mjarchitect_great_host_poi_desc"
			event_chain = mjarchitect_chain
			location = event_target:mjarchitect_great_host
		}
	}
}

# Great Host Restored
country_event = {
	id = mjarchitect_unlock.70
	title = "mjarchitect_unlock.70.name"
	desc = "mjarchitect_unlock.70.desc"
	picture = GFX_evt_cosmic_storms_fleets_of_the_thrice_damned
	show_sound = event_death_cult
	location = event_target:mjarchitect_great_host

	is_triggered_only = yes

	after = {
		mjarchitect_satisfaction_reward_effect_static = {
			SATISFACTION_VALUE = 50
		}
	}

	immediate = {
		random_megastructure = {
			limit = {
				is_megastructure_type = mjarchitect_great_host_ruined
			}
			save_event_target_as = mjarchitect_great_host

			solar_system = {
				spawn_megastructure = {
					type = mjarchitect_great_host
					coords_from = prev
					init_effect = {
						set_megastructure_flag = mjarchitect_great_host_structure
					}
				}
			}
			remove_megastructure = this
		}
	}

	option = {
		name = mjarchitect_unlock.70.a
		set_country_flag = mjarchitect_great_host_restored
	}
