# Event to setup the localization flags on Astral Rifts
# Directly using the exploring leader causes problem in the persistence of the texts
# once the leader has finished the exploration or is dead
# From = Astral Rift being explored
# FromFrom = Exploring fleet
country_event = {
	id = MHMCL_rift_event.9999
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		has_global_flag = MHMCL_system_spawned
		has_astral_planes_dlc = yes
	}

	immediate = {
		from = {
			# Cyborg leader
			if = {
				limit = {
					NOT = { has_astral_rift_flag = MHMCL_description_cyborg }
					leader.species = {
						has_trait = leader_trait_cyborg
					}
				}
				set_astral_rift_flag = MHMCL_description_cyborg
			}
			# Synthetic leader
			if = {
				limit = {
					NOT = { has_astral_rift_flag = MHMCL_description_synthetic }
					leader.species = {
						has_trait = leader_trait_synthetic
					}
				}
				set_astral_rift_flag = MHMCL_description_synthetic
			}
		}
	}
}
