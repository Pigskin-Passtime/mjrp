			}
		}
		text = "MJ25_WH_whispers.201.hivemind.desc"
	}

	show_sound = MJ25_WH_event_song
	
	is_triggered_only = yes
	event_chain = MJ25_WH_whispers_crisis_chain
	
	immediate = {
		set_situation_locked = yes
		owner = {
			capital_scope = {
				leader = {
					save_event_target_as = MJ25_WH_massa_leader_name
				}
			}
		}
	}

	after = {
		set_situation_locked = no
	}

	option = {
		name = MJ25_WH_whispers.201.a
		owner = {
			capital_scope = {
				leader = {
					kill_leader = { show_notification = no }
				}
			}
			hidden_effect = {
				if = {
					limit = {
						NOT = { is_variable_set = MJ25_WH_voices_counter }
					}
					
					set_variable = { which = MJ25_WH_voices_counter value = 0 }
				}
				change_variable = { which = MJ25_WH_voices_counter value = 20000 }
			}
		}
		add_situation_progress = 20
		custom_tooltip = MJ25_WH_whispers.201.a.tooltip
	}

	option = {
		name = MJ25_WH_whispers.201.b
		custom_tooltip = MJ25_WH_whispers.201.b.tooltip
		event_target:MJ25_WH_massa_leader_name = {
