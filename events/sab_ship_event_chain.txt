country_event = {
	id = sab_death_ship.3
	title = "sab_death_ship_3"
	desc = "sab_death_ship_3_desc"
	picture = GFX_evt_sab_fired
	show_sound = event_ship_bridge

	is_triggered_only = yes

	option = {
		name = sab_destroy_ships
	}

	immediate = {

		owner = {
			save_event_target_as = destroyed_ships
			while = {
				count = 15
				random_controlled_ship = {
					limit = {
						OR = {
							is_ship_size = corvette
							is_ship_size = destroyer
							is_ship_size = cruiser
							is_ship_size = frigate
							is_ship_size = battleship
							is_ship_size = science
							is_ship_size = constructor
							is_ship_size = colonizer
							is_ship_size = colony_vessel
							is_ship_size = titan
							is_ship_size = colossus
							is_ship_size = explorer_vessel
							is_ship_size = engineer_vessel
							is_ship_size = payback_warship
						}
					}
					destroy_ship = this
				}
			}	
		}
	}
}
