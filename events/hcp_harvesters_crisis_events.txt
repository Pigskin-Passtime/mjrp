country_event = {
	id = hcp_harvesters_crisis.100
	title = "hcp_harvesters_crisis.100.name"
	desc = {
		trigger = { NOT = { has_ethic = ethic_gestalt_consciousness } }
		text = hcp_harvesters_crisis.100.desc
	}
	desc = {
		trigger = { is_machine_empire = yes }
		text = hcp_harvesters_crisis.100.mach.desc
	}
	desc = {
		trigger = { is_hive_empire = yes }
		text = hcp_harvesters_crisis.100.hive.desc
	}
	picture = GFX_evt_mining_station
	show_sound = event_death_cult
	event_chain = hcp_harvesters_crisis_chain

	is_triggered_only = yes

	immediate = {
		begin_event_chain = {
			event_chain = hcp_harvesters_crisis_chain
			target = root
		}
		every_planet_within_border = {
			limit = {
				has_mining_station = yes
			}
			set_planet_flag = hcp_mining_station_built_@root
		}
		if = {
			limit = {
				hcp_has_valid_silo_site = yes
			}
			hcp_target_valid_silo_site = { TARGET = hcp_first_project_target }
		}
	}

	option = {
		name = hcp_harvesters_crisis.100.a
		trigger = {
			NOT = { has_origin = origin_machine }
		}
		tooltip = {
			begin_event_chain = {
				event_chain = hcp_harvesters_crisis_chain
				target = root
			}
		}
		add_event_chain_counter = {
			event_chain = "hcp_harvesters_crisis_chain"
			counter = "hcp_harvesters_level_reached"
			amount = 1
		}
		if = {
			limit = {
				exists = event_target:hcp_first_project_target
			}
			tooltip = {
				enable_special_project = {
					name = hcp_BUILD_SINGULARITY_SILO_LEVEL_1
					owner = root
				}
			}
			hidden_effect = {
				event_target:hcp_first_project_target = {
					enable_special_project = {
						name = hcp_BUILD_SINGULARITY_SILO_LEVEL_1
						owner = root
					}
				}
			}
		}
		else = {
			enable_special_project = {
				name = hcp_NO_BLACK_HOLE
				owner = root
			}
		}
		custom_tooltip = new_crisis_perks_tt
	}

	option = {
		name = hcp_harvesters_crisis.100.resource_consolidation
		trigger = {
			has_origin = origin_machine
		}
		tooltip = {
			begin_event_chain = {
				event_chain = hcp_harvesters_crisis_chain
				target = root
			}
		}
		add_event_chain_counter = {
			event_chain = "hcp_harvesters_crisis_chain"
			counter = "hcp_harvesters_level_reached"
			amount = 1
		}
		if = {
			limit = {
				exists = event_target:hcp_first_project_target
			}
			event_target:hcp_first_project_target = {
				enable_special_project = {
					name = hcp_BUILD_SINGULARITY_SILO_LEVEL_1
					owner = root
				}
			}
		}
		else = {
			enable_special_project = {
				name = hcp_NO_BLACK_HOLE
				owner = root
			}
		}
		custom_tooltip = new_crisis_perks_tt
	}
}

# Creates a coalition of opposing countries against the crisis
country_event = {
    id = hcp_harvesters_crisis.501
    title = hcp_harvesters_crisis.501.name
    desc = hcp_harvesters_crisis.501.desc
    picture = GFX_evt_announcement
    show_sound = event_sensor_ping
    location = from

    is_triggered_only = yes

	immediate = {
		establish_communications_no_message = from
		add_opinion_modifier = {
			who = from
			modifier = opinion_destroying_galaxy
		}
	}

    # Coalition war
    option = {
        name = hcp_harvesters_crisis.501.a
        custom_tooltip = hcp_harvesters_crisis.501.a.tt
        trigger = {
            is_player_crisis = no
            is_homicidal = no
        }
        ai_chance = {
            modifier = {
                add = 5
                is_xenophobe = yes # No one should become stronger than us
            }
            modifier = {
                add = 5
                is_xenophile = yes # We have to protect the weak
            }
            modifier = {
                add = 5
                is_militarist = yes
            }
            modifier = {
                add = 15
                has_ascension_perk = ap_defender_of_the_galaxy
            }
            modifier = {
                add = 5
                is_galactic_community_member = yes
            }
            modifier = {
                add = 25
                is_fallen_empire = yes
            }
			modifier = {
				add = 10
				any_neighbor_country = {
					is_same_empire = from
				}
			}
            modifier = {
                add = 10
                OR = {
                    is_galactic_custodian = yes
                    is_galactic_emperor = yes
                }
            }
        }
        hidden_effect = { set_country_flag = hcp_anti_harvesters_coalition_@from }
    }

    # Genocidals
    option = {
        name = hcp_harvesters_crisis.501.b
        custom_tooltip = hcp_harvesters_crisis.501.b.tt
        trigger = {
            is_homicidal = yes
        }
    }

    # Neutral
    option = {
        name = hcp_harvesters_crisis.501.c
        custom_tooltip = hcp_harvesters_crisis.501.c.tt
        trigger = {
            is_homicidal = no
        }
        ai_chance = {
            modifier = {
                add = 10
                is_pacifist = yes
            }
            modifier = {
                add = 25
                has_civic = civic_inwards_perfection
            }
            modifier = {
                add = 5
                is_galactic_community_member = no
            }
        }
    }

	after = {
		if = {
			limit = {
				is_country_type = default

		}

		if = {
			limit = {
				has_galactic_emperor = yes
				galactic_emperor = { has_country_flag = hcp_anti_harvesters_coalition_@root }
			}
			galactic_emperor = { save_event_target_as = war_leader }
		}
		else_if = {
			limit = {
				has_galactic_emperor = no
				has_galactic_custodian = yes
				galactic_custodian = { has_country_flag = hcp_anti_harvesters_coalition_@root }
			}
			galactic_custodian = { save_event_target_as = war_leader }
		}

		# An FE will take the lead
		else_if = {
			limit = {
				any_country = {
					is_fallen_empire = yes
					has_country_flag = anti_hyperthermia_fe
				}
			}
			random_country = {
				limit = {
					is_fallen_empire = yes
					has_country_flag = anti_hyperthermia_fe
				}
				save_event_target_as = war_leader
			}
		}

		# Will otherwise select the strongest
		else_if = {
			limit = {
				any_playable_country = {
					is_subject = no
					has_country_flag = hcp_anti_harvesters_coalition_@root
					relative_power = {
						who = prev
						value >= superior
						category = fleet
					}
				}
			}
			random_playable_country = {
				limit = {
					is_subject = no
					has_country_flag = hcp_anti_harvesters_coalition_@root
					relative_power = {
						who = prev
						value >= superior
						category = fleet
					}
				}
				save_event_target_as = war_leader
			}
		}
		else_if = {
			limit = {
				any_playable_country = {
					is_subject = no
					has_country_flag = hcp_anti_harvesters_coalition_@root
					relative_power = {
						who = prev
						value >= equivalent
						category = all
					}
				}
			}
			random_playable_country = {
				limit = {
					is_subject = no
					has_country_flag = hcp_anti_harvesters_coalition_@root
					relative_power = {
						who = prev
						value >= equivalent
						category = all
					}
				}
				save_event_target_as = war_leader
			}
		}
		else = {
			random_playable_country = {
				limit = {
					has_country_flag = hcp_anti_harvesters_coalition_@root
					is_subject = no
				}
				save_event_target_as = war_leader
			}
		}

		# WAR!
		event_target:war_leader = {
			declare_war = {
				target = root
				name = {
					key = "NAME_Declared_Crisis_War"
					variable_string = "[This.MainDefender.GetAdj]"
				}

			}

			limit = {
				has_country_flag = hcp_anti_harvesters_coalition_@root
				NOR = {
					is_same_value = event_target:war_leader
					is_at_war_with = root
					has_subject = root
				}
			}
			join_war = event_target:war_leader
		}

		# Subjects
		every_subject = {
			if = {
				limit = { has_federation = yes }
				leave_alliance = { override_requirements = yes }
			}
			if = {
				limit = {
					is_galactic_community_formed = yes
					is_galactic_community_member = yes
				}
				remove_from_galactic_community = yes
			}
		}

		# Notifications
		country_event = { id = hcp_harvesters_crisis.503 }
		every_playable_country = {
			limit = {
				NOT = { is_same_empire = root }
			}
			country_event = {
				id = hcp_harvesters_crisis.504
				scopes = {
					from = root
				}
			}
		}
	}

			has_special_project = hcp_STEAL_SINGULARITY_SILO_2
		}
	}

	immediate = {
		if = {
			limit = { has_special_project = hcp_STEAL_SINGULARITY_SILO_1 }
			abort_special_project = {
				type = hcp_STEAL_SINGULARITY_SILO_1
			}
		}
		else_if = {
			limit = { has_special_project = hcp_STEAL_SINGULARITY_SILO_2 }
			abort_special_project = {
				type = hcp_STEAL_SINGULARITY_SILO_2
			}
		}
		if = {
			limit = {
				hcp_has_valid_silo_site = yes
			}
			hcp_target_valid_silo_site = { TARGET = hcp_rebuild_project_target }
		}
		if = {
			limit = {
				any_playable_country = {
					any_owned_megastructure = {
						has_megastructure_flag = hcp_crisis_silo
					}
				}
			}
			random_playable_country = {
				limit = {
					any_playable_country = {
						any_owned_megastructure = {
							has_megastructure_flag = hcp_crisis_silo
						}
					}
				}
				random_system_within_border = {
					limit = {
						has_star_flag = hcp_singularity_silo_@prev
						any_system_megastructure = { has_megastructure_flag = hcp_crisis_silo }
					}
					star = {
						save_event_target_as = hcp_silo_to_steal
					}
				}
			}
		}
	}

	option = {
		name = UNFORTUNATE
		trigger = {
			NOT = { exists = event_target:hcp_silo_to_steal }
		}
		if = {
			limit = {
				exists = event_target:hcp_rebuild_project_target
			}
			tooltip = {
				enable_special_project = {
					name = hcp_REBUILD_SINGULARITY_SILO_1
					owner = root
				}
			}
			hidden_effect = {
				event_target:hcp_rebuild_project_target = {
					if = {
						limit = { root = { has_event_chain = hcp_harvesters_crisis_chain } }
						enable_special_project = {
							name = hcp_REBUILD_SINGULARITY_SILO_1
							owner = root
						}
					}
					else_if = {
						limit = { root = { has_event_chain = hcp_harvesters_victory_chain } }
						enable_special_project = {
							name = hcp_REBUILD_SINGULARITY_SILO_2
							owner = root
						}
					}
				}
			}
		}
		else = {
			enable_special_project = {
				name = hcp_NEEDS_NEW_BLACK_HOLE
				owner = root
			}
		}
	}

	option = {
		name = hcp_harvesters_crisis.1015.b
		trigger = {
			exists = event_target:hcp_silo_to_steal
		}
		tooltip = {
			event_target:hcp_silo_to_steal = {
				enable_special_project = {
					name = hcp_STEAL_SINGULARITY_SILO_1
				}
			}
		}
		hidden_effect = {
			event_target:hcp_silo_to_steal = {
				if = {
					limit = { root = { has_event_chain = hcp_harvesters_crisis_chain } }
					enable_special_project = {
						name = hcp_STEAL_SINGULARITY_SILO_1
						owner = root
					}
				}
				else_if = {
					limit = { root = { has_event_chain = hcp_harvesters_victory_chain } }
					enable_special_project = {
						name = hcp_STEAL_SINGULARITY_SILO_2
						owner = root
					}
				}
			}
		}
	}
