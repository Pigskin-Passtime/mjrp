namespace = vgflavor

country_event = {
	id = vgflavor.1
	is_triggered_only = yes
	hide_window = yes

	trigger = {
		 has_country_flag = inanis_owned 
	}

	immediate = {
		if = { 
			limit = { 
				has_country_flag = inanis_owned
				NOT = { has_country_flag = void_governor_first_flavour }
			}
			random_owned_leader = {
				limit = {has_leader_flag = inanis} 
				leader_event = { id = vgflavor.2 days = 1 random = 1800 } 
			} 
		}

		else_if = {
			limit = { 
				has_country_flag = inanis_owned
				mid_game_years_passed >= 50
			} 
			random_owned_leader = {
				limit = {has_leader_flag = inanis} 
				random_list = {
					120 = { leader_event = { id = vgflavor.4 days = 1 random = 1800 } } #upgrades destiny trait
					20 = { leader_event = { id = vgflavor.7 days = 1 random = 1800 } } #dark matter deflector
					20 = { leader_event = { id = vgflavor.8 days = 1 random = 1800 } } #dark matter power
					20 = { leader_event = { id = vgflavor.9 days = 1 random = 1800 } } #dark matter propulsion
					180 = {}
				}
			}	
		}

		else_if = {
			limit = { has_country_flag = inanis_owned} 
			random_owned_leader = {
				limit = {has_leader_flag = inanis} 
				random_list = {
					60 = { leader_event = { id = vgflavor.4 days = 1 random = 1800 } } #upgrades destiny trait
					#5 = { leader_event = { id = vgflavor.7 days = 1 random = 1800 } } #dark matter deflector
					20 = { leader_event = { id = vgflavor.8 days = 1 random = 1800 } } #dark matter power
					#5 = { leader_event = { id = vgflavor.9 days = 1 random = 1800 } } #dark matter propulsion
					280 = {}
				}
			}	
		}
	}
}

	desc = vgflavor.2.desc
	picture = GFX_evt_partition
	show_sound = event_default
	is_triggered_only = yes
	fire_only_once = yes

	trigger = {
		exists = owner
		has_leader_flag = inanis_clone
	}

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_renowned"

	picture_event_data = {
		portrait = root
	}

	option = {
		name = vgflavor.2.a
		owner = {
			add_research_option = tech_void_governor_research			
		}
		hidden_effect = { owner = { set_country_flag = void_governor_first_flavour }}
	}
}
