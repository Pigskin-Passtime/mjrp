#some clone biowaste has started moving...
situation_event = { 
	id = fleshsong.101
    title = fleshsong.101.name
	desc = fleshsong.101.desc
	picture = GFX_evt_clones
	show_sound = event_dangerous_animals
	is_triggered_only = yes

	situation = this

	trigger = {
		owner = {
			any_owned_planet = {
				OR = { 
					has_active_building = building_clone_vats
					has_active_building = building_clone_army_clone_vat
					has_active_building = building_spawning_pool
				}
			}
		}
		has_global_flag = fleshsong_in_progress
	}

	immediate = {
		set_situation_locked = yes
		owner = {
			random_owned_planet = {
				limit = {
					is_colony = yes
					OR = { 
						has_active_building = building_clone_vats
						has_active_building = building_clone_army_clone_vat
						has_active_building = building_spawning_pool
					}
				}

				save_event_target_as = clone_planet

			}
		}
		create_country = {
			name = "NAME_Biowaste_Amalgams"
			type = faction
			effect = {
				establish_communications_no_message = root.owner
				save_event_target_as = fleshsong_amalgams
			}
		}
	}
	after = {
		set_situation_locked = no
	}


    option = {
		name = fleshsong.101.A
		event_target:clone_planet = {
			if = {
				limit = { has_building = building_clone_vats }
				ruin_building = building_clone_vats
			}
			else_if = {
				limit = { has_building = building_clone_army_clone_vat }
				ruin_building = building_clone_army_clone_vat
			}
			else_if = {
				limit = { has_building = building_spawning_pool }
				ruin_building = building_spawning_pool
			}
			while = {
				count = 4
				create_army = {
					owner = event_target:fleshsong_amalgams
					type = fleshsong_amalgam_army
				}
			}
		}
	}
}

#Things have started going awry at one of our gene clinics
situation_event = { 
	id = fleshsong.102
    title = fleshsong.102.name
	desc = fleshsong.102.desc
	picture = GFX_evt_clones
	show_sound = event_dangerous_animals
	is_triggered_only = yes

	situation = this

	trigger = {
		owner = {
			any_owned_planet = {
				OR = { 
					has_active_building = building_clinic
					has_active_building = building_hospital
					has_active_building = building_spawning_pool
				}
			}
		}
		has_global_flag = fleshsong_in_progress
	}

	immediate = {
		set_situation_locked = yes
		owner = {
			random_owned_planet = {
				limit = {
					is_colony = yes
					OR = { 
						has_active_building = building_clinic
						has_active_building = building_hospital
					}
				}

				save_event_target_as = clone_planet

			}
		}
		create_country = {
			name = "NAME_Biowaste_Amalgams"
			type = faction
			effect = {
				establish_communications_no_message = root.owner
				save_event_target_as = fleshsong_amalgams
			}
		}
	}
	after = {
		set_situation_locked = no
	}


    option = {
		name = fleshsong.102.A
		event_target:clone_planet = {
			random_owned_pop = {
				kill_pop = yes
			}
			if = {
				limit = { has_building = building_clinic }
				ruin_building = building_clinic
			}
			else_if = {
				limit = { has_building = building_hospital }
				ruin_building = building_hospital
			}
			while = {
				count = 2
				create_army = {
					owner = event_target:fleshsong_amalgams
					type = fleshsong_amalgam_army
				}
			}
		}
	}
}




#some of out pops have mutated...
situation_event = {
    id = fleshsong.201
    title = "fleshsong.201.name"
    desc = "fleshsong.201.desc"
    picture = GFX_evt_genetic_modification
    show_sound = event_bad_omen
	is_triggered_only = yes

    #limit = {
    #    owner = { has_country_flag = has_fleshsong }
    #}
	trigger = {
		has_global_flag = fleshsong_in_progress
	}

    weight_multiplier = {
        factor = 1
		modifier = {
            factor = 1.6
            owner = { has_country_flag = fleshsong_one_with_flesh }
        }
        modifier = {
            factor = 1.3
            owner = { has_country_flag = fleshsong_encouraging_flesh }
        }
        modifier = {
            factor = 0.6
            owner = { has_country_flag = fleshsong_improved_safety }
        }
		modifier = {
            factor = 0.3
            owner = { has_country_flag = fleshsong_secured_safety }
        }
    }

    
    immediate = {
        owner = { 
			if = {
				limit = {
					has_country_flag = fleshsong_encouraging_flesh
				}
				fleshsong_mutate_pops = { COUNT = 3 }
			}
			else_if = {
				limit = {
					has_country_flag = fleshsong_one_with_flesh
				}
				fleshsong_mutate_pops = { COUNT = 4 }
			}
			else = {
				fleshsong_mutate_pops = { COUNT = 2 }
			}
            
        }
        
    }

    option = {
        name = fleshsong.201.A
        
    }
    
    

}
