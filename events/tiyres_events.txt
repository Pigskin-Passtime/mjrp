country_event = {
	id = tiyres_events.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = tiyres_in_progress
		is_country_type = tiyres_crisis
	}

	immediate = {
		fromfrom = {
			if = {
				limit = {
					NOT = {
						is_variable_set = tiyres_pop_count
					}
				}
				set_variable = {
					which = tiyres_pop_count
					value = 0
				}
			}
			every_owned_pop = {
				prev = {
					change_variable = {
						which = tiyres_pop_count
						value = 1
					}
				}
				kill_pop = yes
			}
			change_pc = pc_infested
			clear_planet_modifiers = yes
			add_modifier = {
				modifier = planet_tiy_mark
			}
		}
		every_owned_army = {
			remove_army = yes
		}
	}
}

country_event = {
	id = tiyres_events.9
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = tiyres_in_progress
		any_owned_planet = {
			has_modifier = planet_tiy_infestation
		}
	}

	immediate = {
		every_owned_planet = {
			limit = {
				has_modifier = planet_tiy_infestation
			}
			if = {
				limit = {
					is_variable_set = tiyres_planet_infestation
				}
				change_variable = {
					which = tiyres_planet_infestation
					value = 1
				}
			}
			else = {
				set_variable = {
					which = tiyres_planet_infestation
					value = 1
				}
			}
			if = {
				limit = {
					colony_age > 12
				}
				random_list = {
					40 = {

					}
					60 = {
						random_owned_pop = {
							if = {
								limit = {
									species = {
										is_robotic = no
									}
								}
								prev = {
									change_variable = {
										which = tiyres_planet_infestation
										value = value:tiyres_pop_infestation
									}
									if = {
										limit = {
											NOT = {
												is_variable_set = tiyres_pop_count
											}
										}
										set_variable = {
											which = tiyres_pop_count
											value = 1
										}
									}
									else = {
										change_variable = {
											which = tiyres_pop_count
											value = 1
										}
									}
								}
							}
							kill_pop = yes
						}
					}
				}
			}
			if = {
				limit = {
					num_assigned_jobs = {
						job = tiyres_infestation_purger
						value > 1
					}
				}
				subtract_variable = {
					which = tiyres_planet_infestation
					value = 6
				}
			}
			else_if = {
				limit = {
					num_assigned_jobs = {
						job = tiyres_infestation_purger
						value > 0
					}
				}
				subtract_variable = {
					which = tiyres_planet_infestation
					value = 3
				}
			}
			remove_modifier = planet_tiy_infestation
			add_modifier = {
				modifier = planet_tiy_infestation
				mult = tiyres_planet_infestation
			}
			if = {
				limit = {
					check_variable = {
						which = tiyres_planet_infestation
						value > 99
					}
				}
				every_owned_pop = {
					prev = {
						change_variable = {
							which = tiyres_pop_count
							value = 1
						}
					}
					kill_pop = yes
				}
				clear_planet_modifiers = yes
				change_pc = pc_infested
				add_modifier = {
					modifier = planet_tiy_mark
				}
			}
			else_if = {
				limit = {
					check_variable = {
						which = tiyres_planet_infestation
						value < 1
					}
				}
				remove_modifier = planet_tiy_infestation
				clear_variable = tiyres_pop_count
				clear_variable = tiyres_planet_infestation
				add_modifier = {
					modifier = planet_tiy_infestation_remains
				}
			}
		}
	}
}
