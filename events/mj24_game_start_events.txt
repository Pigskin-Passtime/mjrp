	trigger = {

	}

	immediate = {
		event_target:global_event_country = {
			random_list = {
				10 = {
					set_variable = { which = num_leader_spawns value = 3 }

			count = event_target:global_event_country.num_leader_spawns

						factor = 0

							has_global_flag = exakades_allowed

						factor = 0
						has_global_flag = exakades_system_spawned

						limit = {
							NOT = {
								is_in_cluster = leader_spawn
							}
						}
						spawn_exakades = yes

							id = leader_spawn
							radius = 150
							center = THIS
						}
					}
