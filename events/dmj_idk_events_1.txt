namespace = dmj_idk_1

country_event = { # Original Empire Victory
	id = dmj_idk_1.1
	title = "dmj.16.name"
	desc = "dmj.16.desc"
	is_triggered_only = yes

	picture = GFX_evt_victorious_army
	show_sound = event_celebration

	trigger = {
		is_ai = no # Only if you are playing as your Original Empire
	}

	immediate = {
		remove_country_flag = dmj_reinhardists_revolting
		if = {
			limit = { any_owned_planet = { has_building = dmj_reinhardist_temple } }
			every_owned_planet = {
				limit = { has_building = dmj_reinhardist_temple }
				remove_building = dmj_reinhardist_temple
			}
		}
	}

	option = {
		name = dmj.16.regular
		hidden_effect = {
			clear_global_event_target = dmj_rebellion_originator
			clear_global_event_target = dmj_reinhardist_uprising
		}
	}
	option = {
		name = dmj.16.authoritarian
		trigger = {
			is_authoritarian = yes
		}
		hidden_effect = {
			clear_global_event_target = dmj_rebellion_originator
			clear_global_event_target = dmj_reinhardist_uprising
		}
	}
}
