situation_event = {
	#Subjugated
	id = coex_situation.103
	title = "coex_situation.103.name"
	desc = coex_situation.103.desc
	picture = GFX_evt_signing
	show_sound = event_death_cult
	is_triggered_only = yes
	immediate = {
		owner = {
			set_country_flag = coex_subjugation_attempted
			save_event_target_as = coex_subjugation_target
		}
		random_country = {
			limit = {
				has_country_flag = coex_crisis_subjugating@root.owner
			}
			country_event = {
				id = coex_situation.110
				days = 1				#Notify subjugator
			}
			root.owner = {
				set_subject_of = {
					who = prev
					allow_instant_negotiation = yes
					preset = preset_vassal_mean_01
				}
			}
		}
	}
	option = {
		name = ACKNOWLEDGED
	}
	after = {
		destroy_situation = this
	}
}

	}

country_event = {
	id = coex_situation.304
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_playable_country = {
			limit = {
				is_ai = no
				NOT = {
					is_same_value = root
				}
			}
			country_event = {
				id = coex_situation.305
			}
		}
		set_update_modifiers_batch = begin
		set_update_modifiers_batch = end
		every_country = {
			limit = {
				NOT = {
					is_country_type = default
				}
			}
			destroy_country = yes
		}
		win = yes
	}
}
