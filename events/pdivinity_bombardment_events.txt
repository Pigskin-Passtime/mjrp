#######################
### planet ruptured ###
#######################

planet_event = {
    id = pdivinity_bombardment.1
    hide_window = yes

    is_triggered_only = yes

    trigger = {
        has_orbital_bombardment_stance = pdivinity_psionic_rupture_bombardment
        habitable_planet = yes
        OR = {
            planet_devastation >= 100
            count_owned_pop_amount = {
                count <= 50
            }
        }
    }

    immediate = {
        random_list = {
            34 = {
                modifier = {
                    factor = 10
                    count_owned_pop_amount = {
                        count > 5000
                    }
                }
                owner = {
                    if = {
                        limit = {
                            owner = {
                                is_at_war_with = from
                            }
                        }
                        add_static_war_exhaustion = {
                            attacker = from
                            location = root
                            value_for_planet_destruction = 0.5
                        }
                    }
                    if = {
                        limit = {
                            NAND = {
                                has_origin = origin_progenitor_hive
                                root = { is_capital = yes }
                            }
                        }
                        country_event = { id = pdivinity_bombardment.2 }
                    }
                    else = {
                        country_event = { id = origin.3265 }
                    }
                }
                from = { country_event = { id = pdivinity_bombardment.3 } }
                change_pc = pc_shrouded
                set_planet_flag = nuked_planet_anomalies_disabled
                remove_all_buildings = yes
                set_planet_flag = armageddon_nuked
            }
            66 = {
                owner = {
                    if = {
                        limit = {
                            owner = {
                                is_at_war_with = from
                            }
                        }
                        add_static_war_exhaustion = {
                            attacker = from
                            location = root
                            value_for_planet_destruction = 0.5
                        }
                    }
                    if = {
                        limit = {
                            NAND = {
                                has_origin = origin_progenitor_hive
                                root = { is_capital = yes }
                            }
                        }
                        country_event = { id = pdivinity_bombardment.4 }
                    }
                    else = {
                        country_event = { id = origin.3265 }
                    }
                }
                from = { country_event = { id = pdivinity_bombardment.5 } }
                add_modifier = {
                    modifier = pdivinity_shroud_phased_modifier
                    days = -1
                }
                set_planet_entity = {
                    picture = pc_pdivinity_shroud
                    entity = shroud_planet_01_entity
                }
                every_owned_pop_group = {
                    kill_single_pop = yes
                }
                destroy_colony = yes
                set_planet_flag = nuked_planet_anomalies_disabled
                remove_all_buildings = yes
                set_planet_flag = armageddon_nuked
            }
        }
    }
}
