# Starts Crisis War
country_event = {
	id = dsPR.50
	hide_window = yes
	is_triggered_only = yes
	immediate = {
	# Min/Max Rifts
		export_trigger_value_to_variable = {
			trigger = num_galaxy_systems
			variable = ds_pr_star_var
		}
		multiply_variable = { which = ds_pr_star_var value = @ds_pr_star_mult }
		round_variable = ds_pr_star_var
		if = {
			limit = { check_variable = { which = ds_pr_star_var value < @ds_pr_star_min} }
			set_variable = { which = ds_pr_star_var value = @ds_pr_star_min }
		}
		if = {
			limit = { check_variable = { which = ds_pr_star_var value > @ds_pr_star_max } }
			set_variable = { which = ds_pr_star_var value = @ds_pr_star_max }
		}
	# Critical Mass
		export_trigger_value_to_variable = {
			trigger = num_galaxy_systems
			variable = ds_pr_crit_var
		}
		multiply_variable = { which = ds_pr_crit_var value = @ds_pr_crit_mult }
		round_variable = ds_pr_crit_var
	# Kick out of alliances
		if = {
			limit = {
				is_galactic_community_formed = yes
				is_galactic_community_member = yes
			}
			remove_from_galactic_community = yes
		}
		if = {
			limit = { has_federation = yes }
			leave_alliance = { override_requirements = yes }
		}
	# War
		ds_pr_crisis_effect = yes
	}
}

# Unbidden Gameplay - pops to ships
# Will create fleets and ships in a pulse, or the game can lag/crash/desync if too many
country_event = {
	id = dsPR.115
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = { check_variable = { which = ds_pr_ship_var value >= 50 } }
			event_target:ds_pr_exodus_system = {
				create_fleet = {
					effect = {
						set_owner = event_target:ds_pr_crisis_country
						set_location = prev
						while = {
							count = 50
							create_ship = {
								name = random
								random_existing_design = ds_pr_reaver
							}
						}
					}
				}
			}
			change_variable = { which = ds_pr_ship_var value = -50 }
		}
		else_if = {
			limit = { check_variable = { which = ds_pr_ship_var value < 50 } }
			event_target:ds_pr_exodus_system = {
				create_fleet = {
					effect = {
						set_owner = event_target:ds_pr_crisis_country
						set_location = prev
						while = {
							count = root.ds_pr_ship_var
							create_ship = {
								name = random
								random_existing_design = ds_pr_reaver
							}
						}
					}
				}
			}
			set_variable = { which = ds_pr_ship_var value = 0 }
		}
		if = {
			limit = { check_variable = { which = ds_pr_ship_var value >= 1 } }
			country_event = { id = dsPR.115 days = 1 }
		}
	}
}

# Unbidden Gameplay - galaxy reset setup
country_event = {
	id= dsPR.125
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		every_system_within_border = {
			limit = { NOT = { is_same_value = event_target:ds_pr_exodus_system } }
			random_fleet_in_system = {
				limit = { is_ship_class = shipclass_starbase }
				destroy_fleet = this
			}
		}
		every_system = {
			limit = { NOT = { is_same_value = event_target:ds_pr_exodus_system } }
			random_list = {
				50 = {}
				50 = { save_global_event_target_as = ds_pr_reset_wave }
			}
		}
		every_system = {
			every_system_colony = {
				remove_all_buildings = yes
				remove_all_districts = yes
				destroy_colony = yes
			}
			every_system_megastructure = {
				remove_megastructure = this
			}
			every_system_planet = {
				limit = { is_artificial = yes }
				remove_planet = yes
			}
		}
		every_system = {
			limit = { is_same_value = event_target:ds_pr_reset_wave }
			ds_pr_crisis_reset = yes
		}
		every_system = {
			limit = {
				NOR = {
					is_same_value = event_target:ds_pr_reset_wave
					is_same_value = event_target:ds_pr_exodus_system
				}
			}
			ds_pr_crisis_reset = yes
		}
		while = {
			count = ds_pr_habitables_var
			random_system = {
				limit = { NOT = { has_star_flag = ds_pr_habitable_world } }
				ds_pr_crisis_habitables = yes
			}
		}
	}
	after = {
		export_trigger_value_to_variable = {
			trigger = check_galaxy_setup_value
			parameters = { setting = num_empires }
			variable = ds_pr_empires_var
		}
		if = {
			limit = { check_variable = { which = ds_pr_empires_var value > 12 } }
			set_variable = { which = ds_pr_empires_var value = 12 }
		}
		if = {
			limit = { check_variable = { which = ds_pr_empires_var value < 5 } }
			set_variable = { which = ds_pr_empires_var value = 5 }
		}
		country_event = { id = dsPR.130 days = 2 }
		country_event = { id = dsPR.135 days = 5 }
	}
}

country_event = {

	is_triggered_only = yes
	trigger = {
		has_ascension_perk = ap_ds_phantom_reality
		has_crisis_level = ds_pr_level_1
	}
	immediate = {

# Objectives
		complete_crisis_objective = crisobj_ds_pr_physics
		complete_crisis_objective = crisobj_ds_pr_rifts
		if = {
			limit = { has_edict = ds_pr_dreamhijacking }
			country_event = { id = dsPR.1030 }
		}
# Reality Breach - Stars
		if = {
			limit = { has_crisis_level = ds_pr_level_3 }
			if = {
				limit = { NOT = { exists = event_target:ds_pr_breach_target@root } }
				capital_star.solar_system = {
					closest_system = {
						limit = {
							exists = owner
							owner = { is_same_value = root }
							NOT = { has_star_flag = ds_pr_breach_star }
						}
						save_global_event_target_as = ds_pr_breach_target@root
						star = { ds_pr_crisis_breach = yes }
					}
				}
			}
			else = {
				limit = { exists = event_target:ds_pr_breach_target@root }
				event_target:ds_pr_breach_target@root = { star = { ds_pr_crisis_breach = yes } }
			}
			# Monthly Pop Conversion
			country_event = { id = dsPR.1001 days = 3 }
		}
# Reality Breach - Planets
		if = {
			limit = { has_crisis_level = ds_pr_level_5 }
			every_owned_planet = {
				limit = { any_owned_pop_group = { NOT = { pop_group_has_trait = trait_ds_exd_phantom } } }
				random_owned_pop_group = {
					limit = { NOT = { pop_group_has_trait = trait_ds_exd_phantom } }
					kill_pop_group = { pop_group = this amount = @ds_pr_rift_pops }
				}
			}
			random_system_within_border = {
				limit = {
					any_neighbor_system = {
						exists = owner
						NOT = { owner = { is_same_value = root } }
						count_system_colony = { count > 0 }
					}
				}
				random_neighbor_system = {
					limit = {
						exists = owner
						NOT = { owner = { is_same_value = root } }
						count_system_colony = { count > 0 }
					}
					every_system_colony = {
						random_owned_pop_group = {
							limit = { NOT = { pop_group_has_trait = trait_ds_exd_phantom } }
							kill_pop_group = { pop_group = this amount = @ds_pr_rift_pops }
						}
					}
				}
			}
		}
# Situation - Debug: effect = { add_situation_progress = 5 }
		if = {
			limit = { has_crisis_level = ds_pr_level_5 }
		# Setup
			export_trigger_value_to_variable = {
				trigger = count_system_within_border
				parameters = { limit = { has_star_flag = ds_pr_breach_star } }
				variable = ds_pr_breach_var
			}
			set_variable = { which = ds_pr_bonus_var value = 0 } # Must be 0 before Conclude, or it progresses without the limit
		# Conclude
			if = {
				limit = { check_variable = { which = ds_pr_breach_var value >= ds_pr_star_var } }
				set_variable = { which = ds_pr_monthly_var value = 1 }
			# Bonus
				set_variable = { which = ds_pr_bonus_var value = ds_pr_breach_var }
				divide_variable = { which = ds_pr_bonus_var value = @ds_pr_bonus_mult }
				multiply_variable = { which = ds_pr_bonus_var value = 0.5 }
				floor_variable = ds_pr_bonus_var
			# Critical Mass
				if = {
					limit = { check_variable = { which = ds_pr_breach_var value >= ds_pr_crit_var } }
					set_timed_global_flag = { flag = ds_pr_crit_flag months = 1 }
				}
			}
			else_if = {
				limit = { check_variable = { which = ds_pr_breach_var value < ds_pr_star_var } }
				if = {
					limit = {
						NOT = {
							any_situation = {
								is_situation_type = situation_ds_phantom_reality
								current_stage = situation_ds_pr_stage_1
							}
						}
					}
					set_variable = { which = ds_pr_monthly_var value = -2 }
				}
				else = { set_variable = { which = ds_pr_monthly_var value = 0 } }
			}
		}
	}
}
