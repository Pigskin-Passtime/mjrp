country_event = { # Create Mircorv
	id = MHMCL_story_event.1
	title = MHMCL_story_event.1.name
	desc = MHMCL_story_event.1.desc
	picture = GFX_evt_board_meeting
	show_sound = par_event_keides # Change to other
	fire_only_once = yes
	is_triggered_only = yes
	event_window_type = leader_story
	#picture = GFX_leader_recruitment_bg_MHMCL_mircorv

	immediate = {
		if = {
			limit = {
				NOT = { exists = event_target:MHMCL_country }
			}
			create_country = {
				name = "NAME_Unknown"
				type = global_event
				auto_delete = yes
				flag = {
					icon = {
						category = "special"
						file = "unknown.dds"
					}
					background = {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors = {
						"red"
						"red"
						"null"
						"null"
					}
				}
				effect = {
					save_global_event_target_as = MHMCL_country
				}
			}
		}
		event_target:MHMCL_country = {
			create_species = {
				name = "NAME_mircorv_race"
				class = "ROBOT"
				namelist = "AI"
				portrait = "ai_crisis_2"
				immortal = yes
				traits = {
					trait = "trait_mechanical"
				}
				effect = { save_event_target_as = MHMCL_species }
			}
			create_leader = {
				class = commander
				species = event_target:MHMCL_species
				name = "NAME_mircorv"
				skill = 4
				gender = female
				event_leader = yes
				skip_background_generation = yes
				custom_catch_phrase = MHMCL_mircorv_catch_phrase
				custom_description = MHMCL_mircorv_desc
				set_age = 89127
				effect = {
					set_leader_flag = legendary_leader
					set_leader_flag = MHMCL_mircorv
					set_leader_flag = immune_to_negative_traits
					set_leader_flag = no_council_traits
					set_leader_flag = no_destiny_traits
					set_leader_flag = cannot_be_ruler
					set_leader_flag = cannot_be_councilor
					set_leader_flag = cannot_explore_rift
					save_global_event_target_as = MHMCL_mircorv_leader
					add_trait_no_notify = MHMCL_leader_trait_ancient_intelligence
					add_trait_no_notify = subclass_commander_admiral
					set_owner = root
					exile_leader_as = MHMCL_mircorv_leader
				}
			}
		}
	}

	picture_event_data = {
		portrait = event_target:MHMCL_mircorv_leader
		room = galactic_custodian_room
	}

	option = {
		name = MHMCL_story_event.1.a
		hidden_effect = {
			event_target:MHMCL_mircorv_leader = {
				change_leader_portrait = sd_mol_robot
			}
			country_event = { id = MHMCL_story_event.2 }
		}
	}
}
