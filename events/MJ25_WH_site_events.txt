							}
						}
						NOR = {
							is_councilor_type = councilor_gestalt_growth
							AND = {
								is_gestalt = yes
								is_ruler = yes
							}
						}
					}
					if = {
						limit = {
							leader_class = scientist
						}
						add_trait = {
							trait = MJ25_WH_leader_trait_truthseeker
							show_message = no
						}
					}
					if = {
						limit = {
							leader_class = official
						}
						add_trait = {
							trait = MJ25_WH_leader_trait_orator
							show_message = no
						}
					}
					if = {
						limit = {
							leader_class = commander
						}
						add_trait = {
							trait = MJ25_WH_leader_trait_speech_shepherd
							show_message = no
						}
					}
				}
				every_pool_leader = {
					limit = {
						species = {
							NOR = {
								has_trait = trait_machine_unit
								has_trait = trait_mechanical
							}
						}
						NOR = {
							is_councilor_type = councilor_gestalt_growth
							AND = {
								is_gestalt = yes

							}
