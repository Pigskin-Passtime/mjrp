# Main Pulse Event
country_event = {
	id = soj_f.2000
	hide_window = yes
	is_triggered_only = yes
	
	trigger = { 
		NOT = { has_country_flag = soj_flag_transcending } 
		has_country_flag = has_soj_leader
	}
	
	immediate = {
		set_variable = { which = soj_f_var_thirst_decision value = 0 }
		random_list = {
			100 = { set_variable = { which = soj_f_var_thirst_decision value = 1 } }
			100 = { set_variable = { which = soj_f_var_thirst_decision value = 2 } }
			90 = { set_variable = { which = soj_f_var_thirst_decision value = 3 } }
			90 = { set_variable = { which = soj_f_var_thirst_decision value = 4 } }
			80 = { set_variable = { which = soj_f_var_thirst_decision value = 5 } }
		}
		country_event = { id = soj_f.200 }
	}
}

country_event = {
	id = soj_f.201
	title = soj_f.201.name
	desc = soj_f.201.desc
	picture = GFX_evt_mysterious_signal

	is_triggered_only = yes

	show_sound = event_air_raid_siren

	# Random Pop
	option = {
		name = WORRYING
		exclusive_trigger = {
			check_variable = { which = soj_f_var_thirst_decision value = 1 }
		}
		while = {
			count = 3
			random_owned_pop = {
				kill_pop = yes
			}
		}
	}

	# Random Building
	option = {
		name = WORRYING
		exclusive_trigger = {
			check_variable = { which = soj_f_var_thirst_decision value = 2 }
		}
		while = {
			count = 3
			random_owned_planet = {
				remove_last_built_district = yes
			}
		}
	}

	# Random Ship
	option = {
		name = WORRYING
		custom_tooltip = SOJ_SHIP_DESTROYED
		exclusive_trigger = {
			check_variable = { which = soj_f_var_thirst_decision value = 3 }
		}
		random_owned_fleet = {
			destroy_fleet = this
		}
	}
	
	# Random Leader
	option = {
		name = WORRYING
		exclusive_trigger = {
			check_variable = { which = soj_f_var_thirst_decision value = 4 }
		}
		random_owned_leader = {
			limit = { leader_class = scientist }
			kill_leader = { show_notification = no }
		}
	}

	# Random Planet
	option = {
		name = WORRYING
		exclusive_trigger = {
			check_variable = { which = soj_f_var_thirst_decision value = 5 }
		}
		random_owned_planet = {
			if = {
				limit = { NOT = { is_planet_class = pc_nuked } }
				change_pc = pc_nuked
				add_planet_devastation = 20
			} else = {
				add_planet_devastation = 50
			}
		}
	}

	after = {
		hidden_effect = { country_event = { id = soj_f.202 } }
	}
}
