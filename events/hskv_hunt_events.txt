namespace = hskv_hunt
# Eve Online time
country_event = {
	id = hskv_hunt.1000
	title = hskv_hunt.1000.name
	desc = hskv_hunt.1000.desc
	picture = GFX_evt_deficit
	show_sound = event_default
	is_triggered_only = yes
	immediate = {
		set_country_flag = hskv_ignore_dismiss_effect
		end_event_chain = hskv_thayo_waiting_chain
		set_variable = {
			which = hskv_total_num_valid_ships
			value = 0
		}
		save_event_target_as = hskv_thayo_country
		every_owned_ship = {
			limit = {
				is_designable = yes
				is_in_combat = no
				is_owned_by = PREV
				OR = {
					is_ship_size = corvette
					is_ship_size = frigate
					is_ship_size = destroyer
					is_ship_size = cruiser
				}
			}
			event_target:hskv_thayo_country = {
				change_variable = {
					which = hskv_total_num_valid_ships
					value = 1
				}
			}
		}
		set_variable = {
			which = hskv_num_targeted_ships
			value = hskv_total_num_valid_ships
		}
		divide_variable = {
			which = hskv_num_targeted_ships
			value = 5
		}
		if = {
			limit = {
				is_at_war = yes
			}
			divide_variable = {
				which = hskv_num_targeted_ships
				value = 4
			}
		}
	}
	option = {
		name = hskv_hunt.1000.response
		custom_tooltip = hskv_hunt.1000.response.tooltip
		hidden_effect = {
			country_event = {
				id = hskv_hunt.1001
				days = 1
			}
		}
		owner = {
			add_resource = {
				energy = -30000
				minerals = -20000
				food = -20000
				consumer_goods = -5000
				alloys = -5000
				rare_crystals = -1500
				exotic_gases = -1500
				volatile_motes = -1500
			}
			random_owned_leader = {
				limit = {
					has_leader_flag = hskv_thayo_everdawn_leader
				}
				kill_leader = {
					show_notification = no
					fire = yes
				}
			}
			random_owned_leader = {
				limit = {
					is_ruler = no
					NOR = {
						has_leader_flag = renowned_leader
						has_leader_flag = legendary_leader
						has_trait = trait_imperial_heir
						has_trait = leader_trait_chosen
						has_trait = leader_trait_eater_chosen
						has_trait = leader_trait_composer_chosen
						has_trait = leader_trait_instrument_chosen
						has_trait = leader_trait_whisperers_chosen
						has_base_skill > 6
					}
				}
				kill_leader = {
					show_notification = yes
					fire = yes
				}
			}
			if = {
				limit = {
					num_owned_leaders > 10
				}
				random_owned_leader = {
					limit = {
						is_ruler = no
						NOR = {
							has_leader_flag = renowned_leader
							has_leader_flag = legendary_leader
							has_trait = trait_imperial_heir
							has_trait = leader_trait_chosen
							has_trait = leader_trait_eater_chosen
							has_trait = leader_trait_composer_chosen
							has_trait = leader_trait_instrument_chosen
							has_trait = leader_trait_whisperers_chosen
							has_base_skill > 6
						}
					}
					kill_leader = {
						show_notification = yes
						fire = yes
					}
				}
			}
			while = {
				count = hskv_num_targeted_ships
				random_owned_ship = {
					limit = {
						is_designable = yes
						is_in_combat = no
						is_owned_by = PREV
						OR = {
							is_ship_size = corvette
							is_ship_size = frigate
							is_ship_size = destroyer
							is_ship_size = cruiser
						}
					}
					delete_ship = this
				}
			}
		}
	}
}
