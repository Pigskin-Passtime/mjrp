# Eve Online time
country_event = {
	id = hskv_hunt.1000
	title = hskv_hunt.1000.name
	desc = hskv_hunt.1000.desc
	picture = GFX_evt_deficit
	show_sound = event_ship_bridge
	is_triggered_only = yes

	immediate = {
		set_country_flag = hskv_ignore_dismiss_effect
		set_variable = {
			which = hskv_total_num_valid_ships
			value = 0
		}
		every_owned_ship = {
			limit = {
				is_designable = yes
				is_in_combat = no
				is_owned_by = PREV
				OR = {
					is_ship_size = constructor
					is_ship_size = corvette
					is_ship_size = frigate
					is_ship_size = destroyer
					is_ship_size = cruiser
				}
			}
			change_variable = {
				which = hskv_total_num_valid_ships
				value = 1
			}
		}
		set_variable = {
			which = hskv_num_targeted_ships
			value = hskv_total_num_valid_ships
		}
		divide_variable = {
			which = hskv_num_targeted_ships
			value = 5
		}
		if = {
			limit = {
				is_at_war = yes
			}
			divide_variable = {
				which = hskv_num_targeted_ships
				value = 4
			}
		}
	}


	option = {
		name = hskv_hunt.1000.response
		custom_tooltip = hskv_hunt.1000.response.tooltip
		hidden_effect = {
			country_event = {
				id = hskv_hunt.1001
			}
		}
		owner = {
			add_resource = {
				energy = -1000
				minerals = -1000
				consumer_goods = -1000
				alloys = -1000
				rare_crystals = -100
				exotic_gases = -100
				volatile_motes = -100
			}
			random_owned_leader = {
				limit = {
					has_leader_flag = hskv_thayo_everdawn_leader
				}
				kill_leader = {
					show_notification = no
				}
			}
			
			random_owned_leader = {
				limit = {
					is_ruler = no
					NOR = {
						has_leader_flag = renowned_leader
						has_leader_flag = legendary_leader
					}
				}
				kill_leader = {
					show_notification = no
				}
			} 
			if = {
				limit = {
					num_owned_leaders > 12
				}
				random_owned_leader = {
					limit = {
						is_ruler = no
						NOR = {
							has_leader_flag = renowned_leader
							has_leader_flag = legendary_leader
						}
					}
					kill_leader = {
						show_notification = no
					}
				} 
			}
			while = {
				count = hskv_num_targeted_ships
				random_owned_ship = {
					limit = {
						is_designable = yes
						is_in_combat = no
						is_owned_by = PREV
						OR = {
							is_ship_size = constructor
							is_ship_size = corvette
							is_ship_size = frigate
							is_ship_size = destroyer
							is_ship_size = cruiser
						}
					}
					delete_ship = this
				}
			}
		}
	}
}
