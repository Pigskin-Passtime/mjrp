# Trigger AOR UI
country_event = {
	id = mjarchitect_ui.0
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		NOT = {
			has_country_flag = mjarchitect_ui_opened
		}
		from = {
			is_country_type = mjarchitect_entity_country
		}
	}

	immediate = {
		set_country_flag = mjarchitect_ui_opened
		if = {
			limit = {
				has_country_flag = mjarchitect_initial_chain_complete
			}
			country_event = {
				id = mjarchitect_ui.100 # Check all values
			}
			country_event = {
				id = mjarchitect_ui.1
				scopes = {
					from = root
				}
			}
		}
		else = {
			country_event = {
				id = mjarchitect_ui.2
			}
		}
	}
}

event = {
	id = mjarchitect_ui.105
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		any_playable_country = {
			has_country_flag = mjarchitect_initial_chain_complete
		}
	}

	immediate = {
		every_playable_country = {
			limit = {
				has_country_flag = mjarchitect_initial_chain_complete
			}

			subtract_variable = {
				which = mjarchitect_satisfaction_display
				value = mjarchitect_satisfaction_decay
			}
			subtract_variable = {
				which = mjarchitect_satisfaction
				value = mjarchitect_satisfaction_decay
			}

			mjarchitect_check_satisfaction = yes
		}
	}
}

# GATEKEEPER - Check Architect's Requests
country_event = {
	id = mjarchitect_ui.110
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		has_country_flag = mjarchitect_initial_chain_complete
	}

	immediate = {
		# Set new request if none already active
		if = {
			limit = {
				NOT = { mjarchitect_any_request_active = yes }
			}
			mjarchitect_set_new_request = yes
		}
		else = {
			# Check values
			mjarchitect_check_request_progress = yes
			# If request exists and target met, trigger success event
			if = {
				limit = {
					check_variable = {
						which = mjarchitect_request_current_num
						value >= mjarchitect_request_target_num
					}
				}
				country_event = {
					id = mjarchitect_ui.115
				}
				country_event = {
					id = mjarchitect_ui.110
					days = 60
					random = 30
				}
			}
			# Otherwise trigger fail event
			else = {
				country_event = {
					id = mjarchitect_ui.120
				}
				country_event = {
					id = mjarchitect_ui.110
					days = 60
					random = 30
				}
			}
		}
	}
}

# Request Period Over - SUCCESS
country_event = {
	id = mjarchitect_ui.115
	is_triggered_only = yes
	diplomatic = yes
	custom_gui = "enclave_artist_window"
	picture_event_data = {
		portrait = shroud1
		room = cosmic_storm_room
    }

	title = mjarchitect_unlock.10.name
	desc = mjarchitect_ui.115.desc
	
	trigger = {
		NOR = {
			has_country_flag = mjarchitect_request_active
			has_country_flag = mjarchitect_interactions_blocked
		}
	}

	after = {
		hidden_effect = {
			mjarchitect_satisfaction_reward_effect_static = {
				SATISFACTION_VALUE = 10
			}
		}
	}

	immediate = {
		set_timed_country_flag = {
			flag = mjarchitect_recently_completed_request
			years = 5
		}
		if = {
			limit = { has_country_flag = mjarchitect_blessing_5_unlocked }
			change_variable = {
				which = mjarchitect_blessing8_requests_counter
				value = 1
			}
			remove_modifier = mjarchitect_blessing_request_boon
			add_modifier = {
				modifier = mjarchitect_blessing_request_boon
				multiplier = mjarchitect_blessing8_requests_counter
			}
		}
		else = {}

		mjarchitect_clear_reward_flags = yes
		if = {
			limit = {
				has_country_flag = mjarchitect_blessing_6_unlocked 
			}
			while = {
				count = 3
				mjarchitect_generate_reward_options = yes
			}
		}
		else = {
			while = {
				count = 2
				mjarchitect_generate_reward_options = yes
			}
		}
	}

	inline_script = {
		script = mjarchitect/mjarchitect_reward_options
	}
}
