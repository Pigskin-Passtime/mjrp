# Monthly check for CM stockpile
event = {
	id = mjarchitect_end.50
	is_triggered_only = yes
	hide_window = yes
	
	trigger = {
		exists = event_target:mjarchitect_crisis_empire
	}

	immediate = {
		event_target:mjarchitect_crisis_empire = {
			export_resource_stockpile_to_variable = {
				resource = mjarchitect_cognitive_matter
				variable = mjarchitect_num_cognitive_matter
			}

			if = {
				limit = {
					OR = {
						has_country_flag = mjarchitect_path_2_sp_completed
						has_country_flag = mjarchitect_path_3_sp_completed
					}
				}
				country_event = {
					id = mjarchitect_end.130
				}
			}
			else = {}

			set_variable = {
				which = mjarchitect_num_cognitive_matter_situation_progress
				value = mjarchitect_num_cognitive_matter
			}
			divide_variable = {
				which = mjarchitect_num_cognitive_matter_situation_progress
				value = 100000
			}

			if = {
				limit = {
					check_variable = {
						which = mjarchitect_num_cognitive_matter_situation_progress
						value > 1
					}
				}
				set_variable = {
					which = mjarchitect_num_cognitive_matter_situation_progress
					value = 1
				}
			}
		}
	}
}

	trigger = {

					}

					which = mjarchitect_num_cognitive_matter_usurpation

				}
			}

			add_modifier = {
				modifier = mjarchitect_usurpation_boon
				multiplier = mjarchitect_num_cognitive_matter_usurpation
			}
	}
}

country_event = {
	id = mjarchitect_end.145
	is_triggered_only = yes
	title = "mjarchitect_end.145.name"
	desc = "mjarchitect_end.145.desc"
	picture = GFX_evt_cosmic_storms_gravity_storm
	
	trigger = {
		mjarchitect_is_crisis = yes
		from = { is_country_type = mjarchitect_entity_country }
		fromfromfrom = { has_ship_flag = mjarchitect_architect_ship }
	}

	immediate = {
		every_playable_country = {
			limit = {
				NOT = { is_same_value = root }
			}
			country_event = {
				id = mjarchitect_end.155
			}
		}
	}

	option = {
		name = "mjarchitect_end.145.a" # Keep the Microcosm
		custom_tooltip = mjarchitect_end.145.tt2
		add_relic = r_mjarchitect_microcosm
		add_monthly_resource_mult = {
				resource = energy
				value = 100
				min = 25000
				max = 50000
		}
		add_monthly_resource_mult = {
				resource = minerals
				value = 100
				min = 25000
				max = 50000
		}
		every_country = {
			set_truce = {
				target = root
				type = war
			}
			add_opinion_modifier = {
				modifier = mjarchitect_microcosm_architect_opinion_modifier
				who = root
			}
		}
		hidden_effect = {
			every_playable_country = {
				limit = {
					NOT = { is_same_value = root }
				}
				set_subject_of = {
					who = root
					preset = preset_vassal
					allow_instant_negotiation = no
				}
			}
			every_agreement = {
				add_loyalty = 300
			}
		}
		win = yes
	}

	option = {
		name = "mjarchitect_end.145.b" # Put it somewhere it will never be found
		custom_tooltip = mjarchitect_end.145.tt
		add_monthly_resource_mult = {
				resource = influence
				value = 240
				min = 500
				max = 1000
		}
		add_monthly_resource_mult = {
				resource = unity
				value = 240
				min = 100000
				max = 900000
		}
		add_monthly_resource_mult = {
				resource = energy
				value = 100
				min = 25000
				max = 50000
		}
		add_monthly_resource_mult = {
				resource = minerals
				value = 100
				min = 25000
				max = 50000
		}
		every_country = {
			set_truce = {
				target = root
				type = war
			}
			add_opinion_modifier = {
				modifier = mjarchitect_stopped_architect_opinion_modifier
				who = root
			}
		}
	}

	after = {
		end_event_chain = "mjarchitect_chain"
		hidden_effect = {
			set_global_flag = mjarchitect_killed
			random_country = {
				limit = {
					is_country_type = mjarchitect_entity_country
				}
				destroy_country = yes
			}
		}
	}
}
