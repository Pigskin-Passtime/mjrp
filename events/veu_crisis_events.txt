# Pile is destroyed
system_event = {
	id = veu_crisis.6100
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		any_system_megastructure = { has_megastructure_flag = veu_pile }
	}

	immediate = {
		save_event_target_as = veu_pile_system
		random_system_megastructure = {
			limit = { has_megastructure_flag = veu_pile }
			prev = {
				spawn_megastructure = {
					type = veu_pile_ruined
					coords_from = prev
					init_effect = {
						if = {
							limit = { exists = fromfrom }
							set_megastructure_flag = veu_pile_of@fromfrom
						}
					}
				}
			}
			switch = {
				trigger = is_megastructure_type
				veu_pile_0 = {
					from = {
						set_variable = {
							which = veu_pile_destroy_reward
							value = 1
						}
					}
				}
				veu_pile_1 = {
					from = {
						set_variable = {
							which = veu_pile_destroy_reward
							value = 1.5
						}
					}
				}
				veu_pile_2 = {
					from = {
						set_variable = {
							which = veu_pile_destroy_reward
							value = 2
						}
					}
				}
				veu_pile_3 = {
					from = {
						set_variable = {
							which = veu_pile_destroy_reward
							value = 3
						}
					}
				}
			}
			remove_megastructure = this
		}
		from = { save_event_target_as = pile_destroyers }
		fromfrom = { save_event_target_as = pile_builders }
		from = { country_event = { id = veu_crisis.6102 } }
		fromfrom = { country_event = { id = veu_crisis.6101 } }
		every_playable_country = {
			limit = {
				is_ai = no
				NOR = {
					is_same_value = from
					is_same_value = fromfrom
				}
			}
			country_event = { id = veu_crisis.6103 }
		}
	}
}

namespace = veu_bombardment

planet_event = {
	id = veu_bombardment.1001
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_orbital_bombardment_stance = veu_waste_bombing
		OR = {
			AND = {
				planet_devastation >= 10
				NOT = { is_variable_set = veu_bomb_blockers }
			}
			AND = {
				is_variable_set = veu_bomb_blockers
				check_variable_arithmetic = {
					which = veu_bomb_blockers
					add = 1
					divide = trigger:planet_devastation
					value <= 0.1
				}
			}
		}
	}

	immediate = {
		# Set variables
		if = {
			limit = {
				NOT = { is_variable_set = veu_bomb_blockers }
			}
			set_variable = {
				which = veu_bomb_blockers
				value = 1
			}
		}
		else = {
			change_variable = {
				which = veu_bomb_blockers
				value = 1
			}
		}
		# Add the blockers
			add_blocker = {
				type = d_veu_bombing_blocker
				blocked_deposit = random
			}
	}
}
