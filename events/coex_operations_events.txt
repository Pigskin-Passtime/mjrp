espionage_operation_event = {
	id = coex_operation.303
	title = coex_operation.303.name
	desc = coex_operation.303.desc
	picture = GFX_evt_one_sided_deal
	show_sound = event_espionage_concluded
	espionage_operation = yes
	is_triggered_only = yes
	immediate = {
		set_espionage_operation_progress_locked = yes
	}
	option = {
		name = LAUNCH_OPERATION
		target = {
			random_system_within_border = {
				limit = {
					has_astral_rift = no
					has_no_non_gate_megastructure = yes
					# count_archaeological_site = {
					# 	limit = {
					# 		is_site_completed = no
					# 	}
					# 	count = 0
					# }
					NOR = {
						any_system_planet = {
							is_astral_scar = yes
						}
						any_system_megastructure = {
							OR = {
								is_megastructure_type = gateway_0
								is_megastructure_type = gateway_final
								is_megastructure_type = gateway_derelict
								is_megastructure_type = gateway_restored
								is_megastructure_type = gateway_ruined
								is_megastructure_type = coex_brain_relay_0
								is_megastructure_type = coex_brain_relay_1
							}
						}
						any_system_planet = {
							OR = {
								is_planet_class = pc_cosmogenesis_world
								is_colonizable = yes
								is_colony = yes
							}
						}
					}
				}
				weights = {
					base = 1
					modifier = {
						factor = 0.2
						has_natural_wormhole = yes
					}
					modifier = {
						factor = 0.3
						starbase = {
							has_starbase_size = starbase_starport
						}
					}
					modifier = {
						factor = 0.01
						starbase = {
							has_starbase_size = starbase_starfortress
						}
					}
					modifier = {
						factor = 0.001
						starbase = {
							has_starbase_size = starbase_citadel
						}
					}
				}
				starbase = {
					set_owner = root.owner
				}
				random_system_planet = {
					save_event_target_as = target_planet
				}
				spawn_megastructure = {
					type = coex_brain_relay_0
					owner = root.owner
					planet = event_target:target_planet
				}
			}
		}
		owner = {
			hidden_effect = {
			if = {
				limit = {
					exists = event_target:target_planet
				}
				country_event = {
					id = coex_operation.310
				}
				prev.target = {
					country_event = {
						id = coex_operation.320
					}
					set_country_flag = coex_espionage_relay_installed
				}
				else = {
					country_event = {
						id = coex_operation.311
					}
				}
			}}
			complete_crisis_objective = coex_crisobj_espionage
		}
	}
	after = {
		set_espionage_operation_progress_locked = no
	}
}
