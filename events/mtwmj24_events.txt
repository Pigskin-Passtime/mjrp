country_event = {
	id = mtwmj24_warden.0100
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = mtwmj24_ongoing_shroud_attack
	}

	immediate = {
		if = {
			limit = { NOT = { exists = event_target:shroud_country } }
			create_country = {
				name = "NAME_Creatures_of_the_Shroud"
				type = shroud_spirits
				flag = {
					icon = {
						category = "special"
						file = "the_shroud.dds"
					}
					background= {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors={
						"dark_purple"
						"black"
						"null"
						"null"
					}
				}
			}
			last_created_country = {
				establish_communications_no_message = root
				save_global_event_target_as = shroud_country
			}
		}

		if = {
			limit = {
				NOT = {
					is_variable_set = shroud_avatar_spawn_rate					
				}
			}
			set_variable = {
				which = shroud_avatar_spawn_rate
				value = 0
			}
		}

		change_variable = {
			which = shroud_avatar_spawn_rate
			value = 1
		}

		while = {
			count = shroud_avatar_spawn_rate
			random_system = {
				random_system_planet = {
					save_event_target_as = shroud_spawn_planet
				}
				create_fleet = {
					name = "NAME_Corrupted_Avatar"
					settings = {
						spawn_debris = no
						is_boss = yes
					}
					effect = {
						set_owner = event_target:shroud_country
						create_ship = {
							name = "NAME_Corrupted_Avatar"
							design = "NAME_Corrupted_Avatar"
						}
						set_location = event_target:shroud_spawn_planet
						set_fleet_stance = aggressive
						set_aggro_range_measure_from = return_point
						set_aggro_range = 300
					}
				}
			}
			if = {
				limit = { 
					exists = event_target:mtwmj24_warden_country 
				}
				event_target:mtwmj24_warden_country = {
					random_system_within_border = {
						random_system_planet = {
							save_event_target_as = shroud_spawn_planet
						}
						create_fleet = {
							name = "NAME_Corrupted_Avatar"
							settings = {
								spawn_debris = no
								is_boss = yes
							}
							effect = {
								set_owner = event_target:shroud_country
								create_ship = {
									name = "NAME_Corrupted_Avatar"
									design = "NAME_Corrupted_Avatar"
								}
								set_location = event_target:shroud_spawn_planet
								set_fleet_stance = aggressive
								set_aggro_range_measure_from = return_point
								set_aggro_range = 300
							}
						}
					}
				}
			}

				## we recursively spawn more shroud avatars, increasing the amount of avatars spawned with each wave
				## This continues until at least 10 Waves have been spawned, and then has a small chance to disappear
				## The Player with the Ascended Legendary Leader can choose to sacrifice the leader to end this attack

			random_country = {
				limit = {
					is_country_type = global_event
				}
				random_list = {
					10 = {
						modifier = {
							factor = 0
							check_variable = {
								which = shroud_avatar_spawn_rate
								value < 10
							}
						}
						remove_global_flag = mtwmj24_ongoing_shroud_attack
					}
					90 = {
						country_event = {
							days = 30
							random = 330
							id = mtwmj24_warden.0100
						}
					}
				}
			}
		}

	}
}
