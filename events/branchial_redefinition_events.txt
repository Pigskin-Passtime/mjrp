country_event = { # Remove Negative Traits
 	id = branchial_redefinition.111
 	title = "branchial_redefinition.1.name"
 	desc = "branchial_redefinition.111.desc"

 	diplomatic = yes

 	picture_event_data = {
 		portrait = root
 		planet_background = root.capital_scope
 		graphical_culture = root
 		room = root
 	}

 	is_triggered_only = yes

	immediate = {
		random_owned_leader = {
			limit = {
				num_leader_traits = {
					value >= 1
					negative = yes
				}
			}
			save_event_target_as = branchial_leader_1
		}
		random_owned_leader = {
			limit = {
				NOR = {
					is_same_value = event_target:branchial_leader_1
				}
				num_leader_traits = {
					value >= 1
					negative = yes
				}
			}
			save_event_target_as = branchial_leader_2
		}
		random_owned_leader = {
			limit = {
				NOR = {
					is_same_value = event_target:branchial_leader_1
					is_same_value = event_target:branchial_leader_2
				}
				num_leader_traits = {
					value >= 1
					negative = yes
				}
			}
			save_event_target_as = branchial_leader_3
		}
	}

	option = {
		name = event_target:branchial_leader_1.name
		trigger = {
			exists = event_target:branchial_leader_1
		}
		allow = {
			AND = {
				resource_stockpile_compare = {
					resource = energy
					value >= 1000
					mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
				}
				resource_stockpile_compare = {
					resource = unity
					value >= 500
				}
				resource_stockpile_compare = {
					resource = branchial_proficiency
					value >= 250
					mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
				}
			}
		}

		event_target:branchial_leader_1 = {
			save_event_target_as = branchial_selected_leader
		}
	}

	option = {
		name = event_target:branchial_leader_2.name
		trigger = {
			exists = event_target:branchial_leader_2
		}
		allow = {
			AND = {
				resource_stockpile_compare = {
					resource = energy
					value >= 1000
					mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
				}
				resource_stockpile_compare = {
					resource = unity
					value >= 500
				}
				resource_stockpile_compare = {
					resource = branchial_proficiency
					value >= 250
					mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
				}
			}
		}

		event_target:branchial_leader_2 = {
			save_event_target_as = branchial_selected_leader
		}
	}

	option = {
		name = event_target:branchial_leader_3.name
		trigger = {
			exists = event_target:branchial_leader_3
		}
		allow = {
			AND = {
				resource_stockpile_compare = {
					resource = energy
					value >= 1000
					mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
				}
				resource_stockpile_compare = {
					resource = unity
					value >= 500
				}
				resource_stockpile_compare = {
					resource = branchial_proficiency
					value >= 250
					mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
				}
			}
		}

		event_target:branchial_leader_3 = {
			save_event_target_as = branchial_selected_leader
		}
	}

	option = { # Reroll
		name = BRANCHIAL_REROLL
		hidden_effect = {
			country_event = { id = branchial_redefinition.111 }
		}
	}

	option = { # Back
 		name = SETTINGS_BACK
 		hidden_effect = {
 			country_event = { id = branchial_redefinition.11 }
 		}
 	}

	after = {
		if = {
			limit = {
				exists = event_target:branchial_selected_leader
			}
			add_resource = {
				energy = -1000
				mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
			}
			add_resource = {
				unity = -500
			}
			add_resource = {
				branchial_proficiency = -250
				mult = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cost_base|
			}
			add_modifier = {
				modifier = "branchial_redefinition_cooldown"
				days = 720
				time_multiplier = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_cooldown_mult|
			}

			event_target:branchial_selected_leader = {
				random_list = {
					90 = { # Success
						leader_event = { branchial_redefinition.1111 }
					}
					10 = { # Failure
						random_list = {
							75 = {
								leader_event = { branchial_redefinition.115 }
							}
							25 = {
								owner.capital_scope = {
									planet_event = { id = branchial_guardian.6 }
								}
							}
						}
						modifier = {
							factor = value:scripted_modifier_mult|MODIFIER|branchial_redefinition_failure_risk_mult|
						}
					}
				}
			}
		}
	}
}
