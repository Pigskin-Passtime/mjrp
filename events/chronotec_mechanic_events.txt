# Creates country and leader at game start
	event = {
		id = chronotec_leader.1
		hide_window = yes
		fire_only_once = yes
		is_triggered_only = yes
		
		trigger = {
			chronotec_trigger = yes
		}
		
		immediate = {
			# Country Created
			create_country = {
				name = "NAME_chronotec_chronolord"
				type = chronotec_event_country
				flag = {
					icon = {
						category = "spherical"
						file = "flag_spherical_20.dds"
					}
					background= {
						category = "backgrounds"
						file = "00_solid.dds"
					}
					colors={
						"purple"
						"black"
						"null"
						"null"
					}
				}
				effect = {
					set_country_flag = suppress_first_contact_comms
					save_global_event_target_as = chronotec_chronolord_country
				}
			}
			# Leader created
			event_target:chronotec_chronolord_country = {
				create_species = {
					name = "NAME_Unknown"
					class = PARAGON
					portrait = "chronotec_chronolord_portrait_standard"
					namelist = "PRT1"
					traits = random
					effect = {
						save_event_target_as = chronotec_chronolord_species
					}
				}
				create_leader = {
					species = event_target:chronotec_chronolord_species
					event_leader = yes
					class = scientist
					name = "NAME_chronotec_chronolord"
					skill = 10
					immortal = yes
					background_ethic = ethic_materialist
					randomize_traits = no
					skip_background_generation = yes
					custom_catch_phrase = chronotec_chronolord_catch_phrase
					custom_description = chronotec_chronolord_backstory
					traits = {
						trait = leader_trait_physics_focus_3
						trait = leader_trait_collaborator_3
						trait = leader_trait_chronotec_chronolord_infusion_neutral
						trait = subclass_scientist_governor
					}
					gender = indeterminable
					set_age = 0
					hide_age = yes
					effect = {
						save_global_event_target_as = chronotec_chronolord_leader
						set_leader_flag = legendary_leader
						set_leader_flag = chronotec_chronolord_leader
						set_leader_flag = immune_to_negative_traits
						set_leader_flag = leader_death_events_blocked
						set_owner = event_target:chronotec_chronolord_country
						exile_leader_as = chronotec_chronolord_leader
					}
				}
			}
		}
	}

		id = chronotec_leader.10
		hide_window = yes
		show_sound = no
		is_triggered_only = yes

		trigger = {
			from = { is_country_type = chronotec_event_country }
			NOT = {
				has_country_flag = chronotec_diplomacy_engaged
			}
		}

		immediate = {
			hidden_effect = {
				set_country_flag = chronotec_diplomacy_engaged
				if = {	# Recruited, but petition is denied
					limit = {
						has_country_flag = chronotec_petition_denied
					}
					country_event = {
						id = chronotec_leader.1010
					}
				}
				else_if = {	# Recruited, but petition is failed
					limit = {
						has_country_flag = chronotec_petition_failed
					}
					country_event = {
						id = chronotec_leader.1020
					}
				}
				else_if = {	# Recruited, but petition is completed
					limit = {
						has_country_flag = chronotec_petition_completed
					}
					country_event = {
						id = chronotec_leader.1030
					}
				}
				else_if = {	# Contact if taken by other empire (former owner)
					limit = {
						NOT = {
							is_same_value = event_target:chronotec_actual_owner_country
						}
					}
					country_event = {
						id = chronotec_leader.1040
					}
				}
				else = {
					country_event = {
						id = chronotec_leader.1030
					}
				}
			}
		}
	}

# Leader death things (30xxx)
 # Leader is restored after death (also as global event target)
	country_event = {	# Saves and Restores ruler on ruler death
		id = chronotec_leader.3010
		is_triggered_only = yes
		hide_window = yes

		trigger = {
			from = {
				has_leader_flag = chronotec_chronolord_leader
				NOT = {	# To avoid duplicated leaders when the planet changes ownership
					has_global_flag = chronotec_chronolord_leaves
				}
			}
		}

		immediate = {
			hidden_effect = {
				clone_leader = {
					target = from
				}
			}
			last_created_leader = {
				copy_flags_and_variables_from = from
				# Redundancy. just in case
				save_global_event_target_as = chronotec_chronolord_leader
				set_leader_flag = legendary_leader
				set_leader_flag = chronotec_chronolord_leader
				set_leader_flag = immune_to_negative_traits
				set_leader_flag = leader_death_events_blocked
			}
			create_message = {
				type = MESSAGE_RECRUITED_LEADER
				localization = chronotec_chronolord_cheated_death
				days = @toast_message_days
				target = event_target:chronotec_chronolord_leader
				variable = {
					type = name
					localization = LEADER
					scope = event_target:chronotec_chronolord_leader
				}
				variable = {
					key = "border"
					value = "GFX_border_legendary"
				}
			}
		}
	}

					limit = {
						has_global_flag = chronotec_corrupted

						has_global_flag = chronotec_perfected
					}
