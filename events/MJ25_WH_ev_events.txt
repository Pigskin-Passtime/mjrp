# Whisper cultists arrive on the planet.
planet_event = {
	id = MJ25_WH.1002
	location = this

	is_triggered_only = yes
	hide_window = yes

	trigger = {
		exists = owner
	}

	immediate = {
		set_planet_flag = MJ25_WH_cultists_set_by_@From.From.Owner
		if = {
			limit = { NOT = { is_variable_set = MJ25_WH_num_whispers_spread } }
			set_variable = {
				which = MJ25_WH_num_whispers_spread
				value = 1
			}
		}
		else = {
			change_variable = {
				which = MJ25_WH_num_whispers_spread
				value = 1
			}
		}
		owner = {
			if = {
				limit = {
					NOT = { has_country_flag = MJ25_WH_cultists_active }
				}
				set_country_flag = MJ25_WH_cultists_active 
			}
		}
		if = {
			limit = {
				owner = { is_gestalt = yes }
			}
			add_modifier = {
				modifier = MJ25_WH_spread_whispers_gestalt
				days = 1800
				mult = MJ25_WH_num_whispers_spread
			}
		}
		else = {
			add_modifier = {
				modifier = MJ25_WH_spread_whispers
				days = 1800
				mult = MJ25_WH_num_whispers_spread
			}
		}
		if = {
			limit = {
				root.owner = {
					NOT = {
						has_country_flag = MJ25_WH_whisper_cult_event_played
					}
				}
			}
			root.owner = {
				set_country_flag = MJ25_WH_whisper_cult_event_played
			}
			planet_event = { id = MJ25_WH_whispers.2004 days = 30 random = 60 }
		}
	}
}

# Spawns Void Echoes if the Aural starmass wins, resets it's counter otherwise.
country_event = {
	id = MJ25_WH.1022

	is_triggered_only = yes
	hide_window = yes

	trigger = {
		OR = {
			has_ascension_perk = MJ25_WH_ap_whispers
			From = { has_ascension_perk = MJ25_WH_ap_whispers }
		}
		OR = {
			FromFrom = {
				any_owned_ship = {
					has_ship_flag = MJ25_WH_aural_starmass
				}
			}
			FromFromFrom = {
				any_owned_ship = {
					has_ship_flag = MJ25_WH_aural_starmass
				}
			}
		}
		
		
	}

	immediate = {
		# If Aural Starmass loses
		if = {
			limit = {
				FromFromFrom = {
					any_owned_ship = {
						has_ship_flag = MJ25_WH_aural_starmass
					}
				}
			}
			FromFromFrom = {
				every_owned_ship = {
					limit = {
						has_ship_flag = MJ25_WH_aural_starmass
					}
					set_variable = { which = MJ25_WH_aural_starmass_ship_counter value = 0 }
				}
			}
		}
		# If Aural Starmass wins
		if = {
			limit = {
				FromFrom = {
					any_owned_ship = {
						has_ship_flag = MJ25_WH_aural_starmass
					}
				}
			}
			FromFrom = {
				root = { set_variable = { which = MJ25_WH_aural_starmass_ship_counter_temp value = 0 } }
				every_owned_ship = {
					limit = {
						has_ship_flag = MJ25_WH_aural_starmass
						is_variable_set = MJ25_WH_aural_starmass_ship_counter
					}
					root = { change_variable = { which = MJ25_WH_aural_starmass_ship_counter_temp value = prev.MJ25_WH_aural_starmass_ship_counter } }
					set_variable = { which = MJ25_WH_aural_starmass_ship_counter value = 0 }
				}
			}
			create_fleet = {
				name = "NAME_MJ25_WH_VOID_ECHO_FLEET"
				settings = {
					spawn_debris = no
				}
				effect = {
					set_owner = root
					while = {
						count = root.MJ25_WH_aural_starmass_ship_counter_temp
						create_ship = {
							name = "NAME_MJ25_WH_VOID_ECHO"
							design = "NAME_MJ25_WH_VOID_ECHO"
							effect = {
								set_ship_flag = MJ25_WH_void_echo
							}
						}
					}
					set_location = {
						target = FromFrom
						distance = 0
						angle = 90
						direction = in_system
					}
					set_fleet_stance = aggressive
					set_aggro_range_measure_from = self
					set_aggro_range = 30000
					set_fleet_bombardment_stance = selective
				}
			}
		}
	}
}

# Conductor confusion
fleet_event = {
	id = MJ25_WH.1041

	is_triggered_only = yes
	hide_window = yes

	immediate = {
		solar_system = {
			random_fleet_in_system = {
				limit = { 
					NOT = { is_same_value = root } 
					is_ship_class = shipclass_military 
					is_in_combat = no
					exists = owner
					owner = {
						OR = {
							NOT = { is_variable_set = MJ25_WH_num_confusions }
							AND = {
								is_variable_set = MJ25_WH_num_confusions
								check_variable = { which = MJ25_WH_num_confusions value < 2 }
							}
						}
						
					}
				}

				# Check to ensure Fallen Empire is not been cheesed too much.
				if = {
					limit = { 
						exists = owner
						owner = { 
							OR = {
								is_country_type = fallen_empire
								is_country_type = awakened_fallen_empire
							}
						}
					}
					owner = {
						if = {
							limit = { NOT = { is_variable_set = MJ25_WH_num_confusions } }
							set_variable = { which = MJ25_WH_num_confusions value = 0 }
						}
						change_variable = { which = MJ25_WH_num_confusions value = 1 }
						if = {
							limit = { check_variable = { which = MJ25_WH_num_confusions value > 1 } }
							save_global_event_target_as = MJ25_WH_temp_fe
							root = { owner = { country_event = { id = MJ25_WH.1170 days = 35 } } }
						}
					}
				}

				create_country = {
					name = "NAME_MJ25_WH_confused_country"
					type = behemoth_hatchling
					flag = {
						icon = {
							category = "zoological"
							file = "flag_zoological_10.dds"
						}
						background= {
							category = "backgrounds"
							file = "new_dawn.dds"
						}
						colors={
							"black"
							"black"
							"null"
							"null"
						}
					}
					effect = {
						every_country = {
							establish_communications_no_message = prev
						}
						save_event_target_as = MJ25_WH_temp_confusion_country
					}
				}
				set_fleet_flag = MJ25_WH_former_owner@owner
				set_owner = event_target:MJ25_WH_temp_confusion_country
				fleet_event = { id = MJ25_WH.1042 days = 30 }

				
			}
		}
	}
}
