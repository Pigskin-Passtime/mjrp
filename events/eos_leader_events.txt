# Border reinforcemnet
country_event = {
    id = eos.1150
    is_triggered_only = yes

    title = eos_event_checkpoint

    immediate = {
        random_system_within_border = {
            limit = {
                any_neighbor_system = {
                    log = "[This.GetName]"
                    not = { is_owned_by = Root }

                }
            }

            create_message = { localization = "UwU [This.GetName]"  recipient = Root days = 69 type = MESSAGE_FRUITFUL_SEED_CRITTER }
            create_fleet = {
                settings = { can_change_composition = no can_change_leader = no can_disband = no can_upgrade = no }
                effect = {
                    set_owner = root
                    set_location = { target = prev.system_star angle = random distance = 100 }
                    create_ship = { random_existing_design = eos_border_checkpoint }
                }

            }
        }
    }
}


country_event = {
    id = eos.1151
    is_triggered_only = yes

    title = OK
    hide_window = yes

    trigger = {
        has_global_flag = eos_leader_spawned
        any_controlled_fleet = {
            is_ship_size = eos_border_checkpoint
        }
    }

    immediate = {

        while = {
            reroll_random = yes
            random_controlled_fleet = {

                #Get a checkpoint in a system with no borders (why is logic hard)
                limit = {

                    is_ship_size = eos_border_checkpoint
                    closest_system = {
                        max_steps = 0
                        limit = {
                            not = {
                                any_neighbor_system = {
                                    not = {
                                        is_owned_by = Root
                                    }
                                }
                            }
                        }
                        log = "[This.GetName]"
                    }
                }


                #Put the checkpoint in its place (on the border)
                root = {
                    random_system_within_border = {
                        limit = {
                            any_neighbor_system = {
                                log = "[This.GetName]"
                                not = { is_owned_by = Root }

                            }
                        }
                        create_message = { type = MESSAGE_FRUITFUL_SEED_PLANTED localization = "[This.GetName]" days = 34 }
                        prevprev = {
                            set_location = { target = prev.system_star angle = random distance = 100 }
                        }
                    }
                }

            }
        }
    }
}
