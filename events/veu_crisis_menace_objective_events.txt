namespace = veu_crisis

event = {
	id = veu_crisis.7001
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		any_playable_country = {
			has_ascension_perk = veu_ap_veu
		}
	}

	immediate = {
		every_playable_country = {
			limit = {
				has_ascension_perk = veu_ap_veu
			}
			if = {
				limit = {
					NOT = { is_variable_set = veu_waste_produced }

				}
				set_variable = {
					which = veu_waste_produced
					value = 0
				}
			}
			export_resource_income_to_variable = {
				resource = veu_waste
				variable = veu_waste_produced
			}
			complete_crisis_objective = veu_crisobj_building
			add_resource = {
    			veu_waste = -1
   				multiplier = veu_waste_produced
			}
		}
	}
}
