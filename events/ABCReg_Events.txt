			}
		}
	}

		name = HIRE
		tag = hire_leader
		custom_tooltip = ABCReg_leader_joins
		sound = par_ABCReg_leader_hire

			set_country_flag = pacified_amoebas  #amoeba pacified

			create_fleet = {
				name = "NAME_Gargantuan_Amoeba_Life-form"
				settings = {
					can_upgrade = no
					can_change_composition = no
					can_change_leader = no
					can_disband = no
					spawn_debris = no

						design = last_created_design
						name = "NAME_Gargantuan_Amoeba_Life-form"

						upgradable = no

							set_ship_flag = is_ABCReg_ship
							save_global_event_target_as = ABCReg_ship
							add_modifier = { modifier = bubble_conclave_modifier days = -1 }
							add_modifier = { modifier = bubble_pirate_modifier days = -1 }
							add_modifier = { modifier = bubble_imperium_modifier days = -1 }
							add_modifier = { modifier = bubble_federation_modifier days = -1 }
							add_modifier = { modifier = bubble_angry_modifier days = -1 }
							add_modifier = { modifier = bubble_safe_belt_modifier days = -1 }
						}
					}
					if = {
						limit = {
							exists = event_target:ABCReg_star     
						}
						set_location = {
							target = event_target:ABCReg_star
							distance = 1
							angle = random
						}
					}
					else = {
						set_location = {
							target = root.capital_scope
							distance = 1
							angle = random
						}
					}

					set_leader = event_target:ABCReg_Leader
				}
			}

				type = MESSAGE_RECRUITED_LEADER
				localization =  MESSAGE_RECRUITED_LEADER_DESC

					type = name
					localization = LEADER
					scope = event_target:ABCReg_Leader
				}
				variable = {
					key = "border"
					value = "GFX_border_legendary"
				}
			}
			country_event = { id = ABCReg.5 days = 1 random = 1 }
			hidden_effect = {
				event_target:ABCReg_cyto = {
						delete_ship = this
			}				}
		}
	}

	option = { # DISMISS
		name = DISMISS
		tag = dismiss_leader
		sound = par_ABCReg_dismiss

		hidden_effect = {
			event_target:ABCReg_Leader = {
				kill_leader = { show_notification = no }
