country_event = { # Revive Reinhard on death, anywhere between 5 and 10 years
	id = dmj_leader.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = event_target:dmj_daniel
		from = { is_same_value = event_target:dmj_daniel }
		has_global_flag = dmj_reinhard_free
		has_country_flag = owns_reinhard
	}

	immediate = {
		if = {
			limit = { has_origin = "origin_legendary_leader_death" }
			if = {
				limit = { has_authority = auth_dictatorial }
				set_origin = "origin_legendary_leader_dictatorial"
			}
			else_if = {
				limit = { has_authority = auth_imperial }
				set_origin = "origin_legendary_leader_imperial"
			}
		}
		country_event = { id = dmj_leader.2 days = 1800 random = 1800 }
	}
}

country_event = {
	id = dmj_leader.2
	title = "dmj_leader.2.name"
	desc = "dmj_leader.2.desc"
	is_triggered_only = yes

	picture = GFX_evt_psionics
	show_sound = event_mystic_reveal

	trigger = {
		has_global_flag = dmj_reinhard_free
		has_country_flag = owns_reinhard
	}

	immediate = {
		create_country = {
			name = "NAME_dmj_daniel"
			type = paragon_event_country
			flag = {
				icon = {
					category = "special"
					file = "unknown.dds"
				}
				background = {
					category = "backgrounds"
					file = "00_solid.dds"
				}
				colors = {
					"yellow"
					"yellow"
					"null"
					"null"
				}
			}
			effect = {
				save_global_event_target_as = dmj_daniel_country
			}
		}
		event_target:dmj_daniel_country = {
			set_country_flag = dmj_daniel_country_flag
			set_faction_hostility = {
				target = root.owner
				set_hostile = no
				set_neutral = yes
				set_friendly = no
			}
			create_species = {
				name = NAME_Human
				plural = NAME_Human_plural
				class = "HUM"
				portrait = dmj_daniel
				traits = {
					trait = "trait_adaptive"
					trait = "trait_nomadic"
					trait = "trait_wasteful"
				}
			}
			last_created_species = {
				save_event_target_as = dmj_daniel_species
			}
			create_leader = {
				class = official
				species = event_target:dmj_daniel_species
				name = "NAME_dmj_daniel"
				skill = 1
				gender = male
				hide_age = yes
				event_leader = yes
				skip_background_generation = yes
				background_ethic = ethic_authoritarian
				background_planet = event_target:sol_system_earth
				custom_catch_phrase = dmj_daniel_catch_phrase
				custom_description = dmj_daniel_desc
				randomize_traits = no
				effect = {
					set_leader_flag = legendary_leader
				}
			}
			last_created_leader = {
				set_leader_flag = dmj_daniel
				set_leader_flag = immune_to_negative_traits
				change_leader_portrait = dmj_daniel
				save_global_event_target_as = dmj_daniel
				set_owner = event_target:dmj_daniel_country
				add_skill_without_trait_selection = 9
				remove_all_traits = yes
				add_trait = leader_trait_chosen
				add_trait = leader_trait_devotion_conduit
				add_trait = leader_trait_supreme_organizer
				add_trait = leader_trait_legendary_military_knowledge_2
				add_trait = leader_trait_legendary_evervigilant_2
				add_trait = leader_trait_legendary_high_king_2
				add_trait = leader_trait_tyrannical
				add_trait = subclass_official_economy_councilor
				set_immortal = yes
				set_variable = { which = dmj_devotion_value_leader value = root.dmj_devotion_value }
				set_owner = root
			}
		}
		event_target:dmj_daniel_country = {
			destroy_country = yes
		}
		if = {
			limit = { has_country_flag = reinhardist_empire }
			set_leader = event_target:dmj_daniel
		}
	}

	option = {
		name = "INCREDIBLE"
	}
}

country_event = { # Send Reinhard to another country when fired, anywhere between 5 and 10 years
	id = dmj_leader.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = event_target:dmj_daniel
		from = { is_same_value = event_target:dmj_daniel }
		has_global_flag = dmj_reinhard_free
		has_country_flag = owns_reinhard
	}

	immediate = {
		remove_country_flag = owns_reinhard
		random_playable_country = {
			limit = {
				NOT = { has_country_flag = owns_reinhard }
				NOT = { has_country_flag = dmj_reimprisoned_reinhard }
				NOT = { is_same_value = event_target:dmj_sol_country }
				dmj_trigger = yes
				dmj_soft_trigger = yes
			}
			country_event = { id = dmj_leader.4 days = 1800 random = 1800 }
		}
	}
}
