country_event = {
    id = pdivinity_crisis.0
    title = pdivinity_crisis.0.name
    desc = pdivinity_crisis.0.desc
    diplomatic = yes
    diplomatic_title = BLANK_STRING
    force_open = yes

    picture_event_data = {
        portrait = shroud1
        room = shroud_room
    }

    is_triggered_only = yes

    option = {
        name = pdivinity_crisis.0.a
        begin_event_chain = {
            event_chain = project_divinity_crisis_chain
            target = root
        }
        hidden_effect = {
            set_variable = {
                which = pdivinity_telepath_jobs
                value = 0
            }
            modify_species = {
                species = this
                add_trait = trait_pdivinity_shroud_touched
            }
            change_dominant_species = {
                species = last_created_species
            }
            while = {
                limit = {
                    any_owned_pop_group = {
                        is_subspecies = root
                        NOT = { has_trait = trait_pdivinity_shroud_touched }
                    }
                }
                weighted_random_owned_pop_group = {
                    limit = {
                        is_subspecies = root
                        NOT = { has_trait = trait_pdivinity_shroud_touched }
                    }
                    root = {
                        modify_species = {
                            species = prev.species
                            add_trait = trait_pdivinity_shroud_touched
                        }
                    }
                }
            }
            create_species = {
                name = "NAME_pdivinity_Shroudpriests"
                class = SHROUDWALKER
                namelist = NECROID1
                portrait = random
                traits = {
                    trait = random_traits
                }
            }
            last_created_species = {
                save_global_event_target_as = pdivinity_avatar_species
            }
            save_global_event_target_as = pdivinity_crisis_empire
        }
        activate_crisis_progression = project_divinity_ascension_path
        shift_ethic = "ethic_fanatic_spiritualist"
    }
}

country_event = {
    id = pdivinity_crisis.101
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_playable_country = {
            has_ascension_perk = ap_pdivinity
        }
    }

    immediate = {
        if = { # checks if a new owner exists
            limit = { exists = FROMFROM }
            if = {
                limit = {
                    is_specialist_subject_type = { TYPE = pdivinity_acolyte }
                    FROMFROM = { NOT = { is_specialist_subject_type = { TYPE = pdivinity_acolyte } } }
                }
                FROM = {
                    if = {
                        limit = { exists = starbase }
                        starbase = { pdivinity_remove_shroud_storm_vassal_array = yes }
                        if = {
                            limit = { ROOT = { any_agreement = { specialist_tier >= 2 } } }
                            starbase = { pdivinity_remove_shroud_storm_vassal_array = yes }
                        }
                    }
                }
            }
            else_if = {
                limit = {
                    is_specialist_subject_tier = { TYPE = pdivinity_acolyte TIER = 2 }
                    FROMFROM = {
                        is_specialist_subject_type = { TYPE = pdivinity_acolyte }
                        NOT = { any_agreement = { specialist_tier = 2 } }
                    }
                }
                FROM = {
                    if = {
                        limit = { exists = starbase }
                        starbase = {
                            pdivinity_remove_shroud_storm_vassal_array = yes
                        }
                    }
                }
            }
        }
        else = { # if no new owner, always remove the disintegrator field (just a failsafe, systems with no onwer shouldn't have a starbase)
            FROM = {
                if = {
                    limit = { exists = starbase }
                    starbase = {
                        pdivinity_remove_shroud_storm_vassal_array = yes
                    }
                }
            }
        }
    }
}
