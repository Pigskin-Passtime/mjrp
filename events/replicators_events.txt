# Creates the country in a distant star system.
# Adds also a secondary spawn in another empire, which isnt immediatly hostile to it (in theory).
country_event = {
	id = replicators.2
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		no_scope = {
			set_spawn_system_batch = begin
				random_rim_system = {
					spawn_system = {
						min_distance >= 170
						max_distance <= 165
						direction = rimwards
						initializer = replicators_arrival_system
						hyperlane = no
						effect = {
							save_event_target_as = replicators_cluster_1
							save_global_event_target_as = replicators_arrival_system
						}
					}
				}
				event_target:replicators_cluster_1 = {
					solar_system = {
						random_system_planet = {
							limit = { is_star = yes }
							save_event_target_as = main_star
							save_global_event_target_as = replicators_star_system
						}
						create_species = {
							name = "Replicators"
							class = ROBOT
							portrait = "replicators_1"
							namelist = "replicators_names"
							traits = { trait = trait_machine_unit }
							effect = { save_event_target_as = replicators_species }
						}
						create_country = {
							name = "Replicators"
							type = "replicators"
							ignore_initial_colony_error = yes
							name_list = "replicators_names"
							species = event_target:replicators_species
							flag = {
								icon= {
									category = "Replicators"
									file = "replicators_flag.dds"
								}
								background= {
									category = "backgrounds"
									file = "flag_BG_39.dds"
								}
								colors={
								"black"
								"medium_steel"
								"black"
								"null"
								}
							}
							effect = {
								save_event_target_as = replicators_country
								save_global_event_target_as = replicators_country
							}
						}
						event_target:replicators_country = {
							create_leader = {
								class = admiral
								species = owner_main_species
								name = random
								skill = 6
								leader_age_min = 25
								leader_age_max = 45
								traits = { trait = random_trait }
							}
						}
						create_fleet = {
							effect = {
								set_owner = event_target:replicators_country
								create_ship = {
									name = random
									design = "NAME_replicators_stellar_engine"
									graphical_culture = "pirate_01"
									effect = { set_ship_flag = replicators_stellar_engine_core }
								}
								set_location = {
									target = event_target:main_star
									distance = 55
									angle = 270
								}
								save_event_target_as = replicators_factory
								save_global_event_target_as = replicators_core_factory
							}
						}
						event_target:replicators_core_factory = { add_modifier = { modifier = d_repl_core } }
						every_country = { establish_communications_no_message = event_target:replicators_country }
						create_fleet = {
							name = "NAME_Replicators_Fleet"
							effect = {
								set_owner = event_target:replicators_country
								create_replicators_ships = yes
								set_formation_scale = 2
								set_location = {
									target = event_target:replicators_factory
									distance = 80
									angle = 0
								}
								assign_leader = last_created_leader
							}
						}
						solar_system = { set_star_flag = replicators_star_system }
					}
				}
			set_spawn_system_batch = end
		}
		set_update_modifiers_batch = begin
			random_system_within_border = {
				limit = { is_capital_system = no }
					random_system_planet = {
						limit = { is_star = yes }
						save_event_target_as = main_star
						save_global_event_target_as = replicators_star_system
					}
				create_fleet = {
					effect = {
						set_owner = event_target:replicators_country
						create_ship = {
							name = random
							design = "NAME_replicators_stellar_engine"
							graphical_culture = "pirate_01"
						}
						set_location = {
							target = event_target:main_star
							distance = 55
							angle = 270
						}
						save_event_target_as = replicators_factory
					}
				}
				create_fleet = {
					name = "NAME_Replicators_Fleet"
					effect = {
						set_owner = event_target:replicators_country
						create_replicators_ships = yes
						set_formation_scale = 2
						set_location = {
							target = event_target:replicators_factory
							distance = 80
							angle = 0
						}
					}
				}
			}
		event_target:replicators_country = { country_event = { id = replicators.10 days = 1825 } }
		set_update_modifiers_batch = end
		every_playable_country = { country_event = { id = replicators.3 } }
#		observer_event = { id = observer.64 } # Whats that?
	}
	after = {
		event_target:global_event_country = {
			set_variable = { which = replicators_power value = 0 }
			change_variable = { which = replicators_power value = trigger:years_passed }
			multiply_variable = { which = replicators_power value = 0.5 }
			ceiling_variable  = replicators_power
		}
#		event_target:global_event_country = {
#			set_variable = {
#				which = replicators_power
#				value = 15
#			}
#		}
	}
}
