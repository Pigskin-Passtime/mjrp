### retain acolyte vassal

event = {
    id = pdivinity_crisis_point.1
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_country = {
            has_ascension_perk = ap_pdivinity
            is_overlord = yes
        }
    }

    immediate = {
        every_playable_country = {
            if = {
                limit = {
                    has_ascension_perk = ap_become_the_crisis
                    is_overlord = yes
                }
                every_subject = {
                    limit = {
                        any_agreement = {
                            has_active_specialization = specialist_pdivinity_acolyte
                        }
                    }
                    prev = {
                        complete_crisis_objective = crisobj_pdivinity_retain_vassals
                    }
                }
            }
        }
    }
}

#### sacrifice pops

planet_event = {
    id = pdivinity_crisis_point.3
    hide_window = yes
    is_triggered_only = yes

    # This = Planet scope
    # From = Country
    # FromFrom = Pop

    trigger = {
        from = {
            has_ascension_perk = ap_pdivinity
        }
        fromfrom = {
            has_purge_type = { country = from type = purge_pdivinity_sacrifice }
        }
    }

    immediate = {
        while = {
            count = value:per_100_local_pop_amount
            from = {
                complete_crisis_objective = crisobj_pdivinity_sacrifice_population
            }
        }
    }
}

## shroudwalker enclaves

event = {
    id = pdivinity_crisis_point.4
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_country = {
            has_ascension_perk = ap_pdivinity
            any_system_within_border = {
                any_fleet_in_system = {
                    is_ship_size = enclave_shroudwalker_station
                }
            }
        }
    }

    immediate = {
        every_playable_country = {
            if = {
                limit = {
                    has_ascension_perk = ap_pdivinity
                }
                every_system_within_border = {
                    limit = {
                        any_fleet_in_system = {
                            is_ship_size = enclave_shroudwalker_station
                        }
                    }
                    prev = {
                        complete_crisis_objective = crisobj_pdivinity_control_shroud_enclave
                    }
                }
            }
        }
    }
}

### build psionic ship

ship_event = {
    id = pdivinity_crisis_point.5
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        owner = {
            has_ascension_perk = ap_pdivinity
        }
        OR = {
            is_ship_size = pdivinity_psionic_cloud
            is_ship_size = pdivinity_psionic_avatar
            is_ship_size = pdivinity_dimensional_tearer
        }
    }

    immediate = {
        while = {
            count = trigger:ship_size_multiplier
            owner = {
                    complete_crisis_objective = crisobj_pdivinity_summon_psionic_ships
            }
        }
    }
}

### employ telepaths

event = {
    id = pdivinity_crisis_point.7
    hide_window = yes
    is_triggered_only = yes

    trigger = {
        any_playable_country = {
            has_ascension_perk = ap_pdivinity
        }
    }

    immediate = {
        every_playable_country = {
            limit = {
                has_ascension_perk = ap_pdivinity
            }
            export_resource_income_to_variable = {
                resource = pdivinity_psionic_strength
                variable = pdivinity_telepath_jobs
            }
            complete_crisis_objective = crisobj_pdivinity_employ_telepaths
            # prevent double
            add_resource = {
                pdivinity_psionic_strength = -1
                multiplier = pdivinity_telepath_jobs
            }
        }
    }
}
