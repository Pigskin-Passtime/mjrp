#If we are about to reach 0 resource in N < 12 months or capped, send notification
country_event = {
    id = TFO.1003
    is_triggered_only = yes
    hide_window = yes

    immediate = {
        if = {
            limit = {
                has_country_flag = TFO_sr_shroud_condensers_max_notification_fired
                check_variable = {
                    which = TFO_sr_shroud_condensers_current
                    value < TFO_sr_shroud_condensers_max
                }
            }

            remove_country_flag = TFO_sr_shroud_condensers_max_notification_fired
        }

        if = {
            limit = {
                check_variable = {
                    which = TFO_sr_shroud_condensers_income
                    value < 0
                }
            }

            set_variable = {
                which = TFO_sr_shroud_condensers_income_abs
                value = TFO_sr_shroud_condensers_income
            }

            multiply_variable = {
                which = TFO_sr_shroud_condensers_income_abs
                value = -1
            }

            set_variable = {
                which = TFO_sr_shroud_condensers_months_left
                value = TFO_sr_shroud_condensers_current
            }

            divide_variable = {
                which = TFO_sr_shroud_condensers_months_left
                value = TFO_sr_shroud_condensers_income_abs
            }
            ceiling_variable = TFO_sr_shroud_condensers_months_left

            if = {
                limit = {
                    check_variable = {
                        which = TFO_sr_shroud_condensers_months_left
                        value <= 12
                    }
                    check_variable = {
                        which = TFO_sr_shroud_condensers_months_left
                        value > 0
                    }
                }

                create_message = {
                    type = TFO_MESSAGE_STOCKPILE_LOW
                    localization = TFO_MESSAGE_STOCKPILE_LOW_DESC
                    days = 10
                    target = this
                }
            }
        }

        if = {
            limit = {
                check_variable = {
                    which = TFO_sr_shroud_condensers_income
                    value > 0
                }
                check_variable = {
                    which = TFO_sr_shroud_condensers_current
                    value = TFO_sr_shroud_condensers_max
                }
                NOT = {
                    has_country_flag = TFO_sr_shroud_condensers_max_notification_fired
                }
            }

            set_country_flag = TFO_sr_shroud_condensers_max_notification_fired
            create_message = {
                type = TFO_MESSAGE_STOCKPILE_MAX
                localization = TFO_MESSAGE_STOCKPILE_MAX_DESC
                days = 15
                target = this
            }
        }
    }
}
