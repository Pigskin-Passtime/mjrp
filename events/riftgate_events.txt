#cycle through all 4 events
country_event = {
	id = riftgate.6
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		fromfrom = {
			OR = {
				has_fleet_flag = trial_1st_wave
				has_fleet_flag = trial_2nd_wave
				has_fleet_flag = trial_3rd_wave
				has_fleet_flag = trial_4th_wave
			}
		}
	}
	immediate = {
		if = {
			limit = {
				FromFrom = {has_fleet_flag = trial_1st_wave}
			}
			from = {
				country_event = {
					id = riftgate.7
				}
			}
		}

		if = {
			limit = {
				FromFrom = {has_fleet_flag = trial_2nd_wave}
			}
			from = {
				country_event = {
					id = riftgate.8
				}
			}
		}

		if = {
			limit = {
				FromFrom = {has_fleet_flag = trial_3rd_wave}
			}
			from = {
				country_event = {
					id = riftgate.9
				}
			}
		}

		if = {
			limit = {
				FromFrom = {has_fleet_flag = trial_4th_wave}
			}
			from = {
				country_event = {
					id = riftgate.11
				}
			}
		}
	}
}
