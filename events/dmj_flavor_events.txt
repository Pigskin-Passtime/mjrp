						}
					}

						factor = 0

							OR = {
								has_country_flag = dmj_reinhardist_empire # Can build temples manually if this is true

									count = all

leader_event = { # A group of pops sacrifice their souls to Reinhard
	id = dmj_flavor.6
	title = "dmj_flavor.6.name"
	desc = "dmj_flavor.6.desc"
	is_triggered_only = yes

	picture = GFX_evt_necrophage
	show_sound = event_death_cult

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_legendary_beholder"

	picture_event_data = {
		portrait = root
	}

	trigger = {
		exists = owner
		owner = {
			any_owned_leader = {
				is_same_value = event_target:dmj_daniel
			}
		}
	}

	immediate = {
		if = {
			limit = {
				exists = planet
				planet = {
					count_owned_pop = {
						limit = {
							pop_faction = {
								is_pop_faction_type = reinhardist
							}
						}
						count >= 5
					}
				}
			}
			planet = { save_event_target_as = dmj_event_sacrifice_planet } # used in loc
		}
		else = {
			owner = {
				random_owned_planet = {
					limit = {
						count_owned_pop = {
							limit = {
								pop_faction = {
									is_pop_faction_type = reinhardist
								}
							}
							count >= 5
						}
					}
					save_event_target_as = dmj_event_sacrifice_planet
				}
			}
		}
		event_target:dmj_event_sacrifice_planet = {
			set_variable_to_random_value = { which = dmj_sacrifice_pops min = 2 max = 5 rounded = yes }
		}
		owner = {
			random_pop_faction = {
				limit = { is_pop_faction_type = reinhardist }
				save_event_target_as = dmj_event_sacrifice_faction # used in loc
			}
		}
	}

	option = {
		name = "dmj_flavor.6.a"
	}

	after = {
		event_target:dmj_event_sacrifice_planet = {
			while = {
				count < dmj_sacrifice_pops
				random_owned_pop = {
					limit = {
						pop_faction = {
							is_pop_faction_type = reinhardist
						}
					}
					kill_pop = yes
				}
			}
		}
		hidden_effect = {
			event_target:dmj_event_sacrifice_planet = {
				clear_variable = dmj_sacrifice_pops
			}
		}
	}
}
