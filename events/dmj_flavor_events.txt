leader_event = { # A group of pops sacrifice their souls to Reinhard
	id = dmj_flavor.6
	name = "dmj_flavor.6.name"
	desc = "dmj_flavor.6.desc"

	picture = GFX_evt_necrophage
	show_sound = event_death_cult

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_renowned"

	trigger = {
		exists = owner
	}

	immediate = {
		if = {
			limit = {
				exists = planet
				planet = {
					count_owned_pop = {
						limit = {
							pop_faction = {
								is_pop_faction_type = reinhardist
							}
						}
						count >= 5
					}
				}
			}
			planet = { save_event_target_as = dmj_event_sacrifice_planet } # used in loc... when I get around to writing it that is
		}
		else = {
			random_owned_planet = {
				limit = {
					count_owned_pop = {
						limit = {
							pop_faction = {
								is_pop_faction_type = reinhardist
							}
						}
						count >= 5
					}
				}
				save_event_target_as = dmj_event_sacrifice_planet
			}
		}
		dmj_event_sacrifice_planet = {
			set_variable_to_random_value = { which = dmj_sacrifice_pops min = 2 max = 5 rounded = yes }
		}
	}

	option = {
		name = "dmj_flavor.6.a"
		event_target:dmj_event_sacrifice_planet = {
			while = {
				count < dmj_sacrifice_pops
				random_owned_pop = {
					limit = {
						pop_faction = {
							is_pop_faction_type = reinhardist
						}
					}
					kill_pop = yes
				}
			}
		}
		hidden_effect = {
			event_target:dmj_event_sacrifice_planet = {
				clear_variable = dmj_sacrifice_pops
			}
		}
	}
}
