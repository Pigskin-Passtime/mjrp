#Infinite Empire Invades and sets up an outpost
planet_event = {
	id = branchial_guardian.1
	hide_window = yes
	is_triggered_only = yes
	fire_only_once = yes
	immediate = {
		set_global_flag = branchial_empire_active
		create_species = {
			name = "Infinites"
			class = random
			namelist = Extradimensional
			portrait = random
			traits = random
			effect = { save_event_target_as = branchial_species }
		}
		create_country = {
			name = "The Infinite Empire"
			type = branchial_virgin
			flag = {
				icon= {
					category = "special"
					file = "unknown.dds"
				}
				background= {
					category = "backgrounds"
					file = "circle.dds"
				}
				colors={
					"blue"
					"blue"
					"blue"
					"blue"
				}
			}
			effect = {
				set_country_flag = branchial_guardian_country
				create_leader = {
					name = random
					class = commander
					species = event_target:branchial_species
					skill = 10
					immortal = yes
				}
				assign_leader = last_created_leader
				create_ship_design = { design = "NAME_Branchial_guardians_station" }
				add_ship_design = last_created_design
				set_hostile = root.owner
			}
		}
		create_fleet = {
			name = "Garrison Fleet"
			settings = { spawn_debris = no }
			effect = {
				set_owner = last_created_country
   	        	while = {
   	    	    	count = 35
   		        	create_ship = {
	       	        	name = "?????"
          		    	design = "NAME_branchial_guardians_large"
    	    		}
   	        	}
    		    while = {
    		    	count = 100
	    	    	create_ship = {
        			    name = "?????"
           			    design = "NAME_branchial_guardians_small"
        	    	}
    		    }
	    	    set_location = root
        	    set_fleet_stance = aggressive
        	    set_fleet_flag = branchial_and_researchable
			}
		}
		create_fleet = {
			name = "Invasion Fleet"
			settings = { spawn_debris = no }
			effect = {
				set_owner = last_created_country
   	        	while = {
   	    	    	count = 50
   		        	create_army_transport = {
						ship_name = "Invasion Transport"
						army_name = "Invasion Force"
						army_type = branchial_empire_event_assault_army
					}
   	        	}
	    	    set_location = root
        	    set_fleet_stance = aggressive
        	    set_fleet_flag = branchial_and_researchable
			}
		}
		solar_system = {
			starbase = { fleet = { destroy_fleet = this } }
			create_starbase = {
				size = branchial_guardians_station_large
				owner = last_created_country
			}
		}
	}
}

#The final attack-triggered using on_build in megastructures
country_event = {
	id = branchial_guardian.18
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		set_country_flag = declared_crisis
		save_event_target_as = branchial_crisis
		country_event = { id = branchial_guardian.19 } #player notification of declared a crisis
		every_playable_country = {
			limit = {
				is_ai = no
				NOT = { is_same_value = prev }
			}
			country_event = { id = branchial_guardian.20 } #notification of galactic crisis war
		}
		if = {
			limit = { has_galactic_custodian = yes }
			random_playable_country = {
				limit = { is_galactic_custodian = yes }
				save_event_target_as = community_war_leader
				declare_war = {
					target = event_target:branchial_crisis
					name = {
						key = "NAME_Declared_Crisis_War"
						variable_string = "[This.MainDefender.GetAdj]"
					}
					attacker_war_goal = "wg_declared_crisis"
				}
			}
			every_playable_country = {
				limit = {
					is_galactic_custodian = no
					is_galactic_community_member = yes
				}
				join_war = event_target:community_war_leader # Should use different effect to ensure they only join the correct war
			}
		}
		else = {
			random_playable_country = {
				limit = {
					is_galactic_community_member = yes
					galactic_community_rank = 1
				}
				save_event_target_as = community_war_leader
				declare_war = {
					target = event_target:branchial_crisis
					name = {
						key = "NAME_Declared_Crisis_War"
						variable_string = "[This.MainDefender.GetAdj]"
					}
					attacker_war_goal = "wg_declared_crisis"
				}
				every_playable_country = {
					limit = {
						is_galactic_community_member = yes
						NOT = { is_same_value = event_target:community_war_leader }
					}
					join_war = event_target:community_war_leader # Should use different	 effect to ensure they only join the correct war
				}
			}
		}
	}

}
