						"null"

			}

		}

			limit = { NOT = { has_global_flag = dmj_no_double_spawn } }

				name = "NAME_dmj_Shroud_Sentinel"
				settings = {
					spawn_debris = no

					set_owner = event_target:shroud_country

						name = "NAME_dmj_Shroud_Sentinel"

				settings = {
					spawn_debris = no
					is_boss = yes
				}

					set_owner = event_target:shroud_country

country_event = { # Detect when the Sentinels are defeated
	id = dmj.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_global_flag = dmj_system_spawned
		exists = event_target:shroud_country
		is_same_value = event_target:shroud_country
		fromfrom = { has_fleet_flag = dmj_sentinel }
		NOT = { has_global_flag = dmj_sentinel_respawn_safety }
	}
	immediate = {
		change_variable = {
			which = dmj_alive_sentinels_count
			value = -1
		}
		if = {
			limit = {
				check_variable = {
					which = dmj_alive_sentinels_count
					value = 0
				}
			}
			remove_global_flag = dmj_no_double_spawn
			if = {
				limit = {
					from = { is_same_value = event_target:dmj_trespasser }
				}
				from = {
					country_event = { id = dmj.3 }
					clear_global_event_target = dmj_trespasser
				}
			}
			event_target:dmj_spawn_system = {
				set_star_flag = dmj_sentinels_cooldown
			}
			country_event = { id = dmj.4 days = 180 }
		}
	}
}

ship_event = { # Destroy the planet and check for remaining bastions
	id = dmj.5
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				exists = event_target:dmj_bastion_1
				fromfrom = { is_same_value = event_target:dmj_bastion_1 }
			}
			fromfrom = {
				clear_global_event_target = dmj_bastion_1
				change_pc = pc_shattered
				clear_deposits = yes
				clear_planet_modifiers = yes
				prevent_anomaly = yes
			}
		}
		else_if = {
			limit = {
				exists = event_target:dmj_bastion_2
				fromfrom = { is_same_value = event_target:dmj_bastion_2 }
			}
			fromfrom = {
				clear_global_event_target = dmj_bastion_2
				change_pc = pc_shattered
				clear_deposits = yes
				clear_planet_modifiers = yes
				prevent_anomaly = yes
			}
		}

		if = {
			limit = {
				OR = {
					exists = event_target:dmj_bastion_1
					exists = event_target:dmj_bastion_2
				}
			}
			owner = { country_event = { id = dmj.6 } }
		}
		else = {
			owner = { country_event = { id = dmj.7 } }
		}
	}
}
