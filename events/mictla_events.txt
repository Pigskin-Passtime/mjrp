country_event = { # If finished situation submitting, Mictla Becomes Ruler
	id = mictla_events.5
	title = "mictla_events.5.name"
	desc = "mictla_events.5.desc"
	picture = GFX_evt_psionics
	show_sound = event_mystic_reveal
	is_triggered_only = yes

	trigger = {
		is_country_type = default
		NOT = { has_country_flag = chosen_one_ruler_event }
		OR = {
			# Either you've not Imperial and have a Chosen One
			AND = {
				OR = {
					has_authority = auth_democratic
					has_authority = auth_oligarchic
					has_authority = auth_imperial
					has_authority = auth_corporate
				}
				any_owned_leader = {
					has_leader_flag = mictla
				}
			}
			# Or you're Imperial and the Chosen One is your current ruler or heir.
			AND = {
				has_authority = auth_dictatorial
				any_owned_leader = {
					has_leader_flag = mictla
					is_ruler = yes
				}
			}
		}
	}

	option = {
		name = "mictla_events.5.b"
		if = {
			limit = {
				has_origin = origin_legendary_leader_imperial
			}
			set_origin = origin_legendary_leader_dictatorial
		}
		shift_ethic = "ethic_authoritarian"
		shift_ethic = "ethic_fanatic_spiritualist"
		change_government = {
			authority = auth_dictatorial
			cooldown = no
			remove_invalid_civics = yes
		}
		enable_special_project = {
			name = MICTLA_MUTUALISM
		}
 		set_government_cooldown = no
		set_country_flag = mictla_leading
		add_modifier = {
			modifier = mictla_accepted_gift
			days = 3600
		}
		# The ruler doesn't need to abdicate if they're being crowned.
		if = {
			limit = {
				ruler = {
					NOT = {
						is_same_value = event_target:mictla_leader
					}
				}
			}
			set_leader = event_target:mictla_leader
		}
		hidden_effect = {
			set_ruler_title_male = "Resonant"
			set_ruler_title_female = "Resonant"
			country_event = { id = utopia.3402 days = 5 }
		}
	}
}
