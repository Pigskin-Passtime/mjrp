country_event = { # Search for Sol
	id = dmj_sol.1
	title = "dmj_sol.1.name"
	desc = "dmj_sol.1.desc"
	desc = {
		trigger = {
			any_system = {
				has_star_flag = sol
				OR = {
					is_in_sensor_range_of_country = root
					any_country = {
						has_communications = root
						any_owned_planet = {
							OR = {
								has_planet_flag = planet_earth
								has_planet_flag = nuked_earth
							}
						}
					}
				}
				NOT = { is_owned_by = root }
			}
		}
		text = "dmj_sol.1.knows_sol_desc"
	}
	desc = {
		trigger = {
			any_system = {
				has_star_flag = sol
				NOT = { is_in_sensor_range_of_country = root }
				NOT = {
					any_country = {
						has_communications = root
						any_owned_planet = {
							OR = {
								has_planet_flag = planet_earth
								has_planet_flag = nuked_earth
							}
						}
					}
				}
				NOT = { is_owned_by = root } # Just in case
			}
		}
		text = "dmj_sol.1.not_knows_sol_desc"
	}
	is_triggered_only = yes

	picture = GFX_evt_ancient_databank
	show_sound = event_scanner

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_legendary_beholder"

	picture_event_data = {
		portrait = event_target:dmj_daniel
	}

	trigger = {
		exists = event_target:dmj_daniel
		any_owned_leader = {
			is_same_value = event_target:dmj_daniel
		}
		NOR = {
			has_communications = event_target:dmj_sol_country
			any_system = {
				AND = {
					has_star_flag = sol
					is_owned_by = root
				}
			}
		}
	}

	immediate = {
		if = {
			limit = {
				any_system = {
					has_star_flag = sol
					OR = {
						is_in_sensor_range_of_country = root
						any_country = {
							has_communications = root
							any_owned_planet = {
								OR = {
									has_planet_flag = planet_earth
									has_planet_flag = nuked_earth
								}
							}
						}
					}
					NOT = { is_owned_by = root }
				}
			}
			root.capital_scope.solar_system = {
				spawn_natural_wormhole = {
					bypass_type = dmj_shroud_tunnel
					random_pos = no
					orbit_angle = 360
				}
			}
			event_target:dmj_terra.solar_system = {
				spawn_natural_wormhole = {
					bypass_type = dmj_shroud_tunnel
					random_pos = no
					orbit_angle = 360
				}
				link_wormholes = root.capital_scope
				set_surveyed = { surveyed = yes surveyor = root }
			}
		}
	}

	option = {
		name = dmj_sol.1.a
		begin_event_chain = { event_chain = dmj_sol_reclamation_chain target = root }
		if = {
			limit = {
				any_system = {
					has_star_flag = sol
					NOT = { is_in_sensor_range_of_country = root }
					NOT = {
						any_country = {
							has_communications = root
							any_owned_planet = {
								OR = {
									has_planet_flag = planet_earth
									has_planet_flag = nuked_earth
								}
							}
						}
					}
				}
			}
			enable_special_project = {
				name = "dmj_SOL_SEARCH"
				owner = root
			}
		}
		else_if = {
			limit = {
				any_system = {
					has_star_flag = sol
					OR = {
						is_in_sensor_range_of_country = root
						any_country = {
							has_communications = root
							any_owned_planet = {
								OR = {
									has_planet_flag = planet_earth
									has_planet_flag = nuked_earth
								}
							}
						}
					}
				}
			}
			if = {
				limit = { NOT = { has_country_flag = reinhardist_empire } }
				country_event = { id = dmj_sol.3 days = 3600 }
			}
			
			random_system = {
				limit = {
					has_star_flag = sol
					OR = {
						is_in_sensor_range_of_country = root
						any_country = {
							has_communications = root
							any_owned_planet = {
								OR = {
									has_planet_flag = planet_earth
									has_planet_flag = nuked_earth
								}
							}
						}
					}
				}
				random_system_planet = {
					limit = {
						OR = {
							has_planet_flag = planet_earth
							has_planet_flag = nuked_earth
						}
					}
					save_event_target_as = dmj_terra
				}
			}
			create_point_of_interest = { id = dmj_terra name = "dmj_terra_poi" desc = "dmj_terra_poi_desc" event_chain = dmj_sol_reclamation_chain location = event_target:dmj_terra }
		}
	}
}
