espionage_operation_event = {
	id = lbd_specialist.2
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_global_flag = lbd_dathnak_fired
		exists = owner
		owner = {
			is_specialist_subject_tier = {
				TYPE = lbd_dathnak_specialist
				TIER = 1
			}
		}
	}
	immediate = {
		owner = {
			save_global_event_target_as = temp_lbd_dathnak_debuff_applier
		}
		from = {
			if = {
				limit = {
					has_modifier = lbd_dathnak_specialist_1_annoying_saboteur_debuff
				}
				remove_modifier = lbd_dathnak_specialist_1_annoying_saboteur_debuff
				add_modifier = {
					modifier = lbd_dathnak_specialist_1_annoying_saboteur_debuff
					days = 180
					time_multiplier = value:lbd_dathnak_calculate_time_multiplier_debuff
				}
				if = {
					limit = {
						has_modifier = lbd_dathnak_specialist_1_annoying_saboteur_debuff_stacking
					}
					change_variable = {
						which = lbd_dathnak_stacking_debuff_counter
						value = 1
					}
					add_modifier = {
						modifier = lbd_dathnak_specialist_1_annoying_saboteur_debuff_stacking
						mult = lbd_dathnak_stacking_debuff_counter
						days = 180
						time_multiplier = value:lbd_dathnak_calculate_time_multiplier_debuff
					}
				}
				else = {
					set_variable = {
						which = lbd_dathnak_stacking_debuff_counter
						value = 1
					}
					add_modifier = {
						modifier = lbd_dathnak_specialist_1_annoying_saboteur_debuff_stacking
						mult = lbd_dathnak_stacking_debuff_counter
						days = 180
						time_multiplier = value:lbd_dathnak_calculate_time_multiplier_debuff
					}
				}
			}
			else = {
				add_modifier = {
					modifier = lbd_dathnak_specialist_1_annoying_saboteur_debuff
					days = 180
					time_multiplier = value:lbd_dathnak_calculate_time_multiplier_debuff
				}
			}
			create_message = {
				type = LBD_DATHNAK_DEBUFF_APPLIED
				localization = LBD_DATHNAK_SPECIALIST_DEBUFFED
				days = 90
				target = this
			}
		}
	}
}
espionage_operation_event = {
	id = lbd_specialist.3
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		has_global_flag = lbd_dathnak_fired
		exists = owner
		owner = {
			is_specialist_subject_tier = {
				TYPE = lbd_dathnak_specialist
				TIER = 2
			}
		}
		from = {
			num_pops > 0
		}
	}
	immediate = {
		
		from = {
			while = {
				count = 3
				random_owned_species = {
					limit = {
						NOT = {
							has_trait = trait_lbd_dathnak_base
						}
					}
					mutate_species = yes
				}
			}
			create_message = {
				type = LBD_DATHNAK_SPECIES_MUTATION_COMPLETE
				localization = LBD_DATHNAK_SPECIALIST_MUTATIONS
				days = 90
				target = this
			}
		}
	}
}
