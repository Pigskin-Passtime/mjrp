namespace = osom_subterranean

# Subterranean Origin
country_event = {
    id = osom_subterranean.1
    is_triggered_only = yes
    hide_window = yes
    
    trigger = {
        has_origin = osom_origin_subterranean
    }
    
    immediate = {
	owner = { give_technology = { tech = osom_tech_subterranean }
        every_owned_planet = {
            limit = {
                is_homeworld = yes
            }
            add_modifier = { 
                modifier = "osom_planet_subterranean_5" 
                days = -1 
            }
        }
    }
}
