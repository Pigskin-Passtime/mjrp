	desc = "tiyres_show_events.9.desc"

	picture = GFX_evt_infested
	show_sound = event_mystic_reveal

	is_triggered_only = yes

	trigger = {
		has_global_flag = tiyres_in_progress
		any_owned_planet = {
			has_modifier = planet_tiy_infestation
		}
		NOT = {
			has_country_flag = tiyres_infestation_project_flag
		}
	}

	option = {
		name = tiyres_show_events.9.a
		hidden_effect = {
			set_country_flag = tiyres_infestation_project_flag
		}
		enable_special_project = {
			name = TIYRES_INFESTATION_PURGING_PROJECT
			location = capital_scope
			owner = root
		}
	}
}
