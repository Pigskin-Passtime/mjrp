country_event = {
	#Cerebral relay captured
	id = coex_crisis.7200
	title = coex_crisis.7200.name
	desc = coex_crisis.7200.desc
	show_sound = event_super_explosion
	picture = GFX_evt_the_beholder
	location = from
	trigger = {
		from = {
			has_megastructure_flag = coex_brain_relay
		}
		if = {
			limit = {
				has_crisis_level = coex_crisis_coexistence_level_1
			}
			NOT = {
				from = {
					has_megastructure_flag = coex_brain_relay_of@root
				}
			}
		}
	}
	abort_trigger = {
		OR = {
			NOT = {
				exists = from
			}
			NOT = {
				exists = from.owner
			}
			from.owner = {
				NOT = {
					is_same_value = root
				}
			}
		}
	}
	is_triggered_only = yes
	immediate = {
		from = {
			solar_system = {
				save_event_target_as = coex_destroyed_relay_system
			}
		}
		if = {
			limit = {
				exists = fromfrom
				fromfrom = {
					has_megastructure_flag = coex_brain_relay_of@prev
				}
			}
			fromfrom = {
				save_event_target_as = coex_relay_builders
			}
		}
		else = {
			#fromfrom is not reliable/doesn't work at all
			random_country = {
				limit = {
					from = {
						has_megastructure_flag = coex_brain_relay_of@prev
					}
				}
				save_event_target_as = coex_relay_builders
			}
		}
	}
	#It must be destroyed
	option = {
		name = coex_crisis.7200.A
		if = {
			limit = {
				has_crisis_level = coex_crisis_coexistence_level_1
			}
			add_resource = {
				coex_mental_authority = 600
				alloys = 400
			}
		}
		else = {
			add_resource = {
				influence = 50
				alloys = 400
			}
		}
		hidden_effect = {
			from = {
				remove_megastructure = this
			}
			if = {
				limit = {
					exists = event_target:coex_relay_builders
				}
				event_target:coex_relay_builders = {
					country_event = {
						id = coex_crisis.7220
					}
				}
			}
			every_playable_country = {
				limit = {
					has_country_flag = coex_relay_affected_by@from
					# NOT = {
					# 	is_same_value = root
					# }
				}
				country_event = {
					id = coex_crisis.7210
				}
			}
		}
		ai_chance = {
			factor = 1
		}
	}
}
