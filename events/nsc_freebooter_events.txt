#pop trait changing event 1
pop_event = {
	id = nsc_freebooter.2
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			AND = {
				has_trait = trait_pc_nsc_freebooter_habitat_preference
				NOT = { has_trait = trait_nsc_freebooter_1 }
				from = { is_planet_class = pc_nsc_freebooter_habitat }
			}
			AND = {
				has_trait = trait_nsc_freebooter_1
				from = {
					NOR = {
						is_planet_class = pc_nsc_freebooter_habitat
						is_planet_class = pc_habitat #check for the vanilla habitat too
					}
				}
			}
		}
	}

	immediate = {
		if = {
			limit = {
				has_trait = trait_pc_nsc_freebooter_habitat_preference
				NOT = { has_trait = trait_nsc_freebooter_1 }
				from = { is_planet_class = pc_nsc_freebooter_habitat }
			}
			if = {
				limit = { has_trait = trait_nsc_freebooter_2 }
				modify_species = {
					species = this
					add_trait = trait_nsc_freebooter_1
					remove_trait = trait_nsc_freebooter_2
				}
				else = {
					modify_species = {
						species = this
						add_trait = trait_nsc_freebooter_1
					}
				}
			}
		}
		else_if = {
			limit = {
				has_trait = trait_nsc_freebooter_1
				from = {
					NOR = {
						is_planet_class = pc_nsc_freebooter_habitat
						is_planet_class = pc_habitat #check for the vanilla habitat too
					}
				}
			}
			modify_species = {
				species = this
				remove_trait = trait_nsc_freebooter_1
				add_trait = trait_nsc_freebooter_2
			}
		}
	}
}
#The Khan conquers Worlds
country_event = {
	id = nsc_freebooter.3
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		from = {
			is_at_war_with = root
			any_war = {
				using_war_goal = {
					type = wg_nsc_freebooter_wrath_of_khan #will not trigger if the war goal is the subjugation (satrapy)
					owner = root
				}
			}
		}
	}

	immediate = {
		from = {
			change_variable = {
				which = "nsc_freebooter_planets_lost_to_khan_var"
				value = 1
			}
		}
		if = {
			limit = {
				from = {
					is_ai = yes
					NOR = {
						has_valid_civic = "civic_fanatic_purifiers"
						has_valid_civic = "civic_hive_devouring_swarm"
						has_valid_civic = "civic_machine_terminator"
					}
					check_variable = {
						which = "nsc_freebooter_planets_lost_to_khan_var"
						value >= 2
					}
					OR = {
						relative_power = {
							who = prev.this
							category = fleet
							value = inferior
						}
						relative_power = {
							who = prev.this
							category = fleet
							value = pathetic
						}
					}
				}
			}
			this = { country_event = { id = nsc_freebooter.4 } }
		}
	}
}

			event_target:nsc_frebooter_empire_fleet_market_target = {

							graphical_culture = root
						}

			}

				limit = { is_ship_size = corvette is_in_combat = no }
				delete_ship = this
			}
			country_event = { id = nsc_freebooter.17 }

					modify_species = {

					}

			}
		}
	}

country_event = {
	id = nsc_freebooter.21
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		NOT = { has_country_flag = nsc_freebooter_the_duel_event_already_triggered }
	}

	immediate = {
		if = {
			limit = { ruler = { has_trait = nsc_freebooter_leader_trait_ruler_great_khan } } #if the khan is not dead yet
			if = {
				limit = { any_owned_leader = { leader_class = general } } #and the player empire has at least one general
				set_country_flag = nsc_freebooter_the_duel_event_already_triggered
				country_event = { id = nsc_freebooter.22 }
			}
			else_if = {
				limit = { NOT = { any_owned_leader = { leader_class = general } } } #and the player empire don't even have at least one general
				country_event = { id = nsc_freebooter.21 random = 180 days = 180 } #recheck the condition again
			}
		}
		if = {
			limit = { ruler = { NOT = { has_trait = nsc_freebooter_leader_trait_ruler_great_khan } } } #if the khan is dead already
			set_country_flag = nsc_freebooter_the_duel_event_already_triggered
		}
	}
}

#the real effect event
planet_event = {
	id = nsc_freebooter.35
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		NOT = { has_planet_flag = nsc_freebooter_currently_under_raiding_planet }
	}

	immediate = {
		set_timed_planet_flag = { flag = nsc_freebooter_currently_under_raiding_planet days = 30 }
		add_planet_devastation = 10
		nsc_freebooter_planet_raiding_planetary_effect = yes
		solar_system = {
			random_fleet_in_system = {
				limit = {
					exists = orbit
					orbit = { is_same_value = root }
					owner = { has_ascension_perk = ap_nsc_freebooter_pirate_raider }
				}
				owner = { #the owner empire (the freebooter empire) of the fleet, which is located in the orbit of the 'root' (the planet where is raided by the empire)
					root.owner = { #the owner empire of the 'root' (the planet where is raided by the freebooter empire)
						add_opinion_modifier = {
							who = prev #one steps back from the 'root.owner' so it's the freebooter empire
							modifier = nsc_freebooter_opinion_they_raided_one_of_our_colony
						}
					}
					if = { #abduct a pop even when the freebooter empire is not at war with the planet's owner
						limit = { NOT = { is_at_war_with = root.owner } }
						root = {
							random_owned_pop = { #a 'random_owned_pop' who is living in the 'root' (the planet where is raided by the empire)
								prevprev = { #the freebooter empire
									random_owned_planet = {
										limit = { free_jobs >= 1 }
										create_pop = {
											species = prevprev #two steps back from the 'random_owned_planet' (one of the freebooter empire's planet) so it's 'random_owned_pop'
											ethos = prevprev
										}
									}
									if = { #if every planet under the freebooter empire has NO free job available at this moment
										limit = { NOT = { any_owned_planet = { free_jobs >= 1 } } }
										capital_scope = {
											create_pop = {
												species = prevprev #two steps back from the 'capital_scope' (the freebooter empire's capital) so it's 'random_owned_pop'
												ethos = prevprev
											}
										}
									}
								}
								kill_pop = yes
							}
						}
					}
					nsc_freebooter_planet_raiding_effect = yes
				}
			}
		}
	}
}
