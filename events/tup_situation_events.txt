situation_event = {
	id = tup_situation_events.120
	title = tup_situation_events.120
	desc = tup_situation_events.120.desc
	picture = GFX_evt_genetic_ascention_cloning
	show_sound = event_default
	is_triggered_only = yes
	immediate = {
		owner = {
			set_variable = {
				which = tup_number_of_pops_to_convert
				value = value:num_pop_groups
			}
			divide_variable = {
				which = tup_number_of_pops_to_convert
				value = 5
			}
			ceiling_variable = tup_number_of_pops_to_convert
		}
	}
	option = {
		name = EXCELLENT
		owner = {
			while = {
				count = tup_number_of_pops_to_convert
				random_owned_pop_group = {
					limit = {
						NOT = {
							has_trait = tup_trait_utopia_ascendant
						}
					}
					modify_species = {
						species = this
						add_trait = tup_trait_utopia_ascendant
					}
				}
			}
			if = {
				limit = {
					has_modifier = tup_situation_communal_reformation_progress_2
				}
				add_modifier = {
					modifier = tup_situation_communal_reformation_progress_3
				}
			}
			else_if = {
				limit = {
					has_modifier = tup_situation_communal_reformation_progress_1
				}
				add_modifier = {
					modifier = tup_situation_communal_reformation_progress_2
				}
			}
			else = {
				add_modifier = {
					modifier = tup_situation_communal_reformation_progress_1
				}
			}
		}
	}
}

	title = "tup_situation_events.200.name"
	desc = tup_situation_events.200.desc
	picture = GFX_evt_pirate_armada
	show_sound = event_space_battle
	is_triggered_only = yes
	immediate = {
		owner = {
			save_event_target_as = pirate_target
			owner_species = {
				save_event_target_as = owner_species
			}
			random_system_within_border = {
				limit = {
					is_capital_system = no
				}
				save_event_target_as = pirate_system
				create_normal_pirate_country = yes
				tup_create_pirate_base = yes
				tup_create_pirate_fleet_2 = yes
			}
		}
	}
	option = {
		name = tup_situation_events.200.a
		hidden_effect = {
			if = {
				limit = {
					NOT = {
						owner = {
							any_system_within_border = {
								any_fleet_in_system = {
									has_fleet_flag = tup_rebellion_pirate_station
								}
							}
						}
					}
				}
				owner = {
					country_event = {
						id = tup_situation_events.206
						days = 1
					}
				}
			}
		}
	}
	after = {
		destroy_situation = this
	}
}

# This = owner of fleet 1 (winner)
# From = owner of fleet 2 (loser)
# FromFrom = fleet 1
# FromFromFrom = fleet 2
# This fires whenever a fleet is finished with a combat through any means (emergency ftl, destroyed, disabled)
country_event = {
	id = tup_situation_events.205
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		FromFromFrom = {
			has_fleet_flag = tup_rebellion_pirate_station
		}
	}
	immediate = {
		FromFromFrom = {
			remove_fleet_flag = tup_rebellion_pirate_station
		}
		every_playable_country = {
			limit = {
				has_country_flag = tup_crisis_chosen
				has_country_flag = tup_crisis_level_2
				NOT = {
					any_system_within_border = {
						any_fleet_in_system = {
							has_fleet_flag = tup_rebellion_pirate_station
						}
					}
				}
			}
			country_event = {
				id = tup_situation_events.206
			}
		}
	}
}
