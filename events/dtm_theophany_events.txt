# Collection of Devotion for Shrouding Worlds
# can be called from shrouding effect
# has to be called from somewhere
country_event = {
	id = dtm_theophany_objectives.6
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		random_owned_planet = {
			limit = {
				has_planet_flag = dtm_theophany_shardworld@root
			}
			owner = {
				create_message = {
					type = DTM_THEOPHANY_PLANET_MESSAGE_TYPE
					#localization = DTM_THEOPHANY_PLANET_MESSAGE_TYPE
					days = 30
					target = prev
				}
			}
			add_deposit = d_dtm_shards
			if = {
				limit = {
					NOT = {
						has_planet_flag = has_dtm_theophany_shardworld_crust
					}
					count_deposit = {
						count > 5
						limit = {
							is_deposit_type = d_dtm_shards
						}
					}
				}
				create_ambient_object = {
					type = dtm_theophany_shardworld_crust
					location = this
					scale = 2					# should be half of normal planets
					use_3d_location = yes
				}
				set_planet_flag = has_dtm_theophany_shardworld_crust
			}
		}
		complete_crisis_objective = crisobj_dtm_theophany_shrouding_worlds
	}
}

namespace = dtm_theophany_emissary
# First Emissary is selected after unlocking stage 2
country_event = {
	id = dtm_theophany_emissary.1
	title = dtm_theophany_emissary.1.name
	desc = dtm_theophany_emissary.1.desc
	picture = GFX_evt_knights
	show_sound = event_mystic_reveal
	is_triggered_only = yes
	event_window_type = leader_story
	event_picture_background = GFX_leader_recruitment_bg_renowned
	picture_event_data = {
		portrait = event_target:dtm_theophany_emissary
	}
	immediate = {
		change_variable = {
			which = dtm_theophany_emissary
			value = 1
		}
		if = {
			limit = {
				any_owned_leader = {
					is_councilor = yes
					has_trait = leader_trait_psionic
				}
			}
			random_owned_leader = {
				limit = {
					is_councilor = yes
					has_trait = leader_trait_psionic
				}
				save_event_target_as = dtm_theophany_emissary
			}
		}
		else_if = {
			limit = {
				any_owned_leader = {
					has_trait = leader_trait_psionic
				}
			}
			random_owned_leader = {
				limit = {
					has_trait = leader_trait_psionic
				}
				save_event_target_as = dtm_theophany_emissary
			}
		}
		else = {
			create_leader = {
				class = official
				skill = 1
				randomize_traits = yes
				effect = {
					save_event_target_as = dtm_theophany_emissary
				}
			}
		}
	}
	option = {
		name = dtm_theophany_emissary.1.a
		event_target:dtm_theophany_emissary = {
			add_trait = {
				trait = leader_trait_dtm_theophany_emissary
			}
			set_leader_flag = dtm_theophany_emissary
		}
	}
}
