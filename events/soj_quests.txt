country_event = {
  id = soj_q.1001
  title = soj_q.1001.name
  desc = soj_q.1001.desc
  picture = GFX_evt_satellite_in_orbit
	show_sound = event_sensor_ping
  event_chain = soj_military_chain

  immediate = {
		save_event_target_as = owner_country
		owner_species = { save_event_target_as = owner_species }
    capital_scope = { save_event_target_as = capital }

    begin_event_chain = {
				event_chain = soj_military_chain
				target = ROOT
		}

    hidden_effect = {
			random_system = {
				limit = {
					distance = {
						source = event_target:capital
						use_bypasses = no
						min_jumps = 2
						max_jumps = 6
					}
					NOR = {
						exists = space_owner
						has_star_flag = hostile_system
					}
				}
				save_event_target_as = soj_mil_system_1
				set_star_flag = soj_mil_system_1
			}
		}

		create_country = {
			name = "NAME_soj_military"
			species = event_target:owner_species
			type = faction
			name_list = ROOT
			auto_delete = no
			effect = {
				set_country_flag = soj_military_country
				establish_communications_no_message = root
				establish_contact = { who = root location = root.capital_scope }
				create_leader = {
					class = commander
					species = owner_main_species
					name = random
					skill = 2
				}
				create_fleet = {
					name = "NAME_soj_fleet"
					effect = {
						set_owner = prev
						create_ship_design = {
							design = "NAME_Dagger"
						}
						create_ship = {
							name = "NAME_Unyielding_Faith"
							design = last_created_design
						}
						create_ship = {
							name = "NAME_Holy_Avatar"
							design = last_created_design
						}
						create_ship = {
							name = "NAME_Divine_Fury"
							design = last_created_design
							effect = {
								set_ship_flag = soj_military_ship
								set_disable_at_health = 0.05
								save_event_target_as = soj_military_ship
							}
						}
						set_location = event_target:soj_mil_system_1
						set_fleet_stance = aggressive
						set_aggro_range_measure_from = return_point
						set_aggro_range = 50
						assign_leader = last_created_leader
						queue_actions = {
							repeat = {
								wait = {
									duration = 100
								}
							}
						}
					}
				}
				add_ship_design = last_created_design
				#save_event_target_as = radical_cult
				set_name = random
				set_country_flag = soj_rebels_of@root
			}
		}
	}

  option = {
    name = soj_q.1001.a
    tooltip = {
			begin_event_chain = {
				event_chain = "soj_military_chain"
				target = ROOT
			}
		}
    hidden_effect = {
      create_point_of_interest = {
				id = soj_military_chain_poi.1
				name = "soj_mil_system_1_poi"
				desc = "soj_mil_system_1_poi_desc"
				event_chain = "soj_military_chain"
				location = event_target:soj_military_ship
			}
    }
  }
}
