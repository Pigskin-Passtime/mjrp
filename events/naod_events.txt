		}
	}

	immediate = {
		every_owned_pop = {
			limit = {
				species = {
					is_robotic = yes
				}
			}

					change_species = event_target:naod_feder
				}
				5 = {
					change_species = event_target:naod_rukh

				5 = {
					change_species = event_target:naod_schwiff
				}

					change_species = root.species
					if = {
						limit = {
							species = {
								has_psionic_species_trait = no
							}
						}
						modify_species = {
							species = this
							add_trait = trait_latent_psionic
						}
					}
					
				}

			}
			
		}
		
	}
}
