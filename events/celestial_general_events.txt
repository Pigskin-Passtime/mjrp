							count = 2

						}
					}

					}
				}
			}
		}
	}

# Tainted Crystal: End of relic activation, remove flags and wormhole (send message?)
country_event = {
	id = celestial.4
	hide_window = yes
	is_triggered_only = yes

	trigger = {
	}
	
	immediate = {
		remove_country_flag = celestial_crystal_target1_set
		remove_country_flag = celestial_tainted_crystal_used_up
		every_owned_megastructure = {
			limit = {
				has_megastructure_flag = celestial_crystal_wormhole
			}
			remove_megastructure = this
		}
		every_planet_within_border = {
			limit = {
				is_star = yes
				OR = {
					has_planet_flag = celestial_crystal_wormhole_star_1
					has_planet_flag = celestial_crystal_wormhole_star_2
				}
			}
			remove_planet_flag = celestial_crystal_wormhole_star_1
			remove_planet_flag = celestial_crystal_wormhole_star_2
		}
		create_message = {
			type = MESSAGE_TAINTED_CRYSTAL_OVER
			localization = MESSAGE_TAINTED_CRYSTAL_OVER
			days = 30
			target = root
		}
	}
}

# Harmonic Pacification
system_event = {
	id = celestial.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
	}
	
	immediate = {
		random_system_megastructure = {
			limit = {
				is_megastructure_type = celestial_harmonic_resonator
			}
			owner = {
				save_event_target_as = celestial_resonator_owner
			}
		}
		every_fleet_in_system = {
			limit = {
				OR = {
					celestial_is_spaceborne_fauna = yes
					celestial_is_leviathan = yes
				}
				owner = {
					is_hostile = event_target:celestial_resonator_owner
				}
			}
			owner = {
				set_faction_hostility = {
					target = event_target:celestial_resonator_owner
					set_hostile = no
					set_neutral = yes
					set_friendly = no
				}
			}
			if = {
				limit = {
					celestial_is_leviathan = yes
				}
				owner = {
					set_timed_country_flag = {
						flag = celestial_recently_pacified
						days = 3600
					}
					country_event = {
						id = celestial.6
						days = 3600
						scopes = {
							from = event_target:celestial_resonator_owner
						}
					}
				}
			}
		}
		every_neighbor_system = {
			every_fleet_in_system = {
				limit = {
					OR = {
						celestial_is_spaceborne_fauna = yes
						celestial_is_leviathan = yes
					}
					owner = {
						NOT = {
							has_country_flag = celestial_recently_pacified
						}
						is_hostile = event_target:celestial_resonator_owner
					}
				}
				fleet.owner = {
					set_faction_hostility = {
						target = event_target:celestial_resonator_owner
						set_hostile = no
						set_neutral = yes
						set_friendly = no
					}
				}
				if = {
					limit = {
						celestial_is_leviathan = yes
					}
					owner = {
						set_timed_country_flag = {
							flag = celestial_recently_pacified
							days = 3600
						}
						country_event = {
							id = celestial.6
							days = 3600
							scopes = {
								from = event_target:celestial_resonator_owner
							}
						}
					}
				}
			}
		}
		event_target:celestial_resonator_owner = {
			create_message = {
				type = MESSAGE_HARMONIC_PACIFIER
				localization = MESSAGE_HARMONIC_PACIFIER
				days = 30
				target = root
				variable = {
					type = name
					localization = SYSTEM
					scope = root
				}
			}
		}
	}
}

# On built Temporal Spire
planet_event = {
	id = celestial.7
	hide_window = yes

	is_triggered_only = yes

	pre_triggers = {
		is_ai = no
	}

	trigger = {
		has_building = building_celestial_temporal_spire
		NOT = {
			has_planet_flag = celestial_temporal_spire_built_here
		}
	}

	immediate = {
		set_timed_planet_flag = {
			flag = celestial_temporal_spire_built_here
			days = 9000 # 25 years
		}
		add_modifier = {
			modifier = celestial_temporal_spire_modifier
			days = 9000 # 25 years
		}
		planet_event = {
			id = celestial.8
			days = 9000 # 25 years
		}
		planet_event = {
			id = celestial.9
			days = 180
			random = 720
		}
	}
}
