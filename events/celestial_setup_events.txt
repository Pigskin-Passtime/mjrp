# Spatial Distortion Occurs
planet_event = {
	id = celestial_setup.5
	hide_window = yes
	
	is_triggered_only = yes
	trigger = {
	}
	
	immediate = {
		add_modifier = {
			modifier = celestial_isolated_star
			# days = 1800
		}
	
		event_target:celestial_fleet.owner = {
			create_message = {
				type = MESSAGE_STAR_ISOLATED
				localization = MESSAGE_STAR_ISOLATED
				days = 30
				target = root
				variable = {
					type = name
					localization = PLANET
					scope = root
				}
			}
		}
		planet_event = { id = celestial_setup.6 days = 1800 }
		create_ambient_object = {
			type = terraforming_planet_effect
			location = this
			use_3d_location = yes
			entity_scale_to_size = yes
			scripted_scale = 5
		}
		last_created_ambient_object = {
			set_ambient_object_flag = celestial_isolating_effect
		}
		create_ambient_object = {
			type = explosion_particle_object
			location = this
			use_3d_location = yes
			duration = 10
			scale = 30
			play_animation_once = yes
		}
		last_created_ambient_object = {
			set_ambient_object_flag = celestial_isolating_effect
		}
		solar_system = {
			every_system_ambient_object = {
				limit = {
					OR = {
						has_ambient_object_flag = celestial_isolating_beam
					}
				}
				destroy_ambient_object = this
			}
			set_star_flag = celestial_distortion_target_system
			every_neighbor_system = {
				limit = {
					NOR = {
						has_star_flag = celestial_distortion_target_system
						has_star_flag = celestial_distortion_affected_system
					}
				}
				set_star_flag = celestial_distortion_affected_system
				remove_hyperlane = {
					from = this
					to = prev
				}
			}
		}
	}
}

					scope = root
				}
			}
		}
		
		solar_system = {
			every_system_ambient_object = {
				limit = {
					has_ambient_object_flag = celestial_isolating_effect
				}
				destroy_ambient_object = this
			}
			
			remove_star_flag = celestial_distortion_target_system

				limit = { has_star_flag = celestial_distortion_affected_system }
				remove_star_flag = celestial_distortion_affected_system
				add_hyperlane = {
					from = this
					to = prev
				}
			}

	}
