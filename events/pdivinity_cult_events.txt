planet_event = {
    id = pdivinity_cults.100

    hide_window = yes

    is_triggered_only = yes

    trigger = {
        exists = fromfrom
        FROMFROM = {
            is_robot_pop_group = no
            is_sapient = yes
            NOT = { has_trait = "trait_hive_mind" }
            has_purge_type = { country = from type = purge_pdivinity_sacrifice }
        }
        is_controlled_by = FROM
    }

    immediate = {
        every_country = {
            limit = {
                has_communications = FROM
                NOT = { is_country = FROM }
                is_subject = yes
                from = {
                    has_ascension_perk = ap_pdivinity
                }
            }
            while = {
                count = root.local_pop_amount
                add_opinion_modifier = { who = FROM modifier = opinion_pdivinity_sacrifices }
            }
        }
    }
}
