		trigger = { has_crisis_level = wiirlak_ep_evergrowth_protocol_level_1 }

		picture = GFX_evt_astral_rift_lonely_planet
	}

	option = {
		name = GOOD

		if = {
			limit = { has_event_chain = wiirlak_ep_evergrowth_protocol_chain }

			add_event_chain_counter = {
				event_chain = wiirlak_ep_evergrowth_protocol_chain
				counter = wiirlak_ep_evergrowth_protocol_levels
				amount = 1
			}
		}
		else = {
			begin_event_chain = {
				event_chain = wiirlak_ep_evergrowth_protocol_chain
			}

			add_event_chain_counter = {
				event_chain = wiirlak_ep_evergrowth_protocol_chain
				counter = wiirlak_ep_evergrowth_protocol_levels
				amount = 1
			}
		}

		custom_tooltip = new_crisis_perks_tt


		# Final Project at level 5
		if = {
			limit = { has_crisis_level = wiirlak_ep_evergrowth_protocol_level_5 }

			enable_special_project = {
				name = WIIRLAK_EP_FINAL_PROJECT_1
			}

			# Warn that someone is going overboard
			hidden_effect = {
				save_event_target_as = wiirlak_ep_crisis_player

				every_country = {
					limit = {
						is_default_or_fallen = yes
						has_communications = ROOT
						NOT = { is_same_empire = ROOT }
					}

					country_event = {
						id = wiirlak_ep.1200	days = 3
					}
				}

				# Choose Allies
				country_event = {
					id = wiirlak_ep.5000	days = 15
				}
			}
		}
		else_if = {
			limit = {
				has_crisis_level = wiirlak_ep_evergrowth_protocol_level_3
				NOT = { has_global_flag = wiirlak_ep_flag_crisis_level_3 }
			}

			hidden_effect = {
				save_event_target_as = wiirlak_ep_crisis_player
				
				set_global_flag = wiirlak_ep_flag_crisis_level_3

				every_country = {
					limit = {
						is_default_or_fallen = yes
						has_communications = ROOT
						NOT = { is_same_empire = ROOT }
					}

					country_event = {
						id = wiirlak_ep.1100	days = 3
					}
				}
			}
		}
	}
}

				if = {

				else = {
					FROM = {
						complete_crisis_objective = wiirlak_ep_crisobj_blooming_planet

						if = {
							limit = { NOT = { has_country_flag = wiirlak_ep_flag_bloomed_once } }

							set_country_flag = wiirlak_ep_flag_bloomed_once
							country_event = {
								id = wiirlak_ep.1516
							}
						}
					}
				}

				
				# Was it inside another empire territory ?
				if = {
					limit = {
						OR = {
							AND = {
								exists = OWNER
								OWNER = { NOT = { is_same_empire = FROM } }
							}
							has_planet_flag = wiirlak_ep_covert_op
						}
					}

					FROM = {
						complete_crisis_objective = wiirlak_ep_crisobj_blooming_outside_border
					}
				}
			}

			# If it was terraformed but not into a Bloom World
			else = {
				remove_planet_flag = wiirlak_ep_adv_planet
			}
		}
	}

				entity_scale_to_size = yes
				scale = 0.5
			}
		}

		# Contingency Final Machine World
		else_if = {
			limit = {

				base_angle_towards = star
				entity_face_object = star

				entity_offset = { min = 0 max = 0 }

				entity_scale_to_size = yes
				scale = 0.5
			}
		}

		# Normal planets
		else_if = {
			limit = { habitable_planet = yes }

			save_event_target_as = wiirlak_ep_planet_targeted
			if = {
				limit = { exists = OWNER }

					country_event = { id = wiirlak_ep.3600 }
				}
			}
			if = {
				limit = { exists = OWNER }
				FROM.OWNER = {
					save_event_target_as = wiirlak_ep_crisis_reward
					country_event = { id = wiirlak_ep.3700 }
				}
			}
			
			wiirlak_ep_colossus_into_bloom = yes
		}

		else_if = {

			change_pc = pc_shattered
		}

				}

				limit = {
					has_ai_personality_behaviour = holy_planets # not awakened
					NOT = { is_at_war_with = FROM.OWNER }
				}
				save_event_target_as = outraged_FE
			}
			FROM.OWNER = {
				log = RAGE
				save_event_target_as = FE_outrager
				country_event = { id = planet_destruction.610 days = 5 }
			}
		}


		# Resolutions breached

			limit = {
				exists = OWNER
				OWNER = {
					NOT = { is_same_value = FROM.OWNER }
				}
				OR = {
					is_active_resolution = "resolution_rulesofwar_independent_tribunals"
					is_active_resolution = "resolution_rulesofwar_last_resort_doctrine"
					is_active_resolution = "resolution_rulesofwar_demobilization_initiative"
				}
			}
			FROM.OWNER = {
				set_timed_country_flag = { flag = resolution_breached_fired_deluge days = 3600 }
			}
		}
		if = { #you've basically shortcutted terraforming. Yep, that's a breach
			limit = {
				exists = SPACE_OWNER
				SPACE_OWNER = {
					is_same_value = FROM.OWNER
				}
				OR = {
					NOT = { exists = OWNER }
					OWNER = { is_same_value = FROM.OWNER }
				}
				OR = {
					is_active_resolution = "resolution_ecology_environmental_control_board"
					is_active_resolution = "resolution_ecology_paradise_initiative"
				}
			}
			FROM.OWNER = {
				set_timed_country_flag = { flag = resolution_breached_terraformed days = 3600 }
			}
		}
		

		# Generate threat
		if = {
			limit = {
				exists = OWNER
				OWNER = {
					NOT = { is_same_value = FROM.OWNER }
					OR = {
						is_country_type = default
						is_country_type = fallen_empire
						is_country_type = awakened_fallen_empire
					}
				}
			}

			every_country = {
				limit = {

						is_same_value = FROM.OWNER
						is_same_value = ROOT.OWNER
						AND = {
							has_federation = yes
							is_in_federation_with = FROM.OWNER
						}
					}
					OR = {
						has_communications = FROM.OWNER
						has_communications = ROOT.OWNER
					}
				}
				add_opinion_modifier = {
					modifier = wiirlak_ep_opinion_seeded_a_world
					who = FROM.OWNER
				}
			}
			# modifiers for victim
			owner = {
				add_opinion_modifier = {
					modifier = wiirlak_ep_opinion_seeded_my_world
					who = FROM.OWNER
				}
				if = {
					limit = { NOT = { has_ethic = ethic_gestalt_consciousness } }
					add_modifier = {
						modifier = colossus_victim
						days = 10800 # 30 years
					}
				}
			}
		}
		else_if = {
			limit = {
				exists = OWNER
				OWNER = { is_country_type = primitive }
			}
			# modifier for allies + those upset by genocide
			every_country = {
				limit = {
					NOR = { is_same_value = FROM.OWNER }
					OR = {
						has_communications = FROM.OWNER
						has_communications = ROOT.OWNER
					}
					is_country_type = default
					NOR = {
						is_xenophobe = yes
						is_homicidal = yes
						is_xenophile = yes
					}
				}
				add_opinion_modifier = {
					modifier = wiirlak_ep_opinion_seeded_a_primitive_world
					who = FROM.OWNER
				}
			}
			every_country = {
				limit = {
					NOR = { is_same_value = FROM.OWNER }
					OR = {
						has_communications = FROM.OWNER
						has_communications = ROOT.OWNER
					}
					OR = {
						has_ai_personality = awakened_fallen_empire_xenophile
						AND = {
							is_country_type = default
							is_xenophile = yes
						}
					}
				}
				add_opinion_modifier = {
					modifier = wiirlak_ep_opinion_seeded_a_primitive_world_phile
					who = FROM.OWNER
				}

		}
	}

# Bloom Trait - Series 4000
# Bi-yearly - Bloom Trait spreading 
country_event = {
	id = wiirlak_ep.4000
	
	hide_window = yes

	is_triggered_only = yes
	trigger = {
		has_ascension_perk = wiirlak_ep_ap_evergrowth_protocol
		has_menace_perk = wiirlak_ep_menap_terraforming_5
	}
	
	immediate = {
		log = "Bi-annual Evergrowth check, starting:"

		# Own Species are forced to transform
		every_owned_species = {
			limit = { NOT = { has_trait = trait_wiirlak_ep_pc_bloom_preference } }

			set_habitability_trait = {
				trait = trait_wiirlak_ep_pc_bloom_preference
			}
		}
		set_variable = {
			which = wiirlak_ep_var_bloomform_count
			value = 0
		}

		# Random Neighbor
		if = {
			limit = {
				count_neighbor_country = {
					count > 0
					limit = { is_country_type = default }
				}
			}

			random_neighbor_country = {
				log = "Neighbor Picked: [THIS.GetName]"
				wiirlak_ep_spread_trait_country = yes
			}
		}


		# Random Country through the galaxy
		random_country = {
			limit = {
				is_country_type = default
				NOT = { is_same_empire = ROOT }
			}

			# Bloom planets would largely increase chance for spread
			weights = {
				factor = 10
				modifier = {
					factor = 10
					any_owned_planet = {
						is_planet_class = wiirlak_ep_pc_bloom
					}
				}
				modifier = {
					factor = 5
					any_planet_within_border = {
						is_planet_class = wiirlak_ep_pc_bloom
					}
				}
			}

			log = "Random Picked: [THIS.GetName]"
			wiirlak_ep_spread_trait_country = yes
		}


		# Spies help us seeding the Galaxy
		random_spynetwork = {
			limit = {
				has_spy_power > 40
				is_running_espionage_operation = no
			}

			TARGET = {
				log = "Spy Network's Target Picked: [THIS.GetName]"

				wiirlak_ep_spread_trait_country = yes
			}
		}
	}

	after = {
		# We're great and connected, grant Unity
		every_country = {
			limit = { is_default_or_fallen = yes }

			ROOT = {
				change_variable = {
					which = wiirlak_ep_var_bloomform_count
					value = PREV.value:wiirlak_ep_bloomform_count
				}
				add_intel = {
					who = PREV
					amount = 10
				}
				log = "Step: [Prev.GetName] => [THIS.wiirlak_ep_var_bloomform_count]"
			}
		}
		log = "Ressources: [THIS.wiirlak_ep_var_bloomform_count]"

		create_message = {
			type = WIIRLAK_EP_MESSAGE_BLOOMFORM
			days = 40
			localization = wiirlak_ep_message_bloomform_unity
			target = ROOT
		}

		add_resource = {
			unity = 1
			multiplier = wiirlak_ep_var_bloomform_count
		}
	}
}
