# Create Archaeological Site for Mission
ship_event = {
	id = celestial_chain.0
	title = "celestial_chain.0.name"
	desc = "celestial_chain.0.desc"
	picture = GFX_evt_wormhole
	is_triggered_only = yes
	location = event_target:celestial_relic_quest_target

	trigger = {
	}
	
	immediate = {
		solar_system = {
			if = {
				limit = { owner = { has_country_flag = celestial_relic_quest1_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_aetherial_codex
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest2_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_tainted_gem
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest3_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_star_core
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest4_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_profane_arca
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest5_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_quantum_gem
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest6_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_spectral_beacon
				}
			}
			random_system_planet = {
				limit = {
					has_planet_flag = celestial_wayfarer_portal
				}
				save_event_target_as = celestial_relic_quest_target
			}
		}
	}

	option = {
		name = "celestial_chain.0.a"
		custom_tooltip = celestial_relic_quest_target_tt
	}
}

# The Wayfarer Escaped
country_event = {
	id = celestial_chain.111
	title = "celestial_chain.109.name"
	desc = "celestial_chain.111.desc"
	diplomatic = yes
	custom_gui = "enclave_caravaneer_window"
	picture_event_data = {
		portrait = event_target:celestial_awol_leader
		room = crystalline_room
		planet_background = pc_ice_asteroid
    }
	event_chain = celestial_interdimensional_pursuit
	
	is_triggered_only = yes
	trigger = {
	}
	
	immediate = {
		event_target:celestial_crux_outpost_target = {
			solar_system = {
				every_system_ambient_object = {
					limit = {
						OR = {
							has_ambient_object_flag = celestial_lock_gfx
						}
					}
					destroy_ambient_object = this
				}
			}
			
		}
		remove_country_flag = celestial_wayfarer_awol
		clone_leader = {
			target = event_target:celestial_awol_leader
		}
		last_created_leader = {
			set_leader_flag = legendary_leader
			set_leader_flag = celestial
			save_global_event_target_as = celestial_leader
			set_leader_flag = immune_to_negative_traits
			set_owner = root
		}
		if = {
			limit = {
				exists = event_target:celestial_fleet
			}
			event_target:celestial_fleet = {
				assign_leader = event_target:celestial_leader
			}
		}
		else = {
			create_fleet = {
				name = "NAME_TheWayfarer"
				effect = {
					set_owner = root
					create_ship = {
						name = "NAME_NomadsPillage"
						design = "NAME_NomadsShip"
						graphical_culture = "extra_dimensional_01"
						prefix = no
						upgradable = no
					}
					last_created_ship = {
						set_ship_flag = iss_wayfarer
						save_global_event_target_as = iss_wayfarer
						set_disable_at_health = 0.05
					}
					set_location = {
						target = root.capital_scope
						distance = 10
						angle = random
					}
					set_fleet_stance = passive
					assign_leader = event_target:celestial_leader
					save_global_event_target_as = celestial_fleet
				}
				settings = {
					can_upgrade = yes
					can_change_composition = no
					can_change_leader = no
					spawn_debris = no
				}
			}
		}
		event_target:celestial_awol_leader = {
			kill_leader = {
				show_notification = no
			}
		}
	}

	option = {
		name = "celestial_chain.111.a"
		custom_gui = "enclave_caravaneer_option"
		is_dialog_only = yes
		response_text = "celestial_chain.111.a.reply"
	}
	option = {
		name = "celestial_chain.111.b"
		custom_gui = "enclave_caravaneer_option"
		set_country_flag = celestial_eventchain_1_completed
		end_event_chain = celestial_interdimensional_pursuit
		custom_tooltip = celestial_wayfarer_returns
		event_target:celestial_leader = {
			add_trait = leader_trait_paranoid
			add_trait = leader_trait_traumatized
			if = {
				limit = { owner = { has_country_flag = celestial_helped_astral_corps_1 } }
				add_age = 30
			}
			else = {
				add_age = 20
			}
		}
	}
	
}
