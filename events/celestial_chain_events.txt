ship_event = {
	id = celestial_chain.0
	title = "celestial_chain.0.name"
	desc = "celestial_chain.0.desc"
	picture = GFX_evt_wormhole
	is_triggered_only = yes
	location = event_target:celestial_relic_quest_target

	trigger = {
	}
	
	immediate = {
		solar_system = {
			if = {
				limit = { owner = { has_country_flag = celestial_relic_quest1_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_aetherial_codex
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest2_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_tainted_gem
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest3_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_star_core
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest4_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_profane_arca
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest5_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_quantum_gem
				}
			}
			else_if = {
				limit = { owner = { has_country_flag = celestial_relic_quest6_selected } }
				create_celestial_mission_vortex = {
					CELESTIAL_MISSION_TYPE = r_celestial_spectral_beacon

				limit = {
					has_planet_flag = celestial_wayfarer_portal

			}

		name = "celestial_chain.0.a"
		custom_tooltip = celestial_relic_quest_target_tt
	}
