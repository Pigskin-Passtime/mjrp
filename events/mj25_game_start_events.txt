country_event = {									# picking out the "host" (in the case of observer host, first empire to spawn will pick)
	id = mj25_game_start.3
	is_triggered_only = yes
	hide_window = yes
	fire_only_once = yes

	trigger = {
		NOT = {
			has_global_flag = host_assigned
		}
	}

	immediate = {	
		if = {
			limit = {
				NOT = {
					has_global_flag = host_assigned
				}
			}
			set_country_flag = game_host
			set_global_flag = host_assigned
		}
	}
}

country_event = {										# everyone gets an advanced start
	id = mj25_game_start.4
	is_triggered_only = yes
	hide_window = yes
	

	trigger = {
		is_country_type = default
		OR = {
			has_global_flag = year_2225_start
			has_global_flag = year_2250_start
			has_global_flag = year_2275_start
			has_global_flag = year_2300_start
		}
	}

	immediate = {
		set_update_modifiers_batch = BEGIN				
		set_spawn_system_batch = BEGIN
		create_tech_country = yes
		copy_techs_from = {
			target = event_target:tech_country
		}
		generate_starting_resources = yes
		generate_starting_borders = yes
#		calculate_modifier = yes
		set_spawn_system_batch = END
		set_update_modifiers_batch = END
		if = {
			limit = {
				has_global_flag = ready_to_generate_empires
			}
			every_country = {
				limit = {
					is_country_type = default
					NOT = {
						has_country_flag = setup_complete
					}
				}
				fire_on_action = {
					on_action = on_ready_for_empire_generation
				}
			}
		}
		#calculate_modifier = yes
		
	}
}
