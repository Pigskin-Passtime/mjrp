				can_assign_to_council = no

					trait = subclass_commander_admiral

					trait = leader_trait_master_gunner
				}
				gender = indeterminable
				set_age = 152
				effect = {
					save_global_event_target_as = wp02_leader

					set_leader_flag = immune_to_negative_traits
					set_leader_flag = leader_death_events_blocked
					set_owner = root
					exile_leader_as = wp02_leader
				}
			}
		}
	}

	option = {
		name = HIRE
		tag = hire_leader
		custom_tooltip = wp02_joins
		sound = par_beholder_hire

		hidden_effect = {
			clone_leader = {
				target = event_target:wp02_leader
			}
			last_created_leader = {
				save_global_event_target_as = wp02_leader
				set_leader_flag = legendary_leader
				set_leader_flag = immune_to_negative_traits
				set_leader_flag = leader_death_events_blocked

					limit = {
						has_paragon_dlc = no
					}
					add_skill = 1
				}
				add_trait_no_notify = leader_trait_wp02
				set_owner = root
			}
			create_fleet = {
				name = "wp02_name"
				settings = {
					can_upgrade = no
					can_change_composition = no
					can_change_leader = no

						name = "NAME_wp02_ship"

						upgradable = no

							set_ship_flag = is_wp02_ship

						}
					}
					if = {
						limit = {
							exists = event_target:wp02_planet
						}
						set_location = {
							target = event_target:wp02_planet
							distance = 5
							angle = 275
						}
						destroy_ambient_object = event_target:wp02_station_object
					}
					else = {
						set_location = {
							target = owner.capital_scope
							distance = 1
							angle = random
						}
					}
					save_global_event_target_as = wp02_fleet
					assign_leader = event_target:wp02_leader
				}
			}
			
			create_message = {
				type = MESSAGE_RECRUITED_LEADER
				localization =  MESSAGE_RECRUITED_LEADER_DESC

					key = "border"
