###Zob adds a mind, Gateway. Cyclical, triggered by hiring Zob. Ends if you lose Zob.
country_event = {
	id = zob.30
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = zob_hired
		is_ai = no
		# exists = event_target:zob_leader
	}

	immediate = {
		count_zob_collective = yes
		#Option to Uplift once you have Epigenetic Triggers
		if = {
			limit = {
				NOT = { has_country_flag = zob_uplift }
				has_technology = tech_epigenetic_triggers
				is_hive_empire = no #Hives can uplift normally?
			}
			set_country_flag = zob_uplift
			country_event = {
				id = zob.40
			}
			country_event = {
				id = zob.30
				days = 1000
				random = 1000
			}
		}
		#Expand the Collective if there is room
		else_if = {
			limit = {
				check_variable = {
					which = zob_collective_size
					value < 4
				}
				any_owned_leader = {
					can_be_zob_collected = yes
				}
			}
			country_event = {
				id = zob.31
			}
		}
		#Check again later
		else = {
			country_event = {
				id = zob.30
				days = 4000
				random = 1000
			}
		}		
	}
}
