fleet_event = {
	id = zob_stage.4
	title = "zob_stage.4.name"
	desc = "zob_stage.4.desc"
	picture = GFX_evt_crusher_droid
	show_sound = event_metal_fatigue

	archaeology = yes

	is_triggered_only = yes

	immediate = {
		from = { set_site_progress_locked = yes }
		leader = {
			save_event_target_as = collected_scientist
		}
	}

	after = {
		from = { set_site_progress_locked = no }
		event_target:collected_scientist = {
			exile_leader_as = collected_scientist
		}
	}

	option = {
		name = ONSCREEN
		owner = {
			country_event = {
				id = zob.10
				days = 3
			}
		}
	}
}


###For Testing, this assigns a collected scientist in case you fired this event from console and dont have an event target already. 
country_event = {
	id = zob.9
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { exists = event_target:collected_scientist }
			}
			random_owned_leader = {
				limit = {
					leader_class = scientist
					is_gestalt_node = no
					is_ruler = no
					is_heir = no
				}
				add_trait = trait_ruler_charismatic
				save_event_target_as = collected_scientist
				exile_leader_as = collected_scientist
			}
		}
		random_galaxy_planet = {
			limit = { 
				has_planet_flag = zob_planet
			}
			save_global_event_target_as = zob_planet
		}
		country_event = {
			id = zob.10
		}
	}
}

###Trait Variable Tracker refresh, this is on level up, adds a delay so you can pick a new trait.
country_event = {
	id = zob.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = zob_hired
		from = {
			OR = {
				has_trait = leader_trait_zob_mind_collector
				has_trait = leader_trait_zob_collected
			}
		}
	}

	immediate = {
		country_event = {
			id = zob.21
			days = 30
		}		
	}
}

###Trait Variable Tracker refresh, this is on yearly pulse, and from zob.20, but only applied if you have Zob.
country_event = {
	id = zob.21
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = zob_hired
	}

	immediate = {
		every_owned_leader = {
			limit = {
				OR = {
					has_trait = leader_trait_zob_mind_collector
					has_trait = leader_trait_zob_collected
				}
			}
			update_zob_variables = yes
		}		
	}
