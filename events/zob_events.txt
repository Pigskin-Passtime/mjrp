	immediate = {
		from = { set_site_progress_locked = yes }
		leader = {
			save_event_target_as = collected_scientist
		}
	}

	after = {
		from = { set_site_progress_locked = no }
		event_target:collected_scientist = {
			exile_leader_as = collected_scientist
		}
	}

	option = {
		name = ONSCREEN
		country_event = {
			id = zob.10
			days = 3
		}
	}
}


###For Testing, this assigns a collected scientist in case you fired this event from console and dont have an event target already. 
country_event = {
	id = zob.9
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		if = {
			limit = {
				NOT = { exists = event_target:collected_scientist }
			}
			random_owned_leader = {
				limit = {
					leader_class = scientist
					is_gestalt_node = no
					is_ruler = no
				}
				add_trait = leader_trait_adaptable
				save_event_target_as = collected_scientist
				exile_leader_as = collected_scientist
			}
		}
		country_event = {
			id = zob.10
		}
	}
}

###Trait Variable Tracker on Leader Level up, (Delayed so you can pick a trait)
country_event = {
	id = zob.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		from = {
			has_trait = leader_trait_zob_collected
		}
	}

	immediate = {
		country_event = {
			id = zob.20
			days = 30
		}
		update_zob_variables = yes
	}
}

###Trait Variable Tracker refresh, triggered above and by yearly backup event in case you time out the trait selection
country_event = {
	id = zob.21
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = zob_hired
	}

	immediate = {
		every_owned_leader = {
			limit = {
				OR = {
					has_trait = leader_trait_zob_mind_collector
					has_trait = leader_trait_zob_collected
				}
			}
			update_zob_variables = yes
		}		
	}
}
