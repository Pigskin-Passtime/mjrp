country_event = {
	id = TILLG.7
	diplomatic = yes
	is_triggered_only = yes
	picture_event_data = {
		city_level = this
		graphical_culture = this
		portrait = event_target:THEFRAGMENT
	}

	title = "TILLG.7.Title"
	desc = "TILLG.7.Desc"

	#We Annex the Struggling Colony
	option = {
		name = "TILLG.7.a"
		hidden_effect = {
			set_country_flag = ANNEXTED_FRAGMENT
			event_target:FINALFRAGMENTSHOWDOWN = {
				modify_species = {
					species = event_target:FRAGMENTSPECIES
					remove_trait = trait_machine_unit
				}
				set_owner = from
			}
		}
	}
	#We vassalise the Struggling colony
	option = {
		name = "TILLG.7.b"

		hidden_effect = {
			set_country_flag = VASSALIZED_FRAGMENT
			#This took Several hours of trial and error and is really Jank, but it works.
			random_country = {
				limit = {
					has_country_flag = hired_THEFRAGMENT
				}
				save_event_target_as = ITISME
			}
			event_target:THEHIDEAWAYS = {
				set_subject_of = {
					who = event_target:ITISME
					preset = preset_vassal
				}
			}

		}
	}
	#The Colony goes back into hiding.
	option = {
		name = "TILLG.7.c"
		hidden_effect = {
			set_country_flag = REFUSED_FRAGMENT
			event_target:THEHIDEAWAYS = {
				destroy_country = yes
			}
			event_target:FINALFRAGMENTSHOWDOWN = {
				destroy_colony = yes
				every_owned_pop = {
					kill_pop = yes
				}
				change_pc = pc_shielded
			}
		}
	}

	after = {
		country_event = {
			id = TILLG.8
		}
	}
}
