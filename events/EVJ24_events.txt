					planet = {
						export_trigger_value_to_variable = {
							trigger = num_assigned_jobs

							variable = EVJ24_timeless_vigil_bonus_planet

						while = {
							count = this.EVJ24_timeless_vigil_bonus_planet
							root = {
								change_variable = { which = EVJ24_timeless_vigil_bonus value = 4 }
							}
						}
					}
				}
				if = {
					limit = { exists = planet.sector }
					planet.sector = {
						every_owned_planet = { 
							limit = { NOT = { is_same_value = event_target:EVJ24_jailer_leader.planet } }
						 	export_trigger_value_to_variable = {
								trigger = num_assigned_jobs
								parameters = { job = soldier }
								variable = EVJ24_timeless_vigil_bonus_planet
							}

							while = {
								count = this.EVJ24_timeless_vigil_bonus_planet
								root = {
									change_variable = { which = EVJ24_timeless_vigil_bonus value = 2 }
								}
							}
						}
					}
				}
			}
		}
		if = {
			limit = { has_modifier = EVJ24_timeless_vigil_bonus }
			remove_modifier = EVJ24_timeless_vigil_bonus
		}
		add_modifier = {
			modifier = EVJ24_timeless_vigil_bonus
			days = -1
			multiplier = this.EVJ24_timeless_vigil_bonus

country_event = {
	id = EVJ24.1000
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		any_owned_leader = { has_leader_flag = EVJ24_jailer }
	}

	immediate = {
		
		if = {
			limit = {
				AND = {
					has_country_flag = EVJ24_middle4_fired 
					mid_game_years_passed >= 0
					NOT = { has_country_flag = EVJ24_end1_fired }
				}
			}
			# Finale event
			
			set_country_flag = EVJ24_end1_fired
			country_event = { id = EVJ24.805 }
			
		}
		else_if = {
			limit = { AND = { has_country_flag = EVJ24_story_event_fired NOT = { has_country_flag = EVJ24_end1_fired } } }
			remove_country_flag = EVJ24_story_event_fired
			
			random_list = {
				# Flavour events - repeatable
				8 = {
					country_event = { id = EVJ24.60 }
				}
				# Flavour events - one off
				12 = {
					if = {
						limit = { NOT = { has_country_flag = EVJ24_30_flavour } }
						set_country_flag = EVJ24_30_flavour
						country_event = { id = EVJ24.30 }
					}
					else_if = {
						limit = { NOT = { has_country_flag = EVJ24_35_flavour } }
						set_country_flag = EVJ24_35_flavour
						country_event = { id = EVJ24.35 }
					}
					else_if = {
						limit = { NOT = { has_country_flag = EVJ24_40_flavour } }
						set_country_flag = EVJ24_40_flavour
						country_event = { id = EVJ24.40 }
					}
					else_if = {
						limit = { NOT = { has_country_flag = EVJ24_50_flavour } }
						set_country_flag = EVJ24_50_flavour
						country_event = { id = EVJ24.50 }
					}
				}
			}
		}
		else_if = {
			
			# Story events
			
			set_country_flag = EVJ24_story_event_fired
			
			if = {
				limit = { AND = { has_country_flag = EVJ24_middle3_fired NOT = { has_country_flag = EVJ24_middle4_fired } } }
				set_country_flag = EVJ24_middle4_fired
				country_event = { id = EVJ24.250 }
			}
			else_if = {
				limit = { AND = { has_country_flag = EVJ24_middle2_fired NOT = { has_country_flag = EVJ24_middle3_fired } } }
				set_country_flag = EVJ24_middle3_fired
				country_event = { id = EVJ24.200 }
			}
			else_if = {
				limit = { AND = { has_country_flag = EVJ24_middle1_fired NOT = { has_country_flag = EVJ24_middle2_fired } } }
				set_country_flag = EVJ24_middle2_fired
				country_event = { id = EVJ24.150 }
			}
			else_if = {
				limit = { NOT = { has_country_flag = EVJ24_middle1_fired } }
				set_country_flag = EVJ24_middle1_fired
				country_event = { id = EVJ24.99 }
			}
		}
		country_event = { id = EVJ24.1000 days = 1000 random = 500 }
	}
}
