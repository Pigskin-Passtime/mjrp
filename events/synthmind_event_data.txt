country_event = {
	id = synthmind.98
	hide_window = yes
	is_triggered_only = yes
	is_test_event = yes
	
	immediate = {
		set_global_flag = synthmind_system_spawned
		
		random_system = {
			save_event_target_as = architects_sphere_spawn
		}

		event_target:architects_sphere_spawn = {
			if = {
				limit = { host_has_dlc = "Ancient Relics Story Pack" }
				spawn_system = {
					min_distance = 10
					max_distance = 30
					max_jumps = 0
					initializer = synthmind_architects_sphere_initializer_relic
				}
				
				
			}
			else = {
				spawn_system = {
					min_distance = 10
					max_distance = 30
					max_jumps = 0
					initializer = synthmind_architects_sphere_initializer_gaia
				}
			}
			if = {
				limit = { last_created_system = { 
					NOT = { has_hyperlane_to = prev } 
				} }
				add_hyperlane = {
					from = this
					to = last_created_system
				}
			}
			else = {
				add_extra_hyperlane_to_spawned_system_effect = yes
			}
			last_created_system = {
				save_global_event_target_as = synthmind_system
			}
		}
	}
}
