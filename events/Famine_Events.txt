country_event = {
	id = Famine.8
	fire_only_once = yes
	hide_window = yes
	is_triggered_only = yes
	
	trigger = {
		has_global_flag = TGF_Main_in_progress
	}
	
	immediate = {

		# Start of system and empire spawning, written by mostly by Exakan#5115 who helped me for the modjam

		no_scope = {
			set_spawn_system_batch = begin
				random_rim_system = {
					if = {
						limit = { galaxy_size = tiny }
						spawn_system = {
							min_distance >= -200								
							max_distance <= -199
							direction = rimwards
							initializer = Famine_system_initializer
							hyperlane = no
							effect = { save_event_target_as = Famine_system }
						}
					}
					else_if = {
						limit = { galaxy_size = small }
						spawn_system = {
							min_distance >= -400								
							max_distance <= -399
							direction = rimwards
							initializer = Famine_system_initializer
							hyperlane = no
							effect = { save_event_target_as = Famine_system }
						}
					}
					else_if = {
						limit = { galaxy_size = medium }
						spawn_system = {
							min_distance >= -600								
							max_distance <= -599
							direction = rimwards
							initializer = Famine_system_initializer
							hyperlane = no
							effect = { save_event_target_as = Famine_system }
						}
					}
					else_if = {
						limit = { galaxy_size = large }
						spawn_system = {
							min_distance >= -800							
							max_distance <= -799
							direction = rimwards
							initializer = Famine_system_initializer
							hyperlane = no
							effect = { save_event_target_as = Famine_system }
						}
					}
					else = {
						spawn_system = {
							min_distance >= -1000					
							max_distance <= -999
							direction = rimwards
							initializer = Famine_system_initializer
							hyperlane = no
							effect = { save_event_target_as = Famine_system }
						}
					}
				}
				event_target:Famine_system = {
					set_name = "Goblon"
					solar_system = {
						spawn_natural_wormhole = {
							bypass_type = TGF_wormhole
								orbit_distance = -10
								orbit_angle = 270
								random_pos = no
						}
						random_system_planet = {
							limit = { is_star = yes }
							save_event_target_as = main_star
							save_global_event_target_as = Famine_system	 #
						}
						
						create_species = {
							name = "Human"						
							class = HUM
							portrait = "human_male_02" 					
							namelist = random
							traits = random
							effect = { save_event_target_as = Famine_species }
						}
						
						create_country = {
							name = "Galacic Empire"								
							type = "enemy"							
							ignore_initial_colony_error = yes
							name_list = "STAR_WARS"			
							species = event_target:Famine_species
							flag = {
								icon= {
									category = "TGF_flag"
									file = "mod_flag_22.dds"
								}
								background = {
									category = "backgrounds"
									file = "circle.dds"
								}
								colors = {
									"red"
									"red"
									"null"
									"null"
								}
							}
							effect = {
								save_event_target_as = Famine_country
								save_global_event_target_as = Famine_country
							}
						}
						
						event_target:Famine_country = {
							set_country_flag = Passageway
							create_leader = {
								class = admiral
								species = owner_main_species
								name = random
								skill = 7
								leader_age_min = 25
								leader_age_max = 45
								traits = { trait = random_trait }
							}
						}
						create_fleet = {
							effect = {
								set_owner = event_target:Famine_country
								create_ship = {
									name = "Neutron Launcher"
									design = "NAME_Swarm_Starbase"		
									#graphical_culture = "mammalian_01"
								}
								set_location = {
									target = event_target:main_star
									distance = 55
									angle = 270
								}
								save_event_target_as = Famine_starbase
							}
						}
						create_fleet = {
							name = "Galacic Death I"									
							effect = {
								set_owner = event_target:Famine_country     
								set_formation_scale = 2
								while = {
									count = 20
									create_ship = {
										name = random
										design = "Galacic_Fleet_2"       # Fleet design
										graphical_culture = "mammalian_01"
									}
								}
								while = {
									count = 15
