# Workplace Accident
country_event = {
	id = celestial_flavour.1
	title = celestial_flavour.1.name
	picture = GFX_evt_friendly_enlightenment
	show_sound = event_default
	is_triggered_only = yes
	fire_only_once = yes

	desc = {
		trigger = {
			is_gestalt = no
		}
		text = celestial_flavour.1.desc
	}
	desc = {
		trigger = {
			is_gestalt = yes
		}
		text = celestial_flavour.1.desc.g
	}

	trigger = {
		exists = event_target:celestial_leader
		any_owned_leader = {
			leader_class = scientist
			NOT = { has_leader_flag = celestial }
		}
	}

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_legendary"

	picture_event_data = {
		portrait = event_target:celestial_leader
	}
	
	immediate = {
		set_country_flag = celestial_flavour_1
		random_owned_leader = {
			limit = {
				leader_class = scientist
				NOT = { has_leader_flag = celestial }
			}
			save_event_target_as = celestial_flavour_event_leader
		}
	}

	option = { # Reprimand them.
		name = celestial_flavour.1.a
		event_target:celestial_flavour_event_leader = {
			add_trait = leader_trait_maimed
		}
		event_target:celestial_leader = { # add tooltip about age
			add_age = 10
		}
		add_monthly_resource_mult = {
			resource = energy
			value = 5
			min = 1000
			max = 50000
		}
		add_monthly_resource_mult = {
			resource = influence
			value = 20
			min = 100
			max = 500
		}
	}
	
	option = { # Science can be dangerous
		name = celestial_flavour.1.b
		event_target:celestial_flavour_event_leader = {
			add_trait = leader_trait_maimed
		}
		event_target:celestial_leader = {
			add_skill = 1
		}
	}
}

# A Most Peculiar Endeavour
country_event = {
	id = celestial_flavour.2
	title = celestial_flavour.2.name
	picture = GFX_evt_astral_rift_desert_ruins_third_chamber
	show_sound = event_default
	is_triggered_only = yes
	fire_only_once = yes

	desc = {
		trigger = {
			is_gestalt = no
		}
		text = celestial_flavour.2.desc
	}
	desc = {
		trigger = {
			is_gestalt = yes
		}
		text = celestial_flavour.2.desc.g
	}

	trigger = {
		exists = event_target:celestial_leader
	}

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_legendary"

	picture_event_data = {
		portrait = event_target:celestial_leader
	}
	
	immediate = {
		set_country_flag = celestial_flavour_2
	}

	option = { # Permit it.
		name = celestial_flavour.2.a
		trigger = {
			is_gestalt = no
		}
		allow = {
			has_resource = {
				type = energy
				amount >= 2000
			}
			has_resource = {
				type = consumer_goods
				amount >= 1000
			}
		}
		add_resource = {
			energy = -2000
			consumer_goods = -1000
		}
		hidden_effect = {
			country_event = {
				id = celestial_flavour.3
				days = 60
				random = 60
			}
		}
	}
	
	option = { # Permit it.
		name = celestial_flavour.2.a
		trigger = {
			is_gestalt = yes
		}
		allow = {
			has_resource = {
				type = energy
				amount >= 2000
			}
			has_resource = {
				type = minerals
				amount >= 3000
			}
		}
		add_resource = {
			energy = -2000
			minerals = -3000
		}
		hidden_effect = {
			country_event = {
				id = celestial_flavour.3
				days = 60
				random = 60
			}
		}
	}
	
	option = { # We cannot afford to do so.
		name = celestial_flavour.2.b
		event_target:celestial_leader = { # add tooltip about age
			add_age = 5
			add_experience = -50
		}
		hidden_effect = {
			country_event = {
				id = celestial_flavour.4
				days = 60
				random = 60
			}
		}
	}
}

	show_sound = event_default
	is_triggered_only = yes
	fire_only_once = yes

	desc = {
		trigger = {
			is_gestalt = no
		}
		text = celestial_flavour.6.desc
	}
	desc = {
		trigger = {
			is_gestalt = yes
		}
		text = celestial_flavour.6.desc.g
	}

	trigger = {
		exists = event_target:celestial_leader
	}

	event_window_type = leader_story
	event_picture_background = "GFX_leader_recruitment_bg_legendary"

	picture_event_data = {
		portrait = event_target:celestial_leader
	}
	
	immediate = {
		set_country_flag = celestial_beasts_invade_country
		capital_scope = {
			save_event_target_as = celestial_incursion_target
			set_planet_flag = celestial_beasts_invade_planet
		}
	}

	option = { # Battlestations!
		name = celestial_flavour.6.a
		hidden_effect = {
			create_country = {
				name = "NAME_Astral_Incursion"
				type = faction
				effect = {
					establish_communications_no_message = root.owner
					set_country_flag = celestial_astral_incursion_country
				}
			}
		}
		event_target:celestial_incursion_target = {
			while = {
				count = 3
				create_army = {
					owner = last_created_country
					type = celestial_legion
				}
			}
		}
	}
}
# Arcane Powers - Invaders Repelled
country_event = {
	id = celestial_flavour.7
	title = celestial_flavour.7.name
	picture = GFX_evt_victorious_army
	show_sound = event_default
	is_triggered_only = yes
	fire_only_once = yes
	location = FROMFROM

	desc = {
		trigger = {
			is_gestalt = no
		}
		text = celestial_flavour.7.desc
	}
	desc = {
		trigger = {
			is_gestalt = yes
		}
		text = celestial_flavour.7.desc.g
	}

	trigger = {
		exists = event_target:celestial_leader
		has_country_flag = celestial_beasts_invade_country
		fromfrom = {
			has_planet_flag = celestial_beasts_invade_planet
		}
	}
	
	immediate = {
		remove_country_flag = celestial_beasts_invade_country
		fromfrom = {
			remove_planet_flag = celestial_beasts_invade_planet
		}
	}

	option = { # Lesson learned.
		name = celestial_flavour.7.a
		if = {
			limit = {
				has_astral_planes_dlc = yes
			}
			add_modifier = {
				modifier = celestial_astral_studies_astralplanes
				days = 3600
			}
		}
		else = {
			add_modifier = {
				modifier = celestial_astral_studies
				days = 3600
			}
		}
		event_target:celestial_leader = { # add tooltip about age?
			add_age = 10
		}
	}
}
