planet_event = {
	id = containment.11
	title = "containment.11.name"
	desc = "containment.11.desc"
	picture = GFX_evt_containment_beta
	location = ROOT


	is_triggered_only = yes

	trigger = {
		# exists = space_owner
		# space_owner = { has_origin = origin_containment }
	}

	immediate = {
		# random_moon = {
			# limit = { has_planet_flag = containment_moon_beta }
		# }
	}

	option = {
		name = containment.11.a
		random_moon = {
			limit = { has_planet_flag = containment_moon_beta }
			change_pc = pc_barren_cold
			clear_deposits = yes
			if = {
				limit = { has_mining_station = yes }
				random_mining_station = { dismantle = yes }
			}
			locked_random_list = {
				10 = { add_deposit = d_physics_2 }
				10 = { add_deposit = d_society_2 }
				10 = { add_deposit = d_engineering_2 }
			}
		}
		hidden_effect = {
			solar_system = {
				random_system_ambient_object = {
					limit = { has_ambient_object_flag = ambient_containment_moon_beta_flag }
					destroy_ambient_object = this
				}
			}
		}
		remove_deposit = d_containment_electro_storm
		planet_event = { id = containment.12 days = 1800 } # TODO  change time!  3 days for testing purposes
	}
}
